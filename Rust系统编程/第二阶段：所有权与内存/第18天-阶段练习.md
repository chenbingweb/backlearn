## 项目：实现线程安全的缓存系统

### 需求

1. 支持多线程并发读写
2. 支持最大容量限制
3. 支持过期时间
4. 线程安全的统计信息

### 完整实现

```rust
use std::collections::HashMap;
use std::sync::{Arc, RwLock};
use std::time::{Duration, Instant};

// ======== 缓存项 ========
#[derive(Clone, Debug)]
struct CacheEntry<V> {
    value: V,
    created_at: Instant,
    ttl: Option<Duration>,
}

impl<V> CacheEntry<V> {
    fn new(value: V, ttl: Option<Duration>) -> CacheEntry<V> {
        CacheEntry {
            value,
            created_at: Instant::now(),
            ttl,
        }
    }

    fn is_expired(&self) -> bool {
        match self.ttl {
            Some(ttl) => self.created_at.elapsed() > ttl,
            None => false,
        }
    }
}

// ======== 统计信息 ========
#[derive(Debug, Default)]
struct Stats {
    hits: u64,
    misses: u64,
    evictions: u64,
}

// ======== 缓存 ========
pub struct Cache<K, V> {
    data: RwLock<HashMap<K, CacheEntry<V>>>,
    stats: RwLock<Stats>,
    max_size: usize,
}

impl<K, V> Cache<K, V>
where
    K: Eq + std::hash::Hash + Clone,
    V: Clone,
{
    pub fn new(max_size: usize) -> Cache<K, V> {
        Cache {
            data: RwLock::new(HashMap::new()),
            stats: RwLock::new(Stats::default()),
            max_size,
        }
    }

    // 获取值
    pub fn get(&self, key: &K) -> Option<V> {
        let data = self.data.read().unwrap();

        match data.get(key) {
            Some(entry) if !entry.is_expired() => {
                self.stats.write().unwrap().hits += 1;
                Some(entry.value.clone())
            }
            _ => {
                self.stats.write().unwrap().misses += 1;
                None
            }
        }
    }

    // 设置值（无过期时间）
    pub fn set(&self, key: K, value: V) {
        self.set_with_ttl(key, value, None);
    }

    // 设置值（有过期时间）
    pub fn set_with_ttl(&self, key: K, value: V, ttl: Option<Duration>) {
        let mut data = self.data.write().unwrap();

        // 如果满了，移除一个条目
        if data.len() >= self.max_size && !data.contains_key(&key) {
            self.evict_one(&mut data);
        }

        data.insert(key, CacheEntry::new(value, ttl));
    }

    // 删除值
    pub fn remove(&self, key: &K) -> Option<V> {
        let mut data = self.data.write().unwrap();
        data.remove(key).map(|entry| entry.value)
    }

    // 清理过期条目
    pub fn cleanup(&self) {
        let mut data = self.data.write().unwrap();
        let before = data.len();
        data.retain(|_, entry| !entry.is_expired());
        let evicted = before - data.len();
        self.stats.write().unwrap().evictions += evicted as u64;
    }

    // 获取统计
    pub fn stats(&self) -> Stats {
        self.stats.read().unwrap().clone()
    }

    // 当前大小
    pub fn len(&self) -> usize {
        self.data.read().unwrap().len()
    }

    // 清空
    pub fn clear(&self) {
        let mut data = self.data.write().unwrap();
        data.clear();
    }

    // 淘汰一个条目（简单策略：随机移除）
    fn evict_one(&self, data: &mut HashMap<K, CacheEntry<V>>) {
        if let Some(key) = data.keys().next().cloned() {
            data.remove(&key);
            self.stats.write().unwrap().evictions += 1;
        }
    }
}

// 线程安全的 Arc 包装
pub type SharedCache<K, V> = Arc<Cache<K, V>>;

pub fn create_cache<K, V>(max_size: usize) -> SharedCache<K, V>
where
    K: Eq + std::hash::Hash + Clone,
    V: Clone,
{
    Arc::new(Cache::new(max_size))
}

// ======== 测试 ========
#[cfg(test)]
mod tests {
    use super::*;
    use std::thread;

    #[test]
    fn test_basic() {
        let cache = Cache::new(10);
        cache.set("key1", "value1");
        assert_eq!(cache.get(&"key1"), Some("value1"));
        assert_eq!(cache.get(&"key2"), None);
    }

    #[test]
    fn test_ttl() {
        let cache = Cache::new(10);
        cache.set_with_ttl("key", "value", Some(Duration::from_millis(50)));
        assert_eq!(cache.get(&"key"), Some("value"));

        std::thread::sleep(Duration::from_millis(100));
        assert_eq!(cache.get(&"key"), None);
    }

    #[test]
    fn test_max_size() {
        let cache = Cache::new(2);
        cache.set(1, "a");
        cache.set(2, "b");
        cache.set(3, "c");  // 应该触发淘汰

        assert!(cache.len() <= 2);
    }

    #[test]
    fn test_concurrent() {
        let cache = create_cache::<i32, String>(100);
        let mut handles = vec![];

        for i in 0..10 {
            let c = Arc::clone(&cache);
            let handle = thread::spawn(move || {
                c.set(i, format!("value-{}", i));
                c.get(&i)
            });
            handles.push(handle);
        }

        for handle in handles {
            assert!(handle.join().unwrap().is_some());
        }
    }
}

fn main() {
    let cache = create_cache::<String, String>(100);

    cache.set(String::from("host"), String::from("localhost"));
    cache.set(String::from("port"), String::from("8080"));

    println!("host: {:?}", cache.get(&String::from("host")));
    println!("port: {:?}", cache.get(&String::from("port")));
    println!("stats: {:?}", cache.stats());

    // TTL 测试
    cache.set_with_ttl(
        String::from("temp"),
        String::from("temporary"),
        Some(Duration::from_secs(1)),
    );
    println!("temp (before): {:?}", cache.get(&String::from("temp")));

    std::thread::sleep(Duration::from_secs(2));
    println!("temp (after): {:?}", cache.get(&String::from("temp")));
}
```

| 知识点 | 应用 |
|--------|------|
| 泛型 | `Cache<K, V>` |
