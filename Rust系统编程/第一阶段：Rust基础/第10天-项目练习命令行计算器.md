use std::env;

// ======== 错误类型 ========
#[derive(Debug)]
enum CalcError {
    DivisionByZero,
    InvalidNumber(String),
    InvalidOperator(char),
    MismatchedParentheses,
    EmptyExpression,
}

impl std::fmt::Display for CalcError {
    fn fmt(&self, f: &mut std::fmt::Formatter) -> std::fmt::Result {
        match self {
            CalcError::DivisionByZero => write!(f, "错误：除数不能为零"),
            CalcError::InvalidNumber(s) => write!(f, "错误：无效数字 '{}'", s),
            CalcError::InvalidOperator(c) => write!(f, "错误：无效运算符 '{}'", c),
            CalcError::MismatchedParentheses => write!(f, "错误：括号不匹配"),
            CalcError::EmptyExpression => write!(f, "错误：表达式为空"),
        }
    }
}

impl std::error::Error for CalcError {}

// ======== Token 类型 ========
#[derive(Debug, Clone)]
enum Token {
    Number(f64),
    Plus,
    Minus,
    Multiply,
    Divide,
    Modulo,
    Power,
    LParen,
    RParen,
    Greater,
    Less,
    Equal,
    GreaterEqual,
    LessEqual,
}

// ======== 词法分析器 ========
struct Lexer {
    input: Vec<char>,
    position: usize,
}

impl Lexer {
    fn new(input: &str) -> Lexer {
        Lexer {
            input: input.chars().collect(),
            position: 0,
        }
    }

    fn peek(&self) -> Option<char> {
        self.input.get(self.position).copied()
    }

    fn next(&mut self) -> Option<char> {
        let ch = self.peek()?;
        self.position += 1;
        Some(ch)
    }

    fn skip_whitespace(&mut self) {
        while let Some(ch) = self.peek() {
            if ch.is_whitespace() {
                self.next();
            } else {
                break;
            }
        }
    }

    fn read_number(&mut self) -> Result<f64, CalcError> {
        let start = self.position;
        let mut has_dot = false;

        while let Some(ch) = self.peek() {
            if ch.is_ascii_digit() {
                self.next();
            } else if ch == '.' && !has_dot {
                has_dot = true;
                self.next();
            } else {
                break;
            }
        }

        let num_str: String = self.input[start..self.position].iter().collect();
        num_str.parse::<f64>()
            .map_err(|_| CalcError::InvalidNumber(num_str))
    }

    fn tokenize(&mut self) -> Result<Vec<Token>, CalcError> {
        let mut tokens = Vec::new();

        loop {
            self.skip_whitespace();

            match self.peek() {
                None => break,
                Some(ch) => {
                    match ch {
                        '+' => { tokens.push(Token::Plus); self.next(); }
                        '-' => { tokens.push(Token::Minus); self.next(); }
                        '*' => { tokens.push(Token::Multiply); self.next(); }
                        '/' => { tokens.push(Token::Divide); self.next(); }
                        '%' => { tokens.push(Token::Modulo); self.next(); }
                        '^' => { tokens.push(Token::Power); self.next(); }
                        '(' => { tokens.push(Token::LParen); self.next(); }
                        ')' => { tokens.push(Token::RParen); self.next(); }
                        '>' => {
                            self.next();
                            if self.peek() == Some('=') {
                                self.next();
                                tokens.push(Token::GreaterEqual);
                            } else {
                                tokens.push(Token::Greater);
                            }
                        }
                        '<' => {
                            self.next();
                            if self.peek() == Some('=') {
                                self.next();
                                tokens.push(Token::LessEqual);
                            } else {
                                tokens.push(Token::Less);
                            }
                        }
                        '=' => {
                            self.next();
                            if self.peek() == Some('=') {
                                self.next();
                                tokens.push(Token::Equal);
                            } else {
                                return Err(CalcError::InvalidOperator(ch));
                            }
                        }
                        _ if ch.is_ascii_digit() => {
                            tokens.push(Token::Number(self.read_number()?));
                        }
                        _ => return Err(CalcError::InvalidOperator(ch)),
                    }
                }
            }
        }

        Ok(tokens)
    }
}

// ======== 语法分析器 / 计算器 ========
struct Calculator {
    tokens: Vec<Token>,
    position: usize,
}

impl Calculator {
    fn new(tokens: Vec<Token>) -> Calculator {
        Calculator { tokens, position: 0 }
    }

    fn peek(&self) -> Option<&Token> {
        self.tokens.get(self.position)
    }

    fn next(&mut self) -> Option<Token> {
        let token = self.tokens.get(self.position).cloned()?;
        self.position += 1;
        Some(token)
    }

    fn expect(&mut self, expected: Token) -> Result<(), CalcError> {
        match self.next() {
            Some(ref t) if std::mem::discriminant(t) == std::mem::discriminant(&expected) => Ok(()),
            _ => Err(CalcError::MismatchedParentheses),
        }
    }

    // parse_expr -> parse_compare
    fn parse_expr(&mut self) -> Result<f64, CalcError> {
        self.parse_compare()
    }

    // 比较运算: add ( (> | < | == | >= | <=) add )*
    fn parse_compare(&mut self) -> Result<f64, CalcError> {
        let left = self.parse_add()?;

        match self.peek() {
            Some(Token::Greater) => {
                self.next();
                let right = self.parse_add()?;
                Ok(if left > right { 1.0 } else { 0.0 })
            }
            Some(Token::Less) => {
                self.next();
                let right = self.parse_add()?;
                Ok(if left < right { 1.0 } else { 0.0 })
            }
            Some(Token::Equal) => {
                self.next();
                let right = self.parse_add()?;
                Ok(if (left - right).abs() < 1e-10 { 1.0 } else { 0.0 })
            }
            Some(Token::GreaterEqual) => {
                self.next();
                let right = self.parse_add()?;
                Ok(if left >= right { 1.0 } else { 0.0 })
            }
            Some(Token::LessEqual) => {
                self.next();
                let right = self.parse_add()?;
                Ok(if left <= right { 1.0 } else { 0.0 })
            }
            _ => Ok(left),
        }
    }

    // 加减: mul ( (+ | -) mul )*
    fn parse_add(&mut self) -> Result<f64, CalcError> {
        let mut left = self.parse_mul()?;

        loop {
            match self.peek() {
                Some(Token::Plus) => {
                    self.next();
                    left += self.parse_mul()?;
                }
                Some(Token::Minus) => {
                    self.next();
                    left -= self.parse_mul()?;
                }
                _ => break Ok(left),
            }
        }
    }

    // 乘除模: power ( (* | / | %) power )*
    fn parse_mul(&mut self) -> Result<f64, CalcError> {
        let mut left = self.parse_power()?;

        loop {
            match self.peek() {
                Some(Token::Multiply) => {
                    self.next();
                    left *= self.parse_power()?;
                }
                Some(Token::Divide) => {
                    self.next();
                    let right = self.parse_power()?;
                    if right == 0.0 {
                        return Err(CalcError::DivisionByZero);
                    }
                    left /= right;
                }
                Some(Token::Modulo) => {
                    self.next();
                    let right = self.parse_power()?;
                    if right == 0.0 {
                        return Err(CalcError::DivisionByZero);
                    }
                    left = left % right;
                }
                _ => break Ok(left),
            }
        }
    }

    // 幂运算: unary ( ^ unary )*
    fn parse_power(&mut self) -> Result<f64, CalcError> {
        let base = self.parse_unary()?;

        match self.peek() {
            Some(Token::Power) => {
                self.next();
                let exp = self.parse_unary()?;
                Ok(base.powf(exp))
            }
            _ => Ok(base),
        }
    }

    // 一元运算: (- | +) primary | primary
    fn parse_unary(&mut self) -> Result<f64, CalcError> {
        match self.peek() {
            Some(Token::Minus) => {
                self.next();
                Ok(-self.parse_primary()?)
            }
            Some(Token::Plus) => {
                self.next();
                self.parse_primary()
            }
            _ => self.parse_primary(),
        }
    }

    // 基本单元: number | ( expr )
    fn parse_primary(&mut self) -> Result<f64, CalcError> {
        match self.next() {
            Some(Token::Number(n)) => Ok(n),
            Some(Token::LParen) => {
                let result = self.parse_expr()?;
                match self.next() {
                    Some(Token::RParen) => Ok(result),
                    _ => Err(CalcError::MismatchedParentheses),
                }
            }
            _ => Err(CalcError::EmptyExpression),
        }
    }

    fn calculate(&mut self) -> Result<f64, CalcError> {
        let result = self.parse_expr()?;
        if self.position < self.tokens.len() {
            return Err(CalcError::InvalidOperator('?'));
        }
        Ok(result)
    }
}

// ======== 主函数 ========
fn evaluate(expr: &str) -> Result<f64, CalcError> {
    if expr.trim().is_empty() {
        return Err(CalcError::EmptyExpression);
    }

    let mut lexer = Lexer::new(expr);
    let tokens = lexer.tokenize()?;

    let mut calc = Calculator::new(tokens);
    calc.calculate()
}

fn main() {
    let args: Vec<String> = env::args().collect();

    if args.len() < 2 {
        println!("Rust 计算器 v1.0");
        println!("用法: calculator \"<表达式>\"");
        println!();
        println!("示例:");
        println!("  calculator \"1 + 2\"");
        println!("  calculator \"(3 + 4) * 5\"");
        println!("  calculator \"2 ^ 10\"");
        println!("  calculator \"10 / 3\"");
        println!("  calculator \"10 % 3\"");
        println!("  calculator \"5 > 3\"");
        return;
    }

    let expr = &args[1];

    match evaluate(expr) {
        Ok(result) => {
            if result == result.trunc() {
                println!("{} = {}", expr, result as i64);
            } else {
                println!("{} = {}", expr, result);
            }
        }
        Err(e) => {
            eprintln!("{}", e);
            std::process::exit(1);
        }
    }
}
