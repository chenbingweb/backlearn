## 1. 响应模型 response_model

### 基础响应模型

```python
from fastapi import FastAPI
from pydantic import BaseModel, EmailStr

app = FastAPI()


class UserIn(BaseModel):
    """创建用户时的输入"""
    username: str
    password: str
    email: EmailStr


class UserOut(BaseModel):
    """返回给客户端的数据（不包含密码）"""
    id: int
    username: str
    email: EmailStr


@app.post("/users/", response_model=UserOut)
def create_user(user: UserIn):
    # 这里保存到数据库，返回新用户
    return {
        "id": 1,
        "username": user.username,
        "email": user.email,
        "password": user.password,  # 即使有也会被过滤掉
    }
```

### 响应模型的作用

1. **过滤敏感字段** — 如密码不会返回
2. **限制输出字段** — 只返回定义的字段
3. **自动序列化** — Pydantic 模型自动转 JSON
4. **文档生成** — Swagger UI 显示正确的响应格式

### 响应模型排除字段

```python
class User(BaseModel):
    id: int
    name: str
    email: str
    password: str


# 方式1：单独定义响应模型（推荐）
class UserResponse(BaseModel):
    id: int
    name: str
    email: str


# 方式2：使用 Field(exclude=True)
class UserAlt(BaseModel):
    id: int
    name: str
    email: str
    password: str = Field(exclude=True)
```

### 排除 None 字段

```python
class Item(BaseModel):
    name: str
    description: str | None = None
    price: float


@app.get("/items/{item_id}", response_model=Item, response_model_exclude_unset=True)
def get_item(item_id: int):
    # 如果字段未设置或值为 None，不返回
    return {"name": "Foo", "price": 50.0}
    # 返回: {"name": "Foo", "price": 50.0}
    # 不会返回: {"description": null}
```

### 排除指定字段

```python
@app.get("/users/{user_id}", response_model=User, response_model_exclude={"email"})
def get_user(user_id: int):
    ...
```

## 2. HTTP 状态码

### 常用状态码

| 状态码 | 含义 | 使用场景 |
|--------|------|----------|
| 200 | OK | GET/PUT 成功 |
| 201 | Created | POST 创建成功 |
| 204 | No Content | DELETE 成功 |
| 400 | Bad Request | 请求参数错误 |
| 401 | Unauthorized | 未认证 |
| 403 | Forbidden | 无权限 |
| 404 | Not Found | 资源不存在 |
| 422 | Unprocessable Entity | 验证失败 |
| 500 | Internal Server Error | 服务器错误 |

### 显式设置状态码

```python
from fastapi import status


@app.post("/users/", status_code=status.HTTP_201_CREATED)
def create_user(user: UserIn):
    return {"id": 1, "username": user.username}


@app.delete("/users/{user_id}", status_code=status.HTTP_204_NO_CONTENT)
def delete_user(user_id: int):
    # 204 不能返回 body
    return None
```

### 条件性状态码

```python
from fastapi import Response


@app.put("/items/{item_id}")
def update_item(item_id: int, item: Item, response: Response):
    exists = check_item_exists(item_id)

    if not exists:
        response.status_code = status.HTTP_201_CREATED
        return create_item(item)

    response.status_code = status.HTTP_200_OK
    return update_existing_item(item_id, item)
```

## 3. 特殊响应类型

### 重定向

```python
from fastapi import FastAPI, RedirectResponse

app = FastAPI()


@app.get("/old-url")
def old_url():
    return RedirectResponse(url="/new-url", status_code=301)
```

### 文件响应

```python
from fastapi.responses import FileResponse


@app.get("/download/{filename}")
def download_file(filename: str):
    return FileResponse(
        path=f"files/{filename}",
        filename=filename,
        media_type="application/octet-stream",
    )
```

### JSON 响应

```python
from fastapi.responses import JSONResponse


@app.get("/custom-json")
def custom_json():
    return JSONResponse(
        content={"message": "hello"},
        status_code=200,
        headers={"X-Custom": "value"},
    )
```

### HTML 响应

```python
from fastapi.responses import HTMLResponse


@app.get("/page", response_class=HTMLResponse)
def read_page():
    return """
    <html>
        <head><title>FastAPI</title></head>
        <body><h1>Hello, FastAPI!</h1></body>
    </html>
    """
```

### 纯文本响应

```python
from fastapi.responses import PlainTextResponse


@app.get("/text", response_class=PlainTextResponse)
def read_text():
    return "Hello, World!"
```

### 流式响应

```python
from fastapi.responses import StreamingResponse
import io


@app.get("/stream")
def stream_data():
    def generate():
        for i in range(100):
            yield f"data: line {i}\n\n"

    return StreamingResponse(
        generate(),
        media_type="text/event-stream",
    )
```

## 4. 自定义响应头

### 设置响应头

```python
from fastapi import Response


@app.get("/items/{item_id}")
def get_item(item_id: int, response: Response):
    response.headers["X-Item-ID"] = str(item_id)
    response.headers["Cache-Control"] = "max-age=3600"
    return {"item_id": item_id}
```

### 设置 Cookie

```python
@app.post("/login")
def login(response: Response):
    response.set_cookie(
        key="session_id",
        value="abc123",
        httponly=True,           # 禁止 JavaScript 访问
        max_age=1800,            # 30分钟
        samesite="lax",
    )
    return {"message": "登录成功"}
```

### 删除 Cookie

```python
@app.post("/logout")
def logout(response: Response):
    response.delete_cookie(key="session_id")
    return {"message": "登出成功"}
```

## 5. 统一响应封装

### 标准响应结构

```python
from typing import Generic, TypeVar
from pydantic import BaseModel

T = TypeVar("T")


class ResponseModel(BaseModel, Generic[T]):
    code: int = 200
    message: str = "success"
    data: T | None = None


class UserOut(BaseModel):
    id: int
    name: str


@app.get("/users/{user_id}", response_model=ResponseModel[UserOut])
def get_user(user_id: int):
    user = UserOut(id=user_id, name="Alice")
    return ResponseModel(data=user)


@app.get("/users/", response_model=ResponseModel[list[UserOut]])
def list_users():
    users = [UserOut(id=1, name="Alice"), UserOut(id=2, name="Bob")]
    return ResponseModel(data=users)
```

返回格式：
```json
{
  "code": 200,
  "message": "success",
  "data": {
    "id": 1,
    "name": "Alice"
  }
}
```

## 实战练习

### 练习 1：完整 CRUD 响应

```python
from fastapi import FastAPI, HTTPException, status
from pydantic import BaseModel, Field
from typing import List

app = FastAPI()

items_db = {}
next_id = 1


class ItemIn(BaseModel):
    name: str = Field(..., min_length=1)
    price: float = Field(..., gt=0)


class ItemOut(BaseModel):
    id: int
    name: str
    price: float


@app.post("/items/", response_model=ItemOut, status_code=status.HTTP_201_CREATED)
def create_item(item: ItemIn):
    global next_id
    new_item = {"id": next_id, **item.model_dump()}
    items_db[next_id] = new_item
    next_id += 1
    return new_item


@app.get("/items/{item_id}", response_model=ItemOut)
def get_item(item_id: int):
    if item_id not in items_db:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail=f"Item {item_id} not found"
        )
    return items_db[item_id]


@app.get("/items/", response_model=List[ItemOut])
def list_items():
    return list(items_db.values())


@app.put("/items/{item_id}", response_model=ItemOut)
def update_item(item_id: int, item: ItemIn):
    if item_id not in items_db:
        raise HTTPException(status_code=404, detail="Not found")
    items_db[item_id].update(item.model_dump())
    return items_db[item_id]


@app.delete("/items/{item_id}", status_code=status.HTTP_204_NO_CONTENT)
def delete_item(item_id: int):
    if item_id not in items_db:
        raise HTTPException(status_code=404, detail="Not found")
    del items_db[item_id]
    return None
```

### 练习 2：文件下载接口

```python
from fastapi.responses import FileResponse
import os


@app.get("/export/csv")
def export_csv():
    # 假设已经生成了 CSV 文件
    file_path = "export/data.csv"
    if not os.path.exists(file_path):
        raise HTTPException(status_code=404, detail="文件不存在")
```

## 今日总结

- [ ] `response_model` 控制返回数据的结构和字段过滤
- [ ] `response_model_exclude_unset=True` 不返回未设置的字段
- [ ] `status_code` 设置响应状态码，推荐用 `fastapi.status`
- [ ] 201 用于创建成功，204 用于删除成功（无 body）
- [ ] `FileResponse` 返回文件，`StreamingResponse` 返回流
- [ ] `response.headers` 设置自定义响应头
- [ ] `response.set_cookie/delete_cookie` 操作 Cookie
- [ ] 统一响应封装让前端处理更一致
