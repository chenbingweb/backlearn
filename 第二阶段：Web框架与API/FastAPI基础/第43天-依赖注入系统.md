## 1. 什么是依赖注入

### 概念

依赖注入（Dependency Injection, DI）是一种设计模式：
- 不直接在函数内部创建依赖，而是**从外部传入**
- 提高代码复用性和可测试性
- 解耦组件之间的依赖关系

```
❌ 传统方式              ✅ 依赖注入
                        
def get_db():           def get_db():
    return connect()        return connect()
                        
def read_user():        def read_user(db = Depends(get_db)):
    db = get_db()           ...
    ...
```

## 2. FastAPI Depends 基础

### 基本用法

```python
from fastapi import FastAPI, Depends

app = FastAPI()


# 定义依赖函数
def common_parameters(q: str | None = None, skip: int = 0, limit: int = 100):
    return {"q": q, "skip": skip, "limit": limit}


# 在路由中使用依赖
@app.get("/items/")
def read_items(commons: dict = Depends(common_parameters)):
    return commons


@app.get("/users/")
def read_users(commons: dict = Depends(common_parameters)):
    return commons
```

### 依赖可以是类

```python
class CommonQueryParams:
    def __init__(self, q: str | None = None, skip: int = 0, limit: int = 100):
        self.q = q
        self.skip = skip
        self.limit = limit


@app.get("/items/")
def read_items(commons: CommonQueryParams = Depends()):
    return commons
```

### 依赖可以是 callable

```python
from typing import Callable

class Paginator:
    def __init__(self, default_size: int = 20):
        self.default_size = default_size

    def __call__(self, page: int = 1, size: int = 20):
        return {"offset": (page - 1) * size, "limit": size}


paginate = Paginator(default_size=20)


@app.get("/items/")
def list_items(pagination: dict = Depends(paginate)):
    return pagination
```

## 3. 常见依赖场景

### 数据库连接依赖

```python
from fastapi import Depends, HTTPException, status


def get_db():
    """创建数据库会话，请求结束后关闭"""
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()


@app.get("/users/{user_id}")
def get_user(user_id: int, db = Depends(get_db)):
    user = db.query(User).filter(User.id == user_id).first()
    if not user:
        raise HTTPException(status_code=404, detail="User not found")
    return user
```

### 认证依赖

```python
from fastapi.security import OAuth2PasswordBearer

oauth2_scheme = OAuth2PasswordBearer(tokenUrl="/token")


def get_current_user(token: str = Depends(oauth2_scheme)):
    """从 token 中解析当前用户"""
    credentials_exception = HTTPException(
        status_code=status.HTTP_401_UNAUTHORIZED,
        detail="Could not validate credentials",
        headers={"WWW-Authenticate": "Bearer"},
    )
    try:
        payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
        user_id = payload.get("sub")
        if user_id is None:
            raise credentials_exception
    except jwt.JWTError:
        raise credentials_exception

    user = get_user_by_id(int(user_id))
    if user is None:
        raise credentials_exception
    return user


@app.get("/me")
def read_me(current_user: User = Depends(get_current_user)):
    return current_user
```

### 权限检查依赖

```python
def require_admin(current_user: User = Depends(get_current_user)):
    if current_user.role != "admin":
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,

    return current_user


@app.delete("/users/{user_id}")

    user_id: int,
```

## 4. 依赖嵌套

### 多级依赖

```python
def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()


def get_current_user(token: str = Depends(oauth2_scheme)):
    # 解析 token
    return user


def get_current_active_user(
    user: User = Depends(get_current_user)
):
    if not user.is_active:
        raise HTTPException(status_code=400, detail="Inactive user")
    return user


def get_current_admin(
    user: User = Depends(get_current_active_user)
):
    if user.role != "admin":
        raise HTTPException(status_code=403, detail="Not admin")
    return user


@app.get("/admin/dashboard")
def admin_dashboard(admin: User = Depends(get_current_admin)):
    return {"admin": admin.username}
```

依赖链：
```
/admin/dashboard
    → get_current_admin
        → get_current_active_user
            → get_current_user
                → oauth2_scheme (从请求头获取 token)
```

## 5. 依赖缓存

### use_cache 参数

```python
@app.get("/items/")
def read_items(
    commons: dict = Depends(common_parameters, use_cache=True)
):
    """use_cache=True: 同一个请求中，依赖只执行一次"""
    return commons
```

### 何时不需要缓存

```python
from fastapi import Depends


def get_dynamic_config():
    """每次调用都可能返回不同值的依赖"""
    return load_config_from_remote()


@app.get("/config")
def read_config(config: dict = Depends(get_dynamic_config, use_cache=False)):
    return config
```

## 6. 路由级别的依赖

### APIRouter 依赖

```python
from fastapi import APIRouter, Depends

router = APIRouter(
    prefix="/admin",
    tags=["admin"],
    dependencies=[Depends(get_current_admin)],  # 所有路由都需要管理员权限

    return {"total_users": 100}
```

## 7. yield 依赖（上下文管理器模式）

### 数据库事务

```python
def get_db_with_transaction():
    db = SessionLocal()
    transaction = db.begin()
    try:
        yield db
        transaction.commit()
    except Exception:
        transaction.rollback()
        raise
    finally:
        db.close()
```

### 异步依赖

```python
async def get_async_db():
    db = AsyncSessionLocal()
    try:
        yield db
    finally:
        await db.close()


@app.get("/items/")
async def read_items(db = Depends(get_async_db)):
    result = await db.execute(select(Item))
    return result.scalars().all()
```

## 实战练习

### 练习 1：完整的依赖体系

```python
from fastapi import FastAPI, Depends, HTTPException, status
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials
from pydantic import BaseModel

app = FastAPI()
security = HTTPBearer()


# 模拟用户数据库
users_db = {
    "1": {"id": 1, "username": "alice", "role": "user"},
    "2": {"id": 2, "username": "bob", "role": "admin"},
}


class User(BaseModel):
    id: int
    username: str
    role: str


# 依赖函数
def verify_token(credentials: HTTPAuthorizationCredentials = Depends(security)):
    """验证 token（简化版）"""
    token = credentials.credentials
    if token not in ["user_token", "admin_token"]:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid token",
        )
    return token


def get_current_user(token: str = Depends(verify_token)) -> User:
    """获取当前用户"""
    if token == "admin_token":
        return User(**users_db["2"])
    return User(**users_db["1"])


def require_admin(user: User = Depends(get_current_user)) -> User:
    """要求管理员权限"""
    if user.role != "admin":
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="Admin required",
        )
    return user


# 公开接口
@app.get("/public")
def public():
    return {"message": "所有人可见"}


# 需要登录
@app.get("/profile")
def profile(user: User = Depends(get_current_user)):
    return user


# 需要管理员权限
@app.get("/admin/stats")
def admin_stats(admin: User = Depends(require_admin)):
    return {"total_users": len(users_db), "admin": admin.username}
```

测试命令：
```bash
# 公开接口
curl http://localhost:8000/public

# 需要 token
curl -H "Authorization: Bearer user_token" http://localhost:8000/profile

# 管理员接口（user_token 会 403）
curl -H "Authorization: Bearer admin_token" http://localhost:8000/admin/stats
```

## 今日总结

- [ ] 依赖注入将依赖从外部传入，提高代码复用性和可测试性
- [ ] `Depends()` 声明依赖，FastAPI 自动解析并注入
- [ ] 依赖可以是函数、类或任何 callable
- [ ] 依赖可以嵌套，形成依赖链
- [ ] `use_cache=True` 同一个请求中只执行一次
- [ ] `yield` 依赖支持上下文管理器模式（如数据库连接）
- [ ] 可在 APIRouter 或 App 级别设置全局依赖
- [ ] 依赖系统非常适合做认证、权限、数据库连接复用
