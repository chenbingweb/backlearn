## 1. 权限模型

### RBAC（基于角色的访问控制）

```
用户 ──属于──→ 角色 ──拥有──→ 权限
         │
         └───属于──→ 用户组 ──属于──→ 角色
```

### 简单模型

| 实体 | 说明 |
|------|------|
| User | 用户 |
| Role | 角色（admin, user, moderator）|
| Permission | 权限（read, write, delete）|
| UserRole | 用户-角色关联 |
| RolePermission | 角色-权限关联 |

### 权限命名约定

```python
# 资源:操作 格式
"users:read"     # 查看用户
"users:write"    # 创建/修改用户
"users:delete"   # 删除用户
"articles:publish"  # 发布文章
"settings:manage"  # 管理设置
```

## 2. 数据库模型

### 定义模型

```python
from sqlalchemy import Column, Integer, String, Boolean, Table, ForeignKey
from sqlalchemy.orm import relationship
from app.database import Base

# 中间表
user_roles = Table(
    "user_roles",
    Base.metadata,
    Column("user_id", Integer, ForeignKey("users.id"), primary_key=True),
    Column("role_id", Integer, ForeignKey("roles.id"), primary_key=True),
)

role_permissions = Table(
    "role_permissions",
    Base.metadata,
    Column("role_id", Integer, ForeignKey("roles.id"), primary_key=True),
    Column("permission_id", Integer, ForeignKey("permissions.id"), primary_key=True),
)


class User(Base):
    __tablename__ = "users"

    id = Column(Integer, primary_key=True, index=True)
    username = Column(String(50), unique=True)
    email = Column(String(100), unique=True)

    roles = relationship("Role", secondary=user_roles, back_populates="users")


class Role(Base):
    __tablename__ = "roles"

    id = Column(Integer, primary_key=True, index=True)
    name = Column(String(50), unique=True, nullable=False)
    description = Column(String(200))

    permissions = relationship("Permission", secondary=role_permissions, back_populates="roles")
    users = relationship("User", secondary=user_roles, back_populates="roles")


class Permission(Base):
    __tablename__ = "permissions"

    id = Column(Integer, primary_key=True, index=True)
    name = Column(String(100), unique=True, nullable=False)
    resource = Column(String(50))
    action = Column(String(20))

    roles = relationship("Role", secondary=role_permissions, back_populates="permissions")
```

## 3. 权限检查函数

### 检查用户角色

```python
async def check_user_role(db: AsyncSession, user_id: int, role_name: str) -> bool:
    result = await db.execute(
        select(Role)
        .join(user_roles)
        .join(User)
        .where(User.id == user_id)
        .where(Role.name == role_name)
    )
    return result.first() is not None


async def get_user_permissions(db: AsyncSession, user_id: int) -> list[str]:
    result = await db.execute(
        select(Permission.name)
        .join(role_permissions)
        .join(Role)
        .join(user_roles)
        .join(User)
        .where(User.id == user_id)
    )
    return [row[0] for row in result.all()]
```

### 检查权限

```python
async def check_permission(db: AsyncSession, user_id: int, permission: str) -> bool:
    permissions = await get_user_permissions(db, user_id)
    return permission in permissions
```

## 4. FastAPI 权限装饰器

### 角色检查依赖

```python
from functools import wraps
from typing import Callable, List

class RoleChecker:
    def __init__(self, allowed_roles: List[str]):
        self.allowed_roles = allowed_roles

    async def __call__(self, user: User = Depends(get_current_user)):
        if user.is_superuser:
            return user

        user_roles = [role.name for role in user.roles]
        for role in self.allowed_roles:
            if role in user_roles:
                return user

        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="权限不足"
        )


# 使用
role_checker = RoleChecker(["admin", "moderator"])

@router.delete("/users/{user_id}")
async def delete_user(
    user_id: int,
    admin: User = Depends(role_checker),
    db: AsyncSession = Depends(get_db),
):
    ...
```

### 权限检查依赖

```python
def require_permission(permission: str):
    async def checker(
        user: User = Depends(get_current_user),
        db: AsyncSession = Depends(get_db),
    ):
        if user.is_superuser:
            return user

        user_permissions = await get_user_permissions(db, user.id)
        if permission not in user_permissions:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail=f"需要权限: {permission}"
            )
        return user
    return checker


# 使用
@router.delete("/users/{user_id}")
async def delete_user(
    user_id: int,
    _: User = Depends(require_permission("users:delete")),
    db: AsyncSession = Depends(get_db),
):
    ...
```

from enum import IntEnum


class UserRole(IntEnum):
    USER = 1
    MODERATOR = 2
    ADMIN = 3


class User(Base):
    __tablename__ = "users"

    id = Column(Integer, primary_key=True)
    username = Column(String(50))
    role = Column(Integer, default=UserRole.USER)
    is_active = Column(Boolean, default=True)


def require_role(min_role: UserRole):
    async def checker(user: User = Depends(get_current_user)):
        if user.role >= min_role:
            return user
        raise HTTPException(status_code=403, detail="权限不足")
    return checker


# 使用
@router.delete("/users/{user_id}")
async def delete_user(
    user_id: int,
    _: User = Depends(require_role(UserRole.ADMIN)),

    ...

async def get_active_user(user: User = Depends(get_current_user)) -> User:
    if not user.is_active:
        raise HTTPException(status_code=400, detail="用户已被禁用")
    return user


# 管理员检查
def require_admin(user: User = Depends(get_current_user)) -> User:
    if not user.is_admin:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="需要管理员权限"
        )
    return user


# 权限检查
def require_permission(permission: str):
    async def checker(
        user: User = Depends(get_current_user),
        db: AsyncSession = Depends(get_db),
    ):
        user_permissions = await get_user_permissions(db, user.id)
        if permission not in user_permissions:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail=f"需要权限: {permission}"
            )
        return user
    return Depends(checker)

## 实战练习

### 练习：完整的权限 API

```python
# app/api/admin.py
from fastapi import APIRouter, Depends, HTTPException, status
from app.dependencies import require_admin, get_current_user
from app.crud import get_users, create_role, assign_role
from app.schemas import RoleCreate, UserAssignRole

router = APIRouter(prefix="/admin", dependencies=[Depends(require_admin)])


@router.get("/users")
async def list_all_users(
    skip: int = 0,
    limit: int = 100,
    db: AsyncSession = Depends(get_db),
):
    return await get_users(db, skip, limit)


@router.post("/roles")
async def create_new_role(
    role: RoleCreate,
    db: AsyncSession = Depends(get_db),
):
    return await create_role(db, role)


@router.post("/users/{user_id}/roles")
async def assign_role_to_user(
    user_id: int,
    data: UserAssignRole,
    db: AsyncSession = Depends(get_db),
):
    await assign_role(db, user_id, data.role_name)
    return {"message": "角色分配成功"}
```
