### 布隆过滤器实现

```python
from bloom_filter2 import BloomFilter

class RedisBloomFilter:
    """Redis + 布隆过滤器实现"""

    def __init__(self, redis_client, name='bloom', expected_items=1000000):
        self.r = redis_client
        self.name = f'bloom:{name}'

        # 使用 Redis Bit Array 模拟布隆过滤器
        # 3个哈希函数，每个需要 2^32 / 3 ≈ 14亿位 ≈ 170MB
        self.bit_size = 1 << 31  # 约 256MB
        self.hash_count = 3

    def _hashes(self, value):
        """生成多个哈希值"""
        import struct

        # 简单的哈希函数
        h1 = hash(value)
        h2 = hash(str(h1))
        h3 = hash(str(h2))

        return (
            h1 % self.bit_size,
            h2 % self.bit_size,
            h3 % self.bit_size
        )

    def add(self, value):
        """添加元素"""
        for bit_pos in self._hashes(str(value)):
            self.r.setbit(self.name, bit_pos, 1)

    def might_contain(self, value):
        """检查元素是否存在"""
        for bit_pos in self._hashes(str(value)):
            if not self.r.getbit(self.name, bit_pos):
                return False
        return True

    def clear(self):
        """清空过滤器"""
        self.r.delete(self.name)


# 使用示例
bf = RedisBloomFilter(r, 'user_ids')

# 初始化：从数据库加载所有有效 ID
for user_id in db.query(User.id).all():
    bf.add(user_id)

# 检查
if bf.might_contain(user_id):
    # 可能存在，需要查数据库确认
    user = get_user_from_db(user_id)
    if user:
        # 第一次查询到，添加到过滤器
        bf.add(user_id)
else:
    # 一定不存在，直接返回
    return None
```

### 增强型空值缓存

```python
class EnhancedNullCache:
    """增强型空值缓存"""

    def __init__(self, redis_client, null_ttl=60, max_null_count=10000):
        self.r = redis_client
        self.null_ttl = null_ttl
        self.max_null_count = max_null_count
        self.null_key = 'null:keys'  # 记录所有空值 key

    def get(self, key, db_func):
        """
        获取数据，支持空值缓存和计数限制
        """
        # 1. 查缓存
        cached = self.r.get(key)
        if cached is not None:
            if cached == '__NULL__':
                # 缓存的空值
                return None
            return json.loads(cached)

        # 2. 查数据库
        data = db_func()

        # 3. 检查空值计数
        if data is None:
            null_count = self.r.scard(self.null_key)
            if null_count >= self.max_null_count:
                # 清理最老的空值
                oldest = self.r.srandmember(self.null_key)
                self.r.srem(self.null_key, oldest)
                self.r.delete(oldest)

            # 缓存空值
            self.r.setex(key, self.null_ttl, '__NULL__')
            self.r.sadd(self.null_key, key)
            self.r.expire(self.null_key, 86400)

        else:
            # 缓存数据
            self.r.setex(key, 3600, json.dumps(data))

        return data

    def invalidate(self, key):
        """失效缓存"""
        self.r.delete(key)
        self.r.srem(self.null_key, key)
```

## 2. 缓存雪崩进阶

### 多级过期策略

```python
import random
import time

class MultiTierCache:
    """多级过期策略"""

    def __init__(self, redis_client):
        self.r = redis_client

    def set(self, key, value, base_ttl=3600):
        """
        设置缓存，使用多级过期策略

        L1: 短期缓存（base_ttl 的 20-40%）
        L2: 中期缓存（base_ttl 的 80-100%）
        L3: 长期缓存（永不过期，逻辑删除）
        """
        jitter = random.randint(int(base_ttl * 0.2), int(base_ttl * 0.4))
        l1_ttl = base_ttl - jitter

        jitter2 = random.randint(int(base_ttl * 0.8), int(base_ttl))
        l2_ttl = jitter2

        # 同时设置多个 key
        pipe = self.r.pipeline()

        # L1 短期缓存
        pipe.setex(f'{key}:L1', l1_ttl, json.dumps(value))
        # L2 中期缓存
        pipe.setex(f'{key}:L2', l2_ttl, json.dumps(value))
        # L3 长期缓存（永不过期）
        pipe.set(f'{key}:L3', json.dumps(value))

        # 设置 L3 的版本号
        pipe.set(f'{key}:version', str(int(time.time())))

        pipe.execute()

    def get(self, key):
        """获取缓存"""
        # 按优先级尝试获取
        for tier in ['L1', 'L2', 'L3']:
            cached = self.r.get(f'{key}:{tier}')
            if cached:
                # 如果是从 L3 获取，异步更新 L1/L2
                if tier == 'L3':
                    self._async_refresh(key)
                return json.loads(cached)
        return None

    def _async_refresh(self, key):
        """异步刷新缓存"""
        # 在后台线程中更新
        threading.Thread(target=self._do_refresh, args=(key,)).start()

    def _do_refresh(self, key):
        """执行刷新"""
        # 获取新数据
        data = self._load_from_db(key)
        if data:
            self.set(key, data)

    def invalidate(self, key):
        """失效缓存"""
        pipe = self.r.pipeline()
        pipe.delete(f'{key}:L1')
        pipe.delete(f'{key}:L2')
        pipe.delete(f'{key}:L3')
        pipe.delete(f'{key}:version')
        pipe.execute()
```

### 熔断降级

```python
import time
from functools import wraps

class CircuitBreaker:
    """熔断器"""

    CLOSED = 'CLOSED'
    OPEN = 'OPEN'
    HALF_OPEN = 'HALF_OPEN'

    def __init__(self, failure_threshold=5, timeout=60):
        self.failure_threshold = failure_threshold
        self.timeout = timeout
        self.failures = 0
        self.last_failure_time = None
        self.state = self.CLOSED

    def call(self, func, fallback=None, *args, **kwargs):
        """执行函数，失败时触发熔断"""
        if self.state == self.OPEN:
            if time.time() - self.last_failure_time > self.timeout:
                self.state = self.HALF_OPEN
            else:
                return fallback() if fallback else None

        try:
            result = func(*args, **kwargs)
            if self.state == self.HALF_OPEN:
                self.state = self.CLOSED
                self.failures = 0
            return result
        except Exception as e:
            self.failures += 1
            self.last_failure_time = time.time()

            if self.failures >= self.failure_threshold:
                self.state = self.OPEN

            return fallback() if fallback else None


def circuit_breaker(failure_threshold=5, timeout=60):
    """熔断装饰器"""
    breaker = CircuitBreaker(failure_threshold, timeout)

    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            return breaker.call(func, *args, **kwargs)
        return wrapper
    return decorator
```

### 分布式锁进阶

```python
import redis
import uuid
import time
import threading

class DistributedLockPro:
    """增强型分布式锁"""

    def __init__(self, redis_client, lock_name, timeout=30, retry=3):
        self.r = redis_client
        self.lock_name = f'lock:{lock_name}'
        self.timeout = timeout
        self.retry = retry
        self.token = str(uuid.uuid4())
        self._watchdog = None

    def acquire(self):
        """获取锁"""
        for _ in range(self.retry):
            result = self.r.set(
                self.lock_name,
                self.token,
                nx=True,
                ex=self.timeout
            )
            if result:
                self._start_watchdog()
                return True
            time.sleep(0.1)
        return False

    def _start_watchdog(self):
        """启动看门狗"""
        self._watchdog = threading.Thread(
            target=self._watchdog_task,
            daemon=True
        )
        self._watchdog.start()

    def _watchdog_task(self):
        """看门狗任务"""
        while True:
            time.sleep(self.timeout / 3)
            # 检查锁是否还在
            current = self.r.get(self.lock_name)
            if current != self.token:
                break
            # 续期
            self.r.expire(self.lock_name, self.timeout)

    def release(self):
        """释放锁"""
        script = """
        if redis.call("get", KEYS[1]) == ARGV[1] then
            return redis.call("del", KEYS[1])
        else
            return 0
        end
        """
        try:
            self.r.eval(script, 1, self.lock_name, self.token)
        except:
            pass

    def extend(self, additional_time=None):
        """延长锁持有时间"""
        if additional_time is None:
            additional_time = self.timeout

        script = """
        if redis.call("get", KEYS[1]) == ARGV[1] then
            return redis.call("expire", KEYS[1], ARGV[2])
        else
            return 0
        end
        """
        return self.r.eval(
            script, 1,
            self.lock_name, self.token,
            additional_time
        ) == 1


class LockFreeCache:
    """无锁缓存（单飞模式）"""

    def __init__(self, redis_client):
        self.r = redis_client
        self.lock_prefix = 'singleflight:'

    def get_or_load(self, key, load_func, ttl=3600):
        """
        单飞模式：同一 key 只有一个请求去加载数据
        其他请求等待该请求完成后直接使用结果
        """
        # 1. 查缓存
        cached = self.r.get(key)
        if cached:
            return json.loads(cached)

        # 2. 检查是否有其他请求在加载
        loading_key = f'{self.lock_prefix}{key}'
        is_loading = self.r.get(loading_key)

        if is_loading:
            # 等待其他请求完成
            for _ in range(100):  # 最多等10秒
                time.sleep(0.1)
                cached = self.r.get(key)
                if cached:
                    return json.loads(cached)

        # 3. 设置加载标记
        self.r.setex(loading_key, 30, '1')

        try:
            # 4. 加载数据
            data = load_func()
            if data:
                self.r.setex(key, ttl, json.dumps(data))
            return data
        finally:
            # 5. 删除加载标记
            self.r.delete(loading_key)
```

## 4. 综合解决方案

### 缓存框架

```python
class CacheSolution:
    """
    综合缓存解决方案
    解决：穿透、雪崩、击穿
    """

    def __init__(self, redis_client):
        self.r = redis_client
        self.bloom = RedisBloomFilter(redis_client, 'data')
        self.lock_prefix = 'lock:'
        self.null_prefix = 'null:'

    def get(self, key, db_loader, bloom_filter_key=None, ttl=3600):
        """
        综合获取缓存

        策略：
        1. 布隆过滤器过滤不存在的数据
        2. 多级缓存加速
        3. 互斥锁防止击穿
        4. 空值缓存防止穿透
        """
        # 1. 如果有布隆过滤器，检查是否可能存在
        if bloom_filter_key:
            if not self.bloom.might_contain(bloom_filter_key):
                return None

        # 2. 尝试获取缓存
        cached = self.r.get(key)
        if cached:
            if cached == '__NULL__':
                return None
            return json.loads(cached)

        # 3. 获取锁
        lock_key = f'{self.lock_prefix}{key}'
        lock_acquired = self.r.set(lock_key, '1', nx=True, ex=10)

        if not lock_acquired:
            # 等待后重试
            for _ in range(50):
                time.sleep(0.1)
                cached = self.r.get(key)
                if cached:
                    if cached == '__NULL__':
                        return None
                    return json.loads(cached)
            return None

        try:
            # 4. 双重检查缓存
            cached = self.r.get(key)
            if cached:
                if cached == '__NULL__':
                    return None
                return json.loads(cached)

            # 5. 加载数据
            data = db_loader()

            # 6. 回填缓存
            if data:
                self.r.setex(key, ttl, json.dumps(data))
            else:
                # 空值缓存，短过期
                self.r.setex(f'{self.null_prefix}{key}', 60, '__NULL__')

            return data

        finally:
            self.r.delete(lock_key)

    def invalidate(self, key):
        """失效缓存"""
        self.r.delete(key)
        self.r.delete(f'{self.null_prefix}{key}')
```
