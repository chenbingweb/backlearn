## 1. EXPLAIN 详解

### PostgreSQL EXPLAIN

```sql
-- 基础分析
EXPLAIN SELECT * FROM orders WHERE status = 'pending';

-- 包含成本估算
EXPLAIN (FORMAT TEXT, COSTS true)
SELECT * FROM orders WHERE status = 'pending';

-- 包含缓冲区和 IO 信息
EXPLAIN (ANALYZE, BUFFERS)
SELECT * FROM orders WHERE status = 'pending';

-- 完整分析
EXPLAIN (ANALYZE, BUFFERS, FORMAT TEXT)
SELECT
    o.order_no,
    u.username,
    SUM(oi.quantity * oi.price) AS total
FROM orders o
JOIN users u ON o.user_id = u.id
JOIN order_items oi ON o.id = oi.order_id
WHERE o.status = 'completed'
GROUP BY o.order_no, u.username;
```

### 输出格式

```sql
-- 成本参数
-- startup_cost: 返回第一行前成本
-- total_cost: 返回所有行成本

-- 典型输出解析
Limit  (cost=0.29..12.50 rows=10 width=89)
  ->  Nested Loop Left Join  (cost=0.29..12.50 rows=10 width=89)
        ->  Seq Scan on orders  (cost=0.00..8.00 rows=100 width=40)
              Filter: (status = 'pending')
        ->  Index Scan using users_pkey on users  (cost=0.29..0.31 rows=1 width=49)
              Index Cond: (id = orders.user_id)

-- 分析：
-- 1. Seq Scan：全表扫描，成本 8.00（高）
-- 2. Index Scan：索引扫描，成本 0.31（低）
-- 3. Nested Loop：连接算法
```

### 关键指标

| 指标 | 说明 | 优化方向 |
|------|------|----------|
| seq_scan | 顺序扫描数 | 考虑加索引 |
| index_scan | 索引扫描数 | 良好 |
| rows | 预估返回行数 | 越小越好 |
| actual_rows | 实际返回行数 | 与预估对比 |
| Buffers | 缓冲区命中 | 高好 |
| actual_time | 实际执行时间 | 越短越好 |

## 2. 常见扫描类型

### Sequential Scan（全表扫描）

```sql
-- 问题：全表扫描
EXPLAIN SELECT * FROM orders WHERE total_amount > 1000;

-- 优化：添加索引
CREATE INDEX idx_orders_amount ON orders(total_amount);

-- 验证
EXPLAIN SELECT * FROM orders WHERE total_amount > 1000;

-- 何时全表扫描更优：
-- 1. 表数据量小（< 1000 行）
-- 2. 查询返回大部分数据（> 50%）
-- 3. 无可用索引
```

### Index Scan（索引扫描）

```sql
-- 使用索引扫描
EXPLAIN SELECT * FROM orders WHERE id = 1;

-- 特点：
-- 1. 先扫描索引
-- 2. 再回表查询数据
-- 3. 适合选择性高的查询
```

### Index Only Scan（仅索引扫描）

```sql
-- 直接从索引返回数据（不回表）
EXPLAIN SELECT user_id, status FROM orders WHERE status = 'pending';

-- 需要覆盖索引
CREATE INDEX idx_orders_status_cover ON orders(status)
INCLUDE (user_id);

-- 验证
EXPLAIN SELECT user_id, status FROM orders WHERE status = 'pending';
```

### Bitmap Scan（位图扫描）

```sql
-- 先用索引找出所有匹配的 row ID，再用 bitmap 合并
EXPLAIN SELECT * FROM orders WHERE status = 'completed' AND total_amount > 1000;

-- 多列查询时可能出现
-- 1. 先扫描 status 索引，记录所有匹配的 row ID
-- 2. 再扫描 amount 索引
-- 3. 合并两个 bitmap
-- 4. 根据 row ID 获取数据
```

### 扫描类型选择

```sql
-- PostgreSQL 自动选择扫描类型
-- 强制使用特定扫描类型
SET enable_seqscan = off;  -- 禁用顺序扫描
SET enable_indexscan = off;  -- 禁用索引扫描
SET enable_bitmapscan = off;  -- 禁用位图扫描

-- 查看各扫描类型统计
SELECT
    name,
    setting
FROM pg_settings
WHERE name LIKE 'enable_%';
```

## 3. 连接类型

### Nested Loop Join

```sql
-- 原理：外层循环驱动，内层索引查找
EXPLAIN
SELECT * FROM orders o
JOIN users u ON o.user_id = u.id;

-- 输出
Nested Loop  (cost=0.29..8.86 rows=100 width=89)
  ->  Seq Scan on users  (cost=0.00..4.00 rows=100 width=49)
  ->  Index Scan using orders_user_id_idx on orders  (cost=0.29..0.31 rows=1 width=40)
        Index Cond: (user_id = u.id)

-- 适用：小表驱动大表，或内表有索引
```

### Hash Join

```sql
-- 原理：先扫描小表建立 hash 表，再扫描大表探测
EXPLAIN
SELECT * FROM orders o
JOIN order_items oi ON o.id = oi.order_id;

-- 输出
Hash Join  (cost=4.00..12.50 rows=100 width=120)
  Hash Cond: (o.id = oi.order_id)
  ->  Seq Scan on orders  (cost=0.00..4.00 rows=100 width=40)
  ->  Hash  (cost=2.50..2.50 rows=100 width=80)
        ->  Seq Scan on order_items  (cost=0.00..2.50 rows=100 width=80)

-- 适用：大表连接
```

### Merge Join

```sql
-- 原理：两个表都排序后合并
EXPLAIN
SELECT * FROM orders o
JOIN order_items oi ON o.id = oi.order_id
ORDER BY o.id;

-- 输出
Merge Join  (cost=0.29..12.50 rows=100 width=120)
  Merge Cond: (o.id = oi.order_id)
  ->  Index Scan using orders_pkey on orders  (cost=0.29..4.00 rows=100 width=40)
  ->  Index Scan using order_items_order_id_idx on order_items  (cost=0.29..4.00 rows=100 width=80)

-- 适用：已排序的数据
```

## 4. 常见优化模式

### 避免 SELECT *

```sql
-- 问题
EXPLAIN ANALYZE
SELECT * FROM orders o
JOIN users u ON o.user_id = u.id
WHERE o.status = 'completed';

-- 优化：只查询需要的列
EXPLAIN ANALYZE
SELECT
    o.order_no,
    o.total_amount,
    u.username,
    u.email
FROM orders o
JOIN users u ON o.user_id = u.id
WHERE o.status = 'completed';
```

### 分页优化

```sql
-- 问题：偏移量大了性能差
EXPLAIN ANALYZE
SELECT * FROM orders
ORDER BY created_at DESC
LIMIT 10 OFFSET 10000;

-- 优化 1：使用游标分页
EXPLAIN ANALYZE
SELECT * FROM orders
WHERE (created_at, id) < ('2024-01-01', 12345)
ORDER BY created_at DESC, id DESC
LIMIT 10;

-- 优化 2：延迟 JOIN
EXPLAIN ANALYZE
SELECT o.*
FROM orders o
INNER JOIN (
    SELECT id FROM orders
    ORDER BY created_at DESC
    LIMIT 10 OFFSET 10000
) t ON o.id = t.id;
```

### IN vs EXISTS

```sql
-- 小数据量：IN
SELECT * FROM products
WHERE category_id IN (SELECT id FROM categories WHERE is_active = TRUE);

-- 大数据量：EXISTS
SELECT * FROM products p
WHERE EXISTS (
    SELECT 1 FROM categories c
    WHERE c.id = p.category_id AND c.is_active = TRUE
);

-- 大数据量：改写为 JOIN
SELECT p.*
FROM products p
JOIN categories c ON p.category_id = c.id
WHERE c.is_active = TRUE;
```

### 避免函数在列上

```sql
-- 问题：索引失效
SELECT * FROM orders
WHERE DATE_TRUNC('month', created_at) = '2024-01-01';

SELECT * FROM products
WHERE LOWER(name) = 'iphone';

-- 优化 1：使用范围查询
SELECT * FROM orders
WHERE created_at >= '2024-01-01'
AND created_at < '2024-02-01';

-- 优化 2：表达式索引
CREATE INDEX idx_orders_month ON orders(DATE_TRUNC('month', created_at));
CREATE INDEX idx_products_name_lower ON products(LOWER(name));
```

## 5. Python 优化实践

```python
import psycopg2
from psycopg2 import sql

# 使用参数化查询（防止 SQL 注入）
def get_orders_by_status(status, limit=100):
    query = """
        SELECT order_no, total_amount, created_at
        FROM orders
        WHERE status = %s
        ORDER BY created_at DESC
        LIMIT %s
    """
    # 始终使用参数化查询
    return execute_query(query, (status, limit))

# 批量插入优化
def batch_insert_orders(orders_data):
    """
    orders_data: [(order_no, total, user_id), ...]
    """
    query = """
        INSERT INTO orders (order_no, total_amount, user_id)
        VALUES %s
    """
    # 使用 psycopg2.extras.execute_values 批量插入
    from psycopg2.extras import execute_values
    with conn.cursor() as cur:
        execute_values(cur, query, orders_data)
    conn.commit()

# 异步查询（大量数据）
import asyncio
import asyncpg

async def fetch_orders():
    conn = await asyncpg.connect('postgresql://user:pass@localhost/test')
    rows = await conn.fetch('''
        SELECT order_no, total_amount
        FROM orders
        WHERE status = $1
    ''', 'completed')
    await conn.close()
    return rows
```

## 6. MySQL EXPLAIN

```sql
-- MySQL 分析
EXPLAIN SELECT * FROM orders WHERE status = 'pending';

-- 完整信息
EXPLAIN FORMAT=JSON
SELECT o.*, u.username
FROM orders o
JOIN users u ON o.user_id = u.id
WHERE o.status = 'completed';

-- 分析结果字段
-- type: ALL（全表）、index、range、ref、eq_ref、const
-- key: 使用的索引
-- rows: 扫描行数
-- Extra: Using filesort、Using temporary、Using index
```

### MySQL 优化要点

```sql
-- 1. 确保使用索引（type 不是 ALL）
EXPLAIN SELECT * FROM orders WHERE order_no = 'xxx';

-- 2. 避免 filesort
EXPLAIN SELECT * FROM orders ORDER BY created_at DESC LIMIT 10;
-- 添加索引
ALTER TABLE orders ADD INDEX idx_created (created_at DESC);

-- 3. 避免临时表
EXPLAIN
SELECT status, COUNT(*) FROM orders GROUP BY status;
```

## 7. 实战练习

### 慢查询优化流程

```sql
-- 1. 识别慢查询
SELECT
    query,
    calls,
    total_time / 1000 AS total_seconds,
    mean_time / 1000 AS mean_ms,
    rows
FROM pg_stat_statements
WHERE total_time > 1000000  -- 超过 1 秒
ORDER BY total_time DESC
LIMIT 10;

-- 2. 分析查询计划
EXPLAIN (ANALYZE, BUFFERS)
SELECT
    u.username,
    o.order_no,
    p.name AS product_name,
    oi.quantity,
    oi.price
FROM orders o
JOIN users u ON o.user_id = u.id
JOIN order_items oi ON o.id = oi.order_id
JOIN products p ON oi.product_id = p.id
WHERE o.created_at > CURRENT_DATE - INTERVAL '30 days'
ORDER BY o.created_at DESC;

-- 3. 识别问题
-- Seq Scan on orders - 可能需要索引
-- Nested Loop - 考虑 Hash Join
-- Sort - 检查是否需要排序

-- 4. 创建优化索引
CREATE INDEX idx_orders_created_user
ON orders(created_at DESC, user_id)
INCLUDE (order_no);

CREATE INDEX idx_order_items_order_product
ON order_items(order_id, product_id)
INCLUDE (quantity, price);

-- 5. 验证优化
EXPLAIN (ANALYZE, BUFFERS)
SELECT
    u.username,
    o.order_no,
    p.name AS product_name,
    oi.quantity,
    oi.price
FROM orders o
JOIN users u ON o.user_id = u.id
JOIN order_items oi ON o.id = oi.order_id
JOIN products p ON oi.product_id = p.id
WHERE o.created_at > CURRENT_DATE - INTERVAL '30 days'
ORDER BY o.created_at DESC;
```
