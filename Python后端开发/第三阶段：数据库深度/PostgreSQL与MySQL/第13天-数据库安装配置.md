## 1. PostgreSQL 安装与配置

### macOS 安装

```bash
# 使用 Homebrew 安装
brew install postgresql@16

# 启动服务
brew services start postgresql@16

# 创建数据库集群（如果需要）
# PostgreSQL 16+ 不需要 initdb

# 连接数据库
psql -h localhost -U $(whoami) -d postgres
```

### Linux 安装（Ubuntu）

```bash
# 安装
sudo apt update
sudo apt install postgresql postgresql-contrib

# 启动服务
sudo systemctl start postgresql
sudo systemctl enable postgresql

# 切换到 postgres 用户
sudo -u postgres psql
```

### Windows 安装

```bash
# 下载安装包
# https://www.postgresql.org/download/windows/

# 或使用 Chocolatey
choco install postgresql
```

### 配置文件

```bash
# PostgreSQL 配置文件位置
# macOS: /opt/homebrew/var/postgresql@16/
# Linux: /etc/postgresql/16/main/

# postgresql.conf - 主配置文件
listen_addresses = 'localhost'      # 监听地址
port = 5432                           # 端口
max_connections = 100                 # 最大连接数
shared_buffers = 128MB               # 共享缓冲区
effective_cache_size = 512MB          # 有效缓存大小
maintenance_work_mem = 64MB           # 维护工作内存
wal_buffers = 16MB                   # WAL 缓冲区
checkpoint_completion_target = 0.9   # 检查点目标
max_wal_size = 1GB                  # 最大 WAL 大小
min_wal_size = 80MB                  # 最小 WAL 大小

# Linux: 编辑配置
sudo nano /etc/postgresql/16/main/postgresql.conf

# 重启服务使配置生效
sudo systemctl restart postgresql
```

### pg_hba.conf 认证配置

```bash
# pg_hba.conf 控制客户端认证

# 添加新的用户和数据库后需要配置认证
# Linux 位置: /etc/postgresql/16/main/pg_hba.conf

# 认证配置示例
# TYPE  DATABASE        USER            ADDRESS                 METHOD

# 本地连接
local   all             postgres                                peer
local   all             all                                     peer

# IPv4 本地连接
host    all             all             127.0.0.1/32            scram-sha-256

# 允许远程连接（开发环境）
host    all             all             0.0.0.0/0              scram-sha-256

# 修改后重载配置
sudo systemctl reload postgresql
```

### 创建用户和数据库

```sql
-- 连接后执行
-- 创建用户
CREATE USER myuser WITH PASSWORD 'mypassword';

-- 创建数据库
CREATE DATABASE mydb OWNER myuser;

-- 授予权限
GRANT ALL PRIVILEGES ON DATABASE mydb TO myuser;

-- 切换到数据库
\c mydb

-- 授予 schema 权限
GRANT ALL ON SCHEMA public TO myuser;
```

## 2. MySQL 安装与配置

### macOS 安装

```bash
# 使用 Homebrew 安装
brew install mysql

# 启动服务
brew services start mysql

# 安全初始化
mysql_secure_installation

# 连接数据库
mysql -u root -p
```

### Linux 安装（Ubuntu）

```bash
# 安装
sudo apt update
sudo apt install mysql-server

# 启动服务
sudo systemctl start mysql
sudo systemctl enable mysql

# 安全初始化
sudo mysql_secure_installation

# 连接数据库
sudo mysql
```

### 配置文件

```bash
# MySQL 配置文件位置
# macOS: /opt/homebrew/etc/mysql/my.cnf
# Linux: /etc/mysql/mysql.conf.d/mysqld.cnf

# my.cnf 配置示例
[mysqld]
# 基本设置
port = 3306
max_connections = 151
max_connect_errors = 100

# 缓冲区设置
innodb_buffer_pool_size = 128M
innodb_log_buffer_size = 16M
key_buffer_size = 32M

# 日志设置
slow_query_log = 1
slow_query_log_file = /var/log/mysql/slow.log
long_query_time = 2

# 字符集
character-set-server = utf8mb4
collation-server = utf8mb4_unicode_ci

# 重启服务
sudo systemctl restart mysql
```

### 创建用户和数据库

```sql
-- 连接后执行
-- 创建用户
CREATE USER 'myuser'@'localhost' IDENTIFIED BY 'mypassword';
CREATE USER 'myuser'@'%' IDENTIFIED BY 'mypassword';  -- 允许远程

-- 创建数据库
CREATE DATABASE mydb CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;

-- 授予权限
GRANT ALL PRIVILEGES ON mydb.* TO 'myuser'@'localhost';
GRANT ALL PRIVILEGES ON mydb.* TO 'myuser'@'%';

-- 刷新权限
FLUSH PRIVILEGES;
```

## 3. Python 连接数据库

### 安装驱动

```bash
# PostgreSQL 驱动
pip install psycopg2-binary
# 或
pip install psycopg2

# MySQL 驱动
pip install pymysql
```

### PostgreSQL 连接

```python
import psycopg2
from contextlib import contextmanager

# 基本连接
conn = psycopg2.connect(
    host="localhost",
    port=5432,
    database="mydb",
    user="myuser",
    password="mypassword"
)

cursor = conn.cursor()
cursor.execute("SELECT version()")
print(cursor.fetchone())

cursor.close()
conn.close()

# 使用 context manager
@contextmanager
def get_connection():
    conn = psycopg2.connect(
        host="localhost",
        port=5432,
        database="mydb",
        user="myuser",
        password="mypassword"
    )
    try:
        yield conn
        conn.commit()
    except Exception as e:
        conn.rollback()
        raise e
    finally:
        conn.close()

# 使用
with get_connection() as conn:
    with conn.cursor() as cursor:
        cursor.execute("SELECT * FROM users")
        for row in cursor.fetchall():
            print(row)
```

### MySQL 连接

```python
import pymysql

conn = pymysql.connect(

    charset="utf8mb4"

        host="localhost",
        port=3306,
        database="mydb",

        yield conn

        conn.close()
```

## 4. SQLAlchemy 连接

### 安装

```bash
pip install sqlalchemy
```

### 连接 PostgreSQL

```python
from sqlalchemy import create_engine

# PostgreSQL
engine = create_engine(
    "postgresql://myuser:mypassword@localhost:5432/mydb"
)

# 或者使用 URL 方式
engine = create_engine(
    "postgresql+psycopg2://myuser:mypassword@localhost:5432/mydb"
)

# 查询
with engine.connect() as conn:
    result = conn.execute("SELECT version()")
    print(result.fetchone())
```

### 连接 MySQL

```python
from sqlalchemy import create_engine

# MySQL
engine = create_engine(
    "mysql+pymysql://myuser:mypassword@localhost:3306/mydb?charset=utf8mb4"
)

# 查询
with engine.connect() as conn:
    result = conn.execute("SELECT version()")
    print(result.fetchone())
```

### 连接池配置

```python
from sqlalchemy import create_engine
from sqlalchemy.pool import QueuePool

engine = create_engine(
    "postgresql://myuser:mypassword@localhost:5432/mydb",
    poolclass=QueuePool,
    pool_size=10,              # 连接池大小
    max_overflow=20,           # 最大溢出
    pool_pre_ping=True,        # 连接前检测
    pool_recycle=3600,         # 回收时间（秒）
    echo=False                  # 是否打印 SQL
)
```

## 5. Docker 部署数据库

### PostgreSQL Docker

```bash
# 启动 PostgreSQL 容器
docker run -d \
  --name postgres \
  -e POSTGRES_USER=myuser \
  -e POSTGRES_PASSWORD=mypassword \
  -e POSTGRES_DB=mydb \
  -p 5432:5432 \
  -v postgres_data:/var/lib/postgresql/data \
  postgres:16-alpine

# 连接
psql -h localhost -U myuser -d mydb
```

### MySQL Docker

```bash
# 启动 MySQL 容器
docker run -d \
  --name mysql \
  -e MYSQL_ROOT_PASSWORD=rootpassword \
  -e MYSQL_DATABASE=mydb \
  -e MYSQL_USER=myuser \
  -e MYSQL_PASSWORD=mypassword \
  -p 3306:3306 \
  -v mysql_data:/var/lib/mysql \
  mysql:8

# 连接
mysql -h localhost -u myuser -p mydb
```

### docker-compose.yml

```yaml
version: '3.8'

services:
  postgres:
    image: postgres:16-alpine
    environment:
      POSTGRES_USER: myuser
      POSTGRES_PASSWORD: mypassword
      POSTGRES_DB: mydb
    ports:
      - "5432:5432"
    volumes:
      - postgres_data:/var/lib/postgresql/data

  mysql:
    image: mysql:8
    environment:
      MYSQL_ROOT_PASSWORD: rootpassword
      MYSQL_DATABASE: mydb
      MYSQL_USER: myuser
      MYSQL_PASSWORD: mypassword
    ports:
      - "3306:3306"
    volumes:
      - mysql_data:/var/lib/mysql

volumes:
  postgres_data:
  mysql_data:
```

## 6. 数据库管理工具

### 命令行工具

```bash
# PostgreSQL
psql -h localhost -U myuser -d mydb    # 交互式连接
pg_dump -h localhost -U myuser mydb > backup.sql  # 导出
pg_restore -h localhost -U myuser -d mydb backup.sql  # 导入

# MySQL
mysql -h localhost -u myuser -p mydb    # 交互式连接
mysqldump -h localhost -u myuser -p mydb > backup.sql  # 导出
mysql -h localhost -u myuser -p mydb < backup.sql  # 导入
```
