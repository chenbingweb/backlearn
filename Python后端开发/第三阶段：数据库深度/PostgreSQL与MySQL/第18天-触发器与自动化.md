## 1. 触发器基础

### 触发器概念

触发器（Trigger）是数据库自动执行的特殊存储过程，在指定事件发生时自动调用。

```
事件 -> 触发器 -> 操作
INSERT/UPDATE/DELETE -> BEFORE/AFTER -> 执行逻辑
```

### 触发器类型

```
BEFORE 触发器：事件执行前触发，可修改数据
AFTER 触发器：事件执行后触发，用于日志记录
INSTEAD OF 触发器：替代原始操作（用于视图）

行级触发器：每行受影响时触发
语句级触发器：每个语句执行一次
```

## 2. PostgreSQL 触发器

### 创建触发器

```sql
-- 创建触发器函数
CREATE OR REPLACE FUNCTION update_modified_column()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = NOW();
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- 为表创建触发器
CREATE TRIGGER update_products_timestamp
    BEFORE UPDATE ON products
    FOR EACH ROW
    EXECUTE FUNCTION update_modified_column();

-- 多个列的更新触发
CREATE OR REPLACE FUNCTION check_price_change()
RETURNS TRIGGER AS $$
BEGIN
    IF OLD.price IS DISTINCT FROM NEW.price THEN
        RAISE NOTICE 'Price changed from % to %', OLD.price, NEW.price;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER check_product_price
    BEFORE UPDATE ON products
    FOR EACH ROW
    EXECUTE FUNCTION check_price_change();
```

### 审计日志触发器

```sql
-- 创建审计日志表
CREATE TABLE audit_log (
    id SERIAL PRIMARY KEY,
    table_name VARCHAR(50) NOT NULL,
    operation VARCHAR(10) NOT NULL,
    old_data JSONB,
    new_data JSONB,
    changed_by VARCHAR(100),
    changed_at TIMESTAMP DEFAULT NOW()
);

-- 创建审计触发器函数
CREATE OR REPLACE FUNCTION audit_trigger_function()
RETURNS TRIGGER AS $$
BEGIN
    IF TG_OP = 'INSERT' THEN
        INSERT INTO audit_log (table_name, operation, new_data, changed_by)
        VALUES (TG_TABLE_NAME, TG_OP, to_jsonb(NEW), current_user);
        RETURN NEW;
    ELSIF TG_OP = 'UPDATE' THEN
        INSERT INTO audit_log (table_name, operation, old_data, new_data, changed_by)
        VALUES (TG_TABLE_NAME, TG_OP, to_jsonb(OLD), to_jsonb(NEW), current_user);
        RETURN NEW;
    ELSIF TG_OP = 'DELETE' THEN
        INSERT INTO audit_log (table_name, operation, old_data, changed_by)
        VALUES (TG_TABLE_NAME, TG_OP, to_jsonb(OLD), current_user);
        RETURN OLD;
    END IF;
    RETURN NULL;
END;
$$ LANGUAGE plpgsql;

-- 为表添加审计触发器
CREATE TRIGGER audit_orders
    AFTER INSERT OR UPDATE OR DELETE ON orders
    FOR EACH ROW
    EXECUTE FUNCTION audit_trigger_function();

-- 查询审计日志
SELECT * FROM audit_log WHERE table_name = 'orders' ORDER BY changed_at DESC;
```

### BEFORE 触发器（数据验证）

```sql
-- 订单验证触发器
CREATE OR REPLACE FUNCTION validate_order()
RETURNS TRIGGER AS $$
BEGIN
    -- 检查订单金额
    IF NEW.total < 0 THEN
        RAISE EXCEPTION 'Order total cannot be negative';
    END IF;

    -- 检查订单日期
    IF NEW.order_date > CURRENT_DATE THEN
        RAISE EXCEPTION 'Order date cannot be in the future';
    END IF;

    -- 确保状态有效
    IF NEW.status NOT IN ('pending', 'paid', 'shipped', 'completed', 'cancelled') THEN
        RAISE EXCEPTION 'Invalid order status: %', NEW.status;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER validate_order_insert
    BEFORE INSERT OR UPDATE ON orders
    FOR EACH ROW
    EXECUTE FUNCTION validate_order();
```

### INSTEAD OF 触发器（视图）

```sql
-- 创建可更新视图
CREATE VIEW active_products AS
SELECT id, name, price, stock, category_id
FROM products
WHERE is_active = TRUE;

-- 为视图创建 INSTEAD OF 触发器
CREATE OR REPLACE FUNCTION insert_active_product()
RETURNS TRIGGER AS $$
BEGIN
    INSERT INTO products (name, price, stock, category_id, is_active)
    VALUES (NEW.name, NEW.price, NEW.stock, NEW.category_id, TRUE);
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER instead_of_insert_product
    INSTEAD OF INSERT ON active_products
    FOR EACH ROW
    EXECUTE FUNCTION insert_active_product();

-- 现在可以向视图插入数据
INSERT INTO active_products (name, price, stock, category_id)
VALUES ('New Product', 99.99, 100, 1);
```

## 3. MySQL 触发器

### 基本语法

```sql
-- 创建触发器
DELIMITER //

CREATE TRIGGER update_products_timestamp
    BEFORE UPDATE ON products
    FOR EACH ROW
BEGIN
    SET NEW.updated_at = CURRENT_TIMESTAMP;
END //

-- 订单审计日志
CREATE TRIGGER audit_orders_insert
    AFTER INSERT ON orders
    FOR EACH ROW
BEGIN
    INSERT INTO audit_log (table_name, operation, new_data, changed_by)
    VALUES ('orders', 'INSERT', NEW.id, current_user());
END //

CREATE TRIGGER audit_orders_update
    AFTER UPDATE ON orders
    FOR EACH ROW
BEGIN
    INSERT INTO audit_log (table_name, operation, old_data, new_data, changed_by)
    VALUES ('orders', 'UPDATE', OLD.id, NEW.id, current_user());
END //

DELIMITER ;
```

### 查看和删除

```sql
-- 查看触发器
SHOW TRIGGERS;
SHOW TRIGGERS LIKE 'orders%';

-- MySQL 8.0+ 信息_schema
SELECT * FROM information_schema.TRIGGERS
WHERE TRIGGER_SCHEMA = 'test';

-- 删除触发器
DROP TRIGGER IF EXISTS update_products_timestamp;
```

## 4. 触发器应用场景

### 自动库存管理

```sql
-- PostgreSQL：库存自动扣减
CREATE OR REPLACE FUNCTION update_inventory()
RETURNS TRIGGER AS $$
BEGIN
    -- 扣减库存
    UPDATE products
    SET stock = stock - NEW.quantity
    WHERE id = NEW.product_id;

    -- 检查库存
    IF (SELECT stock FROM products WHERE id = NEW.product_id) < 0 THEN
        RAISE EXCEPTION 'Insufficient stock for product %', NEW.product_id;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER order_item_after_insert
    AFTER INSERT ON order_items
    FOR EACH ROW
    EXECUTE FUNCTION update_inventory();

-- 订单取消时恢复库存
CREATE OR REPLACE FUNCTION restore_inventory()
RETURNS TRIGGER AS $$
BEGIN
    IF OLD.status != 'cancelled' AND NEW.status = 'cancelled' THEN
        UPDATE products
        SET stock = stock + (
            SELECT COALESCE(SUM(quantity), 0)
            FROM order_items
            WHERE order_id = NEW.id
        )
        WHERE id IN (
            SELECT product_id FROM order_items WHERE order_id = NEW.id
        );
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER restore_inventory_on_cancel
    AFTER UPDATE ON orders
    FOR EACH ROW
    EXECUTE FUNCTION restore_inventory();
```

### 自动创建用户统计

```sql
-- PostgreSQL
CREATE OR REPLACE FUNCTION update_user_stats()
RETURNS TRIGGER AS $$
BEGIN
    -- 更新用户订单数
    UPDATE user_stats
    SET order_count = (
        SELECT COUNT(*) FROM orders WHERE user_id = NEW.user_id
    ),
    total_spent = (
        SELECT COALESCE(SUM(total), 0) FROM orders WHERE user_id = NEW.user_id
    ),
    last_order_date = NEW.created_at
    WHERE user_id = NEW.user_id;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER update_stats_on_order
    AFTER INSERT ON orders
    FOR EACH ROW
    EXECUTE FUNCTION update_user_stats();
```

## 5. 事件调度器

### PostgreSQL pg_cron

```sql
-- 安装 pg_cron
CREATE EXTENSION pg_cron;

-- 配置
SELECT cron.schedule('vacuum-daily', '0 3 * * *', 'VACUUM ANALYZE');

-- 查看调度任务
SELECT * FROM cron.job;

-- 删除任务
SELECT cron.unschedule('vacuum-daily');
```

```sql
-- 定期清理日志
SELECT cron.schedule(
    'cleanup-old-logs',
    '0 2 * * *',
    $$DELETE FROM logs WHERE created_at < NOW() - INTERVAL '30 days'$$
);

-- 定期汇总统计数据
SELECT cron.schedule(
    'daily-sales-summary',
    '0 1 * * *',
    $$INSERT INTO daily_summary (date, total_sales)
      SELECT CURRENT_DATE, COALESCE(SUM(total), 0)
      FROM orders
      WHERE status = 'completed'$$
);
```

### MySQL Event Scheduler

```sql
-- 开启事件调度器
SET GLOBAL event_scheduler = ON;

-- 创建事件
DELIMITER //
CREATE EVENT daily_cleanup
ON SCHEDULE EVERY 1 DAY
STARTS CURRENT_TIMESTAMP
DO
BEGIN
    DELETE FROM logs WHERE created_at < DATE_SUB(NOW(), INTERVAL 30 DAY);
    OPTIMIZE TABLE logs;
END //
DELIMITER ;

-- 查看事件
SHOW EVENTS;
SHOW EVENTS FROM test;

-- 删除事件
DROP EVENT IF EXISTS daily_cleanup;
```

## 6. Python 中的触发器使用

```python
import psycopg2
from datetime import datetime, timedelta

# 触发器在数据库端自动执行，应用层无需特殊处理
# 但可以查询触发器相关信息

conn = psycopg2.connect("dbname=test user=postgres")
cur = conn.cursor()

# 查看触发器
cur.execute("""
    SELECT trigger_name, event_manipulation, action_statement
    FROM information_schema.triggers
    WHERE trigger_schema = 'public'
""")
for row in cur.fetchall():
    print(row)

# 查看触发器依赖关系
cur.execute("""
    SELECT
        t.tgname AS trigger_name,
        c.relname AS table_name,
        p.proname AS function_name
    FROM pg_trigger t
    JOIN pg_class c ON t.tgrelid = c.oid
    JOIN pg_proc p ON t.tgfoid = p.oid
    WHERE c.relname = 'orders'
""")
for row in cur.fetchall():
    print(row)

# 禁用/启用触发器（需要超级用户）
cur.execute("ALTER TABLE orders DISABLE TRIGGER update_products_timestamp")
cur.execute("ALTER TABLE orders ENABLE TRIGGER update_products_timestamp")

cur.close()
conn.close()
```

## 7. 实战练习

### 练习 1：完整的审计系统

```sql
-- 1. 创建审计表
CREATE TABLE audit_log (
    id SERIAL PRIMARY KEY,
    table_name VARCHAR(50) NOT NULL,
    record_id INTEGER NOT NULL,
    operation VARCHAR(10) NOT NULL,
    old_values JSONB,
    new_values JSONB,
    changed_by VARCHAR(100),
    changed_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    ip_address INET
);

-- 2. 创建通用审计函数
CREATE OR REPLACE FUNCTION audit_function()
RETURNS TRIGGER AS $$
DECLARE
    audit_row audit_log;
    old_json JSONB;
    new_json JSONB;
BEGIN
    old_json := NULL;
    new_json := NULL;

    IF TG_OP = 'DELETE' THEN
        old_json := to_jsonb(OLD);
        audit_row := ROW(
            NULL,
            TG_TABLE_NAME,
            OLD.id,
            TG_OP,
            old_json,
            new_json,
            current_user,
            current_timestamp,
            NULL
        )::audit_log;
        INSERT INTO audit_log VALUES (audit_row.*);
        RETURN OLD;
    ELSE
        IF TG_OP = 'UPDATE' THEN
            old_json := to_jsonb(OLD);
        END IF;
        new_json := to_jsonb(NEW);
        audit_row := ROW(
            NULL,
            TG_TABLE_NAME,
            NEW.id,
            TG_OP,
            old_json,
            new_json,
            current_user,
            current_timestamp,
            NULL
        )::audit_log;
        INSERT INTO audit_log VALUES (audit_row.*);
        RETURN NEW;
    END IF;
END;
$$ LANGUAGE plpgsql;

-- 3. 为敏感表添加触发器
CREATE TRIGGER audit_products
    AFTER INSERT OR UPDATE OR DELETE ON products
    FOR EACH ROW
    EXECUTE FUNCTION audit_function();

CREATE TRIGGER audit_orders
    AFTER INSERT OR UPDATE OR DELETE ON orders
    FOR EACH ROW
    EXECUTE FUNCTION audit_function();

CREATE TRIGGER audit_customers
    AFTER INSERT OR UPDATE OR DELETE ON customers
    FOR EACH ROW
    EXECUTE FUNCTION audit_function();
```

## 今日总结

- [ ] 触发器在事件发生时自动执行
- [ ] `BEFORE` 触发器可修改数据，`AFTER` 触发器用于日志
- [ ] `FOR EACH ROW` 逐行触发，`INSTEAD OF` 用于视图
- [ ] `NEW` 和 `OLD` 访问新、旧行数据
- [ ] PostgreSQL 用 `pg_cron` 调度任务
- [ ] MySQL 用 `Event Scheduler` 调度任务
- [ ] 触发器适合审计日志、自动化维护任务
