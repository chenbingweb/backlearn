## 1. 缓存穿透

### 什么是穿透

```
正常请求：
  用户请求 -> Redis缓存 -> 命中 -> 返回数据
                     |
                     v (未命中)
                  数据库 -> 查到数据 -> 回填缓存 -> 返回数据

穿透请求：
  用户请求 -> Redis缓存 -> 未命中
                     |
                     v (恶意构造的不存在的数据)
                  数据库 -> 也查不到
                     |
                     v
                  返回空（但不会回填缓存）
                     |
                     v (大量请求打到数据库)
                  数据库压力过大
```

### 原因

```markdown
1. 恶意攻击：故意构造大量不存在的 Key
2. 业务问题：查询条件导致查询不存在的数据
3. 数据误删：缓存和数据库数据不一致
```

### 解决方案

### 方案1：缓存空值

```python
def get_user_with_null_cache(user_id):
    """缓存空值方案"""
    cache_key = f'user:{user_id}'

    # 1. 查缓存
    cached = r.get(cache_key)

    if cached == 'NULL':
        return None  # 缓存的空值

    if cached:
        return json.loads(cached)

    # 2. 查数据库
    user = db.query(User).get(user_id)

    # 3. 回填缓存（包括空值）
    if user:
        r.setex(cache_key, 3600, json.dumps(user.to_dict()))
    else:
        # 缓存空值，但过期时间短一些
        r.setex(cache_key, 60, 'NULL')

    return user
```

### 方案2：布隆过滤器

```python
from bloom_filter import BloomFilter

class BloomFilterCache:
    def __init__(self, redis_client, expected_items=100000, error_rate=0.01):
        self.r = redis_client
        self.bf_key = 'bloom:user:ids'

        # 初始化布隆过滤器
        self.bloom = BloomFilter(max_elements=expected_items, error_rate=error_rate)
        self._load_bloom()

    def _load_bloom(self):
        """从数据库加载所有合法 ID 到布隆过滤器"""
        # 启动时执行
        all_ids = db.query(User.id).all()
        for (user_id,) in all_ids:
            self.bloom.add(user_id)

    def exists(self, user_id):
        """检查 ID 是否可能存在"""
        return user_id in self.bloom

    def get_user(self, user_id):
        """获取用户"""
        # 先检查布隆过滤器
        if not self.exists(user_id):
            return None  # 一定不存在

        # 布隆过滤器说存在，可能存在，再查缓存/数据库
        cache_key = f'user:{user_id}'
        cached = self.r.get(cache_key)
        if cached:
            return json.loads(cached)

        user = db.query(User).get(user_id)
        if user:
            self.r.setex(cache_key, 3600, json.dumps(user.to_dict()))
            # 别忘了添加到布隆过滤器
            self.bloom.add(user_id)

        return user
```

### 方案3：布隆过滤器 + Redis Bitmap

```python
class RedisBloomFilter:
    def __init__(self, redis_client, name='bloom'):
        self.r = redis_client
        self.name = f'bloom:{name}'

    def _get_bit_index(self, hash_value, bit_size):
        """获取位数组索引"""
        return hash_value % bit_size

    def add(self, value):
        """添加元素"""
        hash1 = hash(value)
        hash2 = hash1 >> 16

        bit_size = 100000000  # 1亿位 ≈ 12MB

        pipe = self.r.pipeline()
        pipe.setbit(self.name, self._get_bit_index(hash1, bit_size), 1)
        pipe.setbit(self.name, self._get_bit_index(hash2, bit_size), 1)
        pipe.execute()

    def might_exist(self, value):
        """检查元素是否存在"""
        hash1 = hash(value)
        hash2 = hash1 >> 16

        bit_size = 100000000

        bit1 = self.r.getbit(self.name, self._get_bit_index(hash1, bit_size))
        bit2 = self.r.getbit(self.name, self._get_bit_index(hash2, bit_size))

        return bit1 and bit2
```

## 2. 缓存雪崩

### 什么是雪崩

```
雪崩场景：
  大量缓存 key 同时过期
       │
       ▼
  大量请求同时到来
       │
       ▼
  请求全部打到数据库
       │
       ▼
  数据库承受不住
       │
       ▼
  系统崩溃
```

### 原因

```markdown
1. 大量 key 同时过期
2. Redis 宕机
3. 缓存服务不可用
```

### 解决方案

### 方案1：过期时间随机化

```python
def set_with_jitter(value, key, base_ttl=3600):
    """设置带抖动的过期时间"""
    import random

    # 过期时间 = 基础时间 + 随机抖动（基础时间的 20%）
    jitter = random.uniform(0, base_ttl * 0.2)
    ttl = int(base_ttl + jitter)

    r.setex(key, ttl, json.dumps(value))

# 使用
set_with_jitter(user_data, 'user:1001', base_ttl=3600)
# 实际过期时间：3600 ~ 4320 秒之间随机
```

### 方案2：多级缓存

```python
class MultiLevelCache:
    """本地缓存 + Redis 缓存"""

    def __init__(self):
        # L1: 本地缓存（进程内）
        self.local_cache = {}

        # L2: Redis 缓存
        self.r = redis.Redis()

    def get(self, key):
        """获取数据"""
        # L1 优先
        if key in self.local_cache:
            return self.local_cache[key]

        # L2 其次
        cached = self.r.get(key)
        if cached:
            data = json.loads(cached)
            # 回填 L1
            self.local_cache[key] = data
            return data

        return None

    def set(self, key, value, ttl=3600):
        """设置数据"""
        # 同时设置 L1 和 L2
        self.local_cache[key] = value
        self.r.setex(key, ttl, json.dumps(value))

    def invalidate(self, key):
        """失效缓存"""
        if key in self.local_cache:
            del self.local_cache[key]
        self.r.delete(key)
```

### 方案3：限流降级

```python
import time
import threading

class RateLimiter:
    """限流器"""

    def __init__(self, redis_client):
        self.r = redis_client
        self.request_counts = {}
        self.lock = threading.Lock()

    def is_allowed(self, key, max_requests, window_seconds):
        """是否允许请求"""
        now = time.time()
        window_key = f'rate:{key}:{int(now / window_seconds)}'

        pipe = self.r.pipeline()
        pipe.incr(window_key)
        pipe.expire(window_key, window_seconds)
        counts = pipe.execute()

        current_count = counts[0]

        return current_count <= max_requests


class CircuitBreaker:
    """熔断器"""

    def __init__(self, failure_threshold=5, timeout_seconds=60):
        self.failure_threshold = failure_threshold
        self.timeout_seconds = timeout_seconds
        self.failures = 0
        self.last_failure_time = None
        self.state = 'CLOSED'  # CLOSED, OPEN, HALF_OPEN

    def call(self, func, *args, **kwargs):
        """执行函数"""
        if self.state == 'OPEN':
            # 检查是否超时
            if time.time() - self.last_failure_time > self.timeout_seconds:
                self.state = 'HALF_OPEN'
            else:
                raise Exception('Circuit is OPEN')

        try:
            result = func(*args, **kwargs)
            if self.state == 'HALF_OPEN':
                self.state = 'CLOSED'
                self.failures = 0
            return result
        except Exception as e:
            self.failures += 1
            self.last_failure_time = time.time()
            if self.failures >= self.failure_threshold:
                self.state = 'OPEN'
            raise e
```

## 3. 缓存击穿

### 什么是击穿

```
击穿场景：
  一个热点 key 突然过期
       │
       ▼
  大量请求同时到来
       │
       ▼
  全部打到数据库
       │
       ▼
  数据库压力过大
```

### 解决方案

### 方案1：互斥锁

```python
import redis
import json
import time

def get_with_lock(key, db_fetch_func, ttl=3600):
    """互斥锁方案获取数据"""
    lock_key = f'lock:{key}'

    # 1. 先查缓存
    cached = r.get(key)
    if cached:
        return json.loads(cached)

    # 2. 获取锁
    lock_acquired = r.set(lock_key, '1', nx=True, ex=10)

    if lock_acquired:
        try:
            # 3. 双重检查缓存
            cached = r.get(key)
            if cached:
                return json.loads(cached)

            # 4. 查数据库
            data = db_fetch_func()

            # 5. 回填缓存
            if data:
                r.setex(key, ttl, json.dumps(data))

            return data
        finally:
            # 6. 释放锁
            r.delete(lock_key)
    else:
        # 7. 没拿到锁，等待后重试
        for _ in range(3):
            time.sleep(0.1)
            cached = r.get(key)
            if cached:
                return json.loads(cached)

        # 8. 兜底：查数据库
        return db_fetch_func()
```

### 方案2：永不过期 + 逻辑过期

```python
class LogicalExpireCache:
    """逻辑过期方案"""

    def __init__(self, redis_client):
        self.r = redis_client
        self.lock_key_prefix = 'lock:'

    def get(self, key, db_fetch_func):
        """获取数据"""
        # 1. 查缓存
        cached = self.r.get(key)

        if not cached:
            return None

        # 2. 解析数据
        data = json.loads(cached)

        # 3. 检查是否过期
        if time.time() > data['expire_time']:
            # 4. 过期了，获取锁
            lock_key = f'{self.lock_key_prefix}{key}'
            lock_acquired = self.r.set(lock_key, '1', nx=True, ex=10)

            if lock_acquired:
                try:
                    # 5. 重新查数据库
                    new_data = db_fetch_func()

                    if new_data:
                        # 6. 写入新数据（新的过期时间）
                        self._set_with_logic_expire(key, new_data, 3600)

                    return new_data
                finally:
                    self.r.delete(lock_key)
            else:
                # 7. 没拿到锁，返回旧数据
                return data['value']

        # 8. 没过期，返回数据
        return data['value']

    def _set_with_logic_expire(self, key, value, ttl):
        """设置带逻辑过期的数据"""
        data = {
            'value': value,
            'expire_time': time.time() + ttl  # 逻辑过期时间
        }
        # 物理过期时间设置较长
        self.r.setex(key, ttl * 2, json.dumps(data))
```

## 5. 完整实战

```python
import redis
import json
import time
import random
import threading
from contextlib import contextmanager

class RedisCache:
    """完整的缓存解决方案"""

    def __init__(self, redis_client):
        self.r = redis_client
        self.local_cache = {}  # L1 本地缓存
        self.lock = threading.Lock()

    def get(self, key, db_func=None, ttl=3600):
        """
        获取缓存，包含以下策略：
        1. 本地缓存快速命中
        2. Redis 缓存命中
        3. 布隆过滤器过滤不存在数据
        4. 互斥锁防止击穿
        5. 逻辑过期处理雪崩
        """
        # L1 本地缓存
        if key in self.local_cache:
            return self.local_cache[key]

        # L2 Redis
        cached = self.r.get(key)
        if cached:
            data = json.loads(cached)
            # 检查逻辑过期
            if isinstance(data, dict) and 'expire_time' in data:
                if time.time() > data['expire_time']:
                    # 逻辑过期，异步更新
                    threading.Thread(
                        target=self._async_refresh,
                        args=(key, db_func, ttl)
                    ).start()
                    return data['value']
            self.local_cache[key] = data
            return data

        # 缓存未命中，查数据库
        if db_func is None:
            return None

        data = db_func()

        if data:
            # 随机过期时间防止雪崩
            actual_ttl = ttl + random.randint(0, int(ttl * 0.2))
            self.r.setex(key, actual_ttl, json.dumps(data))
            self.local_cache[key] = data

        return data

    def set(self, key, value, ttl=3600):
        """设置缓存"""
        # 随机过期时间防止雪崩
        actual_ttl = ttl + random.randint(0, int(ttl * 0.2))
        self.r.setex(key, actual_ttl, json.dumps(value))
        self.local_cache[key] = value

    def invalidate(self, key):
        """失效缓存"""
        if key in self.local_cache:
            del self.local_cache[key]
        self.r.delete(key)

    def _async_refresh(self, key, db_func, ttl):
        """异步刷新缓存"""
        with self._get_lock(key):
            data = db_func()
            if data:
                self.set(key, data, ttl)
```
