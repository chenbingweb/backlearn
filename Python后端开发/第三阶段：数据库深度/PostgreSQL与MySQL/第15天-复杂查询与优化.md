## 1. 查询计划分析

### EXPLAIN 基础

```sql
-- 查看查询计划
EXPLAIN SELECT * FROM products WHERE price > 100;

-- 实际执行计划（包含运行时信息）
EXPLAIN ANALYZE SELECT * FROM products WHERE price > 100;

-- PostgreSQL 详细格式
EXPLAIN (ANALYZE, BUFFERS, FORMAT TEXT)
SELECT * FROM products WHERE price > 100;
```

### 计划节点类型

```
Seq Scan      -- 顺序扫描（全表）
Index Scan    -- 索引扫描
Index Only Scan  -- 仅索引扫描
Bitmap Scan   -- 位图扫描
Nested Loop   -- 嵌套循环连接
Hash Join     -- 哈希连接
Merge Join    -- 合并连接
Hash Aggregate  -- 哈希聚合
Sort          -- 排序
```

### 解读执行计划

```sql
EXPLAIN ANALYZE
SELECT p.name, SUM(oi.quantity) AS total_qty
FROM products p
JOIN order_items oi ON p.id = oi.product_id
WHERE p.category_id = 5
GROUP BY p.name;

-- 分析输出：
-- Seq Scan on products    -- 检查是否使用索引
-- Index Scan on order_items  -- 检查连接是否高效
-- Hash Join              -- 检查连接算法
-- Sort                    -- 检查是否需要排序
```

## 2. 常见性能问题

### 全表扫描

```sql
-- 问题：WHERE 条件无索引
EXPLAIN SELECT * FROM orders WHERE status = 'pending';

-- 解决：创建索引
CREATE INDEX idx_orders_status ON orders(status);

-- 验证优化
EXPLAIN SELECT * FROM orders WHERE status = 'pending';
```

### 索引失效

```sql
-- 导致索引失效的操作
SELECT * FROM products WHERE price * 1.1 > 100;  -- 计算
SELECT * FROM products WHERE UPPER(name) = 'ABC';  -- 函数
SELECT * FROM products WHERE name LIKE '%abc';    -- 前缀通配符

-- 正确的查询方式
SELECT * FROM products WHERE price > 100 / 1.1;
SELECT * FROM products WHERE name = 'ABC';
SELECT * FROM products WHERE name LIKE 'abc%';     -- 前缀匹配
```

### N+1 查询问题

```sql
-- 问题：循环查询
-- Python 示例：
-- for user in users:
--     orders = db.query("SELECT * FROM orders WHERE user_id = ?", user.id)

-- 解决：使用 JOIN 或 IN 查询
SELECT
    u.id,
    u.name,
    o.order_count
FROM users u
LEFT JOIN (
    SELECT user_id, COUNT(*) AS order_count
    FROM orders
    GROUP BY user_id
) o ON u.id = o.user_id;

-- 或
SELECT
    u.id,
    u.name,
    (SELECT COUNT(*) FROM orders WHERE user_id = u.id) AS order_count
FROM users u;
```

### 过度使用子查询

```sql
-- 问题：多层嵌套子查询
SELECT * FROM (
    SELECT * FROM (
        SELECT * FROM products WHERE price > 100
    ) t WHERE category = 'electronics'
) t2;

-- 解决：使用 CTE 或 JOIN
WITH filtered_products AS (
    SELECT * FROM products WHERE price > 100
)
SELECT * FROM filtered_products WHERE category = 'electronics';
```

## 3. 查询优化技巧

### 选择合适列

```sql
-- 问题：SELECT *
SELECT * FROM products JOIN orders ON products.id = orders.product_id;

-- 解决：只查询需要的列
SELECT
    p.name,
    p.price,
    o.order_no,
    o.quantity
FROM products p
JOIN orders o ON p.id = o.product_id;
```

### 使用 LIMIT 限制结果

```sql
-- 分页查询优化
-- 方案 1：基于偏移量（大数据时慢）
SELECT * FROM products ORDER BY id LIMIT 100 OFFSET 10000;

-- 方案 2：基于游标（高效）
SELECT * FROM products
WHERE id > 10000
ORDER BY id
LIMIT 100;

-- 方案 3：延迟 JOIN
SELECT p.* FROM products p
INNER JOIN (
    SELECT id FROM products ORDER BY id LIMIT 100 OFFSET 10000
) t ON p.id = t.id;
```

### 批量操作

```sql
-- 问题：循环单条插入
-- for item in items:
--     cursor.execute("INSERT INTO orders ...", item)

-- 解决：批量插入
INSERT INTO orders (product_id, quantity, price)
VALUES
    (1, 10, 100),
    (2, 20, 200),
    (3, 30, 300);

-- PostgreSQL 批量插入
INSERT INTO orders (product_id, quantity, price)
SELECT product_id, quantity, price FROMunnest($1::int[], $2::int[], $3::numeric[]);
```

### 预计算和缓存

```sql
-- 创建汇总表
CREATE TABLE daily_sales_summary AS
SELECT
    DATE(created_at) AS sale_date,
    COUNT(*) AS order_count,
    SUM(total) AS daily_total
FROM orders
GROUP BY DATE(created_at);

-- 定期刷新
REFRESH MATERIALIZED VIEW daily_sales_summary;

-- 使用汇总表替代实时计算
SELECT * FROM daily_sales_summary WHERE sale_date = CURRENT_DATE;
```

## 4. 索引优化

### 覆盖索引

```sql
-- 创建覆盖索引：包含 WHERE、JOIN、SELECT 所需列
CREATE INDEX idx_orders_status_user_date
ON orders(status, user_id, created_at)
INCLUDE (total);  -- PostgreSQL 12+

-- 查询可直接从索引返回，无需回表
SELECT user_id, created_at, total
FROM orders
WHERE status = 'completed'
AND user_id = 1;
```

### 部分索引

```sql
-- 只索引活跃订单
CREATE INDEX idx_active_orders ON orders(created_at)
WHERE status = 'active';

-- 只索引大额订单
CREATE INDEX idx_large_orders ON orders(user_id)
WHERE total > 10000;

-- 部分索引查询
SELECT * FROM orders WHERE status = 'active' AND created_at > '2024-01-01';
```

### 表达式索引

```sql
-- 创建表达式索引
CREATE INDEX idx_products_price_calc ON products(price * (1 - discount));

-- 使用表达式查询
SELECT * FROM products WHERE price * (1 - discount) > 100;

-- PostgreSQL 支持函数索引
CREATE INDEX idx_users_email_lower ON users(LOWER(email));
SELECT * FROM users WHERE LOWER(email) = 'test@example.com';
```

## 5. 连接优化

### 连接顺序

```sql
-- 小表放左边（驱动表）
EXPLAIN
SELECT *
FROM small_table s
JOIN large_table l ON s.id = l.small_table_id;

-- 检查连接类型
-- Nested Loop：适合小表连接大表（有索引）
-- Hash Join：适合大表连接大表
-- Merge Join：适合已排序的数据
```

### 连接类型选择

```sql
-- 强制使用特定连接
EXPLAIN (SET enable_hashjoin = off)
SELECT * FROM orders o JOIN customers c ON o.customer_id = c.id;

-- 查看统计信息
SELECT
    tablename,
    reltuples AS row_estimate,
    relpages AS page_count
FROM pg_class
WHERE relname IN ('orders', 'customers');
```

## 6. 实战练习

### 练习 1：慢查询分析与优化

```sql
-- 原始慢查询
EXPLAIN ANALYZE
SELECT
    u.username,
    o.order_no,
    p.product_name,
    oi.quantity,
    oi.price
FROM users u
JOIN orders o ON u.id = o.user_id
JOIN order_items oi ON o.id = oi.order_id
JOIN products p ON oi.product_id = p.id
WHERE o.created_at > '2024-01-01'
ORDER BY o.created_at DESC;

-- 优化方案：
-- 1. 添加索引
CREATE INDEX idx_orders_created_at ON orders(created_at DESC);
CREATE INDEX idx_order_items_order_id ON order_items(order_id);
CREATE INDEX idx_order_items_product_id ON order_items(product_id);

-- 2. 限制结果集
LIMIT 1000;

-- 3. 使用覆盖索引
CREATE INDEX idx_orders_date_cover ON orders(created_at DESC)
INCLUDE (user_id, status);
```

### 练习 2：分页优化

```sql
-- 原始分页（OFFSET 性能问题）
EXPLAIN ANALYZE
SELECT * FROM orders
ORDER BY created_at DESC
LIMIT 100 OFFSET 10000;

-- 优化：使用游标分页
CREATE INDEX idx_orders_cursor ON orders(created_at DESC, id);
CREATE INDEX idx_orders_id ON orders(id);

-- 优化查询
EXPLAIN ANALYZE
SELECT * FROM orders
WHERE (created_at, id) < ('2024-01-01', 12345)
ORDER BY created_at DESC, id DESC
LIMIT 100;
```

## 7. PostgreSQL 性能视图

```sql
-- 查看慢查询（PostgreSQL）
SELECT
    query,
    calls,
    total_time,
    mean_time,
    rows
FROM pg_stat_statements
ORDER BY total_time DESC
LIMIT 10;

-- 开启扩展
CREATE EXTENSION IF NOT EXISTS pg_stat_statements;

-- 查看表统计信息
SELECT
    schemaname,
    tablename,
    seq_scan,
    idx_scan,
    n_live_tup AS row_count,
    n_dead_tup AS dead_rows
FROM pg_stat_user_tables
WHERE tablename = 'orders';
```
