## 1. 缓存架构设计

### 多级缓存架构

```
┌─────────────────────────────────────────────────────────┐
│                      用户请求                            │
└───────────────────────┬─────────────────────────────────┘
                        ▼
┌─────────────────────────────────────────────────────────┐

└───────────────────────┬─────────────────────────────────┘
                        ▼
┌─────────────────────────────────────────────────────────┐
│                 L2: Nginx 缓存                          │
│              热点页面、API 响应缓存                       │
└───────────────────────┬─────────────────────────────────┘
                        ▼
┌─────────────────────────────────────────────────────────┐
│                 L3: 本地缓存                            │
│             进程内缓存、热点数据                         │
└───────────────────────┬─────────────────────────────────┘
                        ▼
┌─────────────────────────────────────────────────────────┐
│                 L4: Redis 缓存                          │
│              分布式缓存、会话数据                         │
└───────────────────────┬─────────────────────────────────┘
                        ▼
┌─────────────────────────────────────────────────────────┐
│                 L5: 数据库                              │
│              MySQL / PostgreSQL                         │
└─────────────────────────────────────────────────────────┘
```

### 各层缓存特点

| 层级 | 介质 | 延迟 | 容量 | 一致性 |
|------|------|------|------|--------|
| L1 CDN | 边缘节点 | 1-10ms | TB级 | 静态资源 |
| L2 Nginx | 内存 | 0.1-1ms | MB级 | 页面级 |
| L3 本地 | 内存/进程 | 0.01ms | MB级 | 弱一致 |
| L4 Redis | 内存/SSD | 0.5-2ms | GB级 | 最终一致 |
| L5 DB | 磁盘 | 5-20ms | TB级 | 强一致 |

## 2. 缓存模式

### Cache-Aside（旁路缓存）

```python
"""
Cache-Aside 模式
读：先读缓存，miss 后读数据库，回填缓存
写：先写数据库，再删除缓存
"""

class CacheAside:
    def __init__(self, redis_client, db):
        self.r = redis_client
        self.db = db

    def get_user(self, user_id):
        """读操作"""
        cache_key = f'user:{user_id}'

        # 1. 读缓存
        cached = self.r.get(cache_key)
        if cached:
            return json.loads(cached)

        # 2. 缓存未命中，读数据库
        user = self.db.query(User).get(user_id)

        # 3. 回填缓存
        if user:
            self.r.setex(cache_key, 3600, json.dumps(user.to_dict()))

        return user

    def update_user(self, user_id, data):
        """写操作"""
        # 1. 写数据库
        user = self.db.query(User).get(user_id)
        for key, value in data.items():
            setattr(user, key, value)
        self.db.commit()

        # 2. 删除缓存
        self.r.delete(f'user:{user_id}')

        # 注意：不是更新缓存，而是删除
        # 下次访问会重新加载
```

### Read-Through（读穿透）

```python
"""
Read-Through 模式
读：缓存服务自动加载数据，应用程序只和缓存交互
"""

class ReadThrough:
    def __init__(self, redis_client, db):
        self.r = redis_client
        self.db = db

    def get_user(self, user_id):
        """读操作 - 缓存服务自动处理"""
        cache_key = f'user:{user_id}'

        # 尝试从缓存获取
        cached = self.r.get(cache_key)
        if cached:
            return json.loads(cached)

        # 缓存服务自动加载（这里模拟）
        user = self.db.query(User).get(user_id)
        if user:
            data = json.dumps(user.to_dict())
            self.r.setex(cache_key, 3600, data)
            return user

        return None

    def get_or_load(self, cache_key, load_func, ttl=3600):
        """
        通用的 Read-Through 实现
        load_func: 从数据源加载数据的函数
        """
        cached = self.r.get(cache_key)
        if cached:
            return json.loads(cached)

        data = load_func()
        if data:
            self.r.setex(cache_key, ttl, json.dumps(data))

        return data
```

### Write-Through（写穿透）

"""
Write-Through 模式
写：同时写缓存和数据库

class WriteThrough:

### Write-Behind（写回）

```python
"""
Write-Behind 模式
写：先写缓存，异步写数据库
"""

class WriteBehind:
    def __init__(self, redis_client, db):
        self.r = redis_client
        self.db = db
        self.pending_writes = Queue.Queue()
        self._start_writer_thread()

    def _start_writer_thread(self):
        """启动异步写线程"""
        def writer():
            while True:
                try:
                    # 从队列获取待写入数据
                    task = self.pending_writes.get(timeout=1)
                    self._write_to_db(task)
                except Queue.Empty:
                    continue
                except Exception as e:
                    print(f'Write error: {e}')

        thread = threading.Thread(target=writer, daemon=True)
        thread.start()

    def _write_to_db(self, task):
        """异步写入数据库"""
        table, record_id, data = task
        # 更新数据库
        self.db.query(table).filter_by(id=record_id).update(data)
        self.db.commit()

    def update_user(self, user_id, data):
        """写操作 - 先写缓存"""
        cache_key = f'user:{user_id}'

        # 1. 先更新缓存
        current = self.r.get(cache_key)
        if current:
            user_data = json.loads(current)
            user_data.update(data)
        else:
            user_data = data

        self.r.setex(cache_key, 3600, json.dumps(user_data))

        # 2. 放入异步队列，等待写入数据库
        self.pending_writes.put(('users', user_id, data))
```

## 3. 缓存一致性

### 缓存一致性问题

```markdown
问题场景：

线程A 更新用户：name='Alice' -> 'Bob'
线程B 读取用户：获取到缓存中的 'Alice'
               但数据库已经是 'Bob'

原因：
  线程A：写数据库 → 删除缓存
  线程B：读缓存（命中）→ 返回旧数据
```

### 解决方案

### 延迟双删

```python
def update_user(self, user_id, data):
    """延迟双删：先删除缓存，再更新数据库，再删除缓存"""
    cache_key = f'user:{user_id}'

    # 1. 第一次删除缓存
    self.r.delete(cache_key)

    # 2. 更新数据库
    user = self.db.query(User).get(user_id)
    for key, value in data.items():
        setattr(user, key, value)
    self.db.commit()

    # 3. 延迟一段时间后再删除缓存
    # 等待可能的读请求完成，避免读到脏数据
    import time
    time.sleep(0.5)  # 500ms

    self.r.delete(cache_key)
```

### 订阅 Binlog

```python
"""
Canal + Redis 实现缓存一致性
通过订阅数据库 Binlog，监听数据变更
变更时自动更新/删除缓存
"""

class CanalRedisSync:
    def __init__(self, canal_client, redis_client):
        self.canal = canal_client
        self.r = redis_client
        self._start_sync()

    def _start_sync(self):
        """启动同步"""
        def sync():
            for entry in self.canal.stream():
                if entry.type == 'ROWDATA':
                    self._process_entry(entry)

        thread = threading.Thread(target=sync, daemon=True)
        thread.start()

    def _process_entry(self, entry):
        """处理数据变更"""
        table = entry.table
        action = entry.action  # INSERT, UPDATE, DELETE
        data = entry.row_data

        if table == 'users':
            if action == 'DELETE':
                self.r.delete(f'user:{data["id"]}')
            else:
                user_id = data['id']
                self.r.hset(f'user:{user_id}', mapping=data)
                self.r.expire(f'user:{user_id}', 3600)
```

## 4. 缓存分层设计

### 热点缓存

```python
class HotCache:
    """热点数据缓存"""

    def __init__(self, redis_client, local_cache_size=1000):
        self.r = redis_client
        self.local = {}  # 本地缓存
        self.local_size = local_cache_size
        self.local_hits = 0
        self.local_misses = 0

    def get(self, key):
        """获取数据（本地 + Redis）"""
        # 1. 查本地缓存
        if key in self.local:
            self.local_hits += 1
            return self.local[key]

        self.local_misses += 1

        # 2. 查 Redis
        cached = self.r.get(key)
        if cached:
            data = json.loads(cached)
            # 回填本地缓存
            self._add_to_local(key, data)
            return data

        return None

    def set(self, key, value, ttl=3600):
        """设置数据"""
        # 同时写本地和 Redis
        self._add_to_local(key, value)
        self.r.setex(key, ttl, json.dumps(value))

    def _add_to_local(self, key, value):
        """添加到本地缓存"""
        # 简单 LRU
        if len(self.local) >= self.local_size:
            # 删除最老的
            oldest_key = next(iter(self.local))
            del self.local[oldest_key]

        self.local[key] = value

    def get_hit_rate(self):
        """本地缓存命中率"""
        total = self.local_hits + self.local_misses
        if total == 0:
            return 0
        return self.local_hits / total
```

### 多租户缓存隔离

```python
class TenantCache:
    """多租户缓存隔离"""

    def __init__(self, redis_client):
        self.r = redis_client

    def _tenant_key(self, tenant_id, key):
        """生成租户隔离的 key"""
        return f'tenant:{tenant_id}:{key}'

    def get(self, tenant_id, key):
        """获取租户数据"""
        return self.r.get(self._tenant_key(tenant_id, key))

    def set(self, tenant_id, key, value, ttl=3600):
        """设置租户数据"""
        self.r.setex(
            self._tenant_key(tenant_id, key),
            ttl,
            value
        )

    def invalidate_tenant(self, tenant_id):
        """清除租户所有缓存"""
        pattern = f'tenant:{tenant_id}:*'
        keys = self.r.keys(pattern)
        if keys:
            self.r.delete(*keys)
```

## 5. 缓存监控

### 监控指标

```python
import time

class CacheMonitor:
    """缓存监控"""

    def __init__(self, redis_client):
        self.r = redis_client
        self.request_count = 0
        self.hit_count = 0
        self.miss_count = 0
        self.error_count = 0
        self._start_time = time.time()

    def record_request(self, hit=True):
        """记录请求"""
        self.request_count += 1
        if hit:
            self.hit_count += 1
        else:
            self.miss_count += 1

    def record_error(self):
        """记录错误"""
        self.error_count += 1

    def get_stats(self):
        """获取统计信息"""
        uptime = time.time() - self._start_time
        hit_rate = self.hit_count / self.request_count if self.request_count > 0 else 0

        info = self.r.info('memory')
        redis_memory = info.get('used_memory_human', '0')

        return {
            'uptime_seconds': int(uptime),
            'total_requests': self.request_count,
            'hits': self.hit_count,
            'misses': self.miss_count,
            'hit_rate': f'{hit_rate:.2%}',
            'errors': self.error_count,
            'redis_memory': redis_memory,
        }

    def get_slow_keys(self, count=10):
        """获取慢查询 key（需要开启 slowlog）"""
        # 需要 Redis 配置：slowlog-log-slower-than 1000
        slow_logs = self.r.slowlog_get(count)
        return [
            {'command': log['command'], 'duration': log['duration']}
            for log in slow_logs
        ]


# 使用装饰器自动监控
def monitored(cache_manager):
    """缓存监控装饰器"""
    def wrapper(key, *args, **kwargs):
        try:
            result = cache_manager.get(key)
            if result:
                cache_manager.monitor.record_request(hit=True)
            else:
                cache_manager.monitor.record_request(hit=False)
            return result
        except Exception as e:
            cache_manager.monitor.record_error()
            raise e

    return wrapper
```

### 告警规则

```python
ALERT_RULES = {
    'hit_rate_low': {
        'threshold': 0.8,  # 命中率低于 80%
        'severity': 'warning',
    },
    'memory_high': {
        'threshold': 0.9,  # 内存使用率高于 90%
        'severity': 'critical',
    },
    'error_rate_high': {
        'threshold': 0.01,  # 错误率高于 1%
        'severity': 'critical',
    },
    'latency_high': {
        'threshold': 100,  # 延迟高于 100ms
        'severity': 'warning',
    },
}

def check_alerts(monitor):
    """检查告警"""
    stats = monitor.get_stats()
    alerts = []

    # 检查命中率
    hit_rate = stats['hits'] / stats['total_requests'] if stats['total_requests'] > 0 else 0
    if hit_rate < ALERT_RULES['hit_rate_low']['threshold']:
        alerts.append({
            'type': 'hit_rate_low',
            'severity': ALERT_RULES['hit_rate_low']['severity'],
            'message': f"命中率 {hit_rate:.2%} 低于阈值"
        })

    # 检查错误率
    error_rate = stats['errors'] / stats['total_requests'] if stats['total_requests'] > 0 else 0
    if error_rate > ALERT_RULES['error_rate_high']['threshold']:
        alerts.append({
            'type': 'error_rate_high',
            'severity': ALERT_RULES['error_rate_high']['severity'],
            'message': f"错误率 {error_rate:.2%} 高于阈值"
        })

    return alerts
```

## 6. 完整缓存方案

```python
class CacheManager:
    """
    完整的缓存管理方案
    包含：Cache-Aside + 多级缓存 + 监控 + 故障处理
    """

    def __init__(self, redis_client, db):
        self.r = redis_client
        self.db = db
        self.monitor = CacheMonitor(redis_client)
        self.local = {}  # L1 本地缓存
        self.local_size = 1000

    def _local_get(self, key):
        """L1 本地缓存获取"""
        return self.local.get(key)

    def _local_set(self, key, value):
        """L1 本地缓存设置"""
        if len(self.local) >= self.local_size:
            # LRU 删除
            oldest_key = next(iter(self.local))
            del self.local[oldest_key]
        self.local[key] = value

    def get(self, key):
        """获取数据"""
        # L1 本地缓存
        value = self._local_get(key)
        if value:
            self.monitor.record_request(hit=True)
            return value

        # L2 Redis 缓存
        try:
            cached = self.r.get(key)
            if cached:
                data = json.loads(cached)
                self._local_set(key, data)
                self.monitor.record_request(hit=True)
                return data
        except Exception as e:
            self.monitor.record_error()

        self.monitor.record_request(hit=False)
        return None

    def set(self, key, value, ttl=3600):
        """设置数据"""
        # 同时写 L1 和 L2
        self._local_set(key, value)

        try:
            self.r.setex(key, ttl, json.dumps(value))
        except Exception as e:
            self.monitor.record_error()

    def delete(self, key):
        """删除数据"""
        # 删除 L1
        if key in self.local:
            del self.local[key]

        # 删除 L2
        try:
            self.r.delete(key)
        except Exception as e:
            self.monitor.record_error()

    def get_or_load(self, key, load_func, ttl=3600):
        """获取或加载数据"""
        # 尝试从缓存获取
        value = self.get(key)
        if value:
            return value

        # 缓存未命中，加载数据
        data = load_func()
        if data:
            self.set(key, data, ttl)

        return data

    def get_stats(self):
        """获取统计信息"""
        return self.monitor.get_stats()
```

## 今日总结

- [ ] 多级缓存：CDN → Nginx → 本地 → Redis → DB
- [ ] Cache-Aside：读先缓存 miss 后加载，写先数据库再删缓存
