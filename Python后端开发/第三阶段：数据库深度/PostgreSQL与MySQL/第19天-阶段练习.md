## 2. 创建触发器

```sql
-- 1. 自动更新 updated_at
CREATE OR REPLACE FUNCTION update_updated_at()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = CURRENT_TIMESTAMP;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER update_users_timestamp
    BEFORE UPDATE ON users
    FOR EACH ROW EXECUTE FUNCTION update_updated_at();

CREATE TRIGGER update_products_timestamp
    BEFORE UPDATE ON products
    FOR EACH ROW EXECUTE FUNCTION update_updated_at();

CREATE TRIGGER update_orders_timestamp
    BEFORE UPDATE ON orders
    FOR EACH ROW EXECUTE FUNCTION update_updated_at();

-- 2. 订单完成后更新销量和库存
CREATE OR REPLACE FUNCTION after_order_complete()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.status = 'completed' AND OLD.status != 'completed' THEN
        -- 更新商品销量
        UPDATE products p
        SET sold_count = sold_count + oi.quantity
        FROM order_items oi
        WHERE oi.order_id = NEW.id AND p.id = oi.product_id;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trigger_order_complete
    AFTER UPDATE ON orders
    FOR EACH ROW
    EXECUTE FUNCTION after_order_complete();

-- 3. 订单取消恢复库存
CREATE OR REPLACE FUNCTION restore_stock_on_cancel()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.status = 'cancelled' AND OLD.status != 'cancelled' THEN
        UPDATE products p
        SET stock = stock + oi.quantity
        FROM order_items oi
        WHERE oi.order_id = NEW.id AND p.id = oi.product_id;

        -- 恢复用户积分
        UPDATE users
        SET points = points - (
            SELECT COALESCE(SUM(points_change), 0)
            FROM points_logs
            WHERE order_id = NEW.id AND change_type = 'order_reward'
        )
        WHERE id = NEW.user_id;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trigger_order_cancel
    AFTER UPDATE ON orders
    FOR EACH ROW
    EXECUTE FUNCTION restore_stock_on_cancel();
```

## 3. 创建存储过程

```sql
-- 1. 创建订单（包含库存检查和积分计算）
CREATE OR REPLACE PROCEDURE create_order(
    p_user_id INTEGER,
    p_items JSONB,
    p_shipping_address TEXT,
    p_receiver_name VARCHAR(50),
    p_receiver_phone VARCHAR(20)
)
LANGUAGE plpgsql AS $$
DECLARE
    v_order_id INTEGER;
    v_order_no VARCHAR(50);
    v_total DECIMAL(10,2) := 0;
    v_discount DECIMAL(10,2) := 0;
    v_points INTEGER := 0;
    v_points_before INTEGER;
    v_stock INTEGER;
    v_item JSONB;
BEGIN
    -- 生成订单号
    v_order_no := 'ORD' || TO_CHAR(NOW(), 'YYYYMMDDHH24MISS') || FLOOR(RANDOM() * 10000)::TEXT;

    -- 检查库存并计算总价
    FOR v_item IN SELECT * FROM jsonb_array_elements(p_items)
    LOOP
        SELECT stock INTO v_stock FROM products WHERE id = (v_item->>'product_id')::INTEGER;
        IF v_stock < (v_item->>'quantity')::INTEGER THEN
            RAISE EXCEPTION 'Product % stock insufficient', v_item->>'product_id';
        END IF;
        v_total := v_total + ((v_item->>'price')::DECIMAL * (v_item->>'quantity')::INTEGER);
    END LOOP;

    -- 计算积分抵扣（100积分抵扣1元）
    SELECT points INTO v_points_before FROM users WHERE id = p_user_id;
    v_points := LEAST(v_points_before, FLOOR(v_total)::INTEGER);
    v_discount := v_points / 100.0;

    -- 创建订单
    INSERT INTO orders (
        order_no, user_id, total_amount, discount_amount,
        payment_amount, status, shipping_address,
        receiver_name, receiver_phone
    ) VALUES (
        v_order_no, p_user_id, v_total, v_discount,
        v_total - v_discount, 'pending', p_shipping_address,
        p_receiver_name, p_receiver_phone
    ) RETURNING id INTO v_order_id;

    -- 插入订单明细并扣减库存
    FOR v_item IN SELECT * FROM jsonb_array_elements(p_items)
    LOOP
        INSERT INTO order_items (order_id, product_id, product_name, price, quantity, subtotal)
        SELECT
            v_order_id,
            (v_item->>'product_id')::INTEGER,
            p.name,
            (v_item->>'price')::DECIMAL,
            (v_item->>'quantity')::INTEGER,
            (v_item->>'price')::DECIMAL * (v_item->>'quantity')::INTEGER
        FROM products p WHERE id = (v_item->>'product_id')::INTEGER;

        -- 扣减库存
        UPDATE products
        SET stock = stock - (v_item->>'quantity')::INTEGER
        WHERE id = (v_item->>'product_id')::INTEGER;
    END LOOP;

    -- 扣除积分
    IF v_points > 0 THEN
        UPDATE users SET points = points - v_points WHERE id = p_user_id;
        INSERT INTO points_logs (user_id, order_id, change_type, points_change, points_before, points_after)
        VALUES (p_user_id, v_order_id, 'point_redeem', -v_points, v_points_before, v_points_before - v_points);
    END IF;

    RAISE NOTICE 'Order % created successfully', v_order_no;
END;
$$ LANGUAGE plpgsql;

-- 2. 完成订单（计算积分奖励）
CREATE OR REPLACE PROCEDURE complete_order(p_order_id INTEGER)
LANGUAGE plpgsql AS $$
DECLARE
    v_user_id INTEGER;
    v_payment_amount DECIMAL(10,2);
    v_points_earned INTEGER;
    v_points_before INTEGER;
BEGIN
    SELECT user_id, payment_amount INTO v_user_id, v_payment_amount
    FROM orders WHERE id = p_order_id;

    IF NOT FOUND THEN
        RAISE EXCEPTION 'Order not found';
    END IF;

    -- 计算积分奖励（每消费1元奖励1积分）
    v_points_earned := FLOOR(v_payment_amount);
    SELECT points INTO v_points_before FROM users WHERE id = v_user_id;

    -- 更新订单状态
    UPDATE orders
    SET status = 'completed',
        completed_time = CURRENT_TIMESTAMP,
        updated_at = CURRENT_TIMESTAMP
    WHERE id = p_order_id;

    -- 奖励积分
    UPDATE users SET points = points + v_points_earned WHERE id = v_user_id;
    INSERT INTO points_logs (user_id, order_id, change_type, points_change, points_before, points_after)
    VALUES (v_user_id, p_order_id, 'order_reward', v_points_earned, v_points_before, v_points_before + v_points_earned);

    -- 更新用户等级
    UPDATE users SET level =
        CASE
            WHEN points >= 10000 THEN 'Platinum'
            WHEN points >= 5000 THEN 'Gold'
            WHEN points >= 1000 THEN 'Silver'
            ELSE 'Bronze'
        END
    WHERE id = v_user_id;

    RAISE NOTICE 'Order % completed, % points earned', p_order_id, v_points_earned;
END;
$$ LANGUAGE plpgsql;
```

## 4. 复杂查询练习

### 销售统计

```sql
-- 1. 每日销售报表
SELECT
    DATE(o.created_at) AS order_date,
    COUNT(DISTINCT o.user_id) AS customer_count,
    COUNT(o.id) AS order_count,
    SUM(o.total_amount) AS total_sales,
    SUM(o.discount_amount) AS total_discount,
    SUM(o.payment_amount) AS net_sales,
    AVG(o.payment_amount) AS avg_order_value,
    SUM(oi.quantity) AS items_sold
FROM orders o
JOIN order_items oi ON o.id = oi.order_id
WHERE o.status IN ('completed', 'paid', 'shipped')
AND o.created_at >= DATE_TRUNC('month', CURRENT_DATE)
GROUP BY DATE(o.created_at)
ORDER BY order_date;

-- 2. 商品销量排行榜
SELECT
    p.id,
    p.name,
    c.name AS category,
    SUM(oi.quantity) AS total_sold,
    SUM(oi.subtotal) AS total_revenue,
    ROUND(AVG(r.rating), 2) AS avg_rating
FROM products p
JOIN categories c ON p.category_id = c.id
JOIN order_items oi ON p.id = oi.product_id
JOIN orders o ON oi.order_id = o.id
LEFT JOIN reviews r ON p.id = r.product_id
WHERE o.status = 'completed'
AND o.created_at >= CURRENT_DATE - INTERVAL '30 days'
GROUP BY p.id, p.name, c.name
ORDER BY total_sold DESC
LIMIT 20;

-- 3. 用户消费分析
SELECT
    u.username,
    u.level,
    COUNT(o.id) AS order_count,
    SUM(o.payment_amount) AS total_spent,
    AVG(o.payment_amount) AS avg_order_value,
    MAX(o.created_at) AS last_order_date,
    u.points AS current_points
FROM users u
JOIN orders o ON u.id = o.user_id
WHERE o.status = 'completed'
GROUP BY u.id, u.username, u.level, u.points
ORDER BY total_spent DESC
LIMIT 50;
```

### 库存分析

```sql
-- 1. 库存预警（低于10件）
SELECT
    p.name,
    c.name AS category,
    p.stock,
    p.sold_count,
    CASE
        WHEN p.stock <= 5 THEN '紧急补货'
        WHEN p.stock <= 10 THEN '尽快补货'
        ELSE '正常'
    END AS alert_level
FROM products p
JOIN categories c ON p.category_id = c.id
WHERE p.is_active = TRUE
AND p.stock <= 10
ORDER BY p.stock;

-- 2. 商品周转率分析
SELECT
    p.name,
    c.name AS category,
    p.stock,
    p.sold_count,
    CASE
        WHEN p.sold_count > 0 THEN
            ROUND(p.stock::DECIMAL / (p.sold_count::DECIMAL / 30), 1)
        ELSE NULL
    END AS days_of_stock
FROM products p
JOIN categories c ON p.category_id = c.id
WHERE p.is_active = TRUE
ORDER BY days_of_stock DESC NULLS LAST;
```

### 用户行为分析

```sql
-- 1. 用户购买频次分布
SELECT
    purchase_range,
    COUNT(*) AS user_count
FROM (
    SELECT
        u.id,
        CASE
            WHEN COUNT(o.id) >= 10 THEN '10次以上'
            WHEN COUNT(o.id) >= 5 THEN '5-9次'
            WHEN COUNT(o.id) >= 3 THEN '3-4次'
            WHEN COUNT(o.id) >= 2 THEN '2次'
            ELSE '1次'
        END AS purchase_range
    FROM users u
    LEFT JOIN orders o ON u.id = o.user_id AND o.status = 'completed'
    GROUP BY u.id
) t
GROUP BY purchase_range
ORDER BY MIN(COUNT) DESC;

-- 2. 复购周期分析
WITH user_order_dates AS (
    SELECT
        user_id,
        created_at,
        LAG(created_at) OVER (PARTITION BY user_id ORDER BY created_at) AS prev_order
    FROM orders
    WHERE status = 'completed'
)
SELECT
    CASE
        WHEN AVG(EXTRACT(DAY FROM created_at - prev_order)) <= 7 THEN '周活'
        WHEN AVG(EXTRACT(DAY FROM created_at - prev_order)) <= 30 THEN '月活'
        ELSE '流失'
    END AS user_segment,
    COUNT(DISTINCT user_id) AS user_count
FROM user_order_dates
WHERE prev_order IS NOT NULL
GROUP BY user_segment;
```

## 5. 性能优化练习

```sql
-- 1. 创建物化视图加速统计查询
CREATE MATERIALIZED VIEW monthly_sales_summary AS
SELECT
    DATE_TRUNC('month', o.created_at) AS month,
    c.id AS category_id,
    c.name AS category_name,
    COUNT(DISTINCT o.user_id) AS customer_count,
    COUNT(o.id) AS order_count,
    SUM(oi.quantity) AS items_sold,
    SUM(o.payment_amount) AS total_revenue
FROM orders o
JOIN order_items oi ON o.id = oi.order_id
JOIN products p ON oi.product_id = p.id
JOIN categories c ON p.category_id = c.id
WHERE o.status = 'completed'
GROUP BY DATE_TRUNC('month', o.created_at), c.id, c.name;

CREATE UNIQUE INDEX ON monthly_sales_summary(month, category_id);

-- 刷新物化视图
REFRESH MATERIALIZED VIEW monthly_sales_summary;

-- 2. 创建汇总表
CREATE TABLE daily_product_stats AS
SELECT
    DATE(oi.created_at) AS stat_date,
    p.id AS product_id,
    p.name AS product_name,
    COUNT(DISTINCT o.id) AS order_count,
    SUM(oi.quantity) AS quantity_sold,
    SUM(oi.subtotal) AS revenue
FROM order_items oi
JOIN orders o ON oi.order_id = o.id
JOIN products p ON oi.product_id = p.id
WHERE o.status = 'completed'
GROUP BY DATE(oi.created_at), p.id, p.name;
```
