## 1. PostgreSQL 函数

### 基本函数

```sql
-- 基础函数
CREATE OR REPLACE FUNCTION get_user_count()
RETURNS INTEGER AS $$
DECLARE
    count INTEGER;
BEGIN
    SELECT COUNT(*) INTO count FROM users;
    RETURN count;
END;
$$ LANGUAGE plpgsql;

-- 调用函数
SELECT get_user_count();
-- 或
SELECT * FROM get_user_count();
```

### 参数

```sql
-- 带参数函数
CREATE OR REPLACE FUNCTION get_user_by_id(uid INTEGER)
RETURNS TABLE (
    id INTEGER,
    username VARCHAR(100),
    email VARCHAR(100)
) AS $$
BEGIN
    RETURN QUERY
    SELECT u.id, u.username, u.email
    FROM users u
    WHERE u.id = uid;
END;
$$ LANGUAGE plpgsql;

-- 调用
SELECT * FROM get_user_by_id(1);

-- 参数默认值
CREATE OR REPLACE FUNCTION create_user(
    user_name VARCHAR(100),
    user_email VARCHAR(100) DEFAULT 'unknown@example.com'
)
RETURNS INTEGER AS $$
DECLARE
    new_id INTEGER;
BEGIN
    INSERT INTO users (username, email)
    VALUES (user_name, user_email)
    RETURNING id INTO new_id;
    RETURN new_id;
END;
$$ LANGUAGE plpgsql;

-- OUT 参数
CREATE OR REPLACE FUNCTION get_user_stats(
    uid INTEGER,
    OUT order_count INTEGER,
    OUT total_spent NUMERIC
) AS $$
BEGIN
    SELECT COUNT(*), COALESCE(SUM(total), 0)
    INTO order_count, total_spent
    FROM orders
    WHERE user_id = uid;
END;
$$ LANGUAGE plpgsql;

SELECT * FROM get_user_stats(1);
```

### 条件与循环

```sql
-- IF 条件
CREATE OR REPLACE FUNCTION get_user_level(uid INTEGER)
RETURNS VARCHAR(20) AS $$
DECLARE
    total_spent NUMERIC;
BEGIN
    SELECT COALESCE(SUM(total), 0) INTO total_spent
    FROM orders
    WHERE user_id = uid;

    IF total_spent > 10000 THEN
        RETURN 'VIP';
    ELSIF total_spent > 5000 THEN
        RETURN 'Premium';
    ELSIF total_spent > 0 THEN
        RETURN 'Regular';
    ELSE
        RETURN 'New';
    END IF;
END;
$$ LANGUAGE plpgsql;

-- CASE 语句
CREATE OR REPLACE FUNCTION get_discount_rate(category VARCHAR(50))
RETURNS NUMERIC AS $$
DECLARE
    rate NUMERIC;
BEGIN
    CASE category
        WHEN 'electronics' THEN rate := 0.1;
        WHEN 'books' THEN rate := 0.15;
        WHEN 'food' THEN rate := 0.05;
        ELSE rate := 0;
    END CASE;
    RETURN rate;
END;
$$ LANGUAGE plpgsql;
```

### 循环

```sql
-- FOR 循环
CREATE OR REPLACE FUNCTION generate_series_list(n INTEGER)
RETURNS TABLE (n INTEGER) AS $$
BEGIN
    RETURN QUERY
    SELECT i
    FROM generate_series(1, n) AS t(i);
END;
$$ LANGUAGE plpgsql;

-- WHILE 循环
CREATE OR REPLACE FUNCTION calculate_factorial(n INTEGER)
RETURNS BIGINT AS $$
DECLARE
    result BIGINT := 1;
    i INTEGER := 1;
BEGIN
    WHILE i <= n LOOP
        result := result * i;
        i := i + 1;
    END LOOP;
    RETURN result;
END;
$$ LANGUAGE plpgsql;

-- 游标循环
CREATE OR REPLACE FUNCTION process_users()
RETURNS void AS $$
DECLARE
    user_record RECORD;
BEGIN
    FOR user_record IN SELECT * FROM users WHERE is_active = FALSE LOOP
        RAISE NOTICE 'Processing inactive user: %', user_record.username;
        -- 执行处理逻辑
    END LOOP;
END;
$$ LANGUAGE plpgsql;
```

## 2. PostgreSQL 存储过程

```sql
-- PostgreSQL 存储过程（返回 void）
CREATE OR REPLACE PROCEDURE update_user_status(uid INTEGER, new_status BOOLEAN)
LANGUAGE plpgsql AS $$
BEGIN
    UPDATE users SET is_active = new_status WHERE id = uid;
END;
$$;

-- 调用存储过程
CALL update_user_status(1, TRUE);

-- PostgreSQL 14+ 支持 INOUT 参数
CREATE OR REPLACE PROCEDURE increment_counter(INOUT counter INTEGER)
LANGUAGE plpgsql AS $$
BEGIN
    counter := counter + 1;
END;
$$;

CALL increment_counter(5);  -- 返回 6
```

## 3. MySQL 存储过程

### 基本语法

```sql
-- 创建存储过程
DELIMITER //
CREATE PROCEDURE get_user_by_id(IN uid INT)
BEGIN
    SELECT id, username, email
    FROM users
    WHERE id = uid;
END //
DELIMITER ;

-- 调用
CALL get_user_by_id(1);
```

### 参数类型

```sql
DELIMITER //
CREATE PROCEDURE get_user_stats(
    IN uid INT,
    OUT order_count INT,
    OUT total_spent DECIMAL(10,2)
)
BEGIN
    SELECT
        COUNT(*),
        COALESCE(SUM(total), 0)
    INTO order_count, total_spent
    FROM orders
    WHERE user_id = uid;
END //
DELIMITER ;

-- 调用并获取输出参数
CALL get_user_stats(1, @count, @total);
SELECT @count AS order_count, @total AS total_spent;
```

### 条件和循环

```sql
DELIMITER //
CREATE PROCEDURE get_user_level(
    IN uid INT,
    OUT level_name VARCHAR(20)
)
BEGIN
    DECLARE spent DECIMAL(10,2);

    SELECT COALESCE(SUM(total), 0) INTO spent
    FROM orders
    WHERE user_id = uid;

    IF spent > 10000 THEN
        SET level_name = 'VIP';
    ELSEIF spent > 5000 THEN
        SET level_name = 'Premium';
    ELSEIF spent > 0 THEN
        SET level_name = 'Regular';
    ELSE
        SET level_name = 'New';
    END IF;
END //
DELIMITER ;

-- 循环
DELIMITER //
CREATE PROCEDURE insert_batch(IN count INT)
BEGIN
    DECLARE i INT DEFAULT 1;
    WHILE i <= count DO
        INSERT INTO products (name, price) VALUES (CONCAT('Product', i), i * 10);
        SET i = i + 1;
    END WHILE;
END //
DELIMITER ;

CALL insert_batch(10);
```

## 4. Python 调用

### PostgreSQL

```python
import psycopg2
from psycopg2 import extensions

# 调用函数
conn = psycopg2.connect("dbname=test user=postgres")
cur = conn.cursor()

# 调用函数获取单个值
cur.execute("SELECT get_user_count()")
count = cur.fetchone()[0]
print(f"User count: {count}")

# 调用返回表的函数
cur.execute("SELECT * FROM get_user_by_id(1)")
for row in cur.fetchall():
    print(row)

# 调用存储过程
cur.execute("CALL update_user_status(1, TRUE)")
conn.commit()

# 获取 OUT 参数（PostgreSQL 14+）
cur.execute("""
    CREATE TABLE IF NOT EXISTS counter (value INTEGER);
    INSERT INTO counter VALUES (5);
""")
cur.execute("""
    CALL increment_counter((SELECT value FROM counter));
""")
conn.commit()

cur.close()
conn.close()
```

### MySQL

```python
import pymysql

conn = pymysql.connect(
    host='localhost',
    user='root',
    password='password',
    database='test'
)
cur = conn.cursor()

# 调用存储过程
cur.callproc('get_user_by_id', (1,))

    print(row)

# 调用存储过程获取 OUT 参数
cur.callproc('get_user_stats', (1, 0, 0))
```

## 5. 错误处理

### PostgreSQL

```sql
CREATE OR REPLACE FUNCTION safe_divide(a NUMERIC, b NUMERIC)
RETURNS NUMERIC AS $$
BEGIN
    IF b = 0 THEN
        RAISE WARNING 'Division by zero detected';
        RETURN NULL;
    END IF;
    RETURN a / b;
EXCEPTION
    WHEN OTHERS THEN
        RAISE NOTICE 'Error: %', SQLERRM;
        RETURN NULL;
END;
$$ LANGUAGE plpgsql;
```

```python
# Python 捕获异常
try:
    cur.execute("SELECT safe_divide(10, 0)")
    result = cur.fetchone()[0]
    print(result)
except psycopg2.Error as e:
    print(f"Database error: {e}")
```

### MySQL

```sql
DELIMITER //
CREATE PROCEDURE safe_divide(IN a DECIMAL(10,2), IN b DECIMAL(10,2), OUT result DECIMAL(10,2))
BEGIN
    DECLARE CONTINUE HANDLER FOR SQLEXCEPTION
    BEGIN
        SET result = NULL;
    END;

    IF b = 0 THEN
        SET result = NULL;
    ELSE
        SET result = a / b;
    END IF;
END //
DELIMITER ;
```

## 6. 实战练习

### 练习 1：业务逻辑封装

```sql
-- PostgreSQL：订单完成存储过程
CREATE OR REPLACE PROCEDURE complete_order(order_id INTEGER)
LANGUAGE plpgsql AS $$
DECLARE
    v_status VARCHAR(20);
    v_customer_id INTEGER;
    v_total NUMERIC;
BEGIN
    -- 检查订单状态
    SELECT status, customer_id, total
    INTO v_status, v_customer_id, v_total
    FROM orders WHERE id = order_id;

    IF NOT FOUND THEN
        RAISE EXCEPTION 'Order % not found', order_id;
    END IF;

    IF v_status != 'pending' THEN
        RAISE EXCEPTION 'Order % is not pending', order_id;
    END IF;

    -- 更新订单状态
    UPDATE orders SET status = 'completed', updated_at = NOW()
    WHERE id = order_id;

    -- 更新用户累计消费
    UPDATE customers
    SET total_spent = total_spent + v_total,
        order_count = order_count + 1
    WHERE id = v_customer_id;

    RAISE NOTICE 'Order % completed successfully', order_id;
END;
$$ LANGUAGE plpgsql;
```

### 练习 2：复杂计算函数

```sql
-- PostgreSQL：计算用户信用评分
CREATE OR REPLACE FUNCTION calculate_credit_score(uid INTEGER)
RETURNS INTEGER AS $$
DECLARE
    v_order_count INTEGER;
    v_total_spent NUMERIC;
    v_avg_rating DECIMAL(3,2);
    v_account_age INTEGER;
    v_score INTEGER := 0;
BEGIN
    -- 订单数量得分
    SELECT COUNT(*) INTO v_order_count FROM orders WHERE user_id = uid;
    v_score := v_score + LEAST(v_order_count * 5, 25);

    -- 消费金额得分
    SELECT COALESCE(SUM(total), 0) INTO v_total_spent FROM orders WHERE user_id = uid;
    v_score := v_score + LEAST(v_total_spent / 100, 30);

    -- 平均评分得分
    SELECT COALESCE(AVG(rating), 0) INTO v_avg_rating
    FROM reviews WHERE user_id = uid;
    v_score := v_score + v_avg_rating * 10;

    -- 账户年龄得分
    SELECT EXTRACT(YEAR FROM AGE(NOW(), created_at)) INTO v_account_age
    FROM users WHERE id = uid;
    v_score := v_score + LEAST(v_account_age * 5, 15);

    RETURN LEAST(v_score, 100);
END;
$$ LANGUAGE plpgsql;
```

## 今日总结

- [ ] PostgreSQL 函数用 `CREATE FUNCTION`，存储过程用 `CREATE PROCEDURE`
- [ ] MySQL 存储过程用 `DELIMITER` 定义
- [ ] `IN`/`OUT`/`INOUT` 参数控制参数方向
- [ ] 函数可返回值，存储过程通过 OUT 参数返回
- [ ] Python 用 `cursor.callproc()` 或 `SELECT` 调用
- [ ] 异常处理：`EXCEPTION WHEN ...`（PostgreSQL）或 `DECLARE HANDLER`（MySQL）
