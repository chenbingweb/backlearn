## 1. 数据库操作

### 创建数据库

```sql
-- PostgreSQL
CREATE DATABASE shop;                          -- 基本创建
CREATE DATABASE shop WITH OWNER postgres;      -- 指定所有者
CREATE DATABASE shop ENCODING 'UTF8';         -- 指定编码
CREATE DATABASE shop TEMPLATE template0;       -- 基于模板创建

-- MySQL
CREATE DATABASE shop;                          -- 基本创建
CREATE DATABASE shop CHARACTER SET utf8mb4;    -- 指定字符集
CREATE DATABASE shop COLLATE utf8mb4_unicode_ci;  -- 指定排序规则
```

### 查看和切换数据库

```sql
-- PostgreSQL
\l                                    -- 列出所有数据库
\c shop                               -- 切换到 shop 数据库
\dt                                   -- 列出当前库所有表

-- MySQL
SHOW DATABASES;                       -- 列出所有数据库
USE shop;                             -- 切换到 shop 数据库
SHOW TABLES;                          -- 列出当前库所有表
```

### 删除数据库

```sql
-- 删除数据库（谨慎操作！）
DROP DATABASE shop;                  -- 删除数据库

-- PostgreSQL：删除前检查
SELECT datname FROM pg_database WHERE datname = 'shop';
DROP DATABASE IF EXISTS shop;        -- 安全删除

-- MySQL：删除前检查
DROP DATABASE IF EXISTS shop;
```

### 修改数据库

```sql
-- PostgreSQL
ALTER DATABASE shop RENAME TO new_shop;           -- 重命名
ALTER DATABASE shop OWNER TO new_owner;          -- 修改所有者

-- MySQL
ALTER DATABASE shop CHARACTER SET utf8mb4;        -- 修改字符集
```

## 2. 表操作基础

### 创建表

```sql
-- PostgreSQL
CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    username VARCHAR(50) UNIQUE NOT NULL,
    email VARCHAR(100) NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    age INTEGER,
    status VARCHAR(20) DEFAULT 'active',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- MySQL
CREATE TABLE users (
    id INT AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(50) UNIQUE NOT NULL,
    email VARCHAR(100) NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    age INT,
    status VARCHAR(20) DEFAULT 'active',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_username (username),
    INDEX idx_email (email)
);
```

### 查看表结构

```sql
-- PostgreSQL
\d users                    -- 查看表结构
\d+ users                   -- 查看表结构和大小
\df                         -- 列出所有函数

-- MySQL
DESC users;                 -- 查看表结构
DESCRIBE users;            -- 同上
SHOW CREATE TABLE users;   -- 查看建表语句
```

### 重命名表

```sql
-- PostgreSQL
ALTER TABLE users RENAME TO app_users;

-- MySQL
RENAME TABLE users TO app_users;
ALTER TABLE users RENAME TO app_users;
```

### 删除表

```sql
-- 删除表（谨慎操作！）
DROP TABLE users;                    -- 删除表及其数据

-- 安全删除
DROP TABLE IF EXISTS users;           -- PostgreSQL
DROP TABLE IF EXISTS users;           -- MySQL

-- 级联删除（删除外键关联的表）
DROP TABLE orders CASCADE;            -- PostgreSQL
```

## 3. 约束 Constraints

### 主键约束 PRIMARY KEY

```sql
-- 主键：唯一标识表中每行，不能为空
CREATE TABLE users (
    id SERIAL PRIMARY KEY,           -- 方式1：主键列自增
    username VARCHAR(50) PRIMARY KEY,  -- 方式2：直接指定
    email VARCHAR(100)
);

-- 复合主键
CREATE TABLE order_items (
    order_id INTEGER,
    product_id INTEGER,
    quantity INTEGER,
    PRIMARY KEY (order_id, product_id)  -- 复合主键
);

-- 添加主键
ALTER TABLE users ADD PRIMARY KEY (id);

-- 删除主键
ALTER TABLE users DROP PRIMARY KEY;
```

### 外键约束 FOREIGN KEY

```sql
-- 外键：建立表间关联，保证引用完整性
CREATE TABLE orders (
    id SERIAL PRIMARY KEY,
    user_id INTEGER NOT NULL,
    total DECIMAL(10, 2),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    -- 定义外键
    FOREIGN KEY (user_id) REFERENCES users(id)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);

-- 外键操作选项
-- ON DELETE: 级联删除、置空、拒绝
-- ON UPDATE: 级联更新、置空、拒绝
-- CASCADE: 父表删除/更新，子表跟着删除/更新
-- SET NULL: 父表删除/更新，子表外键设为 NULL
-- SET DEFAULT: 父表删除/更新，子表外键设为默认值
-- NO ACTION: 拒绝删除/更新（默认）
-- RESTRICT: 拒绝删除/更新（立即检查）
```

### 唯一约束 UNIQUE

```sql
-- 唯一约束：字段值不能重复，但可以为 NULL
CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    username VARCHAR(50) UNIQUE NOT NULL,
    email VARCHAR(100) UNIQUE,           -- 单个字段唯一
    phone VARCHAR(20),
    -- 复合唯一约束
    UNIQUE (phone, user_type)
);

-- 添加唯一约束
ALTER TABLE users ADD CONSTRAINT unique_email UNIQUE (email);

-- 删除唯一约束
ALTER TABLE users DROP CONSTRAINT unique_email;
```

### 非空约束 NOT NULL

```sql
-- NOT NULL：字段值不能为空
CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    username VARCHAR(50) NOT NULL,         -- 不能为空
    email VARCHAR(100) NOT NULL,           -- 不能为空
    age INTEGER,                            -- 可以为空
    bio TEXT                               -- 可以为空
);

-- 添加 NOT NULL 约束
ALTER TABLE users ALTER COLUMN bio SET NOT NULL;

-- 删除 NOT NULL 约束
ALTER TABLE users ALTER COLUMN bio DROP NOT NULL;
```

### 检查约束 CHECK

```sql
-- CHECK：限制字段值满足条件
CREATE TABLE products (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    price DECIMAL(10, 2) NOT NULL CHECK (price > 0),
    stock INTEGER NOT NULL CHECK (stock >= 0),
    discount DECIMAL(5, 2) CHECK (discount >= 0 AND discount <= 1)
);

-- 命名的 CHECK 约束
CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    username VARCHAR(50) NOT NULL,
    age INTEGER,
    email VARCHAR(100),

    CONSTRAINT positive_age CHECK (age > 0),
    CONSTRAINT valid_email CHECK (email LIKE '%@%')
);

-- 添加 CHECK 约束
ALTER TABLE products ADD CONSTRAINT positive_price CHECK (price > 0);

-- 删除 CHECK 约束
ALTER TABLE products DROP CONSTRAINT positive_price;
```

### 默认值约束 DEFAULT

```sql
-- DEFAULT：字段未指定时使用默认值
CREATE TABLE orders (
    id SERIAL PRIMARY KEY,
    user_id INTEGER NOT NULL,
    status VARCHAR(20) DEFAULT 'pending',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 使用表达式作为默认值
CREATE TABLE logs (
    id SERIAL PRIMARY KEY,
    message TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    user_id INTEGER DEFAULT (SELECT MAX(id) FROM users)
);

-- PostgreSQL 特殊默认值
CREATE TABLE events (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

## 4. 修改表结构

### 添加列

```sql
-- PostgreSQL & MySQL
ALTER TABLE users ADD COLUMN phone VARCHAR(20);
ALTER TABLE users ADD COLUMN avatar TEXT DEFAULT 'default.png';
```

### 删除列

```sql
-- PostgreSQL & MySQL
ALTER TABLE users DROP COLUMN phone;

-- 删除前检查是否存在
ALTER TABLE users DROP COLUMN IF EXISTS phone;  -- PostgreSQL
```

### 修改列

```sql
-- PostgreSQL：修改列类型
ALTER TABLE users ALTER COLUMN age TYPE DECIMAL(5, 1);
ALTER TABLE users ALTER COLUMN username TYPE VARCHAR(100);

-- PostgreSQL：修改列默认值
ALTER TABLE users ALTER COLUMN status SET DEFAULT 'inactive';
ALTER TABLE users ALTER COLUMN status DROP DEFAULT;

-- PostgreSQL：重命名列
ALTER TABLE users RENAME COLUMN username TO name;

-- MySQL：修改列（需要完整定义）
ALTER TABLE users MODIFY COLUMN age DECIMAL(5, 1);
ALTER TABLE users MODIFY COLUMN username VARCHAR(100) NOT NULL;

-- MySQL：重命名列
ALTER TABLE users CHANGE COLUMN username name VARCHAR(100);
```

### 添加和删除约束

```sql
-- 添加约束
ALTER TABLE orders ADD CONSTRAINT fk_user
    FOREIGN KEY (user_id) REFERENCES users(id);

ALTER TABLE products ADD CONSTRAINT positive_price
    CHECK (price > 0);

-- 删除约束
ALTER TABLE orders DROP CONSTRAINT fk_user;
ALTER TABLE products DROP CONSTRAINT positive_price;
```

## 5. 实战练习

### 练习 1：创建完整的订单系统表

```sql
-- 用户表
CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    username VARCHAR(50) UNIQUE NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 商品表
CREATE TABLE products (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    category VARCHAR(50),
    price DECIMAL(10, 2) NOT NULL CHECK (price > 0),
    stock INTEGER NOT NULL DEFAULT 0 CHECK (stock >= 0),
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 订单表
CREATE TABLE orders (
    id SERIAL PRIMARY KEY,
    order_no VARCHAR(50) UNIQUE NOT NULL,
    user_id INTEGER NOT NULL,
    total DECIMAL(10, 2) NOT NULL,
    status VARCHAR(20) DEFAULT 'pending',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (user_id) REFERENCES users(id)
        ON DELETE RESTRICT
);

-- 订单明细表
CREATE TABLE order_items (
    id SERIAL PRIMARY KEY,
    order_id INTEGER NOT NULL,
    product_id INTEGER NOT NULL,
    quantity INTEGER NOT NULL CHECK (quantity > 0),
    price DECIMAL(10, 2) NOT NULL CHECK (price > 0),

    FOREIGN KEY (order_id) REFERENCES orders(id)
        ON DELETE CASCADE,
    FOREIGN KEY (product_id) REFERENCES products(id)
        ON DELETE RESTRICT
);
```

### 练习 2：修改表结构

## 今日总结

- [ ] `CREATE DATABASE` 创建数据库，`DROP DATABASE` 删除
- [ ] `CREATE TABLE` 创建表，`DROP TABLE` 删除表
- [ ] `PRIMARY KEY` 主键约束，唯一标识每行记录
- [ ] `FOREIGN KEY` 外键约束，建立表间关联
- [ ] `UNIQUE` 唯一约束，字段值不能重复
- [ ] `NOT NULL` 非空约束，字段值不能为空
- [ ] `CHECK` 检查约束，限制字段值范围
- [ ] `DEFAULT` 默认值约束，提供字段默认值
- [ ] `ALTER TABLE` 修改表结构（添加/删除/修改列和约束）
