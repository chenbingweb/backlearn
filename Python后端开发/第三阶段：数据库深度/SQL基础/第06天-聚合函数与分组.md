## 1. 聚合函数

### COUNT 计数

```sql
-- COUNT(*) 统计所有行（包括 NULL）
SELECT COUNT(*) FROM users;                    -- 总用户数
SELECT COUNT(*) FROM orders WHERE status = 'completed';  -- 已完成订单数

-- COUNT(column) 统计非 NULL 行
SELECT COUNT(email) FROM users;              -- 有邮箱的用户数

-- COUNT(DISTINCT column) 统计去重数量
SELECT COUNT(DISTINCT category) FROM products;  -- 商品种类数
SELECT COUNT(DISTINCT user_id) FROM orders;    -- 下过单的用户数
```

### SUM 求和

```sql
-- SUM 求和（忽略 NULL）
SELECT SUM(price) FROM products;                           -- 所有商品价格总和
SELECT SUM(price) FROM orders WHERE status = 'completed';   -- 已完成订单总金额

-- SUM(DISTINCT column) 去重求和
SELECT SUM(DISTINCT price) FROM products;  -- 不同价格的总和

-- 带条件的求和
SELECT SUM(total) FROM orders
WHERE created_at >= '2024-01-01' AND created_at < '2024-02-01';
```

### AVG 平均值

```sql
-- AVG 平均值（忽略 NULL）
SELECT AVG(price) FROM products;                    -- 商品平均价格
SELECT AVG(age) FROM users;                         -- 用户平均年龄

-- AVG(DISTINCT column) 去重平均值
SELECT AVG(DISTINCT price) FROM products;

-- 保留小数位（PostgreSQL）
SELECT ROUND(AVG(price), 2) FROM products;  -- 保留2位小数
```

### MIN 和 MAX

```sql
-- MIN 最小值
SELECT MIN(price) FROM products;         -- 最低价
SELECT MIN(created_at) FROM orders;      -- 最早订单时间

-- MAX 最大值
SELECT MAX(price) FROM products;         -- 最高价
SELECT MAX(created_at) FROM orders;      -- 最新订单时间

-- 组合使用
SELECT
    MIN(price) AS 最低价,
    MAX(price) AS 最高价,
    AVG(price) AS 平均价,
    COUNT(*) AS 商品数
FROM products;
```

### 聚合函数与 NULL

```sql
-- 聚合函数默认忽略 NULL
CREATE TABLE test (
    value INTEGER
);

INSERT INTO test VALUES (1), (2), (NULL), (3), (NULL), (4);

SELECT COUNT(*) FROM test;        -- 6（所有行）
SELECT COUNT(value) FROM test;    -- 4（非 NULL 行）
SELECT SUM(value) FROM test;      -- 10（忽略 NULL：1+2+3+4）
SELECT AVG(value) FROM test;       -- 2.5（10/4）

-- 如果需要包含 NULL 作为 0，可使用 COALESCE
SELECT SUM(COALESCE(value, 0)) FROM test;  -- 10（NULL 当作 0）
```

## 2. GROUP BY 分组

### 基本语法

```sql
-- 按分类统计商品数量
SELECT category, COUNT(*) AS product_count
FROM products
GROUP BY category;

-- 按用户统计订单数
SELECT user_id, COUNT(*) AS order_count
FROM orders
GROUP BY user_id;

-- 按状态统计订单
SELECT status, COUNT(*), SUM(total), AVG(total)
FROM orders
GROUP BY status;
```

### 分组与聚合

```sql
-- 常见分组统计
SELECT
    category,
    COUNT(*) AS product_count,
    SUM(stock) AS total_stock,
    AVG(price) AS avg_price,
    MIN(price) AS min_price,
    MAX(price) AS max_price
FROM products
WHERE is_active = TRUE
GROUP BY category
ORDER BY product_count DESC;
```

### 多字段分组

```sql
-- 按多个字段分组
SELECT
    category,
    status,
    COUNT(*) AS count
FROM products
GROUP BY category, status
ORDER BY category, count DESC;

-- 输出示例：
-- category | status | count
-- ---------+--------+------
-- electronics | active | 10
-- electronics | inactive | 2
-- books | active | 25
```

### GROUP BY 机制说明

```sql
-- GROUP BY 执行过程：
-- 1. 按 GROUP BY 字段分组
-- 2. 每个组内执行聚合函数
-- 3. 返回聚合结果

-- SELECT 中非聚合列必须在 GROUP BY 中
-- ❌ 错误：SELECT name, COUNT(*) FROM products
-- ✅ 正确：SELECT category, COUNT(*) FROM products GROUP BY category

-- PostgreSQL 可以用 ANY_VALUE 取任意值
SELECT
    category,
    ANY_VALUE(name) AS example_name,  -- 取该分组任意一个 name
    COUNT(*) AS count
FROM products
GROUP BY category;
```

## 3. HAVING 过滤分组

### 与 WHERE 的区别

```sql
-- WHERE 过滤行（在分组前）
-- HAVING 过滤分组（在分组后）

SELECT category, COUNT(*) AS count
FROM products
WHERE price > 100           -- 先过滤行
GROUP BY category
HAVING COUNT(*) > 5         -- 再过滤分组
ORDER BY count DESC;
```

### 完整示例

```sql
-- 统计每个用户的订单
SELECT
    user_id,
    COUNT(*) AS order_count,
    SUM(total) AS total_amount,
    AVG(total) AS avg_order_amount
FROM orders
WHERE status IN ('completed', 'shipped')
GROUP BY user_id
HAVING COUNT(*) >= 2 AND SUM(total) > 1000
ORDER BY total_amount DESC;
```

### 多条件 HAVING

```sql
-- HAVING 支持复杂条件
SELECT
    category,
    COUNT(*) AS count,
    AVG(price) AS avg_price
FROM products
GROUP BY category
HAVING
    COUNT(*) >= 3
    AND AVG(price) > 100
    AND SUM(stock) > 50
ORDER BY avg_price DESC;
```

## 4. 完整查询结构

### 查询执行顺序

```sql
SELECT column(s)                    -- 5. 选择列
FROM table                          -- 1. 从表获取数据
WHERE condition                      -- 2. 过滤行
GROUP BY column(s)                  -- 3. 分组
HAVING condition                     -- 4. 过滤分组
ORDER BY column(s) [ASC|DESC]        -- 6. 排序
LIMIT n OFFSET m;                    -- 7. 限制返回
```

### 完整查询示例

```sql
SELECT
    category,
    COUNT(*) AS product_count,
    ROUND(AVG(price), 2) AS avg_price,
    SUM(stock) AS total_stock
FROM products
WHERE is_active = TRUE
  AND stock > 0
GROUP BY category
HAVING COUNT(*) >= 2
ORDER BY avg_price DESC
LIMIT 10;
```

## 5. 实战练习

### 练习 1：订单统计

```sql
-- 按月统计订单
SELECT
    TO_CHAR(created_at, 'YYYY-MM') AS month,
    COUNT(*) AS order_count,
    SUM(total) AS total_amount,
    AVG(total) AS avg_amount
FROM orders
WHERE status = 'completed'
GROUP BY TO_CHAR(created_at, 'YYYY-MM')
ORDER BY month DESC;

-- PostgreSQL 月统计
SELECT
    DATE_TRUNC('month', created_at) AS month,
    COUNT(*) AS order_count,
    SUM(total) AS total_amount
FROM orders
GROUP BY DATE_TRUNC('month', created_at)
ORDER BY month DESC;

-- MySQL 月统计
SELECT
    DATE_FORMAT(created_at, '%Y-%m') AS month,
    COUNT(*) AS order_count,
    SUM(total) AS total_amount
FROM orders
GROUP BY DATE_FORMAT(created_at, '%Y-%m')
ORDER BY month DESC;
```

### 练习 2：用户行为分析

```sql
-- 用户消费统计
SELECT
    u.username,
    COUNT(o.id) AS order_count,
    COALESCE(SUM(o.total), 0) AS total_spent,
    COALESCE(AVG(o.total), 0) AS avg_order_value,
    MAX(o.created_at) AS last_order_date
FROM users u
LEFT JOIN orders o ON u.id = o.user_id AND o.status = 'completed'
GROUP BY u.id, u.username
HAVING COUNT(o.id) > 0
ORDER BY total_spent DESC
LIMIT 20;
```

## 今日总结

- [ ] `COUNT(*)` 统计行数，`COUNT(col)` 统计非 NULL，`COUNT(DISTINCT col)` 去重统计
- [ ] `SUM(col)` 求和，`AVG(col)` 平均值
- [ ] `MIN(col)` 最小值，`MAX(col)` 最大值
- [ ] `GROUP BY column` 分组，配合聚合函数使用
- [ ] `HAVING condition` 在分组后过滤分组
- [ ] `WHERE` 过滤行（在分组前），`HAVING` 过滤分组（在分组后）
- [ ] 执行顺序：FROM → WHERE → GROUP BY → HAVING → SELECT → ORDER BY → LIMIT
