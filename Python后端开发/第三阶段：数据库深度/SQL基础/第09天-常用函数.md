## 1. 字符串函数

### 长度与大小写

```sql
-- 长度函数
SELECT LENGTH('hello');                    -- 5
SELECT CHAR_LENGTH('hello');              -- 5（字符数，MySQL）
SELECT LENGTH('你好');                    -- 6（字节数，UTF8）

SELECT LENGTH(username) FROM users;       -- 用户名长度

-- 大小写转换
SELECT UPPER('hello');                    -- 'HELLO'
SELECT LOWER('HELLO');                    -- 'hello'
SELECT INITCAP('hello world');            -- 'Hello World'（首字母大写）

-- MySQL
SELECT UCASE('hello');                    -- 'HELLO'
SELECT LCASE('HELLO');                    -- 'hello'
```

### 字符串拼接

```sql
-- 拼接
SELECT CONCAT('Hello', ' ', 'World');    -- 'Hello World'
SELECT CONCAT_WS('-', '2024', '01', '15');  -- '2024-01-15'

-- PostgreSQL
SELECT 'Hello' || ' ' || 'World';        -- 'Hello World'
SELECT CONCAT('A', NULL, 'B');           -- 'AB'（忽略 NULL）

-- MySQL
SELECT CONCAT('A', NULL, 'B');           -- NULL（包含 NULL）

-- LPAD / RPAD 填充
SELECT LPAD('5', 3, '0');                -- '005'
SELECT RPAD('hi', 5, '-');              -- 'hi---'
```

### 字符串截取

```sql
-- 截取
SELECT SUBSTRING('Hello World', 1, 5);   -- 'Hello'
SELECT SUBSTRING('Hello World', 7);      -- 'World'
SELECT SUBSTRING('Hello World' FROM 7 FOR 5);  -- SQL 标准语法

-- PostgreSQL
SELECT SUBSTR('Hello World', 1, 5);     -- 'Hello'

-- MySQL
SELECT SUBSTR('Hello World', 1, 5);     -- 'Hello'
SELECT SUBSTRING_INDEX('a.b.c', '.', 2); -- 'a.b'（取前2段）
```

### 字符串替换与查找

```sql
-- 替换
SELECT REPLACE('Hello World', 'World', 'SQL');  -- 'Hello SQL'
SELECT REPLACE('Hello World', 'o', '0');       -- 'Hell0 W0rld'

-- 去空格
SELECT TRIM('  hello  ');                  -- 'hello'
SELECT LTRIM('  hello');                    -- 'hello'
SELECT RTRIM('hello  ');                   -- 'hello'

-- 查找位置
SELECT POSITION('World' IN 'Hello World');  -- 7（PostgreSQL）
SELECT LOCATE('World', 'Hello World');      -- 7（MySQL）

-- 字符 ASCII
SELECT ASCII('A');                          -- 65
```

## 2. 数值函数

### 数学运算

```sql
-- 绝对值
SELECT ABS(-10);                            -- 10
SELECT ABS(price) FROM products;

-- 四舍五入
SELECT ROUND(3.14159, 2);                   -- 3.14
SELECT ROUND(3.14159, 0);                   -- 3
SELECT ROUND(3.5);                          -- 4

-- 向上/向下取整
SELECT CEIL(3.14);                          -- 4
SELECT FLOOR(3.14);                          -- 3

-- PostgreSQL 特有
SELECT CEILING(3.14);                      -- 4

-- 幂运算
SELECT POWER(2, 3);                         -- 8
SELECT 2 ^ 3;                               -- 8（PostgreSQL）

-- 平方根
SELECT SQRT(16);                            -- 4

-- 取模
SELECT MOD(10, 3);                          -- 1
SELECT 10 % 3;                              -- 1
```

### 数值格式化

```sql
-- PostgreSQL 格式化
SELECT TO_CHAR(1234.56, '9999.99');        -- '1234.56'
SELECT TO_CHAR(1234.56, 'FM9999.00');      -- '1234.56'（去前导零）
SELECT TO_CHAR(1234.56, '$9999.99');        -- '$1234.56'

-- MySQL 格式化
SELECT FORMAT(1234.56, 2);                 -- '1,234.56'
```

## 3. 日期时间函数

### 获取当前日期时间

```sql
-- 当前时间
SELECT CURRENT_DATE;                       -- 2024-01-15
SELECT CURRENT_TIME;                       -- 14:30:00
SELECT CURRENT_TIMESTAMP;                  -- 2024-01-15 14:30:00
SELECT NOW();                              -- 2024-01-15 14:30:00
SELECT LOCALTIMESTAMP;                    -- 2024-01-15 14:30:00

-- PostgreSQL 时区
SELECT NOW() AT TIME ZONE 'UTC';          -- UTC 时间
SELECT CURRENT_TIMESTAMP AT TIME ZONE 'Asia/Shanghai';  -- 上海时间
```

### 日期提取

```sql
-- PostgreSQL EXTRACT
SELECT EXTRACT(YEAR FROM created_at) FROM orders;  -- 年
SELECT EXTRACT(MONTH FROM created_at) FROM orders; -- 月
SELECT EXTRACT(DAY FROM created_at) FROM orders;    -- 日
SELECT EXTRACT(HOUR FROM created_at) FROM orders;   -- 时
SELECT EXTRACT(WEEK FROM created_at) FROM orders;   -- 第几周

-- MySQL DATE_PART
SELECT DATE_PART('year', created_at) FROM orders;
SELECT DATE_PART('month', created_at) FROM orders;

-- MySQL 提取函数
SELECT YEAR(created_at), MONTH(created_at), DAY(created_at) FROM orders;
SELECT HOUR(created_at), MINUTE(created_at), SECOND(created_at) FROM orders;
SELECT QUARTER(created_at) FROM orders;  -- 第几季度
```

### 日期计算

```sql
-- 日期加减
SELECT created_at + INTERVAL '1 day' FROM orders;
SELECT created_at - INTERVAL '7 days' FROM orders;
SELECT created_at + INTERVAL '1 month' FROM orders;
SELECT created_at + INTERVAL '1 year' FROM orders;

-- MySQL
SELECT DATE_ADD(created_at, INTERVAL 1 DAY) FROM orders;
SELECT DATE_SUB(created_at, INTERVAL 7 DAY) FROM orders;

-- 计算天数差
SELECT CURRENT_DATE - created_at FROM orders;  -- PostgreSQL
SELECT DATEDIFF(CURRENT_DATE, created_at) FROM orders;  -- MySQL

-- AGE 计算年龄（PostgreSQL）
SELECT AGE(created_at) FROM users;
SELECT AGE(created_at, '2024-01-01') FROM users;
```

### 日期格式化

```sql
-- PostgreSQL TO_CHAR
SELECT TO_CHAR(created_at, 'YYYY-MM-DD') FROM orders;
SELECT TO_CHAR(created_at, 'YYYY年MM月DD日') FROM orders;
SELECT TO_CHAR(created_at, 'HH24:MI:SS') FROM orders;
SELECT TO_CHAR(created_at, 'YYYY-MM-DD HH24:MI:SS') FROM orders;

-- MySQL DATE_FORMAT
SELECT DATE_FORMAT(created_at, '%Y-%m-%d') FROM orders;
SELECT DATE_FORMAT(created_at, '%Y年%m月%d日') FROM orders;
SELECT DATE_FORMAT(created_at, '%H:%i:%s') FROM orders;

-- 解析字符串为日期（PostgreSQL）
SELECT TO_DATE('2024-01-15', 'YYYY-MM-DD');
SELECT TO_TIMESTAMP('2024-01-15 14:30:00', 'YYYY-MM-DD HH24:MI:SS');

-- MySQL STR_TO_DATE
SELECT STR_TO_DATE('2024-01-15', '%Y-%m-%d');
```

## 4. 条件表达式

### CASE 表达式

```sql
-- 简单 CASE
SELECT
    name,
    category,
    CASE category
        WHEN 'electronics' THEN '电子产品'
        WHEN 'books' THEN '图书'
        WHEN 'food' THEN '食品'
        ELSE '其他'
    END AS category_cn
FROM products;

-- 搜索 CASE
SELECT
    username,
    age,
    CASE
        WHEN age < 18 THEN '未成年'
        WHEN age >= 18 AND age < 30 THEN '青年'
        WHEN age >= 30 AND age < 60 THEN '中年'
        ELSE '老年'
    END AS age_group
FROM users;
```

### COALESCE 和 NULLIF

```sql
-- COALESCE：返回第一个非 NULL 值
SELECT COALESCE(NULL, NULL, 'default');       -- 'default'
SELECT COALESCE(email, phone, 'unknown') FROM users;
SELECT COALESCE(SUM(total), 0) FROM orders;  -- NULL 转为 0

-- NULLIF：两值相等返回 NULL，否则返回第一个
SELECT NULLIF(1, 1);                          -- NULL
SELECT NULLIF(1, 2);                          -- 1
SELECT NULLIF('', '');                         -- NULL

-- 避免除零错误
SELECT total / NULLIF(count, 0) FROM stats;  -- count 为 0 时返回 NULL 而非错误
```

### NVL / IFNULL（数据库特定）

```sql
-- Oracle NVL
SELECT NVL(email, 'unknown') FROM users;

-- MySQL IFNULL
SELECT IFNULL(email, 'unknown') FROM users;

-- PostgreSQL COALESCE
SELECT COALESCE(email, 'unknown') FROM users;
```

## 5. 类型转换

```sql
-- PostgreSQL 类型转换
SELECT '123'::INTEGER;                    -- 字符串转整数
SELECT CAST('123' AS INTEGER);              -- 同上
SELECT 123::VARCHAR;                       -- 整数转字符串

-- MySQL 类型转换
SELECT CAST('123' AS UNSIGNED);            -- 字符串转整数
SELECT CONVERT('123', SIGNED);              -- 同上
SELECT CONVERT('2024-01-15', DATE);         -- 字符串转日期
```

## 6. 实战练习

### 练习 1：数据格式化报表

```sql
-- 用户注册统计
SELECT
    TO_CHAR(created_at, 'YYYY年MM月') AS month,
    COUNT(*) AS user_count,
    SUM(CASE WHEN status = 'active' THEN 1 ELSE 0 END) AS active_count,
    ROUND(
        SUM(CASE WHEN status = 'active' THEN 1 ELSE 0 END)::DECIMAL /
        NULLIF(COUNT(*), 0) * 100,
        2
    ) AS activation_rate
FROM users
WHERE created_at >= '2024-01-01'
GROUP BY TO_CHAR(created_at, 'YYYY年MM月')
ORDER BY month;
```

### 练习 2：订单状态转换

```sql
-- 订单状态转换
SELECT
    order_no,
    total,
    status,
    CASE
        WHEN status = 'pending' AND created_at < CURRENT_TIMESTAMP - INTERVAL '7 days'
            THEN '超时待处理'
        WHEN status = 'pending' THEN '待处理'
        WHEN status = 'paid' THEN '已付款'
        WHEN status = 'shipped' THEN '已发货'
        WHEN status = 'completed' THEN '已完成'
        WHEN status = 'cancelled' THEN '已取消'
        ELSE '未知状态'
    END AS status_description,
    TO_CHAR(created_at, 'YYYY-MM-DD HH24:MI') AS created_time
FROM orders
ORDER BY created_at DESC;
```
