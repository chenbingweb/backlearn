## 1. pytest 断言详解

pytest 使用原生 `assert`，但失败时会提供详细的上下文。

### 基本断言

```python
def test_assertions():
    # 相等
    assert 1 + 1 == 2

    # 不相等
    assert 1 + 1 != 3

    # 真值
    assert True
    assert "hello"   # 非空字符串为真
    assert [1, 2]    # 非空列表为真

    # 假值
    assert not False
    assert not []
    assert not None
```

### 复杂断言

```python
def test_complex_assertions():
    # in / not in
    assert 3 in [1, 2, 3]
    assert "py" not in "java"

    # is / is not（身份比较）
    assert None is None
    assert True is True

    # 比较
    assert 5 > 3
    assert 3 <= 5

    # 近似相等（浮点数）
    assert 0.1 + 0.2 == pytest.approx(0.3)
    assert 3.14159 == pytest.approx(3.14, abs=0.01)
```

### 集合比较

```python
def test_collections():
    # 列表比较（失败时会显示差异）
    expected = [1, 2, 3]
    result = [1, 2, 4]
    assert result == expected
    # E       assert [1, 2, 4] == [1, 2, 3]
    # E         At index 2 diff: 4 != 3

    # 字典比较
    expected = {"name": "Alice", "age": 25}
    result = {"name": "Alice", "age": 26}
    assert result == expected

    # 集合比较（忽略顺序）
    assert {1, 2, 3} == {3, 2, 1}
```

### 自定义失败消息

```python
def test_with_message():
    user = {"name": "Bob", "age": 20}
    assert user["age"] >= 21, f"用户 {user['name']} 未满21岁"
```

## 2. 参数化测试

用不同参数多次运行同一个测试函数。

```python
import pytest

# 基本参数化
@pytest.mark.parametrize("a,b,expected", [
    (1, 2, 3),
    (5, 5, 10),
    (-1, 1, 0),
    (0, 0, 0),
])
def test_add(a, b, expected):
    assert a + b == expected

# 多个参数
@pytest.mark.parametrize("name", [
    "Alice",
    "Bob",
    "Charlie",
])
def test_name_length(name):
    assert len(name) > 0

# 组合参数（笛卡尔积）
@pytest.mark.parametrize("x", [1, 2])
@pytest.mark.parametrize("y", ["a", "b"])
def test_combinations(x, y):
    print(f"x={x}, y={y}")
    # 会运行 4 次：1a, 1b, 2a, 2b
```

## 3. 标记（Markers）

### 自定义标记

```python
import pytest

@pytest.mark.slow
def test_heavy_computation():
    # 耗时操作
    pass

@pytest.mark.integration
def test_database_connection():
    pass

@pytest.mark.skip(reason="功能尚未实现")
def test_future_feature():
    pass

# 条件跳过
@pytest.mark.skipif(sys.platform == "win32", reason="Windows 不支持")
def test_unix_feature():
    pass

# 预期失败
@pytest.mark.xfail(reason="已知 bug，待修复")
def test_buggy_feature():
    assert 1 / 0 == 0
```

### 运行标记的测试

```bash
# 只运行 slow 测试
pytest -m slow

# 排除 slow 测试
pytest -m "not slow"

# 运行 integration 但不运行 slow
pytest -m "integration and not slow"
```

### pyproject.toml 注册标记

```toml
[tool.pytest.ini_options]
markers = [
    "slow: marks tests as slow (deselect with '-m \"not slow\"')",
    "integration: marks tests as integration tests",
]
```

## 4. 测试组织

### 测试类

```python
class TestCalculator:
    """Calculator 测试类"""

    def test_add(self):
        assert 1 + 1 == 2

    def test_subtract(self):
        assert 5 - 3 == 2

    @pytest.mark.slow
    def test_heavy_operation(self):
        pass

class TestStringUtils:
    """String 工具测试类"""

    def test_upper(self):
        assert "hello".upper() == "HELLO"
```

### 用类组织共享 setup

```python
class TestShoppingCart:
    def setup_method(self):
        """每个测试方法前运行"""
        self.cart = ShoppingCart()
        self.cart.add("apple", 5)

    def teardown_method(self):
        """每个测试方法后运行"""
        self.cart.clear()

    def test_total(self):
        assert self.cart.total() == 5

    def test_add_item(self):
        self.cart.add("banana", 3)
        assert self.cart.total() == 8
```

## 实战练习

### 练习：测试密码强度验证

```python
# password_validator.py
import re

def validate_password(password):
    if len(password) < 8:
        return False, "密码至少8位"
    if not re.search(r"[A-Z]", password):
        return False, "需要大写字母"
    if not re.search(r"[a-z]", password):
        return False, "需要小写字母"
    if not re.search(r"\d", password):
        return False, "需要数字"
    return True, "密码强度OK"
```

```python
# test_password_validator.py
import pytest
from password_validator import validate_password

@pytest.mark.parametrize("password,expected_valid,expected_msg", [
    ("Short1", False, "密码至少8位"),
    ("lowercase1", False, "需要大写字母"),
    ("UPPERCASE1", False, "需要小写字母"),
    ("NoDigits!", False, "需要数字"),
    ("Valid1Pass", True, "密码强度OK"),
    ("MyP@ssw0rd", True, "密码强度OK"),
])
def test_validate_password(password, expected_valid, expected_msg):
    valid, msg = validate_password(password)
    assert valid == expected_valid
    assert msg == expected_msg
```
