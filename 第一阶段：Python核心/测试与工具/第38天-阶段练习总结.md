### test_todo.py

```python
# tests/test_todo.py
import pytest
from todo import TodoApp


class TestTodoApp:
    def test_add_task(self, temp_db):
        task = temp_db.add("买牛奶", "normal")
        assert task.content == "买牛奶"
        assert task.priority == "normal"
        assert not task.done

    def test_add_task_persists(self, temp_db):
        temp_db.add("买牛奶")
        # 重新加载
        app2 = TodoApp(data_file=temp_db.data_file)
        tasks = app2.list()
        assert len(tasks) == 1
        assert tasks[0].content == "买牛奶"

    def test_mark_done(self, temp_db):
        task = temp_db.add("任务")
        result = temp_db.done(task.id)
        assert result.done is True
        assert result.done_at is not None

    def test_mark_done_nonexistent(self, temp_db):
        result = temp_db.done(999)
        assert result is None

    def test_delete_task(self, temp_db):
        task = temp_db.add("待删除")
        assert temp_db.delete(task.id) is True
        assert len(temp_db.list()) == 0

    def test_list_filter(self, sample_tasks):
        # 完成一个
        sample_tasks.done(1)

        all_tasks = sample_tasks.list("all")
        done_tasks = sample_tasks.list("done")
        pending_tasks = sample_tasks.list("pending")

        assert len(all_tasks) == 3
        assert len(done_tasks) == 1
        assert len(pending_tasks) == 2

    def test_stats(self, sample_tasks):
        sample_tasks.done(1)
        stats = sample_tasks.stats()

        assert stats["total"] == 3
        assert stats["done"] == 1
        assert stats["pending"] == 2
        assert stats["completion_rate"] == "33.3%"

    def test_clear_done(self, sample_tasks):
        sample_tasks.done(1)
        count = sample_tasks.clear_done()

        assert count == 1
        assert len(sample_tasks.list()) == 2


class TestTask:
```

### test_main.py（带 mock）

```python
# tests/test_main.py
from unittest.mock import patch, MagicMock
import pytest


class TestCLI:
    @patch("sys.argv", ["main.py", "add", "买牛奶"])
    def test_cli_add(self, capsys):
        from main import main
        from todo import TodoApp

        # 使用临时文件
        import tempfile
        import os

        with tempfile.NamedTemporaryFile(mode="w", delete=False, suffix=".json") as f:
            db_file = f.name

        try:
            with patch("main.TodoApp") as mock_app_class:
                mock_app = MagicMock()
                mock_task = MagicMock()
                mock_task.id = 1
                mock_task.content = "买牛奶"
                mock_task.__str__ = MagicMock(return_value="[ ] #1 买牛奶")

                mock_app.add.return_value = mock_task
                mock_app_class.return_value = mock_app

                main()

                mock_app.add.assert_called_once_with("买牛奶", "normal")
        finally:
            os.unlink(db_file)
```
