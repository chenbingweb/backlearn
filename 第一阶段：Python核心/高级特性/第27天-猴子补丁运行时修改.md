## 1. 什么是猴子补丁

猴子补丁（Monkey Patching）是在运行时动态修改类或模块的行为。

```python
# 原始类
class Dog:
    def bark(self):
        return "汪汪"

# 猴子补丁：运行时修改方法
def new_bark(self):
    return "汪汪汪！（更大声）"

Dog.bark = new_bark

dog = Dog()
print(dog.bark())  # 汪汪汪！（更大声）
```

## 2. 常见应用场景

### 修复第三方库 Bug

```python
import some_library

# 修复第三方库的某个方法
def fixed_method(self):
    # 修复后的实现
    pass

some_library.SomeClass.broken_method = fixed_method
```

### 测试中的 Mock

```python
import datetime

# 测试中固定时间
class MockDateTime:
    @classmethod
    def now(cls):
        return datetime.datetime(2024, 1, 1, 12, 0, 0)

# 猴子补丁
datetime.datetime = MockDateTime

print(datetime.datetime.now())  # 2024-01-01 12:00:00
```

### 添加缺失功能

```python
# 给内置类型添加方法（不推荐但可行）
def str_reverse(self):
    return self[::-1]

str.reverse = str_reverse

print("hello".reverse())  # olleh
```

## 4. 安全的猴子补丁

```python
import functools

def patch(target, name):
    """安全的猴子补丁装饰器"""
    def decorator(func):
        original = getattr(target, name)
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            return func(original, *args, **kwargs)
        setattr(target, name, wrapper)
        return wrapper
    return decorator

# 使用
class Calculator:
    def add(self, a, b):
        return a + b

@patch(Calculator, "add")
def logged_add(original, self, a, b):
    print(f"调用 add({a}, {b})")
    result = original(self, a, b)
    print(f"结果: {result}")
    return result

calc = Calculator()
calc.add(2, 3)
# 调用 add(2, 3)
# 结果: 5
```

## 5. 运行时属性添加

```python
class User:
    def __init__(self, name):
        self.name = name

user = User("Alice")

# 动态添加属性
user.age = 25
user.email = "alice@example.com"

from types import MethodType

def greet(self):
    return f"Hello, I'm {self.name}"

user.greet = MethodType(greet, user)
```

## 6. 使用场景与风险

### 适合使用

- 单元测试中的 Mock
- 临时修复第三方库 Bug
- 插件系统中扩展功能

### 风险

- 代码难以理解和维护
- 可能破坏依赖该代码的其他部分
- 升级库时可能失效

### 替代方案

```python
# 不用猴子补丁，用组合
class FixedDog:
    def __init__(self):
        self._dog = Dog()

    def bark(self):
        # 修改行为而不动原始类
        return self._dog.bark().upper()
```

## 实战练习

### 练习 1：请求缓存补丁

```python
import functools

# 模拟 requests
class requests:
    @staticmethod
    def get(url):
        print(f"真实请求: {url}")
        return f"响应: {url}"

# 缓存补丁
cache = {}
_original_get = requests.get

def cached_get(url):
    if url not in cache:
        cache[url] = _original_get(url)
    else:
        print(f"缓存命中: {url}")
    return cache[url]

requests.get = cached_get

# 测试
print(requests.get("https://api.com/data"))
print(requests.get("https://api.com/data"))  # 缓存命中
```

### 练习 2：性能计时补丁

```python
import time

class TimerPatch:
    """为所有方法添加计时"""
    def __init__(self, cls):
        self.cls = cls
        self._patch_methods()

    def _patch_methods(self):
        for name in dir(self.cls):
            if not name.startswith("_"):
                attr = getattr(self.cls, name)
                if callable(attr):
                    setattr(self.cls, name, self._timer_wrapper(attr))

    def _timer_wrapper(self, func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            start = time.time()
            result = func(*args, **kwargs)
            print(f"{func.__name__}: {time.time() - start:.4f}s")
            return result
        return wrapper

# 使用
class DataProcessor:
    def process(self):
        time.sleep(0.1)

    def save(self):
        time.sleep(0.05)

TimerPatch(DataProcessor)

dp = DataProcessor()
dp.process()
dp.save()
```
