```python
import asyncio
import threading
import multiprocessing
import queue
import time
import json
import hashlib
from abc import ABC, abstractmethod
from urllib.parse import urlparse
from dataclasses import dataclass, asdict
from typing import List, Set, Dict, Optional
import urllib.request


@dataclass
class CrawlResult:
    url: str
    status: int
    title: str
    elapsed: float
    error: Optional[str] = None

    def to_dict(self):
        return asdict(self)


class BaseCrawler(ABC):
    """爬虫基类"""

    def __init__(self, max_workers: int = 5, delay: float = 0):
        self.max_workers = max_workers
        self.delay = delay
        self.visited: Set[str] = set()
        self.results: List[CrawlResult] = []

    def _normalize_url(self, url: str) -> str:
        return hashlib.md5(url.encode()).hexdigest()

    def _is_visited(self, url: str) -> bool:
        url_hash = self._normalize_url(url)
        if url_hash in self.visited:
            return True
        self.visited.add(url_hash)
        return False

    def _fetch(self, url: str) -> CrawlResult:
        """模拟 HTTP 请求"""
        start = time.time()
        try:
            if self.delay > 0:
                time.sleep(self.delay)

            # 模拟请求
            req = urllib.request.Request(
                url,
                headers={"User-Agent": "Python-Crawler/1.0"}
            )
            with urllib.request.urlopen(req, timeout=10) as response:
                html = response.read().decode("utf-8", errors="ignore")
                # 提取 title
                title = self._extract_title(html)
                return CrawlResult(
                    url=url,
                    status=response.status,
                    title=title,
                    elapsed=time.time() - start
                )
        except Exception as e:
            return CrawlResult(
                url=url,
                status=0,
                title="",
                elapsed=time.time() - start,
                error=str(e)
            )

    def _extract_title(self, html: str) -> str:
        import re
        match = re.search(r"<title>(.*?)</title>", html, re.IGNORECASE)
        return match.group(1).strip() if match else "No Title"

    @abstractmethod
    def crawl(self, urls: List[str]) -> List[CrawlResult]:
        pass

    def save_results(self, filename: str):
        with open(filename, "w", encoding="utf-8") as f:
            json.dump([r.to_dict() for r in self.results], f, indent=2)

    def statistics(self) -> Dict:
        total = len(self.results)
        success = sum(1 for r in self.results if r.status == 200)
        failed = total - success
        avg_time = sum(r.elapsed for r in self.results) / total if total > 0 else 0

        return {
            "total": total,
            "success": success,
            "failed": failed,
            "avg_time": round(avg_time, 4),
            "workers": self.max_workers
        }


class ThreadCrawler(BaseCrawler):
    """多线程爬虫"""

    def crawl(self, urls: List[str]) -> List[CrawlResult]:
        result_queue = queue.Queue()
        url_queue = queue.Queue()

        for url in urls:
            if not self._is_visited(url):
                url_queue.put(url)

        def worker():
            while True:
                try:
                    url = url_queue.get(timeout=1)
                except queue.Empty:
                    break
                result = self._fetch(url)
                result_queue.put(result)
                url_queue.task_done()

        threads = []
        for _ in range(min(self.max_workers, len(urls))):
            t = threading.Thread(target=worker)
            t.start()
            threads.append(t)

        for t in threads:
            t.join()

        while not result_queue.empty():
            self.results.append(result_queue.get())

        return self.results


class ProcessCrawler(BaseCrawler):
    """多进程爬虫"""

    def _fetch_wrapper(self, url: str) -> CrawlResult:
        return self._fetch(url)

    def crawl(self, urls: List[str]) -> List[CrawlResult]:
        unique_urls = [u for u in urls if not self._is_visited(u)]

        with multiprocessing.Pool(self.max_workers) as pool:
            self.results = pool.map(self._fetch, unique_urls)

        return self.results


class AsyncCrawler(BaseCrawler):
    """异步爬虫"""

    async def _fetch_async(self, session, url: str) -> CrawlResult:
        start = time.time()
        try:
            import aiohttp
            async with session.get(url, timeout=10) as response:
                html = await response.text()
                title = self._extract_title(html)
                return CrawlResult(
                    url=url,
                    status=response.status,
                    title=title,
                    elapsed=time.time() - start
                )
        except Exception as e:
            return CrawlResult(
                url=url,
                status=0,
                title="",
                elapsed=time.time() - start,
                error=str(e)
            )

    def crawl(self, urls: List[str]) -> List[CrawlResult]:
        unique_urls = [u for u in urls if not self._is_visited(u)]
        asyncio.run(self._crawl(unique_urls))
        return self.results

    async def _crawl(self, urls: List[str]):
        import aiohttp
        semaphore = asyncio.Semaphore(self.max_workers)

        async def bounded_fetch(session, url):
            async with semaphore:
                return await self._fetch_async(session, url)

        async with aiohttp.ClientSession() as session:
            tasks = [bounded_fetch(session, url) for url in urls]
            self.results = await asyncio.gather(*tasks)


class CrawlerFactory:
    """爬虫工厂"""

    _crawlers = {
        "thread": ThreadCrawler,
        "process": ProcessCrawler,
        "async": AsyncCrawler,
    }

    @classmethod
    def create(cls, mode: str, **kwargs) -> BaseCrawler:
        if mode not in cls._crawlers:
            raise ValueError(f"未知模式: {mode}")
        return cls._crawlers[mode](**kwargs)


# ============== 演示 ==============

def demo():
    urls = [
        "https://httpbin.org/get",
        "https://httpbin.org/ip",
        "https://httpbin.org/user-agent",
        "https://httpbin.org/headers",
        "https://httpbin.org/delay/1",
    ] * 2  # 10 个 URL

    for mode in ["thread", "process"]:
        print(f"\n{'='*40}")
        print(f"模式: {mode}")
        print(f"{'='*40}")

        crawler = CrawlerFactory.create(mode, max_workers=3, delay=0.5)

        start = time.time()
        results = crawler.crawl(urls)
        elapsed = time.time() - start

        stats = crawler.statistics()
        print(f"总耗时: {elapsed:.2f}s")
        print(f"统计: {stats}")

        for r in results[:3]:
            print(f"  {r.url} -> {r.status} ({r.elapsed:.2f}s)")

        crawler.save_results(f"results_{mode}.json")


if __name__ == "__main__":
    demo()
```
