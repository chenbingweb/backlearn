
### 带参数的异常

```python
class BusinessError(Exception):
    """业务逻辑错误"""
    def __init__(self, message, code=None):
        super().__init__(message)
        self.code = code
        self.message = message

    def __str__(self):
        if self.code:
            return f"[{self.code}] {self.message}"
        return self.message

# 使用
try:
    raise BusinessError("用户不存在", code=1001)
except BusinessError as e:
    print(e)           # [1001] 用户不存在
    print(e.code)      # 1001
    print(e.message)   # 用户不存在
```

## 3. 实际应用：API 异常处理

```python
class APIError(Exception):
    """API 错误基类"""
    status_code = 500

    def __init__(self, message, details=None):
        self.message = message
        self.details = details or {}
        super().__init__(message)

    def to_dict(self):
        return {
            "error": {
                "message": self.message,
                "code": self.status_code,
                "details": self.details
            }
        }

class BadRequest(APIError):
    status_code = 400

class Unauthorized(APIError):
    status_code = 401

class Forbidden(APIError):
    status_code = 403

class NotFound(APIError):
    status_code = 404

class Conflict(APIError):
    status_code = 409

# 使用（模拟 Flask 处理）
def handle_api_error(error: APIError):
    return error.to_dict(), error.status_code

# 模拟端点
def get_user(user_id):
    if not user_id.isdigit():
        raise BadRequest("用户ID必须是数字")
    user = find_user(int(user_id))
    if not user:
        raise NotFound(f"用户 {user_id} 不存在")
    return user
```

## 4. 异常链

### 保留原始异常

```python
def parse_config(filepath):
    try:
        with open(filepath) as f:
            import json
            return json.load(f)
    except FileNotFoundError as e:
        raise ConfigError(f"配置文件不存在: {filepath}") from e
    except json.JSONDecodeError as e:
        raise ConfigError(f"配置文件格式错误: {filepath}") from e

class ConfigError(Exception):
    pass
```

## 实战练习

### 练习 1：完整的验证异常

```python
class ValidationError(Exception):
    def __init__(self, field, message, value=None):
        self.field = field
        self.message = message
        self.value = value
        super().__init__(f"{field}: {message}")

class Validator:
    @staticmethod
    def required(value, field=""):
        if value is None or value == "":
            raise ValidationError(field, "必填")
        return value

    @staticmethod
    def min_length(value, length, field=""):
        if len(value) < length:
            raise ValidationError(field, f"最少{length}个字符", value)
        return value

    @staticmethod
    def range(value, min_val, max_val, field=""):
        if not (min_val <= value <= max_val):
            raise ValidationError(
                field,
                f"必须在 {min_val} 和 {max_val} 之间",
                value
            )
        return value

# 使用
try:
    Validator.required("", "用户名")
except ValidationError as e:
    print(f"字段: {e.field}, 错误: {e.message}")
```

### 练习 2：HTTP 状态码异常

```python
class HTTPException(Exception):
    status_code = 500

    def __init__(self, message=None):
        self.message = message or "服务器错误"
        super().__init__(self.message)

class HTTP_400(HTTPException):
    status_code = 400
    def __init__(self, message="请求参数错误"):
        super().__init__(message)

class HTTP_401(HTTPException):
    status_code = 401
    def __init__(self, message="未授权"):
        super().__init__(message)

class HTTP_404(HTTPException):
    status_code = 404
    def __init__(self, message="资源不存在"):
        super().__init__(message)

# 统一处理
def api_response(func):
    def wrapper(*args, **kwargs):
        try:
            result = func(*args, **kwargs)
            return {"code": 200, "data": result}
        except HTTPException as e:
            return {"code": e.status_code, "error": e.message}
    return wrapper

@api_response
def get_user(user_id):
    if user_id < 0:
        raise HTTP_400("用户ID不能为负数")
    if user_id == 999:
        raise HTTP_404()
    return {"id": user_id, "name": "Alice"}
```
