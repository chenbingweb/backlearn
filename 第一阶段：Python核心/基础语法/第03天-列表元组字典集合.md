## 1. 列表 (List)

### 创建列表

```python
# 空列表
empty = []
empty = list()

# 带初始值
numbers = [1, 2, 3, 4, 5]
mixed = [1, "hello", 3.14, True]
nested = [[1, 2], [3, 4], [5, 6]]
```

### 索引和切片

```python
fruits = ["apple", "banana", "cherry", "date", "elderberry"]

print(fruits[0])            # "apple"
print(fruits[-1])           # "elderberry"
print(fruits[1:3])           # ["banana", "cherry"]
print(fruits[:2])           # ["apple", "banana"]
print(fruits[3:])           # ["date", "elderberry"]
print(fruits[::2])          # ["apple", "cherry", "elderberry"]  步长为2
print(fruits[::-1])         # ["elderberry", "date", "cherry", "banana", "apple"]  反转
```

### 修改列表

```python
fruits = ["apple", "banana", "cherry"]

fruits[0] = "avocado"       # 修改元素
fruits.append("date")       # 末尾添加
fruits.insert(1, "apricot") # 指定位置插入
fruits.extend(["elderberry", "fig"])  # 末尾添加多个元素
fruits.remove("banana")     # 删除指定元素（第一个匹配）
fruits.pop()                # 删除末尾元素
fruits.pop(0)               # 删除指定位置元素
fruits.clear()              # 清空列表
```

### 列表操作

```python
# 合并列表
list1 = [1, 2, 3]
list2 = [4, 5, 6]
combined = list1 + list2    # [1, 2, 3, 4, 5, 6]

# 重复列表
repeated = [1, 2] * 3        # [1, 2, 1, 2, 1, 2]

# 检查元素是否存在
fruits = ["apple", "banana", "cherry"]
print("apple" in fruits)    # True
print("grape" in fruits)    # False
```

### 列表方法

```python
fruits = ["banana", "apple", "cherry", "banana"]

print(len(fruits))          # 4  长度
print(fruits.count("banana"))  # 2  计数
print(fruits.index("apple"))   # 1  查找索引
fruits.sort()               # 排序（原地修改）
fruits.reverse()           # 反转（原地修改）
sorted_fruits = sorted(fruits)  # 返回排序后的新列表
copy = fruits.copy()        # 浅拷贝
```

### 列表常用函数

```python
numbers = [1, 2, 3, 4, 5]

print(len(numbers))         # 5
print(max(numbers))         # 5
print(min(numbers))         # 1
print(sum(numbers))         # 15
print(any([0, False]))       # False  任意为真
print(all([1, 2, 3]))       # True   全部为真
```

## 2. 元组 (Tuple)

### 创建元组

```python
# 空元组
empty = ()
empty = tuple()

# 单元素元组（必须加逗号）
single = (1,)              # 不是 (1)

# 多元素元组
point = (1, 2, 3)
mixed = (1, "hello", 3.14)
nested = ((1, 2), (3, 4))

# 省略括号（打包）
coordinates = 1, 2, 3      # (1, 2, 3)
```

### 元组操作

```python
point = (1, 2, 3)

print(point[0])            # 1
print(point[-1])           # 3
print(point[0:2])          # (1, 2)

# 元组不可修改
# point[0] = 10            # TypeError

# 连接元组
point2 = (4, 5, 6)
combined = point + point2  # (1, 2, 3, 4, 5, 6)

# 重复元组
repeated = point * 2       # (1, 2, 3, 1, 2, 3)
```

### 元组方法

```python
point = (1, 2, 3, 2, 2)

print(len(point))          # 5
print(point.count(2))      # 3
print(point.index(2))     # 1  第一个匹配的索引
```

### 元组 vs 列表

| 特性 | 元组 | 列表 |
|------|------|------|
| 语法 | `(1, 2, 3)` | `[1, 2, 3]` |
| 可变性 | 不可修改 | 可修改 |
| 性能 | 更快 | 较慢 |
| 用途 | 固定数据 | 动态数据 |
| 字典键 | 可以 | 不可以 |

### 拆包

```python
# 基本拆包
point = (1, 2, 3)
x, y, z = point           # x=1, y=2, z=3

# 使用 * 进行拆包
a, *b, c = [1, 2, 3, 4, 5]  # a=1, b=[2,3,4], c=5

# 交换变量
a, b = b, a               # 交换

# 函数返回元组
def get_stats(numbers):
    return min(numbers), max(numbers), sum(numbers)

min_val, max_val, total = get_stats([1, 2, 3, 4, 5])
```

## 3. 字典 (Dict)

### 创建字典

```python
# 空字典
empty = {}
empty = dict()

# 基本字典
person = {
    "name": "Alice",
    "age": 25,
    "city": "Beijing"
}

# dict() 构造函数
person2 = dict(name="Bob", age=30, city="Shanghai")

# 从键值对列表创建
pairs = [["name", "Alice"], ["age", 25]]
person3 = dict(pairs)
```

### 访问字典

```python
person = {"name": "Alice", "age": 25, "city": "Beijing"}

print(person["name"])              # "Alice"
print(person.get("name"))           # "Alice"
print(person.get("country", "中国"))  # "中国"  默认值
```

### 修改字典

```python
person = {"name": "Alice", "age": 25}

person["name"] = "Bob"             # 修改值
person["city"] = "Shanghai"       # 添加新键值对
person.update({"age": 26, "city": "Beijing"})  # 批量更新
del person["city"]                 # 删除键值对
age = person.pop("age")           # 删除并返回值
person.clear()                     # 清空字典
```

### 字典方法

```python
person = {"name": "Alice", "age": 25, "city": "Beijing"}

print(len(person))                 # 3
print(person.keys())              # dict_keys(['name', 'age', 'city'])
print(person.values())            # dict_values(['Alice', 25, 'Beijing'])
print(person.items())             # dict_items([('name','Alice'), ...])
print(person.copy())              # 浅拷贝
print(person.setdefault("country", "China"))  # 设置默认值
```

### 遍历字典

```python
person = {"name": "Alice", "age": 25, "city": "Beijing"}

# 遍历键
for key in person:
    print(key)

for key in person.keys():
    print(key)

# 遍历值
for value in person.values():
    print(value)

# 遍历键值对
for key, value in person.items():
    print(f"{key}: {value}")
```

### 字典推导式

```python
# 基本用法
squares = {x: x**2 for x in range(5)}
# {0: 0, 1: 1, 2: 4, 3: 9, 4: 16}

# 带条件
even_squares = {x: x**2 for x in range(10) if x % 2 == 0}
# {0: 0, 2: 4, 4: 16, 6: 36, 8: 64}

# 交换键值
original = {"a": 1, "b": 2, "c": 3}
swapped = {v: k for k, v in original.items()}
# {1: 'a', 2: 'b', 3: 'c'}
```

### defaultdict

```python
from collections import defaultdict

# 字典值为列表
groups = defaultdict(list)
groups["fruits"].append("apple")
groups["fruits"].append("banana")
groups["vegetables"].append("carrot")
# {'fruits': ['apple', 'banana'], 'vegetables': ['carrot']}

# 字典值为整数（计数）
counts = defaultdict(int)
words = ["apple", "banana", "apple", "cherry", "banana", "apple"]
for word in words:
    counts[word] += 1
# {'apple': 3, 'banana': 2, 'cherry': 1}
```

## 4. 集合 (Set)

### 创建集合

```python
# 空集合
empty = set()

# 基本集合
fruits = {"apple", "banana", "cherry"}

# 从列表创建（去重）
numbers = [1, 2, 2, 3, 3, 4, 5]
unique_numbers = set(numbers)  # {1, 2, 3, 4, 5}
```

### 集合操作

```python
set1 = {1, 2, 3, 4, 5}
set2 = {4, 5, 6, 7, 8}

# 并集
print(set1 | set2)              # {1, 2, 3, 4, 5, 6, 7, 8}
print(set1.union(set2))        # 同上

# 交集
print(set1 & set2)              # {4, 5}
print(set1.intersection(set2))  # 同上

# 差集
print(set1 - set2)              # {1, 2, 3}
print(set1.difference(set2))    # 同上

# 对称差集（不同时出现在两个集合中）
print(set1 ^ set2)              # {1, 2, 3, 6, 7, 8}
print(set1.symmetric_difference(set2))  # 同上
```

### 修改集合

```python
fruits = {"apple", "banana", "cherry"}

fruits.add("date")              # 添加元素
fruits.update(["elderberry", "fig"])  # 添加多个元素
fruits.remove("banana")        # 删除元素（不存在会报错）
fruits.discard("grape")        # 删除元素（不存在不报错）
fruits.pop()                   # 删除并返回任意元素
fruits.clear()                 # 清空集合
```

### 集合方法

```python
set1 = {1, 2, 3}
set2 = {1, 2, 4}

print(len(set1))               # 3
print(2 in set1)              # True
print(set1.issubset(set2))    # False  set1 是否是 set2 的子集
print(set1.issuperset(set2))  # False  set1 是否是 set2 的超集
print(set1.isdisjoint(set2))  # False  是否有交集
```

### frozenset（不可变集合）

```python
immutable_fruits = frozenset(["apple", "banana", "cherry"])
# immutable_fruits.add("date")  # 报错，frozenset 不可修改

# 可以作为字典键
mapping = {immutable_fruits: "fruits"}
```

## 5. 数据结构选择

| 场景 | 推荐数据结构 |
|------|-------------|
| 有序列表，可修改 | `list` |
| 固定数据，只读 | `tuple` |
| 键值对映射 | `dict` |
| 去重，无顺序 | `set` |
| 不可变的键值对 | `frozenset` |

## 6. 实战练习

### 练习 1：列表去重

```python
numbers = [1, 2, 2, 3, 3, 3, 4, 4, 4, 5]
# 方法1：使用 set
unique = list(set(numbers))
print(unique)  # [1, 2, 3, 4, 5]

# 方法2：保留顺序的去重
seen = []
for num in numbers:
    if num not in seen:
        seen.append(num)
print(seen)  # [1, 2, 3, 4, 5]
```

### 练习 2：词频统计

```python
text = "apple banana apple cherry banana apple"
words = text.split()

# 使用 defaultdict
from collections import defaultdict
counts = defaultdict(int)
for word in words:
    counts[word] += 1
print(dict(counts))  # {'apple': 3, 'banana': 2, 'cherry': 1}
```

### 练习 3：两个集合的交集、并集、差集

```python
set_a = {1, 2, 3, 4, 5}
set_b = {4, 5, 6, 7, 8}

print(f"交集: {set_a & set_b}")
print(f"并集: {set_a | set_b}")
print(f"差集 (A-B): {set_a - set_b}")
print(f"差集 (B-A): {set_b - set_a}")
```
