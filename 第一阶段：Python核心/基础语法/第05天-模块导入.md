## 1. 模块基础

### 什么是模块

模块是一个 `.py` 文件，包含 Python 代码（函数、类、变量等）。用于组织代码，提高复用性。

### 导入整个模块

```python
import math

print(math.pi)        # 3.141592653589793
print(math.sqrt(16))  # 4.0
print(math.pow(2, 3)) # 8.0
```

### 导入特定内容

```python
from math import pi, sqrt

print(pi)             # 3.141592653589793
print(sqrt(16))       # 4.0
```

### 使用别名

```python
import math as m
from math import sqrt as s

print(m.pi)           # 3.141592653589793
print(s(16))          # 4.0
```

### 导入所有内容（不推荐）

```python
from math import *  # 导入所有内容到当前命名空间

print(pi)            # 3.141592653589793
print(sqrt(16))      # 4.0
```

### 模块搜索路径

```python
import sys
print(sys.path)
# 包含：
# 1. 当前目录
# 2. PYTHONPATH 环境变量
# 3. Python 安装目录
# 4. 第三方包目录
```

## 2. 自定义模块

### 创建模块

my_module.py:
```python
"""我的工具模块"""

def greet(name):
    """问候函数"""
    return f"Hello, {name}!"

def add(a, b):
    """加法函数"""
    return a + b

VERSION = "1.0.0"
```

### 使用自定义模块

```python
import my_module

print(my_module.greet("Alice"))  # "Hello, Alice!"
print(my_module.add(1, 2))      # 3
print(my_module.VERSION)         # "1.0.0"
```

### 模块的 `__name__`

```python
# my_module.py
def greet(name):
    return f"Hello, {name}!"

if __name__ == "__main__":
    # 只有直接运行此文件时才执行
    print(greet("World"))
else:
    # 被导入时不执行
    pass
```

## 3. 包 (Package)

### 创建包

```
my_package/
    __init__.py          # 包初始化文件
    module1.py           # 模块1
    module2.py           # 模块2
    sub_package/
        __init__.py
        module3.py
```

### `__init__.py`

```python
# my_package/__init__.py
from .module1 import function1
from .module2 import function2

__all__ = ["function1", "function2"]  # 指定公开接口
VERSION = "1.0.0"
```

### 导入包

```python
import my_package
my_package.function1()

from my_package import function1
function1()

from my_package import function1 as f1
f1()
```

## 4. 常用标准库模块

### os 模块（操作系统）

```python
import os

os.getcwd()                  # 获取当前目录
os.chdir("/path/to/dir")     # 切换目录
os.listdir(".")              # 列出目录内容
os.mkdir("new_dir")          # 创建目录
os.rmdir("dir")              # 删除空目录
os.remove("file.txt")        # 删除文件
os.path.exists("file.txt")   # 检查路径是否存在
os.path.join("a", "b")       # 拼接路径
os.path.split("/a/b/file.txt")  # 分割路径
os.path.basename("/a/b/file.txt")  # 获取文件名
os.path.dirname("/a/b/file.txt")   # 获取目录名
```

### sys 模块（系统）

```python
import sys

sys.argv                    # 命令行参数列表
sys.version                 # Python 版本
sys.path                    # 模块搜索路径
sys.exit(0)                 # 退出程序
sys.stdout                  # 标准输出
sys.stdin                   # 标准输入
```

### datetime 模块（日期时间）

```python
from datetime import datetime, date, time, timedelta

# 获取当前时间
now = datetime.now()
print(now)                   # 2024-01-15 10:30:45.123456

# 格式化
print(now.strftime("%Y-%m-%d %H:%M:%S"))

# 解析字符串
dt = datetime.strptime("2024-01-15 10:30:00", "%Y-%m-%d %H:%M:%S")

# 日期计算
tomorrow = now + timedelta(days=1)
yesterday = now - timedelta(days=1)

# 日期属性
print(now.year)             # 2024
print(now.month)            # 1
print(now.day)              # 15
print(now.hour)             # 10
```

### json 模块（JSON）

```python
import json

# Python 对象转 JSON 字符串
data = {"name": "Alice", "age": 25, "scores": [90, 85, 88]}
json_str = json.dumps(data, indent=2, ensure_ascii=False)
print(json_str)

# JSON 字符串转 Python 对象
parsed = json.loads(json_str)
print(parsed["name"])       # "Alice"

# 写入文件
with open("data.json", "w", encoding="utf-8") as f:
    json.dump(data, f, indent=2, ensure_ascii=False)

# 读取文件
with open("data.json", "r", encoding="utf-8") as f:
    loaded = json.load(f)
```

### random 模块（随机数）

```python
import random

random.random()             # 0-1 之间的随机浮点数
random.randint(1, 100)      # 1-100 之间的随机整数
random.choice(["a", "b", "c"])  # 随机选择
random.shuffle([1, 2, 3, 4, 5]) # 打乱顺序
random.sample([1, 2, 3, 4, 5], 3)  # 随机选3个不重复
```

### collections 模块

```python
from collections import Counter, defaultdict, OrderedDict

# Counter 计数
cnt = Counter(["a", "b", "a", "c", "a", "b"])
print(cnt)                  # Counter({'a': 3, 'b': 2, 'c': 1})
print(cnt.most_common(2))   # [('a', 3), ('b', 2)]

# defaultdict 默认字典
dd = defaultdict(int)
dd["key"] += 1
print(dd)                   # defaultdict(<class 'int'>, {'key': 1})

# OrderedDict 有序字典（Python 3.7+ 普通字典已有序）
od = OrderedDict()
od["a"] = 1
od["b"] = 2
```

## 6. 实战练习

### 练习 1：创建工具模块

创建 my_utils.py：
```python
"""工具模块"""

def calculate_stats(numbers):
    """计算统计信息"""
    return {
        "count": len(numbers),
        "sum": sum(numbers),
        "average": sum(numbers) / len(numbers) if numbers else 0,
        "min": min(numbers) if numbers else None,
        "max": max(numbers) if numbers else None
    }

def format_time(seconds):
    """格式化时间"""
    minutes = seconds // 60
    secs = seconds % 60
    return f"{minutes}分{secs}秒"
```

使用：
```python
import my_utils

stats = my_utils.calculate_stats([1, 2, 3, 4, 5])
print(stats)
print(my_utils.format_time(125))
```

### 练习 2：使用 json 和 os

```python
import json
import os

def save_data(filename, data):
    """保存数据到 JSON 文件"""
    with open(filename, "w", encoding="utf-8") as f:
        json.dump(data, f, indent=2, ensure_ascii=False)

def load_data(filename):
    """从 JSON 文件加载数据"""
    if os.path.exists(filename):
        with open(filename, "r", encoding="utf-8") as f:
            return json.load(f)
    return None
```
