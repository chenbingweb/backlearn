## 1. 条件语句

### 基础 if 语句

```python
age = 18

if age >= 18:
    print("已成年")
```

### if-else 语句

```python
age = 16

if age >= 18:
    print("已成年")
else:
    print("未成年")
```

### if-elif-else 语句

```python
score = 85

if score >= 90:
    print("优秀")
elif score >= 80:
    print("良好")
elif score >= 60:
    print("及格")
else:
    print("不及格")
```

### 嵌套条件

```python
age = 25
has_ticket = True

if age >= 18:
    if has_ticket:
        print("允许入场")
    else:
        print("需要购票")
else:
    print("未成年禁止入场")
```

### 简化的条件表达式

```python
# 三元运算符
age = 20
status = "成年" if age >= 18 else "未成年"
print(status)           # "成年"
```

## 2. 循环语句

### while 循环

```python
# 基本语法
count = 0
while count < 5:
    print(count)
    count += 1

# while 循环计算 1-100 的和
sum_num = 0
i = 1
while i <= 100:
    sum_num += i
    i += 1
print(f"1+2+...+100 = {sum_num}")
```

### for 循环

```python
# 遍历列表
fruits = ["apple", "banana", "orange"]
for fruit in fruits:
    print(fruit)

# 遍历字符串
for char in "Python":
    print(char)

# 使用 range()
for i in range(5):
    print(i)            # 0, 1, 2, 3, 4

# range(start, stop, step)
for i in range(0, 10, 2):
    print(i)            # 0, 2, 4, 6, 8

# 遍历字典
person = {"name": "Alice", "age": 25, "city": "Beijing"}
for key in person:
    print(f"{key}: {person[key]}")

# 遍历键值对
for key, value in person.items():
    print(f"{key}: {value}")
```

### enumerate 函数

```python
fruits = ["apple", "banana", "orange"]
for index, fruit in enumerate(fruits):
    print(f"{index}: {fruit}")
# 输出:
# 0: apple
# 1: banana
# 2: orange
```

### zip 函数

```python
names = ["Alice", "Bob", "Charlie"]
ages = [25, 30, 35]

for name, age in zip(names, ages):
    print(f"{name}: {age}")
# 输出:
# Alice: 25
# Bob: 30
# Charlie: 35
```

## 3. 循环控制语句

### break 语句

```python
# 找到第一个偶数
numbers = [1, 3, 5, 6, 7, 9]
for num in numbers:
    if num % 2 == 0:
        print(f"找到第一个偶数: {num}")
        break
```

### continue 语句

```python
# 跳过所有奇数
for i in range(1, 10):
    if i % 2 == 1:
        continue
    print(i)            # 2, 4, 6, 8
```

### else 语句（循环后执行）

```python
# 循环正常结束才会执行 else
numbers = [1, 3, 5, 7, 9]
for num in numbers:
    if num % 2 == 0:
        print("找到偶数")
        break
else:
    print("没有找到偶数")

# 如果找到偶数，不会打印"没有找到偶数"
# 如果没找到偶数，才会打印"没有找到偶数"
```

### pass 语句

```python
# 占位符，不做任何事情
for i in range(5):
    pass                # 以后再实现

# 用于空函数或空类
def future_function():
    pass

class FutureClass:
    pass
```

## 4. 嵌套循环

```python
# 打印九九乘法表
for i in range(1, 10):

        print(f"{j}x{i}={i*j}", end="\t")
    print()
```

## 5. 列表推导式

### 基本语法

```python
# 传统方式
squares = []
for x in range(10):
    squares.append(x ** 2)

# 列表推导式
squares = [x ** 2 for x in range(10)]
```

### 带条件

```python
# 只保留偶数
even_squares = [x ** 2 for x in range(10) if x % 2 == 0]
# [0, 4, 16, 36, 64]
```

### 多层循环

```python
# 生成所有两两组合
pairs = [(a, b) for a in [1, 2, 3] for b in [4, 5, 6]]
# [(1,4), (1,5), (1,6), (2,4), (2,5), (2,6), (3,4), (3,5), (3,6)]
```

### 字典推导式

```python
words = ["apple", "banana", "cherry"]
word_lengths = {word: len(word) for word in words}
# {'apple': 5, 'banana': 6, 'cherry': 6}
```

## 6. 实战练习

### 练习 1：猜数字游戏

```python
import random

target = random.randint(1, 100)
attempts = 0

while True:
    guess = int(input("请猜一个 1-100 之间的数字: "))
    attempts += 1

    if guess == target:
        print(f"恭喜猜对了！共猜测 {attempts} 次")
        break
    elif guess < target:
        print("太小了，再试试")
    else:
        print("太大了，再试试")
```

### 练习 2：判断素数

```python
num = int(input("请输入一个正整数: "))

if num < 2:
    print(f"{num} 不是素数")
else:
    is_prime = True
    for i in range(2, int(num ** 0.5) + 1):
        if num % i == 0:
            is_prime = False
            break

    if is_prime:
        print(f"{num} 是素数")
    else:
        print(f"{num} 不是素数")
```

### 练习 3：斐波那契数列

```python
n = int(input("请输入斐波那契数列的项数: "))

fibonacci = []
a, b = 0, 1
for i in range(n):
    fibonacci.append(a)
    a, b = b, a + b

print(f"斐波那契数列前 {n} 项: {fibonacci}")
```
