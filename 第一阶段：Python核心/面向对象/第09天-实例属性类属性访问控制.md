## 1. 实例属性 vs 类属性

### 实例属性

```python
class Dog:
    def __init__(self, name):
        self.name = name  # 实例属性

dog1 = Dog("Buddy")
dog2 = Dog("Max")

print(dog1.name)  # Buddy
print(dog2.name)  # Max

# 修改互不影响
dog1.name = "Buddy2"
print(dog1.name)  # Buddy2
print(dog2.name)  # Max
```

### 类属性

```python
class Dog:
    species = "Canis familiaris"  # 类属性
    count = 0                      # 类属性

    def __init__(self, name):
        self.name = name           # 实例属性
        Dog.count += 1             # 修改类属性

dog1 = Dog("Buddy")
dog2 = Dog("Max")

print(Dog.species)   # Canis familiaris
print(dog1.species)  # Canis familiaris（实例可访问类属性）
print(Dog.count)     # 2
```

### 关键区别

```python
class Example:
    class_attr = []  # 类属性

    def __init__(self):
        self.instance_attr = []  # 实例属性

a = Example()
b = Example()

# 类属性共享
a.class_attr.append(1)
print(b.class_attr)  # [1]  ← b 也能看到！

# 实例属性独立
a.instance_attr.append(1)
print(b.instance_attr)  # []
```

### 正确的类属性用法

```python
class Dog:
    species = "Canis familiaris"  # 所有狗共有的属性

    def __init__(self, name):
        self.name = name          # 每只狗不同的属性
        self.tricks = []          # 每只狗独立的列表

dog1 = Dog("Buddy")
dog2 = Dog("Max")

dog1.tricks.append("sit")
print(dog1.tricks)  # ['sit']
print(dog2.tricks)  # []
```

## 2. 访问控制

### Python 的命名约定

| 命名方式 | 含义 | 访问权限 |
|---------|------|---------|
| `name` | 公有 | 任意访问 |
| `_name` | 保护（约定） | 建议不要外部访问 |
| `__name` | 私有（名称改写） | 类内部访问 |

### 公有属性

```python
class Person:
    def __init__(self, name):
        self.name = name  # 公有

p = Person("Alice")
print(p.name)    # Alice
p.name = "Bob"   # 随意修改
print(p.name)    # Bob
```

### 保护属性（单下划线）

```python
class Person:
    def __init__(self, name, age):
        self.name = name
        self._age = age  # 保护属性（约定）

p = Person("Alice", 25)
print(p._age)     # 25（可以访问，但不建议）
p._age = 26       # 可以修改，但不建议
```

### 私有属性（双下划线）

```python
class Person:
    def __init__(self, name, password):
        self.name = name
        self.__password = password  # 私有属性

    def check_password(self, pwd):
        return self.__password == pwd

p = Person("Alice", "secret123")

print(p.name)           # Alice
# print(p.__password)   # AttributeError!

# 名称改写后仍可访问（不要这样做）
print(p._Person__password)  # secret123

# 正确方式
print(p.check_password("wrong"))     # False
print(p.check_password("secret123")) # True
```

## 3. @property 装饰器

### 为什么需要 @property

```python
# 问题：直接暴露属性
class Circle:
    def __init__(self, radius):
        self.radius = radius

c = Circle(5)
c.radius = -10  # 不合理的值！
```

### 使用 getter/setter

```python
class Circle:
    def __init__(self, radius):
        self._radius = radius

    @property
    def radius(self):
        """getter"""
        return self._radius

    @radius.setter
    def radius(self, value):
        """setter"""
        if value <= 0:
            raise ValueError("半径必须大于0")
        self._radius = value

    @property
    def area(self):
        """只读属性"""
        return 3.14159 * self._radius ** 2

c = Circle(5)
print(c.radius)   # 5
print(c.area)     # 78.53975

c.radius = 10     # 通过 setter
print(c.area)     # 314.159

# c.radius = -5   # ValueError!
# c.area = 100    # AttributeError! 只读
```

### 完整示例

```python
class Temperature:
    def __init__(self, celsius=0):
        self._celsius = celsius

    @property
    def celsius(self):
        return self._celsius

    @celsius.setter
    def celsius(self, value):
        self._celsius = value

    @property
    def fahrenheit(self):
        return self._celsius * 9 / 5 + 32

    @fahrenheit.setter
    def fahrenheit(self, value):
        self._celsius = (value - 32) * 5 / 9

    @property
    def kelvin(self):
        return self._celsius + 273.15

t = Temperature(25)
print(f"{t.celsius}°C = {t.fahrenheit}°F")
t.fahrenheit = 212
print(f"{t.celsius}°C = {t.fahrenheit}°F")  # 100.0°C = 212.0°F
```

### @property 实现只读属性

```python
class Employee:
    def __init__(self, name, salary):
        self.name = name
        self._salary = salary

    @property
    def salary(self):
        """只读属性"""
        return self._salary

    def give_raise(self, amount):
        """通过方法修改"""
        if amount > 0:
            self._salary += amount

e = Employee("Alice", 5000)
print(e.salary)      # 5000
# e.salary = 6000    # AttributeError!
e.give_raise(500)
print(e.salary)      # 5500
```

## 4. 私有方法

```python
class BankAccount:
    def __init__(self, balance=0):
        self.__balance = balance

    def deposit(self, amount):
        if amount > 0:
            self.__balance += amount
            self.__log(f"存款: +{amount}")

    def withdraw(self, amount):
        if 0 < amount <= self.__balance:
            self.__balance -= amount
            self.__log(f"取款: -{amount}")
            return amount
        return 0

    def __log(self, message):
        """私有方法"""
        print(f"[LOG] {message}")

    @property
    def balance(self):
        return self.__balance

account = BankAccount(1000)
account.deposit(500)
account.withdraw(200)
print(account.balance)
# account.__log("test")  # AttributeError!
```

## 5. __slots__

### 限制属性

```python
class Person:
    __slots__ = ['name', 'age']  # 只允许这些属性

    def __init__(self, name, age):
        self.name = name
        self.age = age

p = Person("Alice", 25)
print(p.name)

# p.city = "Beijing"  # AttributeError!
```

### 优点

- 节省内存
- 防止随意添加属性
- 访问速度更快

## 实战练习

### 练习 1：安全的银行账户

```python
class BankAccount:
    interest_rate = 0.05  # 类属性：利率

    def __init__(self, owner, balance=0):
        self.owner = owner
        self.__balance = balance  # 私有
        self.__transactions = []   # 私有

    @property
    def balance(self):
        return self.__balance

    def deposit(self, amount):
        if amount <= 0:
            raise ValueError("存款金额必须大于0")
        self.__balance += amount
        self.__transactions.append(f"+{amount}")
        return self.__balance

    def withdraw(self, amount):
        if amount <= 0:
            raise ValueError("取款金额必须大于0")
        if amount > self.__balance:
            raise ValueError("余额不足")
        self.__balance -= amount
        self.__transactions.append(f"-{amount}")
        return amount

    def apply_interest(self):
        interest = self.__balance * self.interest_rate
        self.__balance += interest
        self.__transactions.append(f"利息:{interest:.2f}")

    def history(self):
        return self.__transactions.copy()

# 使用
acc = BankAccount("Alice", 1000)
acc.deposit(500)
acc.withdraw(200)
acc.apply_interest()
print(f"余额: {acc.balance}")
print(f"流水: {acc.history()}")
```

### 练习 2：矩形类

```python
class Rectangle:
    def __init__(self, width=1, height=1):
        self.width = width
        self.height = height

    @property
    def width(self):
        return self._width

    @width.setter
    def width(self, value):
        if value <= 0:
            raise ValueError("宽度必须大于0")
        self._width = value

    @property
    def height(self):
        return self._height

    @height.setter
    def height(self, value):
        if value <= 0:
            raise ValueError("高度必须大于0")
        self._height = value

    @property
    def area(self):
        return self._width * self._height

    @property
    def perimeter(self):
        return 2 * (self._width + self._height)

    def __str__(self):
        return f"Rectangle({self.width}x{self.height})"

r = Rectangle(5, 3)
print(r)              # Rectangle(5x3)
print(r.area)         # 15
print(r.perimeter)    # 16
r.width = 10
print(r.area)         # 30
```

## 今日总结

- [ ] 实例属性每个实例独立，类属性所有实例共享

- [ ] 私有方法用双下划线开头
