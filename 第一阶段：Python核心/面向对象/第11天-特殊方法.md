## 1. 什么是特殊方法

特殊方法（Magic Methods / Dunder Methods）是双下划线开头和结尾的方法，如 `__init__`。

它们让你在自定义类中实现 Python 内置类型的行为。

## 2. 字符串表示

### __str__ vs __repr__

```python
class Person:
    def __init__(self, name, age):
        self.name = name
        self.age = age

    def __str__(self):
        """用户友好的字符串表示"""
        return f"{self.name} ({self.age}岁)"

    def __repr__(self):
        """开发者友好的字符串表示"""
        return f"Person(name='{self.name}', age={self.age})"

p = Person("Alice", 25)
print(str(p))    # Alice (25岁)
print(repr(p))   # Person(name='Alice', age=25)
print(p)         # Alice (25岁)  ← 调用 __str__
```

### 区别

| 方法 | 用途 | 调用方式 |
|-----|------|---------|
| `__str__` | 给用户看 | `str(obj)`, `print(obj)` |
| `__repr__` | 给开发者看 | `repr(obj)`, 交互式输出 |

**最佳实践**：`__repr__` 应该返回可以重新创建对象的字符串。

## 3. 比较运算符

```python
class Rectangle:
    def __init__(self, width, height):
        self.width = width
        self.height = height

    @property
    def area(self):
        return self.width * self.height

    # 等于
    def __eq__(self, other):
        if not isinstance(other, Rectangle):
            return NotImplemented
        return self.area == other.area

    # 小于
    def __lt__(self, other):
        if not isinstance(other, Rectangle):
            return NotImplemented
        return self.area < other.area

    # 小于等于
    def __le__(self, other):
        return self == other or self < other

    def __repr__(self):
        return f"Rectangle({self.width}x{self.height})"

r1 = Rectangle(4, 5)   # area=20
r2 = Rectangle(2, 10)  # area=20
r3 = Rectangle(3, 4)   # area=12

print(r1 == r2)   # True
print(r1 > r3)    # True（Python 自动推导 __gt__）
print(r1 >= r3)   # True
```

### 自动推导的关系

实现部分比较方法后，Python 可以自动推导其他方法：

| 你实现 | Python 推导 |
|-------|-----------|
| `__eq__` | |
| `__ne__` | 自动从 `__eq__` 推导 |
| `__lt__` | `__gt__`（交换操作数） |
| `__le__` | `__ge__`（交换操作数） |

**推荐**：使用 `@functools.total_ordering` 简化。

```python
from functools import total_ordering

@total_ordering
class Rectangle:
    def __init__(self, width, height):
        self.width = width
        self.height = height

    @property
    def area(self):
        return self.width * self.height

    def __eq__(self, other):
        return self.area == other.area

    def __lt__(self, other):
        return self.area < other.area
```

## 4. 算术运算符

```python
class Vector:
    def __init__(self, x, y):
        self.x = x
        self.y = y

    # 加法：vector + other
    def __add__(self, other):
        return Vector(self.x + other.x, self.y + other.y)

    # 减法：vector - other
    def __sub__(self, other):
        return Vector(self.x - other.x, self.y - other.y)

    # 乘法：vector * scalar
    def __mul__(self, scalar):
        return Vector(self.x * scalar, self.y * scalar)

    # 反向乘法：scalar * vector
    def __rmul__(self, scalar):
        return self.__mul__(scalar)

    # 长度
    def __abs__(self):
        return (self.x ** 2 + self.y ** 2) ** 0.5

    # 负数
    def __neg__(self):
        return Vector(-self.x, -self.y)

    def __str__(self):
        return f"Vector({self.x}, {self.y})"

    def __repr__(self):
        return f"Vector({self.x}, {self.y})"

v1 = Vector(1, 2)
v2 = Vector(3, 4)

print(v1 + v2)      # Vector(4, 6)
print(v2 - v1)      # Vector(2, 2)
print(v1 * 3)       # Vector(3, 6)
print(3 * v1)       # Vector(3, 6)  ← __rmul__
print(abs(v1))      # 2.236...
```

## 5. 容器类型方法

```python
class Deck:
    """一副牌"""
    def __init__(self):
        self._cards = []

    # 长度
    def __len__(self):
        return len(self._cards)

    # 索引访问
    def __getitem__(self, index):
        return self._cards[index]

    # 切片赋值
    def __setitem__(self, index, value):
        self._cards[index] = value

    # 删除
    def __delitem__(self, index):
        del self._cards[index]

    # 迭代
    def __iter__(self):
        return iter(self._cards)

    # 包含检查
    def __contains__(self, item):
        return item in self._cards

    def add(self, card):
        self._cards.append(card)

deck = Deck()
for i in range(1, 6):
    deck.add(f"Card-{i}")

print(len(deck))        # 5
print(deck[0])          # Card-1
print(deck[-1])         # Card-5
print("Card-3" in deck) # True

for card in deck:
    print(card)
```

## 6. 对象生命周期

```python
class DatabaseConnection:
    def __new__(cls, *args, **kwargs):
        """创建实例（很少需要重写）"""
        print("1. __new__: 创建对象")
        instance = super().__new__(cls)
        return instance

    def __init__(self, host):
        """初始化实例"""
        print("2. __init__: 初始化对象")
        self.host = host
        self.connected = False

    def connect(self):
        self.connected = True
        print(f"连接到 {self.host}")

    def __del__(self):
        """析构方法（对象被销毁时调用）"""
        print("3. __del__: 对象被销毁")
        if self.connected:
            print("断开连接")

# 使用
db = DatabaseConnection("localhost")
db.connect()
del db  # 触发 __del__
```

## 7. 上下文管理器

```python
class FileManager:
    def __init__(self, filename, mode):
        self.filename = filename
        self.mode = mode
        self.file = None

    def __enter__(self):
        """进入 with 块时调用"""
        self.file = open(self.filename, self.mode)
        print(f"打开文件: {self.filename}")
        return self.file

    def __exit__(self, exc_type, exc_val, exc_tb):
        """退出 with 块时调用"""
        if self.file:
            self.file.close()
            print(f"关闭文件: {self.filename}")
        # 返回 True 抑制异常
        return False

# 使用
with FileManager("test.txt", "w") as f:
    f.write("Hello")
# 自动关闭文件
```

## 8. 可调用对象

```python
class Counter:
    def __init__(self, start=0):
        self.count = start

    def __call__(self, step=1):
        """让对象可以像函数一样调用"""
        self.count += step
        return self.count

    def __str__(self):
        return f"Counter({self.count})"

counter = Counter(10)
print(counter())      # 11
print(counter())      # 12
print(counter(5))     # 17
print(counter)        # Counter(17)
```

## 9. 属性访问控制

```python
class Config:
    def __init__(self):
        self._data = {}

    def __getattr__(self, name):
        """访问不存在的属性时调用"""
        return self._data.get(name, f"'{name}' 未设置")

    def __setattr__(self, name, value):
        """设置属性时调用"""
        if name.startswith("_"):
            super().__setattr__(name, value)
        else:
            self._data[name] = value

    def __getitem__(self, key):
        return self._data.get(key)

config = Config()
config.debug = True
config.host = "localhost"

print(config.debug)   # True
print(config.host)    # localhost
print(config.port)    # 'port' 未设置
```

## 实战练习

### 练习 1：自定义列表

```python
class MyList:
    def __init__(self, items=None):
        self._items = list(items) if items else []

    def __len__(self):
        return len(self._items)

    def __getitem__(self, index):
        return self._items[index]

    def __setitem__(self, index, value):
        self._items[index] = value

    def __iter__(self):
        return iter(self._items)

    def __contains__(self, item):
        return item in self._items

    def __add__(self, other):
        return MyList(self._items + list(other))

    def __str__(self):
        return str(self._items)

    def append(self, item):
        self._items.append(item)

ml = MyList([1, 2, 3])
print(len(ml))        # 3
print(ml[0])          # 1
print(2 in ml)        # True

ml2 = ml + [4, 5]
print(ml2)            # [1, 2, 3, 4, 5]
```

### 练习 2：分数类

```python
import math

class Fraction:
    def __init__(self, numerator, denominator):
        if denominator == 0:
            raise ValueError("分母不能为0")
        g = math.gcd(numerator, denominator)
        self.numerator = numerator // g
        self.denominator = denominator // g
        if self.denominator < 0:
            self.numerator = -self.numerator
            self.denominator = -self.denominator

    def __str__(self):
        if self.denominator == 1:
            return str(self.numerator)
        return f"{self.numerator}/{self.denominator}"

    def __repr__(self):
        return f"Fraction({self.numerator}, {self.denominator})"

    def __eq__(self, other):
        return (self.numerator == other.numerator and
                self.denominator == other.denominator)

    def __add__(self, other):
        n = self.numerator * other.denominator + other.numerator * self.denominator
        d = self.denominator * other.denominator
        return Fraction(n, d)

    def __sub__(self, other):
        n = self.numerator * other.denominator - other.numerator * self.denominator
        d = self.denominator * other.denominator
        return Fraction(n, d)

    def __mul__(self, other):
        return Fraction(self.numerator * other.numerator,
                       self.denominator * other.denominator)

    def __truediv__(self, other):
        return Fraction(self.numerator * other.denominator,
                       self.denominator * other.numerator)

    def __float__(self):
        return self.numerator / self.denominator

f1 = Fraction(1, 2)
f2 = Fraction(1, 3)

print(f"{f1} + {f2} = {f1 + f2}")   # 1/2 + 1/3 = 5/6
print(f"{f1} - {f2} = {f1 - f2}")   # 1/2 - 1/3 = 1/6
print(f"{f1} * {f2} = {f1 * f2}")   # 1/2 * 1/3 = 1/6
print(f"{f1} / {f2} = {f1 / f2}")   # 1/2 / 1/3 = 3/2
print(float(f1))                     # 0.5
```

## 特殊方法速查表

| 类别 | 方法 | 说明 |
|-----|------|------|
| 初始化 | `__init__` | 构造函数 |
| 字符串 | `__str__`, `__repr__` | 字符串表示 |
| 比较 | `__eq__`, `__lt__`, `__gt__` | 比较运算符 |
| 算术 | `__add__`, `__sub__`, `__mul__` | 算术运算符 |
| 容器 | `__len__`, `__getitem__`, `__contains__` | 容器操作 |
| 生命周期 | `__new__`, `__del__` | 创建和销毁 |
| 上下文 | `__enter__`, `__exit__` | with 语句 |
| 可调用 | `__call__` | 函数式调用 |
| 类型转换 | `__int__`, `__float__`, `__bool__` | 类型转换 |
