## 1. 带参数的装饰器详解

### 三层嵌套结构

```python
import functools

# 装饰器工厂（接收装饰器参数）
def repeat(times=2):
    # 装饰器（接收被装饰函数）
    def decorator(func):
        # 包装函数（接收原函数参数）
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            for _ in range(times):
                result = func(*args, **kwargs)
            return result
        return wrapper
    return decorator

@repeat(times=3)
def greet(name):
    print(f"Hello, {name}!")

greet("Alice")
# Hello, Alice!
# Hello, Alice!
# Hello, Alice!
```

### 带参数的计时装饰器

```python
import time

def timer(prefix="[TIMER]"):
    def decorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            start = time.time()
            result = func(*args, **kwargs)
            elapsed = time.time() - start
            print(f"{prefix} {func.__name__}: {elapsed:.4f}s")
            return result
        return wrapper
    return decorator

@timer(prefix="[性能]")
def slow_op():
    time.sleep(0.5)
    return "done"

slow_op()  # [性能] slow_op: 0.5012s
```

## 2. 方法装饰器

### 实例方法装饰器

```python
def auth_required(func):
    """要求登录的装饰器"""
    @functools.wraps(func)
    def wrapper(self, *args, **kwargs):
        if not getattr(self, 'is_authenticated', False):
            raise PermissionError("请先登录")
        return func(self, *args, **kwargs)
    return wrapper

class UserService:
    def __init__(self):
        self.is_authenticated = False

    def login(self, password):
        if password == "correct":
            self.is_authenticated = True

    @auth_required
    def get_secret_data(self):
        return "绝密数据"

service = UserService()
# service.get_secret_data()  # PermissionError!

service.login("correct")
print(service.get_secret_data())  # 绝密数据
```

### 通用方法装饰器

```python
def method_timer(func):
    """适用于方法的计时装饰器"""
    @functools.wraps(func)
    def wrapper(self, *args, **kwargs):
        import time
        start = time.time()
        result = func(self, *args, **kwargs)
        print(f"{self.__class__.__name__}.{func.__name__}: "
              f"{time.time() - start:.4f}s")
        return result
    return wrapper

class DataProcessor:
    @method_timer
    def process(self, data):
        import time
        time.sleep(0.1)
        return [x * 2 for x in data]

processor = DataProcessor()
processor.process([1, 2, 3])
# DataProcessor.process: 0.1012s
```

### 保留函数签名

```python
import inspect

def bad_decorator(func):
    def wrapper(*args, **kwargs):
        return func(*args, **kwargs)
    return wrapper

def good_decorator(func):
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        return func(*args, **kwargs)
    return wrapper

@bad_decorator
def f1(x: int, y: str) -> bool:
    pass

@good_decorator
def f2(x: int, y: str) -> bool:
    pass

print(inspect.signature(f1))  # (*args, **kwargs)  ← 签名丢失了！
print(inspect.signature(f2))  # (x: int, y: str) -> bool
```

## 5. 参数验证装饰器（完整版）

```python
import functools
import inspect

def validate(**validators):
    """
    参数验证装饰器

    用法:
    @validate(x=lambda v: v > 0, y=lambda v: isinstance(v, str))
    def func(x, y):
        pass
    """
    def decorator(func):
        sig = inspect.signature(func)
        param_names = list(sig.parameters.keys())

        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            # 绑定参数
            bound = sig.bind(*args, **kwargs)
            bound.apply_defaults()

            # 验证
            for name, validator in validators.items():
                if name in bound.arguments:
                    value = bound.arguments[name]
                    if not validator(value):
                        raise ValueError(
                            f"参数 '{name}'={value!r} 验证失败"
                        )

            return func(*args, **kwargs)
        return wrapper
    return decorator

@validate(
    age=lambda v: v >= 0,
    name=lambda v: len(v) > 0
)
def create_user(name, age):
    print(f"创建用户: {name}, {age}岁")

create_user("Alice", 25)    # OK
# create_user("", 25)       # ValueError!
# create_user("Bob", -1)    # ValueError!
```

## 6. 类属性装饰器

### 用类实现装饰器

```python
class CountCalls:
    """统计调用次数的装饰器类"""
    def __init__(self, func):
        functools.update_wrapper(self, func)
        self.func = func
        self.count = 0

    def __call__(self, *args, **kwargs):
        self.count += 1
        print(f"第 {self.count} 次调用")
        return self.func(*args, **kwargs)

    def reset(self):
        self.count = 0

@CountCalls
def process():
    print("处理中...")

process()  # 第 1 次调用
process()  # 第 2 次调用
process.count  # 2
process.reset()
```

## 7. 实用装饰器库

### 自己实现 cache

```python
def lru_cache(maxsize=128):
    """简单的 LRU 缓存装饰器"""
    def decorator(func):
        cache = {}
        order = []

        @functools.wraps(func)
        def wrapper(*args):
            if args in cache:
                # 移到末尾（最近使用）
                order.remove(args)
                order.append(args)
                return cache[args]

            result = func(*args)
            cache[args] = result
            order.append(args)

            # 淘汰最旧的
            if len(cache) > maxsize:
                oldest = order.pop(0)
                del cache[oldest]

            return result

        wrapper.cache_info = lambda: f"Cache: {len(cache)} items"
        wrapper.cache_clear = lambda: (cache.clear(), order.clear())
        return wrapper
    return decorator

@lru_cache(maxsize=3)
def fib(n):
    if n < 2:
        return n
    return fib(n - 1) + fib(n - 2)

print(fib(10))  # 55
print(fib.cache_info())  # Cache: 3 items
```

## 实战练习

### 练习 1：路由装饰器（模拟 Flask）

```python
class Router:
    def __init__(self):
        self.routes = {}

    def route(self, path):
        """路由装饰器"""
        def decorator(func):
            self.routes[path] = func
            return func
        return decorator

    def dispatch(self, path):
        if path in self.routes:
            return self.routes[path]()
        return "404 Not Found"

app = Router()

@app.route("/")
def home():
    return "首页"

@app.route("/about")
def about():
    return "关于我们"

print(app.dispatch("/"))      # 首页
print(app.dispatch("/about")) # 关于我们
print(app.routes)
```

### 练习 2：速率限制装饰器

```python
import time
from collections import deque

def rate_limit(max_calls=5, period=60):
    """限制调用频率的装饰器"""
    def decorator(func):
        calls = deque()

        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            now = time.time()

            # 移除过期的调用记录
            while calls and calls[0] < now - period:
                calls.popleft()

            if len(calls) >= max_calls:
                raise RuntimeError(
                    f"超过频率限制: {max_calls}次/{period}秒"
                )

            calls.append(now)
            return func(*args, **kwargs)
        return wrapper
    return decorator

@rate_limit(max_calls=3, period=10)
def api_call():
    return "API 响应"

for i in range(5):
    try:
        print(f"请求 {i+1}: {api_call()}")
    except RuntimeError as e:
        print(f"请求 {i+1}: {e}")
```
