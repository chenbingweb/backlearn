## 1. 工厂模式

### 简单工厂

```python
class Dog:
    def speak(self):
        return "汪汪"

class Cat:
    def speak(self):
        return "喵喵"

class Duck:
    def speak(self):
        return "嘎嘎"

# 简单工厂
def get_pet(pet_type):
    pets = {
        "dog": Dog,
        "cat": Cat,
        "duck": Duck
    }
    return pets.get(pet_type, Dog)()

pet = get_pet("cat")
print(pet.speak())  # 喵喵
```

### 工厂方法模式

```python
from abc import ABC, abstractmethod

class Pet(ABC):
    @abstractmethod
    def speak(self):
        pass

    @abstractmethod
    def move(self):
        pass

class Dog(Pet):
    def speak(self):
        return "汪汪"

    def move(self):
        return "狗在跑"

class Cat(Pet):
    def speak(self):
        return "喵喵"

    def move(self):
        return "猫在跳"

class PetFactory(ABC):
    @abstractmethod
    def create_pet(self) -> Pet:
        pass

class DogFactory(PetFactory):
    def create_pet(self) -> Pet:
        return Dog()

class CatFactory(PetFactory):
    def create_pet(self) -> Pet:
        return Cat()

# 使用
def interact_with_pet(factory: PetFactory):
    pet = factory.create_pet()
    print(pet.speak())
    print(pet.move())

interact_with_pet(DogFactory())
interact_with_pet(CatFactory())
```

## 2. 策略模式

### 什么是策略模式

定义一系列算法，把它们一个个封装起来，并且使它们可以互相替换。

### 实现

```python
from abc import ABC, abstractmethod
from typing import List

class SortStrategy(ABC):
    @abstractmethod
    def sort(self, data: List[int]) -> List[int]:
        pass

class BubbleSort(SortStrategy):
    def sort(self, data: List[int]) -> List[int]:
        result = data.copy()
        n = len(result)
        for i in range(n):
            for j in range(0, n - i - 1):
                if result[j] > result[j + 1]:
                    result[j], result[j + 1] = result[j + 1], result[j]
        return result

class QuickSort(SortStrategy):
    def sort(self, data: List[int]) -> List[int]:
        if len(data) <= 1:
            return data
        pivot = data[len(data) // 2]
        left = [x for x in data if x < pivot]
        middle = [x for x in data if x == pivot]
        right = [x for x in data if x > pivot]
        return self.sort(left) + middle + self.sort(right)

class PythonSort(SortStrategy):
    def sort(self, data: List[int]) -> List[int]:
        return sorted(data)

class Sorter:
    def __init__(self, strategy: SortStrategy):
        self._strategy = strategy

    def set_strategy(self, strategy: SortStrategy):
        self._strategy = strategy

    def sort(self, data: List[int]) -> List[int]:
        return self._strategy.sort(data)

# 使用
data = [64, 34, 25, 12, 22, 11, 90]

sorter = Sorter(BubbleSort())
print(sorter.sort(data))

sorter.set_strategy(QuickSort())
print(sorter.sort(data))
```

### 支付策略示例

```python
from abc import ABC, abstractmethod

class PaymentStrategy(ABC):
    @abstractmethod
    def pay(self, amount: float) -> str:
        pass

class Alipay(PaymentStrategy):
    def __init__(self, phone: str):
        self.phone = phone

    def pay(self, amount: float) -> str:
        return f"支付宝支付 ¥{amount} (账号: {self.phone})"

class WeChatPay(PaymentStrategy):
    def __init__(self, openid: str):
        self.openid = openid

    def pay(self, amount: float) -> str:
        return f"微信支付 ¥{amount} (OpenID: {self.openid})"

class CreditCard(PaymentStrategy):
    def __init__(self, card_number: str, cvv: str):
        self.card_number = card_number[-4:]  # 只保留后四位
        self.cvv = cvv

    def pay(self, amount: float) -> str:
        return f"信用卡支付 ¥{amount} (卡号尾号: {self.card_number})"

class ShoppingCart:
    def __init__(self):
        self.items = []
        self._payment_strategy = None

    def add_item(self, name: str, price: float):
        self.items.append({"name": name, "price": price})

    def total(self) -> float:
        return sum(item["price"] for item in self.items)

    def set_payment_strategy(self, strategy: PaymentStrategy):
        self._payment_strategy = strategy

    def checkout(self) -> str:
        if not self._payment_strategy:
            raise ValueError("请先选择支付方式")
        return self._payment_strategy.pay(self.total())

# 使用
cart = ShoppingCart()
cart.add_item("iPhone", 5999)
cart.add_item("AirPods", 999)

cart.set_payment_strategy(Alipay("13800138000"))
print(cart.checkout())

cart.set_payment_strategy(CreditCard("1234567890123456", "123"))
print(cart.checkout())
```

## 3. 用函数实现策略模式

Python 中可以用更简单的函数实现：

```python
from typing import Callable, List

def bubble_sort(data: List[int]) -> List[int]:
    result = data.copy()
    n = len(result)
    for i in range(n):
        for j in range(0, n - i - 1):
            if result[j] > result[j + 1]:
                result[j], result[j + 1] = result[j + 1], result[j]
    return result

def quick_sort(data: List[int]) -> List[int]:
    if len(data) <= 1:
        return data
    pivot = data[len(data) // 2]
    return (quick_sort([x for x in data if x < pivot]) +
            [x for x in data if x == pivot] +
            quick_sort([x for x in data if x > pivot]))

def sort_data(data: List[int], strategy: Callable[[List[int]], List[int]]) -> List[int]:
    return strategy(data)

# 使用
data = [64, 34, 25, 12, 22, 11, 90]

print(sort_data(data, bubble_sort))
print(sort_data(data, quick_sort))
print(sort_data(data, sorted))  # 直接用内置函数
```

### 练习 1：文件解析器工厂

```python
import json
import csv
from abc import ABC, abstractmethod
from typing import List, Dict

class FileParser(ABC):
    @abstractmethod
    def parse(self, filepath: str) -> List[Dict]:
        pass

class JSONParser(FileParser):
    def parse(self, filepath: str) -> List[Dict]:
        with open(filepath, 'r') as f:
            return json.load(f)

class CSVParser(FileParser):
    def parse(self, filepath: str) -> List[Dict]:
        with open(filepath, 'r') as f:
            reader = csv.DictReader(f)
            return list(reader)

class ParserFactory:
    _parsers = {
        '.json': JSONParser,
        '.csv': CSVParser
    }

    @classmethod
    def get_parser(cls, filepath: str) -> FileParser:
        import os
        ext = os.path.splitext(filepath)[1].lower()
        parser_class = cls._parsers.get(ext)
        if not parser_class:
            raise ValueError(f"不支持的文件格式: {ext}")
        return parser_class()

# 使用
def process_file(filepath: str):
    parser = ParserFactory.get_parser(filepath)
    data = parser.parse(filepath)
    print(f"从 {filepath} 解析了 {len(data)} 条记录")
    return data

```

### 练习 2：折扣策略

```python
from abc import ABC, abstractmethod
from datetime import datetime

class DiscountStrategy(ABC):
    @abstractmethod
    def calculate_discount(self, amount: float) -> float:
        pass

class NoDiscount(DiscountStrategy):
    def calculate_discount(self, amount: float) -> float:
        return 0

class PercentageDiscount(DiscountStrategy):
    def __init__(self, percentage: float):
        self.percentage = percentage

    def calculate_discount(self, amount: float) -> float:
        return amount * self.percentage / 100

class FixedDiscount(DiscountStrategy):
    def __init__(self, discount: float):
        self.discount = discount

    def calculate_discount(self, amount: float) -> float:
        return min(self.discount, amount)

class SeasonalDiscount(DiscountStrategy):
    def __init__(self, season: str, discount_map: dict):
        self.season = season
        self.discount_map = discount_map

    def calculate_discount(self, amount: float) -> float:
        percentage = self.discount_map.get(self.season, 0)
        return amount * percentage / 100

class Order:
    def __init__(self, amount: float, strategy: DiscountStrategy):
        self.amount = amount
        self.strategy = strategy

    def final_amount(self) -> float:
        discount = self.strategy.calculate_discount(self.amount)
        return self.amount - discount

# 使用
order1 = Order(1000, PercentageDiscount(20))
print(f"原价: ¥1000, 折扣: ¥{order1.amount - order1.final_amount()}, 实付: ¥{order1.final_amount()}")

order2 = Order(1000, FixedDiscount(300))
print(f"原价: ¥1000, 折扣: ¥{order2.amount - order2.final_amount()}, 实付: ¥{order2.final_amount()}")
```
