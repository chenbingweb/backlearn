## 1. 什么是单例模式

单例模式（Singleton Pattern）确保一个类只有一个实例，并提供一个全局访问点。

### 使用场景

- 数据库连接池
- 配置文件管理
- 日志记录器
- 缓存系统
- 线程池

---

## 2. 实现方式一：模块

Python 最简单自然的单例方式。

```python
# singleton_module.py
class DatabaseConfig:
    def __init__(self):
        self.host = "localhost"
        self.port = 3306
        self.user = "root"

# 模块只会被导入一次
config = DatabaseConfig()
```

```python
# main.py
from singleton_module import config

config.host = "192.168.1.1"

# 其他地方导入
from singleton_module import config as config2

print(config2.host)  # 192.168.1.1 ← 同一个对象！
print(config is config2)  # True
```

**优点**：简单、线程安全、Pythonic  
**缺点**：导入时即创建，不够惰性

## 3. 实现方式二：__new__

```python
class Singleton:
    _instance = None

    def __new__(cls, *args, **kwargs):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
        return cls._instance

# 使用
s1 = Singleton()
s2 = Singleton()

print(s1 is s2)  # True
s1.x = 10
print(s2.x)      # 10
```

### 带初始化的单例

```python
class Database:
    _instance = None

    def __new__(cls, *args, **kwargs):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
        return cls._instance

    def __init__(self, connection_string=None):
        # 注意：每次创建都会调用 __init__
        if not hasattr(self, '_initialized'):
            self.connection_string = connection_string or "default"
            self._initialized = True

    def query(self, sql):
        print(f"查询: {sql}")

db1 = Database("mysql://localhost")
db2 = Database("postgres://remote")

print(db1 is db2)                      # True
print(db1.connection_string)           # mysql://localhost（不会被覆盖）
```

## 4. 实现方式三：装饰器

```python
import functools

def singleton(cls):
    """单例装饰器"""
    instances = {}
    @functools.wraps(cls)
    def wrapper(*args, **kwargs):
        if cls not in instances:
            instances[cls] = cls(*args, **kwargs)
        return instances[cls]
    return wrapper

@singleton
class Logger:
    def __init__(self):
        self.level = "INFO"
        self.handlers = []

    def log(self, message):
        print(f"[{self.level}] {message}")

logger1 = Logger()
logger2 = Logger()

print(logger1 is logger2)  # True
logger1.level = "DEBUG"
print(logger2.level)       # DEBUG
```

## 5. 实现方式四：元类

```python
class SingletonMeta(type):
    """单例元类"""
    _instances = {}

    def __call__(cls, *args, **kwargs):
        if cls not in cls._instances:
            cls._instances[cls] = super().__call__(*args, **kwargs)
        return cls._instances[cls]

class Config(metaclass=SingletonMeta):
    def __init__(self):
        self.debug = False
        self.database_url = "sqlite:///app.db"

class Cache(metaclass=SingletonMeta):
    def __init__(self):
        self.data = {}

# 每个类都是独立的单例
c1 = Config()
c2 = Config()
print(c1 is c2)  # True

cache1 = Cache()
cache2 = Cache()
print(cache1 is cache2)  # True

print(c1 is cache1)  # False（不同类）
```

## 6. 线程安全的单例

### 线程不安全的版本

```python
import threading
import time

class UnsafeSingleton:
    _instance = None

    def __new__(cls):
        if cls._instance is None:
            time.sleep(0.001)  # 模拟耗时操作
            cls._instance = super().__new__(cls)
        return cls._instance
```

### 加锁实现

```python
import threading

class ThreadSafeSingleton:
    _instance = None
    _lock = threading.Lock()

    def __new__(cls):
        if cls._instance is None:
            with cls._lock:
                # 双重检查
                if cls._instance is None:
                    cls._instance = super().__new__(cls)
        return cls._instance
```

### 使用装饰器实现线程安全

```python
import functools
import threading

def thread_safe_singleton(cls):
    instances = {}
    lock = threading.Lock()

    @functools.wraps(cls)
    def wrapper(*args, **kwargs):
        if cls not in instances:
            with lock:
                if cls not in instances:
                    instances[cls] = cls(*args, **kwargs)
        return instances[cls]
    return wrapper
```

## 7. 实际应用：配置管理器

```python
import threading
from typing import Any, Optional

class AppConfig(metaclass=SingletonMeta):
    """应用配置单例"""

    def __init__(self):
        self._config = {}
        self._lock = threading.Lock()

    def get(self, key: str, default: Any = None) -> Any:
        return self._config.get(key, default)

    def set(self, key: str, value: Any) -> None:
        with self._lock:
            self._config[key] = value

    def load_from_dict(self, config: dict) -> None:
        with self._lock:
            self._config.update(config)

    def __getitem__(self, key: str) -> Any:
        return self._config[key]

    def __setitem__(self, key: str, value: Any) -> None:
        self.set(key, value)

# 全局使用
config = AppConfig()
config.load_from_dict({
    "DEBUG": True,
    "DATABASE_URL": "mysql://localhost:3306/mydb",
    "SECRET_KEY": "my-secret-key",
    "MAX_CONNECTIONS": 100
})

# 任何地方获取的都是同一个实例
config2 = AppConfig()
print(config2["DEBUG"])  # True
print(config is config2)  # True
```

## 8. 实际应用：数据库连接池

```python
import threading
from queue import Queue

class ConnectionPool:
    _instance = None
    _lock = threading.Lock()

    def __new__(cls, max_connections=10):
        if cls._instance is None:
            with cls._lock:
                if cls._instance is None:
                    cls._instance = super().__new__(cls)
                    cls._instance._initialized = False
        return cls._instance

    def __init__(self, max_connections=10):
        if self._initialized:
            return

        self.max_connections = max_connections
        self._pool = Queue(maxsize=max_connections)
        self._lock = threading.Lock()
        self._initialized = True

        # 初始化连接
        for _ in range(max_connections):
            self._pool.put(self._create_connection())

    def _create_connection(self):
        return {"id": id(object()), "status": "ready"}

    def get_connection(self):
        return self._pool.get()

    def release_connection(self, conn):
        self._pool.put(conn)

    @property
    def size(self):
        return self._pool.qsize()

# 使用
pool1 = ConnectionPool(max_connections=5)
pool2 = ConnectionPool(max_connections=20)  # 参数被忽略

print(pool1 is pool2)        # True
print(pool1.max_connections) # 5

conn = pool1.get_connection()
print(pool1.size)  # 4
pool1.release_connection(conn)
print(pool1.size)  # 5
```

## 9. 单例的优缺点

### 优点

- 控制资源访问
- 节省内存
- 避免重复创建
- 全局访问点

### 缺点

- 违反单一职责原则
- 难以测试（全局状态）
- 隐藏依赖关系
- 线程安全问题

### 替代方案

```python
# 依赖注入（推荐）
class Service:
    def __init__(self, config, logger):
        self.config = config
        self.logger = logger

# 使用工厂函数
def create_service():
    config = load_config()
    logger = get_logger()
    return Service(config, logger)
```

## 实战练习

### 练习 1：日志单例

```python
import datetime
from typing import List

class Logger(metaclass=SingletonMeta):
    def __init__(self):
        self.logs: List[str] = []
        self.level = "INFO"

    def _log(self, level: str, message: str):
        timestamp = datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        log_entry = f"[{timestamp}] [{level}] {message}"
        self.logs.append(log_entry)
        print(log_entry)

    def debug(self, message: str):
        if self.level in ("DEBUG",):
            self._log("DEBUG", message)

    def info(self, message: str):
        if self.level in ("DEBUG", "INFO"):
            self._log("INFO", message)

    def error(self, message: str):
        self._log("ERROR", message)

    def get_logs(self) -> List[str]:
        return self.logs.copy()

# 使用
logger = Logger()
logger.info("系统启动")
logger.error("连接失败")

# 其他地方
logger2 = Logger()
logger2.info("用户登录")
print(f"总日志数: {len(logger2.get_logs())}")
```
