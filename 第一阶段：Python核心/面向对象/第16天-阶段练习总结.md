from abc import ABC, abstractmethod
from datetime import datetime, timedelta
from typing import List, Optional, Dict
import json
import os

# ============== 数据模型 ==============

class Book:
    def __init__(self, isbn: str, title: str, author: str,
                 category: str, total_copies: int = 1):
        self.isbn = isbn
        self.title = title
        self.author = author
        self.category = category
        self.total_copies = total_copies
        self.available_copies = total_copies

    @property
    def is_available(self) -> bool:
        return self.available_copies > 0

    def borrow(self) -> bool:
        if self.available_copies > 0:
            self.available_copies -= 1
            return True
        return False

    def return_book(self):
        if self.available_copies < self.total_copies:
            self.available_copies += 1

    def __str__(self) -> str:
        return f"《{self.title}》{self.author} [{self.available_copies}/{self.total_copies}]"

    def __repr__(self) -> str:
        return f"Book('{self.isbn}', '{self.title}', '{self.author}')"

    def to_dict(self) -> dict:
        return {
            "isbn": self.isbn,
            "title": self.title,
            "author": self.author,
            "category": self.category,
            "total_copies": self.total_copies,
            "available_copies": self.available_copies
        }

    @classmethod
    def from_dict(cls, data: dict) -> "Book":
        book = cls(
            data["isbn"],
            data["title"],
            data["author"],
            data["category"],
            data["total_copies"]
        )
        book.available_copies = data["available_copies"]
        return book


class Reader:
    def __init__(self, reader_id: str, name: str):
        self.reader_id = reader_id
        self.name = name
        self._borrowed: Dict[str, datetime] = {}  # isbn -> 借阅时间

    @property
    def borrowed_count(self) -> int:
        return len(self._borrowed)

    def borrow(self, isbn: str) -> bool:
        if isbn not in self._borrowed:
            self._borrowed[isbn] = datetime.now()
            return True
        return False

    def return_book(self, isbn: str) -> bool:
        if isbn in self._borrowed:
            del self._borrowed[isbn]
            return True
        return False

    def has_borrowed(self, isbn: str) -> bool:
        return isbn in self._borrowed

    def __str__(self) -> str:
        return f"{self.name}({self.reader_id}) [已借{self.borrowed_count}本]"

    def to_dict(self) -> dict:
        return {
            "reader_id": self.reader_id,
            "name": self.name,
            "borrowed": {k: v.isoformat() for k, v in self._borrowed.items()}
        }


class BorrowRecord:
    def __init__(self, reader_id: str, isbn: str,
                 borrow_time: datetime, return_time: Optional[datetime] = None):
        self.reader_id = reader_id
        self.isbn = isbn
        self.borrow_time = borrow_time
        self.return_time = return_time

    @property
    def is_returned(self) -> bool:
        return self.return_time is not None

    @property
    def duration_days(self) -> int:
        end = self.return_time or datetime.now()
        return (end - self.borrow_time).days

    def __str__(self) -> str:
        status = "已归还" if self.is_returned else "借阅中"
        return f"{self.reader_id} 借 {self.isbn} [{status}, {self.duration_days}天]"


# ============== 管理类 ==============

class Library:
    _instance = None

    def __new__(cls, *args, **kwargs):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
            cls._instance._initialized = False
        return cls._instance

    def __init__(self, data_file: str = "library.json"):
        if self._initialized:
            return

        self.data_file = data_file
        self._books: Dict[str, Book] = {}
        self._readers: Dict[str, Reader] = {}
        self._records: List[BorrowRecord] = []
        self._initialized = True

        self._load_data()

    # ---- 图书管理 ----
    def add_book(self, isbn: str, title: str, author: str,
                 category: str, copies: int = 1) -> Book:
        if isbn in self._books:
            self._books[isbn].total_copies += copies
            self._books[isbn].available_copies += copies
        else:
            self._books[isbn] = Book(isbn, title, author, category, copies)
        self._save_data()
        return self._books[isbn]

    def remove_book(self, isbn: str) -> bool:
        if isbn in self._books:
            del self._books[isbn]
            self._save_data()
            return True
        return False

    def get_book(self, isbn: str) -> Optional[Book]:
        return self._books.get(isbn)

    def search_books(self, keyword: str = "", category: str = "") -> List[Book]:
        results = []
        for book in self._books.values():
            match_keyword = (not keyword or
                           keyword.lower() in book.title.lower() or
                           keyword.lower() in book.author.lower())
            match_category = not category or book.category == category
            if match_keyword and match_category:
                results.append(book)
        return results

    def list_books(self) -> List[Book]:
        return list(self._books.values())

    # ---- 读者管理 ----
    def register_reader(self, reader_id: str, name: str) -> Reader:
        if reader_id not in self._readers:
            self._readers[reader_id] = Reader(reader_id, name)
            self._save_data()
        return self._readers[reader_id]

    def get_reader(self, reader_id: str) -> Optional[Reader]:
        return self._readers.get(reader_id)

    # ---- 借阅管理 ----
    def borrow_book(self, reader_id: str, isbn: str) -> str:
        reader = self._readers.get(reader_id)
        if not reader:
            return "读者不存在"

        book = self._books.get(isbn)
        if not book:
            return "图书不存在"

        if not book.is_available:
            return "图书已借完"

        if reader.has_borrowed(isbn):
            return "不能重复借阅同一本书"

        if book.borrow() and reader.borrow(isbn):
            record = BorrowRecord(reader_id, isbn, datetime.now())
            self._records.append(record)
            self._save_data()
            return f"借阅成功: 《{book.title}》"

        return "借阅失败"

    def return_book(self, reader_id: str, isbn: str) -> str:
        reader = self._readers.get(reader_id)
        if not reader:
            return "读者不存在"

        book = self._books.get(isbn)
        if not book:
            return "图书不存在"

        if not reader.has_borrowed(isbn):
            return "没有借阅记录"

        if reader.return_book(isbn):
            book.return_book()

            # 更新记录
            for record in reversed(self._records):
                if (record.reader_id == reader_id and
                    record.isbn == isbn and
                    not record.is_returned):
                    record.return_time = datetime.now()
                    break

            self._save_data()
            return f"归还成功: 《{book.title}》"

        return "归还失败"

    def get_reader_records(self, reader_id: str) -> List[BorrowRecord]:
        return [r for r in self._records if r.reader_id == reader_id]

    # ---- 统计 ----
    def statistics(self) -> dict:
        return {
            "total_books": len(self._books),
            "total_readers": len(self._readers),
            "total_borrowed": sum(1 for r in self._records if not r.is_returned),
            "categories": list(set(b.category for b in self._books.values()))
        }

    # ---- 持久化 ----
    def _save_data(self):
        data = {
            "books": [b.to_dict() for b in self._books.values()],
            "readers": [r.to_dict() for r in self._readers.values()],
            "records": [
                {
                    "reader_id": r.reader_id,
                    "isbn": r.isbn,
                    "borrow_time": r.borrow_time.isoformat(),
                    "return_time": r.return_time.isoformat() if r.return_time else None
                }
                for r in self._records
            ]
        }
        with open(self.data_file, "w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=2)

    def _load_data(self):
        if not os.path.exists(self.data_file):
            return

        try:
            with open(self.data_file, "r", encoding="utf-8") as f:
                data = json.load(f)

            for book_data in data.get("books", []):
                book = Book.from_dict(book_data)
                self._books[book.isbn] = book

            for reader_data in data.get("readers", []):
                reader = Reader(reader_data["reader_id"], reader_data["name"])
                self._readers[reader.reader_id] = reader

        except (json.JSONDecodeError, KeyError):
            pass


# ============== 演示 ==============

def demo():
    library = Library()

    # 添加图书
    library.add_book("978-7-111-1", "Python编程", "张三", "计算机", 3)
    library.add_book("978-7-111-2", "算法导论", "李四", "计算机", 2)
    library.add_book("978-7-111-3", "三体", "刘慈欣", "科幻", 5)
    library.add_book("978-7-111-4", "红楼梦", "曹雪芹", "文学", 2)

    # 注册读者
    library.register_reader("R001", "Alice")
    library.register_reader("R002", "Bob")

    # 借书
    print(library.borrow_book("R001", "978-7-111-1"))
    print(library.borrow_book("R001", "978-7-111-3"))
    print(library.borrow_book("R002", "978-7-111-1"))
    print(library.borrow_book("R002", "978-7-111-3"))

    # 搜索
    print("\n搜索 'Python':")
    for book in library.search_books(keyword="Python"):
        print(f"  {book}")

    print("\n计算机类图书:")
    for book in library.search_books(category="计算机"):
        print(f"  {book}")

    # 统计
    print(f"\n统计: {library.statistics()}")

    # 还书
    print(f"\n{library.return_book('R001', '978-7-111-1')}")

    # 再次借书（现在有空位了）
    print(library.borrow_book("R002", "978-7-111-1"))


if __name__ == "__main__":
    demo()
