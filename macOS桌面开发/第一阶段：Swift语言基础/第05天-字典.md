## 2. 访问字典

### 下标访问

```swift
var ages = ["Alice": 25, "Bob": 30, "Charlie": 22]

// 返回 Optional
let aliceAge = ages["Alice"]       // Optional(25)
let unknownAge = ages["David"]     // nil

// 解包
if let age = ages["Alice"] {
    print("Alice 的年龄是 \(age)")
}

// 提供默认值
let davidAge = ages["David", default: 0]
print(davidAge)  // 0
```

### 检查键是否存在

```swift
let hasAlice = ages.keys.contains("Alice")   // true
let hasDavid = ages.keys.contains("David")   // false
```

## 3. 修改字典

### 添加/更新

```swift
var scores: [String: Int] = [:]

// 添加新键值对
scores["Alice"] = 95
scores["Bob"] = 87

// 更新值
scores["Alice"] = 98

// 使用 updateValue（返回旧值）
let oldScore = scores.updateValue(100, forKey: "Alice")
print(oldScore ?? "无旧值")  // Optional(98)

// 如果不存在则添加
if scores["Charlie"] == nil {
    scores["Charlie"] = 90
}
```

### 删除

```swift
var fruits = ["Apple": 5, "Banana": 3, "Orange": 4]

// 通过键删除
fruits["Apple"] = nil

// 使用 removeValue
let removed = fruits.removeValue(forKey: "Banana")
print(removed ?? "无")  // Optional(3)

// 删除不存在的键
let notFound = fruits.removeValue(forKey: "Grape")
print(notFound ?? "不存在")  // "不存在"

// 清空
fruits.removeAll()
```

## 4. 字典遍历

### 遍历键值对

```swift
let capitals = [
    "China": "Beijing",
    "USA": "Washington",
    "Japan": "Tokyo"
]

// 遍历键值对
for (country, capital) in capitals {
    print("\(country) 的首都是 \(capital)")
}

// 只遍历键
for country in capitals.keys {
    print(country)
}

// 只遍历值
for capital in capitals.values {
    print(capital)
}

// 排序后遍历
for country in capitals.keys.sorted() {
    print("\(country): \(capitals[country]!)")
}
```

## 5. 字典常用操作

### 合并字典

```swift
var dict1 = ["a": 1, "b": 2]
var dict2 = ["b": 3, "c": 4]

// 方法1：遍历合并
for (key, value) in dict2 {
    dict1[key] = value
}
// dict1 = ["a": 1, "b": 3, "c": 4]

// 方法2：Swift 4.0+ merge
dict1.merge(dict2) { (current, _) in current }
// 保留当前值
```

### 过滤

```swift
let scores = [
    "Alice": 95,
    "Bob": 72,
    "Charlie": 88,
    "David": 91
]

// 过滤及格的学生
let passed = scores.filter { $0.value >= 80 }
print(passed)  // ["Alice": 95, "Charlie": 88, "David": 91]
```

### 映射

```swift
// 值映射
let doubled = scores.mapValues { $0 * 2 }
print(doubled)

// 转换为数组
let scoreList = scores.map { (name, score) in
    "\(name): \(score)"
}
print(scoreList)
```

### 分组

```swift
let names = ["Alice", "Bob", "Charlie", "David", "Eve"]

// 按首字母分组
let grouped = Dictionary(grouping: names) { $0.first! }
print(grouped)
// ["A": ["Alice"], "B": ["Bob"], "C": ["Charlie"], "D": ["David"], "E": ["Eve"]]
```

## 6. 字典与数组的转换

### 数组转字典

```swift
let pairs = [("Alice", 95), ("Bob", 87), ("Charlie", 92)]
let dict = Dictionary(uniqueKeysWithValues: pairs)
// ["Alice": 95, "Bob": 87, "Charlie": 92]
```

### 字典转数组

```swift
let scores = ["Alice": 95, "Bob": 87]

// 键数组
let names = Array(scores.keys)      // ["Alice", "Bob"]

// 值数组
let values = Array(scores.values)   // [95, 87]

// 元组数组
let pairs = Array(scores)           // [("Alice", 95), ("Bob", 87)]
```

## 7. 常用字典类型

### 计数字典

```swift
let words = ["apple", "banana", "apple", "cherry", "banana", "apple"]

var counts: [String: Int] = [:]
for word in words {
    counts[word, default: 0] += 1
}
print(counts)  // ["apple": 3, "banana": 2, "cherry": 1]
```

### 嵌套字典

```swift
// 二维数据
var students: [String: [String: Int]] = [
    "Alice": ["Math": 95, "English": 88],
    "Bob": ["Math": 78, "English": 92]
]

// 访问
let aliceMath = students["Alice"]?["Math"]
print(aliceMath ?? 0)  // 95

// 修改
students["Bob"]?["Math"] = 85
```

## 实战练习

### 练习 1：通讯录

```swift
var contacts: [String: String] = [
    "张三": "13800138000",
    "李四": "13900139000",
    "王五": "13700137000"
]

// 添加联系人
contacts["赵六"] = "13600136000"

// 查找
if let phone = contacts["张三"] {
    print("张三的电话: \(phone)")
}

// 删除
contacts.removeValue(forKey: "王五")

// 打印所有联系人
print("\n通讯录列表:")
for (name, phone) in contacts.sorted(by: { $0.key < $1.key }) {
    print("\(name): \(phone)")
}
```

### 练习 2：词频统计

```swift
let text = """
    Swift is a powerful and intuitive programming language.
    Swift is designed to work with Apple's Cocoa and Cocoa Touch frameworks.
    """

let words = text.lowercased()
    .components(separatedBy: CharacterSet.alphanumerics.inverted)
    .filter { !$0.isEmpty }

var frequency: [String: Int] = [:]
for word in words {
    frequency[word, default: 0] += 1
}

// 排序输出
let sorted = frequency.sorted { $0.value > $1.value }
for (word, count) in sorted {
    print("\(word): \(count)")
}
```

### 练习 3：成绩管理

```swift
struct Student {
    let name: String
    var scores: [String: Int]  // 科目: 分数
}

var students: [String: Student] = [:]

// 添加学生
students["S001"] = Student(name: "Alice", scores: ["Math": 95, "English": 88])
students["S002"] = Student(name: "Bob", scores: ["Math": 78, "English": 92])

// 计算平均分
func averageScore(student: Student) -> Double {
    let total = student.scores.values.reduce(0, +)
    return Double(total) / Double(student.scores.count)
}

// 打印成绩单
for (id, student) in students {
    let avg = averageScore(student: student)
    print("\(student.name) (\(id)): 平均分 \(String(format: "%.1f", avg))")
}
```

## 今日总结

- [ ] 字典使用键值对存储数据，键必须唯一
- [ ] 访问返回 Optional，可使用默认值
- [ ] updateValue 返回旧值，removeValue 返回被删除的值
- [ ] 支持 filter、mapValues 等高阶函数
- [ ] 可用 Dictionary(grouping:) 进行分组
