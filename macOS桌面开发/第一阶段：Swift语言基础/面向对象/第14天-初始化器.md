## 1. 类的初始化器

### 指定初始化器

```swift
class Person {
    var name: String
    var age: Int

    // 指定初始化器：设置所有属性
    init(name: String, age: Int) {
        self.name = name
        self.age = age
    }
}

let person = Person(name: "Alice", age: 30)
print(person.name)  // Alice
```

### 默认初始化器

```swift
class User {
    var username: String = "anonymous"
    var email: String = ""
    var isActive: Bool = true
}

// 如果所有属性都有默认值，Swift 自动生成默认初始化器
let user = User()
print(user.username)  // anonymous
```

## 2. 便捷初始化器

### convenience init

```swift
class Person {
    var name: String
    var age: Int

    // 指定初始化器
    init(name: String, age: Int) {
        self.name = name
        self.age = age
    }

    // 便捷初始化器：调用指定初始化器
    convenience init(name: String) {
        self.init(name: name, age: 0)
    }

    convenience init(age: Int) {
        self.init(name: "Unknown", age: age)
    }
}

let p1 = Person(name: "Alice", age: 30)
let p2 = Person(name: "Bob") // age = 0
let p3 = Person(age: 25)               // name = "Unknown"
```

### 规则

1. 便捷初始化器必须调用同一个类的指定初始化器
2. 便捷初始化器不能调用 super.init

## 3. 结构体初始化器

### 成员逐一初始化器

```swift
struct Point {
    var x: Double
    var y: Double
}

// 自动生成成员逐一初始化器
let p1 = Point(x: 0, y: 0)
let p2 = Point(x: 1.5, y: 2.5)
```

### 自定义初始化器

```swift
struct Point {
    var x: Double
    var y: Double

    // 自定义初始化器（会覆盖自动生成的）
    init(x: Double, y: Double) {
        self.x = x
        self.y = y
    }

    // 额外初始化器
    init() {
        self.init(x: 0, y: 0)  // 委托给主初始化器
    }

    init(from origin: Point) {
        self.init(x: origin.x, y: origin.y)
    }
}

let p1 = Point() // (0, 0)
let p2 = Point(x: 1, y: 2)           // (1, 2)
let p3 = Point(from: p2)             // (1, 2)
```

## 4. 可选属性初始化

### 初始化器中的可选属性

```swift
class User {
    var username: String
    var email: String?
    var phone: String?

    init(username: String) {
        self.username = username
        // email 和 phone 默认 nil
    }
}

let user = User(username: "alice")
print(user.email)  // nil
user.email = "alice@example.com"
print(user.email!) // alice@example.com
```

### 可选类型的解包初始化

```swift
class Config {
    var apiKey: String
    var timeout: Int?

    init(apiKey: String, timeout: Int? = 30) {
        self.apiKey = apiKey
        self.timeout = timeout
    }

    func getTimeout() -> Int {
        // 使用 nil 合并运算符
        return timeout ?? 30
    }
}

let config1 = Config(apiKey: "abc123")
let config2 = Config(apiKey: "xyz789", timeout: 60)

print(config1.getTimeout())  // 30
print(config2.getTimeout()) // 60
```

## 5. 初始化器委托

### 值类型的初始化器委托

```swift
struct Size {
    var width: Double
    var height: Double

    // 主初始化器
    init(width: Double, height: Double) {
        self.width = width
        self.height = height
    }

    // 委托给主初始化器
    init(value: Double) {
        self.init(width: value, height: value)
    }

    init() {
        self.init(value: 0)
    }
}

let s1 = Size(width: 10, height: 20)  // (10, 20)
let s2 = Size(value: 50)            // (50, 50)
let s3 = Size()                     // (0, 0)
```

### 类的初始化器委托（两阶段初始化）

```swift
class Address {
    var street: String
    var city: String

    init(street: String, city: String) {
        self.street = street
        self.city = city
    }
}

class Person {
    var name: String
    var address: Address?

    // 指定初始化器
    init(name: String) {
        self.name = name
    }

    // 便捷初始化器
    convenience init(name: String, city: String) {
        let address = Address(street: "未知", city: city)
        self.init(name: name)
        self.address = address
    }
}

let p1 = Person(name: "Alice")
let p2 = Person(name: "Bob", city: "Beijing")
print(p2.address?.city)  // Optional("Beijing")
```

## 6. init? 可失败初始化器

### 基本用法

```swift
struct Animal {
    var name: String

    // 可失败初始化器：返回 nil 表示初始化失败
    init?(name: String) {
        if name.isEmpty {
            return nil
        }
        self.name = name
    }
}

if let animal = Animal(name: "Dog") {
    print(animal.name)  // Dog
}

if let animal = Animal(name: "") {
    print(animal.name)  // 不会执行
} else {
    print("初始化失败")  // 初始化失败
}
```

### 类的可失败初始化器

```swift
class Product {
    var name: String
    var price: Double

    init?(name: String, price: Double) {
        if name.isEmpty || price < 0 {
            return nil
        }
        self.name = name
        self.price = price
    }
}

if let product = Product(name: "iPhone", price: 999) {
    print(product.price)  // 999.0
}

if let product = Product(name: "", price: 999) {
    print(product.name)  // 不会执行
}
// nil
```

## 实战练习

### 练习 1：分数类

```swift
struct Fraction {
    var numerator: Int
    var denominator: Int

    // 规范化分数
    init?(numerator: Int, denominator: Int) {
        guard denominator != 0 else { return nil }
        guard numerator >= 0 else { return nil }

        self.numerator = numerator
        self.denominator = denominator
    }

    // 便捷初始化器
    convenience init(_ numerator: Int) {
        self.init(numerator: numerator, denominator: 1)!
    }

    var value: Double {
        return Double(numerator) / Double(denominator)
    }

    func description() -> String {
        if denominator == 1 {
            return "\(numerator)"
        }
        return "\(numerator)/\(denominator)"
    }
}

if let frac1 = Fraction(numerator: 3, denominator: 4) {
    print(frac1.description())  // 3/4
    print(frac1.value)          // 0.75
}

if let frac2 = Fraction(numerator: 5, denominator: 0) {
    print(frac2.description())  // 不会执行
} else {
    print("分母不能为 0")  // 分母不能为 0
}

let frac3 = Fraction(5)!
print(frac3.description())  // 5
```

### 练习 2：用户配置类

```swift
class UserConfig {
    var username: String
    var email: String
    var theme: String
    var language: String
    var notificationsEnabled: Bool

    // 指定初始化器
    init(username: String, email: String, theme: String = "light",
         language: String = "zh-CN", notificationsEnabled: Bool = true) {
        self.username = username
        self.email = email
        self.theme = theme
        self.language = language
        self.notificationsEnabled = notificationsEnabled
    }

    // 便捷初始化器
    convenience init(username: String) {
        self.init(username: username, email: "\(username)@example.com")
    }
}

let config1 = UserConfig(username: "alice", email: "alice@example.com")
let config2 = UserConfig(username: "bob")  // 使用默认值
print(config2.theme)       // light
print(config2.email)      // bob@example.com
```

## 今日总结

- [ ] 指定初始化器 `init` 设置所有属性
- [ ] 便捷初始化器 `convenience init` 调用指定初始化器
- [ ] 结构体自动生成成员逐一初始化器
- [ ] 可选属性默认 `nil`
- [ ] 初始化器可以委托给其他初始化器
- [ ] `init?` 可失败初始化器可返回 `nil`
- [ ] 可失败初始化器适合需要验证输入的场景
