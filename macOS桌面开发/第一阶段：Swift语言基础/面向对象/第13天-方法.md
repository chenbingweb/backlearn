## 1. 实例方法

### 基本定义

```swift
class Counter {
    var count: Int = 0

    // 实例方法
    func increment() {
        count += 1
    }

    func reset() {
        count = 0
    }

    func getValue() -> Int {
        return count
    }
}

let counter = Counter()
counter.increment()
counter.increment()
counter.increment()
print(counter.getValue())  // 3
counter.reset()
print(counter.getValue())  // 0
```

### self 属性

```swift
class Point {
    var x: Double
    var y: Double

    init(x: Double, y: Double) {
        self.x = x
        self.y = y
    }

    // 使用 self 区分属性和参数
    func setX(_ x: Double) {
        self.x = x
    }

    // 返回自身
    func offsetBy(dx: Double, dy: Double) -> Point {
        return Point(x: self.x + dx, y: self.y + dy)
    }
}

let point = Point(x: 1, y: 2)
let newPoint = point.offsetBy(dx: 10, dy: 20)
print("(\(newPoint.x), \(newPoint.y))")  // (11.0, 22.0)
```

## 2. mutating 方法

### 结构体的 mutating 方法

```swift
struct Point {
    var x: Double
    var y: Double

    // mutating 表示会修改结构体本身
    mutating func moveBy(dx: Double, dy: Double) {
        x += dx
        y += dy
    }

    mutating func reset() {
        x = 0
        y = 0
    }
}

var point = Point(x: 1, y: 2)
point.moveBy(dx: 10, dy: 20)
print("(\(point.x), \(point.y))")  // (11.0, 22.0)
```

### 类的 mutating 问题

```swift
class PointClass {
    var x: Double
    var y: Double

    init(x: Double, y: Double) {
        self.x = x
        self.y = y
    }

    // 类不需要 mutating，因为类用引用类型
    func moveBy(dx: Double, dy: Double) {
        x += dx
        y += dy
    }
}
```

## 3. 类型方法

### static 方法

```swift
struct Math {
    // 类型方法：通过类型名调用
    static func abs(_ value: Int) -> Int {
        return value < 0 ? -value : value
    }

    static func max(_ a: Int, _ b: Int) -> Int {
        return a > b ? a : b
    }

    static func min(_ a: Int, _ b: Int) -> Int {
        return a < b ? a : b
    }
}

// 调用类型方法
print(Math.abs(-5))      // 5
print(Math.max(3, 7))   // 7
print(Math.min(3, 7))    // 3
```

### class 方法（可被继承）

```swift
class Vehicle {
    var speed: Double = 0

    // class 方法可被子类重写
    class func description() -> String {
        return "交通工具"
    }

    // static 方法不能被重写
    static func category() -> String {
        return "一般交通工具"
    }
}

class Car: Vehicle {
    var brand: String = ""

    // 重写 class 方法
    override class func description() -> String {
        return "汽车"
    }
}

print(Vehicle.description())  // 交通工具
print(Car.description())     // 汽车
```

## 4. 方法参数

### 参数标签

```swift
struct Rectangle {
    var width: Double
    var height: Double

    // 参数标签：外部用 "byWidth"，内部用 "width"
    func scale(byWidth width: Double, height: Double) {
        self.width *= width
        self.height *= height
    }

    // 下划线表示省略外部参数名
    func scaleBy(_ factor: Double) {
        width *= factor
        height *= factor
    }
}

var rect = Rectangle(width: 10, height: 5)
rect.scale(byWidth: 2, height: 3)
print("\(rect.width), \(rect.height)")  // 20.0, 15.0

rect.scaleBy(0.5)
print("\(rect.width), \(rect.height)")  // 10.0, 7.5
```

### 默认参数值

```swift
struct Point {
    var x: Double
    var y: Double

    // 默认参数
    func offset(by dx: Double = 0, _ dy: Double = 0) -> Point {
        return Point(x: x + dx, y: y + dy)
    }
}

let point = Point(x: 1, y: 2)
print(point.offset()) // (1.0, 2.0)
print(point.offset(by: 10))     // (11.0, 2.0)
print(point.offset(by: 10, 20))  // (11.0, 22.0)
```

## 5. 方法与函数的区别

|特性 | 方法 | 函数 |
|------|------|------|
| 关联 | 属于类型 | 独立存在 |
| 调用 | `instance.method()` | `function()` |
| self | 有隐式 self | 无 |
| mutating | struct/class 可区分 | 无 |

### 方法作为第一公民

```swift
class Calculator {
    var result: Double = 0

    func add(_ value: Double) {
        result += value
    }

    func subtract(_ value: Double) {
        result -= value
    }
}

// 方法可以作为闭包传递
let calc = Calculator()
let operations: [() -> Void] = [
    { calc.add(5) },
    { calc.add(10) },
    { calc.subtract(3) }
]

for op in operations {
    op()
}
print(calc.result)  // 12
```

## 实战练习

### 练习 1：栈操作

```swift
struct Stack<Element> {
    private var items: [Element] = []

    mutating func push(_ item: Element) {
        items.append(item)
    }

    mutating func pop() -> Element? {
        return items.popLast()
    }

    func peek() -> Element? {
        return items.last
    }

    var isEmpty: Bool {
        return items.isEmpty
    }

    var count: Int {
        return items.count
    }
}

var stack = Stack<Int>()
stack.push(1)
stack.push(2)
stack.push(3)
print(stack.peek()!) // 3
print(stack.count)      // 3
print(stack.pop()!)     // 3
print(stack.count)      // 2
```

### 练习 2：日期计算

```swift
class DateHelper {
    var day: Int
    var month: Int
    var year: Int

    init(day: Int, month: Int, year: Int) {
        self.day = day
        self.month = month
        self.year = year
    }

    func addDays(_ days: Int) -> DateHelper {
        var newDay = self.day + days
        var newMonth = self.month
        var newYear = self.year

        let daysInMonth = daysInMonth(newMonth, year: newYear)
        while newDay > daysInMonth {
            newDay -= daysInMonth
            newMonth += 1
            if newMonth > 12 {
                newMonth = 1
                newYear += 1
            }
        }

        return DateHelper(day: newDay, month: newMonth, year: newYear)
    }

    private func daysInMonth(_ month: Int, year: Int) -> Int {
        switch month {
        case 1, 3, 5, 7, 8, 10, 12: return 31
        case 4, 6, 9, 11: return 30
        case 2:
            return isLeapYear(year) ? 29 : 28
        default: return 30
        }
    }

    private func isLeapYear(_ year: Int) -> Bool {
        return (year % 4 == 0 && year % 100 != 0) || (year % 400 == 0)
    }

    func description() -> String {
        return String(format: "%04d-%02d-%02d", year, month, day)
    }
}

let date = DateHelper(day: 28, month: 2, year: 2024)
print(date.description())  // 2024-02-28

let newDate = date.addDays(1)
print(newDate.description())  // 2024-02-29
```

## 今日总结

- [ ] 实例方法：属于类型实例，通过 `instance.method()` 调用
- [ ] `self` 指向当前实例，用于区分属性和参数
- [ ] `mutating` 用于结构体/枚举的方法，可修改属性
- [ ] `static` 类型方法属于类型本身，不可被继承
- [ ] `class` 类型方法属于类型，可被继承重写
- [ ] 方法参数可使用外部标签和默认参数值
- [ ] 类作为引用类型不需要 `mutating`，结构体需要
