## 1. 可选链基础

### 问题引出

```swift
class Person {
    var name: String
    var residence: Residence?

    init(name: String) {
        self.name = name
    }
}

class Residence {
    var address: Address?
}

class Address {
    var street: String
    var city: String

    init(street: String, city: String) {
        self.street = street
        self.city = city
    }
}

// 传统写法：层层判断
if let person = person {
    if let residence = person.residence {
        if let address = residence.address {
            print(address.street)
        }
    }
}

// 可选链写法：优雅简洁
if let street = person?.residence?.address?.street {
    print(street)
}
```

## 2. 可选链语法

### 基本语法 `?.`

```swift
let street = person?.residence?.address?.street
// 如果任意一环为 nil，整个表达式返回 nil
```

### 调用方法

```swift
class Person {
    var name: String

    init(name: String) {
        self.name = name
    }

    func greet() -> String {
        return "Hello, \(name)"
    }
}

let people: [Person] = [
    Person(name: "Alice"),
    Person(name: "Bob"),
    nil,
    Person(name: "Carol")
]

for person in people {
    // 可选链调用方法
    print(person?.greet() ?? "nil")
}
// Hello, Alice
// Hello, Bob
// nil
// Hello, Carol
```

## 3. 可选链赋值

### 条件赋值

```swift
person?.residence?.address?.street = "Main Street"

// 等价于：
if let residence = person?.residence,
   let address = residence.address {
    address.street = "Main Street"
}
```

## 4. 可选链与下标

### 可选链下标访问

```swift
var scores: [String: [Int]] = [
    "Alice": [90, 85, 88],
    "Bob": [92, 80]
]

// 获取第一个分数
let firstScore = scores["Alice"]?[0]
print(firstScore)  // Optional(90)

// 访问不存在的键
let missing = scores["Charlie"]?[0]
print(missing)  // nil
```

## 5. 多层可选链

### 多层访问

```swift
class Dog {
    var name: String
    var owner: Person?

    init(name: String) {
        self.name = name
    }
}

// 获取狗主人的朋友的姓名
let dogName = dog?.owner?.friend?.name
// dog -> Person? -> Person? -> String?
```

### 结合不同类型

```swift
struct Door {
    var lock: Lock?
}

struct Lock {
    var isLocked: Bool

    func unlock() -> Bool {
        isLocked = false
        return true
    }
}

let door = Door(lock: Lock(isLocked: true))

// 可选链调用方法
if let success = door?.lock?.unlock() {
    print("解锁成功: \(success)")
}
```

## 6. 可选绑定与可选链

### 对比

```swift
// 可选绑定
if let residence = person.residence {
    if let address = residence.address {
        print(address.city)
    }
}

// 可选链 + 空合运算符
print(person?.residence?.address?.city ?? "未知城市")

// 多层可选链
if let city = person?.residence?.address?.city,
   let street = person?.residence?.address?.street {
    print("\(street), \(city)")
}
```

## 实战练习

### 练习 1：安全的数组访问

```swift
extension Array {
    subscript(safe index: Index) -> Element? {
        return indices.contains(index) ? self[index] : nil
    }
}

let numbers = [1, 2, 3]
print(numbers[safe: 0])   // Optional(1)
print(numbers[safe: 10])  // nil
```

### 练习 2：配置对象链式访问

```swift
class AppConfig {
    var api: APIConfig?
}

class APIConfig {
    var endpoint: EndpointConfig?
}

class EndpointConfig {
    var baseURL: String
    var timeout: Int

    init(baseURL: String, timeout: Int = 30) {
        self.baseURL = baseURL
        self.timeout = timeout
    }
}

let config = AppConfig()
config.api = APIConfig()
config.api?.endpoint = EndpointConfig(baseURL: "https://api.example.com")

// 安全获取配置
let timeout = config?.api?.endpoint?.timeout ?? 30
let url = config?.api?.endpoint?.baseURL ?? "https://default.com"

print("Timeout: \(timeout)s")  // Timeout: 30s
print("URL: \(url)")           // URL: https://api.example.com
```
