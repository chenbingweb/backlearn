## 1. 属性包装器基础

### 基本结构

```swift
@propertyWrapper
struct Uppercased {
    private var value: String = ""

    var wrappedValue: String {
        get { value }
        set { value = newValue.uppercased() }
    }

    init(wrappedValue: String) {
        self.wrappedValue = wrappedValue
    }
}
```

### 使用

```swift
struct Person {
    @Uppercased var name: String
}

var person = Person()
person.name = "alice"
print(person.name)  // ALICE
```

## 2. 带参数的属性包装器

### 带配置参数

```swift
@propertyWrapper
struct RangeLimited {
    private var value: Int = 0
    let minValue: Int
    let maxValue: Int

    var wrappedValue: Int {
        get { value }
        set { value = min(max(newValue, minValue), maxValue) }
    }

    init(wrappedValue: Int, minValue: Int, maxValue: Int) {
        self.minValue = minValue
        self.maxValue = maxValue
        self.wrappedValue = wrappedValue
    }
}

struct User {
    @RangeLimited(minValue: 0, maxValue: 100) var score: Int = 0
}

var user = User()
user.score = 50
print(user.score)  // 50

user.score = 150
print(user.score)  // 100（被限制）

user.score = -10
print(user.score)  // 0（被限制）
```

### 带默认值的配置

```swift
@propertyWrapper
struct Clamped<Value: Comparable> {
    private var value: Value
    let minValue: Value
    let maxValue: Value

    var wrappedValue: Value {
        get { value }
        set { value = min(max(newValue, minValue), maxValue) }
    }

    init(wrappedValue: Value, minValue: Value, maxValue: Value) {
        self.minValue = minValue
        self.maxValue = maxValue
        self.value = min(max(wrappedValue, minValue), maxValue)
    }
}
```

## 3. 属性包装器的多种形式

### 只读属性包装器

```swift
@propertyWrapper
struct Capitalized {
    private var value: String = ""

    var wrappedValue: String {
        get { value.capitalized }
        // 没有 set，只有初始化时可设置
    }

    init(wrappedValue: String) {
        self.value = wrappedValue
    }
}

struct Book {
    @Capitalized var title: String
}

var book = Book(title: "swift programming")
print(book.title)  // Swift Programming
// book.title = "new title"  // 不能赋值
```

### 带验证的属性包装器

```swift
@propertyWrapper
struct Email {
    private var email: String = ""

    var wrappedValue: String {
        get { email }
        set {
            if newValue.contains("@") && newValue.contains(".") {
                email = newValue
            } else {
                print("无效的邮箱格式: \(newValue)")
            }
        }
    }

    init(wrappedValue: String) {
        self.wrappedValue = wrappedValue
    }
}

struct User {
    @Email var email: String
}

var user = User()
user.email = "alice@example.com"
print(user.email)  // alice@example.com

user.email = "invalid"
print(user.email)  // 无效的邮箱格式
                  // 保持原值
```

## 4. 多个属性包装器组合

### 不能直接组合

```swift
// 错误：Swift 不支持直接组合多个 @propertyWrapper
// struct User {
//     @Clamped(min: 0, max: 100)
//     @Uppercased var name: String  // 不能叠加
// }
```

### 解决方案：合并包装器

```swift
@propertyWrapper
struct Validated {
    private var value: String = ""
    private let transform: (String) -> String
    private let validator: (String) -> Bool

    var wrappedValue: String {
        get { value }
        set {
            if validator(newValue) {
                value = transform(newValue)
            }
        }
    }

    init(
        wrappedValue: String,
        transform: @escaping (String) -> String,
        validator: @escaping (String) -> Bool
    ) {
        self.transform = transform
        self.validator = validator
        self.wrappedValue = wrappedValue
    }
}
```

## 5. 属性包装器在 SwiftUI 中的应用

### @State 模拟

```swift
@propertyWrapper
struct MyState<Value> {
    private var value: Value

    var wrappedValue: Value {
        get { value }
        mutating set { value = newValue }
    }

    mutating func update(_ newValue: Value) {
        value = newValue
    }

    init(wrappedValue: Value) {
        self.value = wrappedValue
    }
}
```

### @Binding 模拟

```swift
@propertyWrapper
struct MyBinding<Value> {
    private var getValue: () -> Value
    private var setValue: (Value) -> Void

    var wrappedValue: Value {
        get { getValue() }
        nonmutating set { setValue(newValue) }
    }

    init(getValue: @escaping () -> Value, setValue: @escaping (Value) -> Void) {
        self.getValue = getValue
        self.setValue = setValue
    }
}
```

## 6. 属性包装器与投影

### 投影属性 $

```swift
@propertyWrapper
struct Logged {
    private var value: Int = 0

    var wrappedValue: Int {
        get { value }
        set {
            print("设置值: \(newValue)")
            value = newValue
        }
    }

    // 投影属性
    var projectedValue: Logged { self }
}

struct Counter {
    @Logged var count: Int = 0
}

var counter = Counter()
counter.count = 5  // 输出: 设置值: 5
counter.count = 10 // 输出: 设置值: 10
```

### 更复杂的投影

```swift
@propertyWrapper
struct Username {
    private var value: String = ""
    private var isValid: Bool = false

    var wrappedValue: String {
        get { value }
        set {
            value = newValue
            isValid = newValue.count >= 3 && newValue.allSatisfy { $0.isLetter }
        }
    }

    var projectedValue: UsernameProjection { UsernameProjection(isValid: isValid) }
}

struct UsernameProjection {
    let isValid: Bool
}

struct User {
    @Username var username: String
}

var user = User()
user.username = "alice"
print(user.username)  // alice
print(user.$username.isValid)  // true
```

## 实战练习

### 练习 1：防抖属性包装器

```swift
@propertyWrapper
struct Debounced {
    private var value: String = ""
    private var lastUpdate: Date = Date()

    var wrappedValue: String {
        get { value }
        set {
            let now = Date()
            if now.timeIntervalSince(lastUpdate) > 0.5 {
                value = newValue
                lastUpdate = now
            }
        }
    }

    init(wrappedValue: String) {
        self.wrappedValue = wrappedValue
    }
}

struct SearchBar {
    @Debounced var query: String = ""
}

var searchBar = SearchBar()
searchBar.query = "a"
searchBar.query = "ab"  // 被忽略（时间间隔太短）
searchBar.query = "abc"
print(searchBar.query)  // abc
```

### 练习 2：持久化属性包装器

```swift
@propertyWrapper
struct UserDefaultsBacked<T> {
    private let key: String
    private let defaultValue: T
    private var cached: T?

    var wrappedValue: T {
        get {
            if let cached = cached {
                return cached
            }
            let value = UserDefaults.standard.object(forKey: key) as? T ?? defaultValue
            cached = value
            return value
        }
        set {
            cached = newValue
            UserDefaults.standard.set(newValue, forKey: key)
        }
    }

    init(key: String, defaultValue: T) {
        self.key = key
        self.defaultValue = defaultValue
        self.cached = UserDefaults.standard.object(forKey: key) as? T ?? defaultValue
    }
}

struct Settings {
    @UserDefaultsBacked(key: "username", defaultValue: "Guest") var username: String
    @UserDefaultsBacked(key: "notifications_enabled", defaultValue: true) var notificationsEnabled: Bool
}

var settings = Settings()
print(settings.username)  // Guest（首次）
settings.username = "Alice"
print(settings.username)  // Alice
```

## 今日总结

- [ ] `@propertyWrapper` 封装属性访问逻辑
- [ ] `wrappedValue` 是核心，定义存储和读取
- [ ] 可带配置参数（init 有参数）
