## 1. 错误表示

### Error 协议

```swift
enum NetworkError: Error {
    case invalidURL
    case noData
    case decodingFailed
    case serverError(Int)
    case unknown
}
```

### 抛出错误

```swift
func divide(_ a: Double, by b: Double) throws -> Double {
    guard b != 0 else {
        throw MathError.divisionByZero
    }
    return a / b
}

enum MathError: Error {
    case divisionByZero
    case overflow
    case underflow
}
```

## 2. throws 函数

### 基本用法

```swift
func validate(age: Int) throws {
    guard age >= 0 else {
        throw ValidationError.negativeAge
    }
    guard age <= 150 else {
        throw ValidationError.tooOld
    }
}

enum ValidationError: Error {
    case negativeAge
    case tooOld
    case emptyName
}
```

### 调用 throws 函数

```swift
// 使用 try
do {
    try validate(age: 25)
    print("验证通过")
} catch {
    print("验证失败: \(error)")
}
```

## 3. try-catch

### 基本语法

```swift
do {
    let result = try divide(10, by: 2)
    print("结果: \(result)")
} catch {
    print("错误: \(error)")
}
```

### 匹配特定错误

```swift
do {
    try validate(age: -5)
} catch ValidationError.negativeAge {
    print("年龄不能为负数")
} catch ValidationError.tooOld {
    print("年龄太大")
} catch {
    print("未知错误: \(error)")
}
```

### 多错误处理

```swift
enum FileError: Error {
    case fileNotFound
    case permissionDenied
    case readError
}

func readFile(at path: String) throws -> String {
    guard path != "" else {
        throw FileError.fileNotFound
    }
    // 模拟读取
    return "file contents"
}

do {
    let content = try readFile(at: "")
} catch let error as FileError {
    switch error {
    case .fileNotFound: print("文件不存在")
    case .permissionDenied: print("权限被拒绝")
    case .readError: print("读取错误")
    }
} catch {
    print("其他错误: \(error)")
}
```

## 4. try? 和 try!

### try? 返回可选值

```swift
func parse(_ string: String) -> Int? {
    return Int(string)
}

// 使用 try?
let result = try? parse("42")
print(result)  // Optional(42)

let invalid = try? parse("abc")
print(invalid)  // nil
```

### try! 强制解包

```swift
// 只有确定不会失败时才用 try!
let number = try! parse("123")
print(number)  // 123

// 如果失败会崩溃
// let invalid = try! parse("abc")  // 崩溃
```

## 5. defer

### 基本用法

```swift
func processFile() {
    let file = openFile()

    defer {
        closeFile(file)  // 无论何时退出都会执行
    }

    // 处理文件
    writeToFile(file)

    // 可以有多个 defer，按逆序执行
    defer {
        print("最后一个 defer")
    }

    // 函数结束前，defer 逆序执行
}

func openFile() -> String { "file handle" }
func closeFile(_: String) { print("文件已关闭") }
func writeToFile(_: String) { print("写入文件") }
```

### defer 注意事项

```swift
func example() {
    defer {
        print("1. defer 执行")
    }

    if true {
        defer {
            print("2. if 内的 defer 执行")
        }
    }

    defer {
        print("3. 最后的 defer 执行")
    }

    print("函数主体执行")
}
// 输出顺序: 函数主体执行 -> 3. 最后的 defer 执行 -> 2. if 内的 defer 执行 -> 1. defer 执行
```

## 6. 自定义错误处理

### 错误传播

```swift
func validateAndProcess(age: Int) throws -> String {
    guard age >= 0 else {
        throw ValidationError.negativeAge
    }
    guard age < 18 else {
        throw ValidationError.tooYoung
    }
    return "处理成功"
}

// 调用者处理
do {
    let result = try validateAndProcess(age: 25)
    print(result)
} catch {
    print("失败: \(error)")
}
```

### Result 类型

```swift
func fetchUser(id: Int, completion: (Result<String, Error>) -> Void) {
    if id > 0 {
        completion(.success("User_\(id)"))
    } else {
        completion(.failure(NetworkError.invalidURL))
    }
}

// 使用
fetchUser(id: 1) { result in
    switch result {
    case .success(let user):
        print("获取用户: \(user)")
    case .failure(let error):
        print("错误: \(error)")
    }
}
```

## 7. 错误与泛型

### 泛型函数中的错误

```swift
func retry<T>(_ operation: () throws -> T, times: Int) -> Result<T, Error> {
    for _ in 0..<times {
        do {
            return .success(try operation())
        } catch {
            continue
        }
    }
    return .failure(NetworkError.unknown)
}

let result = retry({ 42 }, times: 3)
print(result)  // Success(42)
```

## 实战练习

### 练习 1：密码验证

```swift
enum PasswordError: Error, CustomStringConvertible {
    case tooShort
    case noUppercase
    case noLowercase
    case noDigit

    var description: String {
        switch self {
        case .tooShort: return "密码太短"
        case .noUppercase: return "需要大写字母"
        case .noLowercase: return "需要小写字母"
        case .noDigit: return "需要数字"
        }
    }
}

func validatePassword(_ password: String) throws {
    guard password.count >= 8 else {
        throw PasswordError.tooShort
    }
    guard password.contains(where: { $0.isUppercase }) else {
        throw PasswordError.noUppercase
    }
    guard password.contains(where: { $0.isLowercase }) else {
        throw PasswordError.noLowercase
    }
    guard password.contains(where: { $0.isNumber }) else {
        throw PasswordError.noDigit
    }
}

do {
    try validatePassword("Abc123")
    print("密码有效")
} catch PasswordError.tooShort {
    print("密码至少8位")
} catch {
    print("错误: \(error)")
}
```

### 练习 2：JSON 解析模拟

```swift
enum ParseError: Error {
    case invalidJSON
    case missingField(String)
    case typeMismatch
}

struct User {
    let name: String
    let age: Int
}

func parseUser(from json: [String: Any]) throws -> User {
    guard let name = json["name"] as? String else {
        throw ParseError.missingField("name")
    }
    guard let age = json["age"] as? Int else {
        throw ParseError.typeMismatch
    }
    return User(name: name, age: age)
}

let validJson: [String: Any] = ["name": "Alice", "age": 30]
let invalidJson: [String: Any] = ["name": "Bob"]

do {
    let user = try parseUser(from: validJson)
    print("用户: \(user.name), \(user.age)岁")
} catch ParseError.missingField(let field) {
    print("缺少字段: \(field)")
} catch ParseError.typeMismatch {
    print("类型不匹配")
} catch {
    print("解析错误")
}
```
