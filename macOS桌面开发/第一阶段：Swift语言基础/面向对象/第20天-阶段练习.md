### 2. 定义任务优先级协议

```swift
protocol Prioritized {
    var priority: Priority { get }
}

enum Priority: Int, Comparable, CustomStringConvertible {
    case low = 1
    case medium = 2
    case high = 3
    case urgent = 4

    static func < (lhs: Priority, rhs: Priority) -> Bool {
        return lhs.rawValue < rhs.rawValue
    }

    var description: String {
        switch self {
        case .low: return "低"
        case .medium: return "中"
        case .high: return "高"
        case .urgent: return "紧急"
        }
    }
}
```

### 3. 定义任务状态

```swift
enum TaskStatus: String, CustomStringConvertible {
    case pending = "待办"
    case inProgress = "进行中"
    case completed = "已完成"

    var description: String {
        return self.rawValue
    }
}
```

### 4. 任务结构体

```swift
struct Task: Prioritized, Equatable, CustomStringConvertible {
    let id: UUID
    var title: String
    var description: String
    var priority: Priority
    var status: TaskStatus
    let createdAt: Date

    init(title: String, description: String = "", priority: Priority = .medium) {
        guard !title.trimmingCharacters(in: .whitespaces).isEmpty else {
            fatalError("任务标题不能为空")
        }
        self.id = UUID()
        self.title = title
        self.description = description
        self.priority = priority
        self.status = .pending
        self.createdAt = Date()
    }

    var isCompleted: Bool {
        status == .completed
    }

    var description: String {
        return "[\(priority.description)][\(status.description)] \(title)"
    }

    mutating func complete() {
        status = .completed
    }

    mutating func start() {
        status = .inProgress
    }

    static func == (lhs: Task, rhs: Task) -> Bool {
        return lhs.id == rhs.id
    }
}
```

### 5. 任务管理器类

```swift
class TaskManager {
    private var tasks: [Task] = []
    static let shared = TaskManager()

    private init() {}

    // MARK: - CRUD 操作

    func add(_ task: Task) {
        tasks.append(task)
    }

    func remove(_ task: Task) throws {
        guard let index = tasks.firstIndex(of: task) else {
            throw TaskError.taskNotFound
        }
        tasks.remove(at: index)
    }

    func find(byId id: UUID) -> Task? {
        return tasks.first { $0.id == id }
    }

    // MARK: - 状态操作

    func complete(_ task: Task) throws {
        guard var found = find(byId: task.id) else {
            throw TaskError.taskNotFound
        }
        found.complete()
        try update(found)
    }

    func start(_ task: Task) throws {
        guard var found = find(byId: task.id) else {
            throw TaskError.taskNotFound
        }
        found.start()
        try update(found)
    }

    private func update(_ task: Task) throws {
        guard let index = tasks.firstIndex(where: { $0.id == task.id }) else {
            throw TaskError.taskNotFound
        }
        tasks[index] = task
    }

    // MARK: - 查询

    func allTasks() -> [Task] {
        return tasks
    }

    func pendingTasks() -> [Task] {
        return tasks.filter { $0.status == .pending }
    }

    func completedTasks() -> [Task] {
        return tasks.filter { $0.isCompleted }
    }

    func tasks(withPriority priority: Priority) -> [Task] {
        return tasks.filter { $0.priority == priority }
    }

    var count: Int { tasks.count }
    var isEmpty: Bool { tasks.isEmpty }

    // MARK: - 排序

    func sortedByPriority() -> [Task] {
        return tasks.sorted { $0.priority > $1.priority }
    }

    func sortedByCreatedDate() -> [Task] {
        return tasks.sorted { $0.createdAt < $1.createdAt }
    }

    // MARK: - 统计

    func statistics() -> TaskStatistics {
        return TaskStatistics(
            total: tasks.count,
            pending: pendingTasks().count,
            inProgress: tasks.filter { $0.status == .inProgress }.count,
            completed: completedTasks().count,
            highPriority: tasks.filter { $0.priority >= .high }.count
        )
    }
}
```

### 7. 使用示例

```swift
// 创建任务管理器
let manager = TaskManager.shared

// 添加任务
let task1 = Task(title: "完成项目报告", description: "撰写技术文档", priority: .high)
let task2 = Task(title: "会议讨论", priority: .medium)
let task3 = Task(title: "代码review", priority: .low)
let task4 = Task(title: "紧急修复", priority: .urgent)

manager.add(task1)
manager.add(task2)
manager.add(task3)
manager.add(task4)

// 更新任务状态
if let found = manager.find(byId: task1.id) {
    try? manager.start(found)
}

// 完成紧急任务
try? manager.complete(task4)

// 打印所有任务
print("=== 所有任务 ===")
for task in manager.sortedByPriority() {
    print(task)
}

// 打印统计
print(manager.statistics())
```

```

[高][进行中] 完成项目报告
[中][待办] 会议讨论
[低][待办] 代码review

=== 任务统计 ===
总数: 4
待办: 2
进行中: 1
已完成: 1
高优先级: 2
完成率: 25.0%
```

### 练习 1：添加分类功能

```swift
protocol Categorizable {
    var category: String { get }
}

extension Task {
    var category: String {
        switch priority {
        case .urgent: return "紧急"
        case .high: return "重要"
        case .medium: return "普通"
        case .low: return "可选"
        }
    }
}

extension TaskManager {
    func tasks(inCategory category: String) -> [Task] {
        return tasks.filter { $0.category == category }
    }

    func groupedByCategory() -> [String: [Task]] {
        return Dictionary(grouping: tasks) { $0.category }
    }
}
```

### 练习 2：添加筛选器协议

```swift
protocol TaskFilter {
    func filter(_ tasks: [Task]) -> [Task]
}

struct PriorityFilter: TaskFilter {
    let priority: Priority

    func filter(_ tasks: [Task]) -> [Task] {
        return tasks.filter { $0.priority == self.priority }
    }
}

struct StatusFilter: TaskFilter {
    let status: TaskStatus

    func filter(_ tasks: [Task]) -> [Task] {
        return tasks.filter { $0.status == self.status }
    }
}

extension TaskManager {
    func filter(using filter: TaskFilter) -> [Task] {
        return filter.filter(tasks)
    }
}
```

### 练习 3：泛型堆栈扩展

```swift
extension TaskManager {
    func topTasks(_ count: Int) -> [Task] {
        return Array(manager.sortedByPriority().prefix(count))
    }
}

// 或者使用泛型
extension Array where Element: Comparable {
    func top(_ count: Int) -> [Element] {
        return Array(self.sorted().suffix(count))
    }
}
```
