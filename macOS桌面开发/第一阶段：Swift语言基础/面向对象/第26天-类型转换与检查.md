## 1. 类型检查 is

### 基本用法

```swift
class Animal {
    var name: String
    init(name: String) {
        self.name = name
    }
}

class Dog: Animal {
    func bark() {
        print("\(name): 汪汪！")
    }
}

class Cat: Animal {
    func meow() {
        print("\(name): 喵喵！")
    }
}

let animals: [Animal] = [
    Dog(name: "Max"),
    Cat(name: "Kitty"),
    Dog(name: "Buddy")
]

for animal in animals {
    if animal is Dog {
        print("\(animal.name) 是狗")
    } else if animal is Cat {
        print("\(animal.name) 是猫")
    }
}
```

---

## 2. 类型转换 as

### 强制转换 as!

```swift
let dog = animals[0] as! Dog
dog.bark()  // Max: 汪汪！
```

### 安全转换 as?

```swift
for animal in animals {
    if let dog = animal as? Dog {
        dog.bark()
    } else if let cat = animal as? Cat {
        cat.meow()
    }
}

// 输出:
// Max: 汪汪！
// Kitty: 喵喵！
// Buddy: 汪汪！
```

### 转换后调用特定方法

```swift
for animal in animals {
    // 直接转换并调用
    (animal as? Dog)?.bark()
    (animal as? Cat)?.meow()
}
```

## 3. Any 和 AnyObject

### Any 任意类型

```swift
var things: [Any] = []

things.append(42)
things.append(3.14)
things.append("hello")
things.append(Dog(name: "Max"))
things.append({ (x: Int) -> Int in x * 2 })

// 类型检查
for thing in things {
    if thing is Int {
        print("整数: \(thing)")
    } else if thing is Double {
        print("浮点数: \(thing)")
    } else if thing is String {
        print("字符串: \(thing)")
    } else if thing is Dog {
        (thing as? Dog)?.bark()
    }
}
```

### AnyObject 任意类实例

```swift
var objects: [AnyObject] = []

let dog = Dog(name: "Max")
let cat = Cat(name: "Kitty")

objects.append(dog)
objects.append(cat)

for obj in objects {
    if let dog = obj as? Dog {
        dog.bark()
    } else if let cat = obj as? Cat {
        cat.meow()
    }
}
```

## 4. 协议类型的类型检查

### 检查协议遵循

```swift
protocol Drawable {
    func draw()
}

class Circle: Drawable {
    var radius: Double
    init(radius: Double) {
        self.radius = radius
    }
    func draw() {
        print("画圆，半径 \(radius)")
    }
}

class Rectangle: Drawable {
    var width: Double
    var height: Double
    init(width: Double, height: Double) {
        self.width = width
        self.height = height
    }
    func draw() {
        print("画矩形")
    }
}

let shapes: [Any] = [
    Circle(radius: 5),
    Rectangle(width: 10, height: 5),
    Circle(radius: 3)
]

for shape in shapes {
    if let drawable = shape as? Drawable {
        drawable.draw()
    }
}
```

## 5. 嵌套类型检查

### 多层类型检查

```swift
class Shape {
    var color: String
    init(color: String) {
        self.color = color
    }
}

class Circle: Shape {
    var radius: Double
    init(radius: Double, color: String) {
        self.radius = radius
        super.init(color: color)
    }
}

class Rectangle: Shape {
    var width: Double
    var height: Double
    init(width: Double, height: Double, color: String) {
        self.width = width
        self.height = height
        super.init(color: color)
    }
}

let shapes: [Shape] = [
    Circle(radius: 5, color: "红色"),
    Rectangle(width: 10, height: 5, color: "蓝色")
]

for shape in shapes {
    if shape is Circle {
        let circle = shape as! Circle
        print("圆形，半径 \(circle.radius)，颜色 \(circle.color)")
    } else if shape is Rectangle {
        let rect = shape as! Rectangle
        print("矩形，\(rect.width)x\(rect.height)，颜色 \(rect.color)")
    }
}
```

## 6. 元类型

### 类型的类型

```swift
class Person {
    var name: String
    var type: Person.Type {
        return Person.self
    }

    init(name: String) {
        self.name = name
    }

    class func description() -> String {
        return "人类"
    }
}

let personType: Person.Type = Person.self
let person = personType.init(name: "Alice")
print(person.name)  // Alice
print(Person.description())  // 人类
```

### 元类型比较

```swift
protocol Shape {
    func draw()
}

class Circle: Shape {
    func draw() { print("圆形") }
}

class Rectangle: Shape {
    func draw() { print("矩形") }
}

let shapes: [Shape] = [Circle(), Rectangle()]

for shape in shapes {
    // 判断具体类型
    if type(of: shape) == Circle.self {
        print("这是圆形")
    } else if type(of: shape) == Rectangle.self {
        print("这是矩形")
    }
}
```

## 实战练习

### 练习 1：JSON 数据解析

```swift
let json: [String: Any] = [
    "name": "Alice",
    "age": 30,
    "address": ["city": "Beijing", "zip": "100000"],
    "scores": [95, 88, 92]
]

// 解析不同类型
if let name = json["name"] as? String {
    print("姓名: \(name)")
}

if let age = json["age"] as? Int {
    print("年龄: \(age)")
}

if let address = json["address"] as? [String: String],
   let city = address["city"] {
    print("城市: \(city)")
}

if let scores = json["scores"] as? [Int] {
    let avg = Double(scores.reduce(0, +)) / Double(scores.count)
    print("平均分: \(avg)")
}
```

### 练习 2：动物园模拟

```swift
class Animal {
    var name: String
    init(name: String) {
        self.name = name
    }
    func speak() {
        print("...")
    }
}

class Lion: Animal {
    override func speak() {
        print("\(name): 吼！")
    }
}

class Monkey: Animal {
    override func speak() {
        print("\(name): 吱吱！")
    }
}

class Elephant: Animal {
    override func speak() {
        print("\(name): 哞！")
    }
}

let zoo: [Animal] = [
    Lion(name: "辛巴"),
    Monkey(name: "孙悟空"),
    Elephant(name: "艾玛"),
    Lion(name: "刀疤")
]

// 分组统计
var lionCount = 0
var monkeyCount = 0
var elephantCount = 0

for animal in zoo {
    if animal is Lion {
        lionCount += 1
        (animal as! Lion).speak()
    } else if animal is Monkey {
        monkeyCount += 1
        (animal as! Monkey).speak()
    } else if animal is Elephant {
        elephantCount += 1
        (animal as! Elephant).speak()
    }
}

print("狮子: \(lionCount), 猴子: \(monkeyCount), 大象: \(elephantCount)")
```
