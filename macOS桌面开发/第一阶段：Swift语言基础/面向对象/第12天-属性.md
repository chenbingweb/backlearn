## 1. 存储属性

### 基本定义

```swift
class User {
    var username: String     // 变量存储属性，可修改
    let email: String        // 常量存储属性，不可修改
    var age: Int = 0         // 带默认值的存储属性
}

let user = User(username: "alice", email: "alice@example.com")
print(user.username)  // alice
user.username = "bob"  // ✅ 可以修改
// user.email = "new@example.com"  // ❌ 不能修改常量属性
```

### 结构体中的存储属性

```swift
struct Address {
    var street: String
    var city: String
    var zipCode: String
}

var home = Address(street: "123 Main St", city: "Beijing", zipCode: "100000")
home.city = "Shanghai"  // ✅
```

## 2. 计算属性

### 基本定义

```swift
struct Rectangle {
    var width: Double
    var height: Double

    // 计算属性：通过 get 返回
    var area: Double {
        return width * height
    }

    // 计算属性：读写 get/set
    var perimeter: Double {
        get {
            return (width + height) * 2
        }
        set {
            // newValue 是传入的新值
            let half = newValue / 2
            width = half / 2
            height = half / 2
        }
    }
}

let rect = Rectangle(width: 10, height: 5)
print(rect.area)       // 50.0
print(rect.perimeter)   // 30.0
```

### 只读计算属性

```swift
struct Circle {
    var radius: Double

    // 只读计算属性（可以省略 get）
    var diameter: Double {
        return radius * 2
    }

    var area: Double {
        return .pi * radius * radius
    }
}

let circle = Circle(radius: 5)
print(circle.diameter)  // 10.0
print(circle.area)      // 78.539816...
```

### 简写语法

```swift
struct Circle {
    var radius: Double

    //完整写法
    var diameter: Double {
        get { return radius * 2 }
        set { radius = newValue / 2 }
    }

    // 简写（只读时）
    var area: Double { radius * radius * .pi }
}
```

## 3. 属性观察器

### willSet / didSet

```swift
class Temperature {
    var celsius: Double = 0 {
        willSet {
            print("即将变为 \(newValue)°C")
        }
        didSet {
            print("已变为 \(celsius)°C")
            if celsius >= 100 {
                print("水开了！")
            }
        }
    }
}

var temp = Temperature()
temp.celsius = 50
// 即将变为 50.0°C
// 已变为 50.0°C

temp.celsius = 100
// 即将变为 100.0°C
// 已变为 100.0°C
// 水开了！
```

###观察器用途

```swift
class BankAccount {
    var balance: Double = 0 {
        didSet {
            if balance < 0 {
                print("警告：余额为负！")
            }
        }
    }
}

var account = BankAccount()
account.balance = 100 // 无输出
account.balance = -50 // 警告：余额为负！
```

## 4. 延迟属性 lazy

### 基本用法

```swift
class DataLoader {
    // 延迟属性：第一次访问时才初始化
    lazy var data: [String] = {
        print("正在加载数据...")
        return ["item1", "item2", "item3"]
    }()

    init() {
        print("DataLoader 已创建")
    }
}

print("创建 DataLoader")
let loader = DataLoader()
print("准备访问 data")
print(loader.data[0])
print(loader.data[0])  // 第二次访问不打印"正在加载"
```

输出：
```
创建 DataLoader
准备访问 data
正在加载数据...
item1
item1
```

###适用场景

```swift
class HeavyObject {
    init() {
        print("HeavyObject 初始化（耗时）")
    }

    func process() {
        print("处理中...")
    }
}

class Container {
    // 如果不用 lazy，每次创建 Container 都会初始化 HeavyObject
    lazy var heavy = HeavyObject()

    var isEnabled = false
}

var container = Container()
print("Container 创建完成")
// Container 创建完成（HeavyObject 还未创建）

container.heavy.process()
// HeavyObject 初始化（耗时）
// 处理中...
```

## 5. 类型属性 static

### 定义类型属性

```swift
struct MathConstants {
    static let pi: Double = 3.1415926
    static let e: Double = 2.7182818

    // 计算类型属性
    static var goldenRatio: Double {
        return (1 + sqrt(5)) / 2
    }
}

// 通过类型名访问
print(MathConstants.pi)           // 3.1415926
print(MathConstants.goldenRatio)   // 1.6180339
```

### 类中的类型属性

```swift
class Configuration {
    static var shared = Configuration()

    static var defaultTimeout: Int = 30

    var apiEndpoint: String = "https://api.example.com"

    private init() { }
}

// 使用
print(Configuration.defaultTimeout)  // 30
let config = Configuration.shared
print(config.apiEndpoint)
```

## 6. 属性包装器 @PropertyWrapper

### 基本用法

```swift
@propertyWrapper
struct Uppercased {
    private var value: String = ""

    var wrappedValue: String {
        get { value }
        set { value = newValue.uppercased() }
    }

    init(wrappedValue: String) {
        self.wrappedValue = wrappedValue
    }
}

struct Person {
    @Uppercased var name: String
}

var person = Person()
person.name = "alice"
print(person.name)  // ALICE
```

### 带参数的属性包装器

```swift
@propertyWrapper
struct Clamped {
    private var value: Int = 0
    var wrappedValue: Int {
        get { value }
        set { value = min(max(newValue, minValue), maxValue) }
    }

    let minValue: Int
    let maxValue: Int

    init(wrappedValue: Int, minValue: Int, maxValue: Int) {
        self.minValue = minValue
        self.maxValue = maxValue
        self.wrappedValue = wrappedValue
    }
}

struct Score {
    @Clamped(minValue: 0, maxValue: 100) var score: Int = 0
}

var score = Score()
score.score = 50 // 50
score.score = 150   // 100（被限制）
score.score = -10 // 0（被限制）
print(score.score)  // 0
```

## 实战练习

### 练习1：完整的模型类

```swift
class Product {
    // 存储属性
    var name: String
    var price: Double {
        didSet {
            if price < 0 {
                price = 0
                print("价格不能为负，已设为0")
            }
        }
    }
    var quantity: Int

    // 计算属性
    var totalPrice: Double {
        return price * Double(quantity)
    }

    var isExpensive: Bool {
        return price > 1000
    }

    // 初始化器
    init(name: String, price: Double, quantity: Int = 1) {
        self.name = name
        self.price = price
        self.quantity = quantity
    }

    // 方法
    func description() -> String {
        return "\(name)，单价\(price)，数量\(quantity)，总计\(totalPrice)"
    }
}

var product = Product(name: "iPhone", price: 999.99, quantity: 2)
print(product.description())
// iPhone，单价999.99，数量2，总计1999.98

product.price = -100
// 价格不能为负，已设为 0
print(product.price)  // 0.0
```

### 练习 2：使用属性包装器验证

```swift
@propertyWrapper
struct Email {
    private var email: String = ""

    var wrappedValue: String {
        get { email }
        set {
            if newValue.contains("@") && newValue.contains(".") {
                email = newValue
            } else {
                print("无效的邮箱格式")
            }
        }
    }

    init(wrappedValue: String) {
        self.wrappedValue = wrappedValue
    }
}

struct User {
    @Email var email: String
    var name: String
}

var user = User(name: "Alice")
user.email = "alice@example.com"
print(user.email)  // alice@example.com

user.email = "invalid-email"
print(user.email)  // 无效的邮箱格式
                 // 保持原值 alice@example.com
```

## 今日总结

- [ ] 存储属性：保存具体值，`var` 可变，`let` 不可变
- [ ] 计算属性：通过 `get`/`set` 计算，不存储值
- [ ] 属性观察器：`willSet`（变化前）、`didSet`（变化后）
- [ ] 延迟属性：`lazy` 第一次访问时才初始化
- [ ] 类型属性：`static` 属于类型本身，非实例
- [ ] 属性包装器：`@propertyWrapper` 封装属性逻辑
- [ ] 属性包装器可用于验证、转换、限制值
