## 1. 扩展基础

### 基本语法

```swift
extension SomeType {
    // 新增功能
}
```

### 扩展现有类型

```swift
// 扩展 String，添加新方法
extension String {
    var isBlank: Bool {
        return self.trimmingCharacters(in: .whitespacesAndNewlines).isEmpty
    }

    func greet() -> String {
        return "你好，\(self)！"
    }
}

let name = "Alice"
print(name.isBlank)  // false
print(name.greet())   // 你好，Alice！
```

## 2. 扩展计算属性

### 实例计算属性

```swift
extension Double {
    var km: Double { self / 1000.0 }
    var m: Double { self }
    var cm: Double { self * 100.0 }
    var mm: Double { self * 1000.0 }
}

let distance = 5.0.km
print(distance)  // 0.005

let length = 1000.0.mm
print(length)    // 1000.0
```

### 类型计算属性

```swift
extension Int {
    static var zero: Int { 0 }

    var isEven: Bool {
        return self % 2 == 0
    }
}

print(Int.zero)  // 0
print(42.isEven)  // true
```

## 3. 扩展方法

### 实例方法

```swift
extension Array {
    func average() -> Double? {
        guard !isEmpty else { return nil }
        return Double(reduce(0, +)) / Double(count)
    }

    func contains(_ element: Element) -> Bool {
        return self.contains { $0 == element }
    }
}

let numbers = [1, 2, 3, 4, 5]
print(numbers.average())  // 3.0
print(numbers.contains(3)) // true
```

### mutating 方法

```swift
extension Array where Element: Comparable {
    mutating func insertionSort() {
        for i in 1..<count {
            var j = i
            while j > 0 && self[j-1] > self[j] {
                self.swapAt(j-1, j)
                j -= 1
            }
        }
    }
}

var nums = [5, 3, 1, 4, 2]
nums.insertionSort()
print(nums)  // [1, 2, 3, 4, 5]
```

## 4. 扩展初始化器

### 便捷初始化器

```swift
struct Point {
    var x: Double
    var y: Double
}

extension Point {
    init(xy: Double) {
        self.init(x: xy, y: xy)
    }

    init(from string: String) {
        let parts = string.split(separator: ",")
        self.init(
            x: Double(parts[0]) ?? 0,
            y: Double(parts[1]) ?? 0
        )
    }
}

let p1 = Point(xy: 5)           // (5, 5)
let p2 = Point(from: "3,7")     // (3, 7)
```

## 5. 扩展下标

### 自定义下标

```swift
extension String {
    subscript(index: Int) -> Character? {
        guard index >= 0 && index < count else { return nil }
        return self[self.index(startIndex, offsetBy: index)]
    }

    subscript(start: Int, end: Int) -> String? {
        guard start >= 0 && end <= count && start < end else { return nil }
        let s = self.index(startIndex, offsetBy: start)
        let e = self.index(startIndex, offsetBy: end)
        return String(self[s..<e])
    }
}

let str = "Hello"
print(str[0])           // Optional('H')
print(str[1, 4])        // Optional("ell")
```

## 6. 扩展嵌套类型

### 在扩展中添加嵌套类型

```swift
extension Int {
    enum Kind {
        case negative, zero, positive
    }

    var kind: Kind {
        if self < 0 { return .negative }
        if self == 0 { return .zero }
        return .positive
    }
}

func describe(_ number: Int) {
    switch number.kind {
    case .negative: print("负数")
    case .zero: print("零")
    case .positive: print("正数")
    }
}

describe(-5)  // 负数
describe(0)   // 零
describe(10)  // 正数
```

## 7. 扩展协议

### 让类型遵循协议

```swift
protocol Loggable {
    func log() -> String
}

extension Double: Loggable {
    func log() -> String {
        return "数值: \(self)"
    }
}

let value: Double = 3.14
print(value.log())  // 数值: 3.14
```

## 实战练习

### 练习 1：扩展日期类型

```swift
extension Date {
    var year: Int {
        Calendar.current.component(.year, from: self)
    }

    var month: Int {
        Calendar.current.component(.month, from: self)
    }

    var day: Int {
        Calendar.current.component(.day, from: self)
    }

    func addDays(_ days: Int) -> Date {
        Calendar.current.date(byAdding: .day, value: days, to: self) ?? self
    }

    var isToday: Bool {
        Calendar.current.isDateInToday(self)
    }

    var description: String {
        String(format: "%04d-%02d-%02d", year, month, day)
    }
}

let today = Date()
print(today.description)  // 2024-01-01
print(today.isToday)     // true

let nextWeek = today.addDays(7)
print(nextWeek.description)
```

### 练习 2：扩展集合

```swift
extension Collection {
    var isNotEmpty: Bool {
        !isEmpty
    }

    subscript(safe index: Index) -> Element? {
        indices.contains(index) ? self[index] : nil
    }
}

let numbers = [1, 2, 3]
print(numbers.isNotEmpty)  // true
print(numbers[safe: 5])     // nil
print(numbers[safe: 1])     // Optional(2)
```

## 今日总结

- [ ] 扩展为现有类型添加新功能，无需继承
- [ ] 扩展可以添加计算属性、方法、初始化器、下标
- [ ] 扩展可以添加嵌套类型
- [ ] 扩展可以让类型遵循新协议
- [ ] 扩展不能添加存储属性
- [ ] 扩展不能重写现有方法（但可以覆盖通过 extension）
