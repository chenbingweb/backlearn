## 1. 泛型函数

### 基本语法

```swift
func swapTwoValues<T>(_ a: inout T, _ b: inout T) {
    let temp = a
    a = b
    b = temp
}

var x = 5
var y = 10
swapTwoValues(&x, &y)
print("x = \(x), y = \(y)")  // x = 10, y = 5

var s1 = "hello"
var s2 = "world"
swapTwoValues(&s1, &s2)
print("s1 = \(s1), s2 = \(s2)")  // s1 = world, s2 = hello
```

### 多个类型参数

```swift
func createPair<A, B>(_ first: A, _ second: B) -> (A, B) {
    return (first, second)
}

let pair = createPair("age", 25)
print(pair)  // ("age", 25)
```

## 2. 泛型类型

### 泛型结构体

```swift
struct Stack<Element> {
    private var items: [Element] = []

    mutating func push(_ item: Element) {
        items.append(item)
    }

    mutating func pop() -> Element? {
        items.popLast()
    }

    var top: Element? {
        items.last
    }

    var isEmpty: Bool {
        items.isEmpty
    }

    var count: Int {
        items.count
    }
}

var intStack = Stack<Int>()
intStack.push(1)
intStack.push(2)
intStack.push(3)
print(intStack.pop()!)  // 3

var stringStack = Stack<String>()
stringStack.push("Hello")
stringStack.push("World")
print(stringStack.top!)  // World
```

### 泛型类

```swift
class Box<T> {
    var value: T

    init(_ value: T) {
        self.value = value
    }

    func get() -> T {
        return value
    }
}

let intBox = Box(42)
print(intBox.get())  // 42

let stringBox = Box("Hello")
print(stringBox.get())  // Hello
```

## 3. 泛型约束

### 类型约束

```swift
// 要求 T 必须遵循 Comparable
func findMax<T: Comparable>(_ array: [T]) -> T? {
    guard var max = array.first else { return nil }
    for item in array {
        if item > max {
            max = item
        }
    }
    return max
}

print(findMax([1, 5, 3, 9, 2]))  // Optional(9)
print(findMax(["a", "z", "m"]))  // Optional("z")
```

### 多个约束

```swift
// 要求 T 同时遵循 Comparable 和 CustomStringConvertible
func printSorted<T: Comparable & CustomStringConvertible>(_ array: [T]) {
    let sorted = array.sorted()
    for item in sorted {
        print(item)
    }
}

printSorted(["banana", "apple", "cherry"])
// apple
// banana
// cherry
```

### where 子句

```swift
func allEqual<T, U>(_ first: [T], _ second: [U]) -> Bool
    where T: Equatable, U: Equatable
{
    return first.count == second.count &&
           zip(first, second).allSatisfy { $0 == $1 }
}

print(allEqual([1, 2, 3], [1, 2, 3]))  // true
print(allEqual([1, 2], [1, 2, 3]))     // false
```

## 5. 泛型协议

### 关联类型

```swift
protocol Container {
    associatedtype Item

    var count: Int { get }
    mutating func append(_ item: Item)
    subscript(i: Int) -> Item { get }
}

struct IntList: Container {
    private var items: [Int] = []

    typealias Item = Int

    var count: Int { items.count }

    mutating func append(_ item: Int) {
        items.append(item)
    }

    subscript(i: Int) -> Int {
        return items[i]
    }
}

var list = IntList()
list.append(1)
list.append(2)
list.append(3)
print(list[0])  // 1
print(list.count)  // 3
```

### 泛型协议约束

```swift
protocol Summable {
    static func + (lhs: Self, rhs: Self) -> Self
    static var zero: Self { get }
}

extension Int: Summable {
    static var zero: Int { 0 }
}

extension Double: Summable {
    static var zero: Double { 0 }
}

func sum<T: Summable>(_ values: [T]) -> T {
    return values.reduce(T.zero, +)
}

print(sum([1, 2, 3, 4, 5]))      // 15
print(sum([1.5, 2.5, 3.0]))     // 7.0
```

## 6. 泛型扩展

### 扩展泛型类型

```swift
extension Stack {
    var peek: Element? {
        items.last
    }

    func contains(_ item: Element) -> Bool {
        items.contains(item)
    }
}

var stack = Stack<Int>()
stack.push(10)
stack.push(20)
print(stack.contains(10))  // true
print(stack.peek!)          // 20
```

## 实战练习

### 练习 1：泛型队列

```swift
struct Queue<T> {
    private var items: [T] = []

    mutating func enqueue(_ item: T) {
        items.append(item)
    }

    mutating func dequeue() -> T? {
        guard !items.isEmpty else { return nil }
        return items.removeFirst()
    }

    var front: T? {
        items.first
    }

    var isEmpty: Bool {
        items.isEmpty
    }

    var count: Int {
        items.count
    }
}

var queue = Queue<String>()
queue.enqueue("first")
queue.enqueue("second")
queue.enqueue("third")
print(queue.dequeue()!)  // first
print(queue.front!)      // second
print(queue.count)      // 2
```

### 练习 2：泛型映射

```swift
extension Array {
    func map<T>(_ transform: (Element) -> T) -> [T] {
        var result: [T] = []
        for item in self {
            result.append(transform(item))
        }
        return result
    }

    func filter(_ predicate: (Element) -> Bool) -> [Element] {
        var result: [Element] = []
        for item in self {
            if predicate(item) {
                result.append(item)
            }
        }
        return result
    }
}

let numbers = [1, 2, 3, 4, 5]
let doubled = numbers.map { $0 * 2 }
print(doubled)  // [2, 4, 6, 8, 10]

let evens = numbers.filter { $0 % 2 == 0 }
print(evens)  // [2, 4]
```
