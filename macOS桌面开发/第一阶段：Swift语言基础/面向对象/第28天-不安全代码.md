## 1. 不安全代码概述

### 为什么需要不安全代码

- 与 C 语言 API 交互
- 高性能底层操作

### 启用方式


## 2. 指针类型

### 指针类型对照表

| Swift 类型 | C 类型 | 说明 |
|-----------|--------|------|
| UnsafePointer<T> | const T* | 只读指针 |
| UnsafeMutablePointer<T> | T* | 可变指针 |
| UnsafeRawPointer | const void* | 原始只读指针 |
| UnsafeMutableRawPointer | void* | 原始可变指针 |

### 创建指针

```swift
var value = 42

// 获取指针
withUnsafePointer(to: &value) { ptr in
    print(ptr.pointee)  // 42
}

// withUnsafeMutablePointer 可变版本
withUnsafeMutablePointer(to: &value) { ptr in
    ptr.pointee = 100
}
print(value)  // 100
```

## 3. 不安全字节操作

### 直接内存访问

```swift
var data = Data([0x01, 0x02, 0x03, 0x04])

data.withUnsafeBytes { bytes in
    // bytes 是 UnsafeRawBufferPointer
    for byte in bytes {
        print(String(format: "0x%02X", byte))
    }
}
// 0x01
// 0x02
// 0x03
// 0x04
```

### 字节序转换

```swift
let value: UInt32 = 0x12345678

withUnsafeBytes(of: value) { bytes in
    for byte in bytes {
        print(String(format: "%02X", byte), terminator: " ")
    }
}
// 78 56 34 12（小端）或 12 34 56 78（大端）
```

## 4. 不安全转换 unsafeBitCast

### 类型转换

```swift
let doubleValue: Double = 3.14159

// 将 Double 的位模式当作 UInt64 解读
let bits = doubleValue.bitPattern
print(String(format: "0x%016llX", bits))
// 0x400921FB54442D18

// 安全转换
let intValue = unsafeBitCast(doubleValue, to: UInt64.self)
print(String(format: "0x%016llX", intValue))
// 0x400921FB54442D18
```

### 用途

```swift
// 检测 NaN
func isNaN(_ value: Double) -> Bool {
    let bits = value.bitPattern
    return (bits & 0x7FF0000000000000) == 0x7FF0000000000000 &&
           (bits & 0x000FFFFFFFFFFFFF) != 0
}

print(isNaN(Double.nan))  // true
print(isNaN(3.14))        // false
```

## 5. Memory Layout

### 查看内存布局

```swift
struct MyStruct {
    var a: Int8    // 1 byte
    var b: Int32    // 4 bytes
    var c: Double   // 8 bytes
}

let size = MemoryLayout<MyStruct>.size
let stride = MemoryLayout<MyStruct>.stride
let alignment = MemoryLayout<MyStruct>.alignment

print("size: \(size), stride: \(stride), alignment: \(alignment)")
// size: 13, stride: 16, alignment: 8
```

### 紧凑布局

```swift
struct CompactStruct {
    var a: Int8
    var b: Int8
    var c: Int8
    var d: Int8
}

let size = MemoryLayout<CompactStruct>.size  // 4
let stride = MemoryLayout<CompactStruct>.stride  // 4
```

## 6. 手动内存管理

### 分配和释放

```swift
// 分配内存
let count = 5
let ptr = UnsafeMutablePointer<Int>.allocate(capacity: count)
ptr.initialize(to: 0, count: count)

// 使用内存
for i in 0..<count {
    ptr[i] = i * 2
}

for i in 0..<count {
    print(ptr[i])
}

// 清理
ptr.deinitialize(count: count)
ptr.deallocate()

// 输出: 0 2 4 6 8
```

### 重新解释内存

```swift
var bytes: [UInt8] = [0x12, 0x34, 0x56, 0x78]

bytes.withUnsafeBytes { rawBuffer in
    let intValue = rawBuffer.load(as: UInt32.self)
    print(String(format: "0x%08X", intValue))
    // 小端: 0x78563412
    // 大端: 0x12345678
}
```

## 7. 与 C 互操作

### 调用 C 函数

```swift
// 假设有一个 C 函数
// extern int strlen(const char *s);

// 在 Swift 中调用
let cString = "Hello, Swift!"
let length = cString.withCString { ptr in
    strlen(ptr)
}
print(length)  // 13
```

### 处理 C 数组

```swift
let array = [1, 2, 3, 4, 5]

array.withUnsafeBufferPointer { buffer in
    // buffer.baseAddress 是 UnsafePointer<Int>
    for i in 0..<buffer.count {
        print(buffer[i])
    }
}
```

## 实战练习

### 练习 1：字节序检测

```swift
func hostByteOrder() -> String {
    let value: UInt32 = 0x01020304
    var bytes = [UInt8](repeating: 0, count: 4)

    withUnsafeBytes(of: value) { ptr in
        for i in 0..<4 {
            bytes[i] = ptr[i]
        }
    }

    if bytes[0] == 0x01 {
        return "大端序"
    } else {
        return "小端序"
    }
}

print(hostByteOrder())
```

### 练习 2：内存拷贝

```swift
func memcpyDemo() {
    let source: [Int8] = [65, 66, 67, 68, 69]  // "ABCDE"
    var destination = [Int8](repeating: 0, count: 5)

    source.withUnsafeBufferPointer { srcBuffer in
        destination.withUnsafeMutableBufferPointer { dstBuffer in
            // 拷贝 5 个字节
            dstBuffer.baseAddress!.assign(from: srcBuffer.baseAddress!, count: 5)
        }
    }

    print(String(bytes: destination, encoding: .utf8)!)  // ABCDE
}

memcpyDemo()
```
