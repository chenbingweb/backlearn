## 1. 继承基础

### 基本语法

```swift
// 父类/基类
class Animal {
    var name: String

    init(name: String) {
        self.name = name
    }

    func speak() {
        print("...")
    }
}

// 子类
class Dog: Animal {
    var breed: String

    init(name: String, breed: String) {
        self.breed = breed
        super.init(name: name)  // 调用父类初始化器
    }

    override func speak() {
        print("\(name) 说：汪汪！")
    }
}

let dog = Dog(name: "Max", breed: "金毛")
dog.speak()  // Max 说：汪汪！
```

### Swift 不支持多继承

```swift
// ❌ 错误：Swift 不支持多继承
class A { }
class B { }
class AB: A, B { }  // 编译错误
```

## 2. 重写 override

### 重写方法

```swift
class Shape {
    func draw() {
        print("绘制图形")
    }

    func area() -> Double {
        return 0
    }
}

class Circle: Shape {
    var radius: Double

    init(radius: Double) {
        self.radius = radius
    }

    override func draw() {
        print("绘制圆形，半径 \(radius)")
    }

    override func area() -> Double {
        return .pi * radius * radius
    }
}

let circle = Circle(radius: 5)
circle.draw()  // 绘制圆形，半径 5.0
print(circle.area())  // 78.539816...
```

### 重写属性

```swift
class Person {
    var name: String { "人" }
    var age: Int {0 }
}

class Student: Person {
    var name: String { "学生" }
    var grade: String

    init(grade: String) {
        self.grade = grade
    }

    var age: Int { 18 }
}

let student = Student(grade: "A")
print(student.name)  // 学生
print(student.age)   // 18
```

### super 调用父类实现

```swift
class Cat: Animal {
    var color: String

    init(name: String, color: String) {
        self.color = color
        super.init(name: name)
    }

    override func speak() {
        super.speak()  // 先调用父类方法
        print("\(name) 说：喵喵！")
    }
}

let cat = Cat(name: "Kitty", color: "白色")
cat.speak()
// ...
// Kitty 说：喵喵！
```

## 3. 防止重写 final

### final class

```swift
// 防止继承
final class Constants {
    static let pi = 3.14159
    static let gravity = 9.8
}

// ❌ 错误：不能继承 final 类
class MyConstants: Constants { }
```

### final 方法/属性

```swift
class Base {
    final func method() {
        print("不能被重写")
    }

    var property: Int {
        return 1
    }
}

class Derived: Base {
    // ❌ 错误：不能重写 final 方法
    // override func method() { }

    // 可以重写非 final 属性
    override var property: Int {
        return 2
    }
}
```

## 4. 多态

### 运行时多态

```swift
class Shape {
    func draw() {
        print("绘制形状")
    }

    func area() -> Double {
        return 0
    }
}

class Rectangle: Shape {
    var width: Double
    var height: Double

    init(width: Double, height: Double) {
        self.width = width
        self.height = height
    }

    override func draw() {
        print("绘制矩形")
    }

    override func area() -> Double {
        return width * height
    }
}

class Triangle: Shape {
    var base: Double
    var height: Double

    init(base: Double, height: Double) {
        self.base = base
        self.height = height
    }

    override func draw() {
        print("绘制三角形")
    }

    override func area() -> Double {
        return base * height / 2
    }
}

// 多态：不同类型的对象用同一接口
func printShapeInfo(_ shape: Shape) {
    shape.draw()
    print("面积: \(shape.area())")
}

let shapes: [Shape] = [
    Rectangle(width: 10, height: 5),
    Triangle(base: 10, height: 5)
]

for shape in shapes {
    printShapeInfo(shape)
}
//绘制矩形
// 面积: 50.0
// 绘制三角形
// 面积: 25.0
```

### 类型检查 is / as

```swift
let shapes: [Shape] = [
    Rectangle(width: 10, height: 5),
    Triangle(base: 10, height: 5)
]

for shape in shapes {
    if shape is Rectangle {
        print("这是矩形")
    } else if shape is Triangle {
        print("这是三角形")
    }

    // 安全转换
    if let triangle = shape as? Triangle {
        print("底边: \(triangle.base), 高: \(triangle.height)")
    }
}
```

## 5. 类型转换

### is 检查类型

```swift
class Animal { }
class Dog: Animal { }
class Cat: Animal { }

let dog = Dog()
print(dog is Animal)  // true（Dog 是 Animal 的子类）
print(dog is Dog)     // true
print(dog is Cat)     // false
```

### as强制转换

```swift
let animal: Animal = Dog(name: "Max")

// 强制转换（不安全，可能崩溃）
let dog = animal as! Dog

// 安全转换（失败返回 nil）
if let dog = animal as? Dog {
    print("转换成功: \(dog.name)")
} else {
    print("转换失败")
}
```

### Any 和 AnyObject

```swift
var things: [Any] = []

things.append(42)
things.append("hello")
things.append(Dog(name: "Max"))

for thing in things {
    if let number = thing as? Int {
        print("整数: \(number)")
    } else if let str = thing as? String {
        print("字符串: \(str)")
    } else if let dog = thing as? Dog {
        print("狗: \(dog.name)")
    }
}
```

## 实战练习

### 练习 1：形状系统

```swift
class Shape {
    var name: String { "形状" }

    func describe() {
        print("这是一个 \(name)")
    }

    func area() -> Double {
        return 0
    }
}

class Square: Shape {
    var side: Double

    init(side: Double) {
        self.side = side
    }

    override var name: String { "正方形" }

    override func area() -> Double {
        return side * side
    }
}

class Circle: Shape {
    var radius: Double

    init(radius: Double) {
        self.radius = radius
    }

    override var name: String { "圆形" }

    override func area() -> Double {
        return .pi * radius * radius
    }
}

// 多态使用
let shapes: [Shape] = [
    Square(side: 10),
    Circle(radius: 5)
]

for shape in shapes {
    shape.describe()
    print("面积: \(shape.area())")
}
// 这是一个正方形
// 面积: 100.0
// 这是一个圆形
// 面积: 78.539816...
```

### 练习 2：员工系统

```swift
class Employee {
    var name: String
    var salary: Double

    init(name: String, salary: Double) {
        self.name = name
        self.salary = salary
    }

    func description() -> String {
        return "\(name)，薪资 \(salary)"
    }

    func bonus() -> Double {
        return salary * 0.1
    }
}

class Manager: Employee {
    var teamSize: Int

    init(name: String, salary: Double, teamSize: Int) {
        self.teamSize = teamSize
        super.init(name: name, salary: salary)
    }

    override func bonus() -> Double {
        return salary * 0.2 // 经理多拿20%
    }
}

// 使用
let employees: [Employee] = [
    Employee(name: "Alice", salary: 10000),
    Manager(name: "Bob", salary: 15000, teamSize: 5)
]

for emp in employees {
    print("\(emp.description())，奖金 \(emp.bonus())")
}
// Alice，薪资 10000.0，奖金 1000.0
// Bob，薪资 15000.0，奖金 3000.0
```
