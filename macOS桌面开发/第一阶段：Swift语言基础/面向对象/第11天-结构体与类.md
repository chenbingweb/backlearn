## 1. 结构体 struct

### 定义结构体

```swift
struct Person {
    var name: String
    var age: Int
}

// 创建实例
let person = Person(name: "Alice", age: 30)

//访问属性
print(person.name)  // Alice
print(person.age)   // 30

// 修改属性（var变量）
var person2 = Person(name: "Bob", age: 25)
person2.age = 26
```

### 初始化器

结构体自动生成成员逐一初始化器：

```swift
struct Point {
    var x: Double
    var y: Double
}

// 使用初始化器
let point1 = Point(x: 0.0, y: 0.0)
let point2 = Point(x: 1.0, y: 2.0)
```

### 定义方法

```swift
struct Rectangle {
    var width: Double
    var height: Double

    // 计算属性（不是方法）
    var area: Double {
        return width * height
    }

    // 方法
    func describe() -> String {
        return "宽\(width)，高\(height)，面积\(area)"
    }
}

let rect = Rectangle(width: 10, height: 5)
print(rect.describe())  // 宽10.0，高5.0，面积50.0
```

## 2. 类 class

### 定义类

```swift
class Dog {
    var name: String
    var age: Int

    // 初始化器
    init(name: String, age: Int) {
        self.name = name
        self.age = age
    }

    func bark() {
        print("\(name) 说：汪汪！")
    }
}

// 创建实例
let dog = Dog(name: "Max", age: 3)
dog.bark()  // Max 说：汪汪！
```

### 构造函数 init

```swift
class User {
    var username: String
    var email: String
    var isActive: Bool

    // 带默认值的初始化器
    init(username: String, email: String) {
        self.username = username
        self.email = email
        self.isActive = true
    }
}
```

## 3. 结构体 vs 类

### 核心区别

| 特性 | struct | class |
|------|---------|-------|
| 类型 | 值类型 | 引用类型 |
| 继承 | 不能继承 | 可以继承 |
| 初始化器 | 自动生成 memberwise init | 需要手动编写 |
| 内存管理 | 栈上分配，复制 | 堆上分配，引用计数 |
| 修饰符 | let不可变属性 | let 不可变引用 |

### 值类型 vs 引用类型

```swift
// 结构体（值类型）
struct Point {
    var x: Int
    var y: Int
}

var point1 = Point(x: 1, y: 2)
var point2 = point1  // 复制一份
point2.x = 10

print(point1.x)  // 1（不变）
print(point2.x)  // 10

// 类（引用类型）
class PointClass {
    var x: Int
    var y: Int

    init(x: Int, y: Int) {
        self.x = x
        self.y = y
    }
}

var p1 = PointClass(x: 1, y: 2)
var p2 = p1       // 共享引用
p2.x = 10

print(p1.x)  // 10（变了，因为共享引用）
print(p2.x)  // 10
```

### 选择建议

- **用结构体**：简单数据类型（Point、Size、Date）、不需要继承、值语义
- **用类**：需要继承、引用语义、需要引用计数管理 identity

## 4. 命名与约定

###命名规则

```swift
// 结构体/类名：大写开头（UpperCamelCase）
struct UserAccount { }
class PaymentProcessor { }

// 属性/方法名：小写开头（lowerCamelCase）
var userName: String
func processPayment() { }
```

### 默认值与懒加载

```swift
struct Config {
    var apiKey: String = ""
    var timeout: Int = 30
    var retryCount: Int = 3

    // 懒加载属性
    lazy var cache = [String: Any]()
}

var config = Config()
print(config.timeout)  // 30（使用默认值）
```

## 5. mutating 方法

### 结构体的方法修改属性

```swift
struct Counter {
    var count: Int = 0

    // mutating 表示会修改属性
    mutating func increment() {
        count += 1
    }

    mutating func reset() {
        count = 0
    }
}

var counter = Counter()
counter.increment()
counter.increment()
print(counter.count)  // 2
counter.reset()
print(counter.count)  // 0
```

### 为什么需要 mutating

```swift
// 普通方法默认不能修改 self
struct Point {
    var x: Int
    var y: Int

    func moveBy(dx: Int, dy: Int) {
        // ❌ 错误：不能修改属性
        x += dx
        y += dy
    }

    mutating func moveBy(dx: Int, dy: Int) {
        // ✅ 正确
        x += dx
        y += dy
    }
}
```

## 实战练习

### 练习 1：定义一个结构体

```swift
struct BankAccount {
    var balance: Double
    let accountNumber: String

    var description: String {
        return "账号 \(accountNumber)，余额 \(balance)"
    }

    mutating func deposit(_ amount: Double) {
        if amount > 0 {
            balance += amount
        }
    }

    mutating func withdraw(_ amount: Double) -> Bool {
        if amount > 0 && balance >= amount {
            balance -= amount
            return true
        }
        return false
    }
}

var account = BankAccount(balance: 1000, accountNumber: "ACC001")
print(account.description)
// 账号 ACC001，余额 1000.0

account.deposit(500)
print(account.balance)  // 1500.0

if account.withdraw(200) {
    print("取款成功，余额：\(account.balance)")
}
// 取款成功，余额：1300.0
```

### 练习 2：定义一个类

```swift
class Shape {
    var name: String

    init(name: String) {
        self.name = name
    }

    func describe() -> String {
        return "图形：\(name)"
    }

    func area() -> Double {
        return 0.0
    }
}

class Rectangle: Shape {
    var width: Double
    var height: Double

    init(width: Double, height: Double) {
        self.width = width
        self.height = height
        super.init(name: "矩形")
    }

    override func area() -> Double {
        return width * height
    }

    override func describe() -> String {
        return "\(super.describe())，宽\(width)，高\(height)，面积\(area())"
    }
}

let rect = Rectangle(width: 10, height: 5)
print(rect.describe())
// 图形：矩形，宽10.0，高5.0，面积50.0
```

## 今日总结

- [ ] 结构体是值类型，类是引用类型
- [ ] 结构体自动生成 `memberwise init`，类需要手动编写
- [ ] 值类型赋值给另一个变量时复制，引用类型共享引用
- [ ] 结构体不能继承，类可以继承
- [ ] `mutating` 方法可以修改结构体的属性
- [ ] 简单数据结构优先使用结构体
- [ ] 需要继承或引用语义时使用类
