## 1. ARC 基础

### 引用计数

```swift
class Person {
    var name: String

    init(name: String) {
        self.name = name
        print("\(name) 被初始化")
    }

    deinit {
        print("\(name) 被销毁")
    }
}

var person1: Person? = Person(name: "Alice")  // count = 1
var person2 = person1                          // count = 2
var person3 = person1                          // count = 3

person1 = nil  // count = 2
person2 = nil  // count = 1
person3 = nil  // count = 0，触发 deinit
// 输出: Alice 被初始化
//       Alice 被销毁
```

## 2. 强引用循环

### 问题示例

```swift
class Person {
    var name: String
    var apartment: Apartment?  // 强引用

    init(name: String) {
        self.name = name
    }

    deinit {
        print("\(name) 被销毁")
    }
}

class Apartment {
    var number: Int
    var tenant: Person?  // 强引用

    init(number: Int) {
        self.number = number
    }

    deinit {
        print("公寓 \(number) 被销毁")
    }
}

var alice: Person? = Person(name: "Alice")
var apt: Apartment? = Apartment(number: 101)

alice!.apartment = apt  // Alice -> Apartment (强引用)
apt!.tenant = alice     // Apartment -> Alice (强引用)

// 即使设为 nil，对象也无法释放
alice = nil
apt = nil
// 两个对象都在内存中，形成循环引用！
```

## 3. 弱引用 weak

### 使用 weak 打破循环

```swift
class Person {
    var name: String
    var apartment: Apartment?

    init(name: String) {
        self.name = name
    }

    deinit {
        print("\(name) 被销毁")
    }
}

class Apartment {
    var number: Int
    weak var tenant: Person?  // 弱引用，不增加引用计数

    init(number: Int) {
        self.number = number
    }

    deinit {
        print("公寓 \(number) 被销毁")
    }
}

var alice: Person? = Person(name: "Alice")
var apt: Apartment? = Apartment(number: 101)

alice!.apartment = apt
apt!.tenant = alice

alice = nil  // Person 被销毁，apartment.tenant 自动变 nil
apt = nil    // Apartment 被销毁
// 输出:
// Alice 被销毁
// 公寓 101 被销毁
```

## 4. 无主引用 unowned

### 使用 unowned

```swift
class Customer {
    var name: String
    var card: CreditCard?

    init(name: String) {
        self.name = name
    }

    deinit {
        print("Customer \(name) 被销毁")
    }
}

class CreditCard {
    var number: String
    unowned var owner: Customer  // 无主引用，假设必有主人

    init(number: String, owner: Customer) {
        self.number = number
        self.owner = owner
    }

    deinit {
        print("信用卡 \(number) 被销毁")
    }
}

var customer: Customer? = Customer(name: "Alice")
customer!.card = CreditCard(number: "1234-5678", owner: customer!)

customer = nil  // Customer 销毁，CreditCard 随之销毁
// 输出:
// Customer Alice 被销毁
// 信用卡 1234-5678 被销毁
```

### weak vs unowned

| 特性 | weak | unowned |
|------|------|--------|
| 可选性 | 必须是可选类型 | 可以非可选 |
| 初始化后可为 nil | 是 | 否（假设始终有值） |
| 使用场景 | 可为空的关系 | 始终有值的关系 |

## 5. 闭包中的引用

### 闭包捕获列表

```swift
class ViewController {
    var name = "ViewController"

    lazy var closure: () -> String = { [weak self] in
        guard let self = self else { return "" }
        return "Hello, \(self.name)"
    }

    deinit {
        print("\(name) 被销毁")
    }
}
```

### 完整示例

```swift
class Person {
    var name: String
    var hobbies: [String]

    // 闭包捕获 self
    lazy var hobbyDescription: () -> String = { [weak self] in
        guard let self = self else { return "" }
        return "\(self.name) 的爱好: \(self.hobbies.joined(separator: ", "))"
    }

    // 使用 unowned（如果确定 self 不会释放）
    lazy var hobbyDescription2: () -> String = { [unowned self] in
        return "\(self.name) 的爱好: \(self.hobbies.joined(separator: ", "))"
    }

    init(name: String, hobbies: [String]) {
        self.name = name
        self.hobbies = hobbies
    }

    deinit {
        print("\(name) 被销毁")
    }
}

var person: Person? = Person(name: "Alice", hobbies: ["阅读", "游泳"])
print(person!.hobbyDescription())
person = nil
// 输出:
// Alice 的爱好: 阅读, 游泳
// Alice 被销毁
```

## 6. 解决循环引用的模式

### 三个原则

1. **用 weak**: 父子关系，子可不存在（可选）
2. **用 unowned**: 父子关系，子必定存在（非可选）
3. **用捕获列表**: 闭包中使用 `[weak self]` 或 `[unowned self]`

### 常见模式

```swift
// 委托模式（delegate）- 通常用 weak
protocol Delegate: AnyObject {
    func didFinish()
}

class Manager {
    weak var delegate: Delegate?
}

// 闭包回调用 - 用 weak self
class NetworkService {
    var onComplete: ((Result<String, Error>) -> Void)?

    func fetch() {
        // 使用 [weak self] 避免循环引用
        onComplete = { [weak self] result in
            guard let self = self else { return }
            self.handleResult(result)
        }
    }
}

// 数据源引用 - 通常用 weak
class TableView {
    weak var dataSource: DataSource?
}
```

## 实战练习

### 练习 1：优化引用关系

```swift
class Author {
    var name: String
    var books: [Book]
    weak var bestFriend: Author?

    init(name: String) {
        self.name = name
        self.books = []
    }

    deinit {
        print("Author \(name) 被销毁")
    }
}

class Book {
    var title: String
    unowned var author: Author  // 书必定有作者

    init(title: String, author: Author) {
        self.title = title
        self.author = author
    }

    deinit {
        print("Book \(title) 被销毁")
    }
}

var author1: Author? = Author(name: "鲁迅")
var author2: Author? = Author(name: "周作人")

author1!.bestFriend = author2
author2!.bestFriend = author1

let book = Book(title: "呐喊", author: author1!)

author1 = nil
author2 = nil

print("book still exists: \(book.title)")
// Author 鲁迅 被销毁
// Author 周作人 被销毁
// book still exists: 呐喊
```

### 练习 2：闭包中的循环引用

```swift
class ImageLoader {
    var imageName: String
    var onLoad: (() -> Void)?

    init(imageName: String) {
        self.imageName = imageName
    }

    func loadImage() {
        // 错误写法：循环引用
        // onLoad = { [self] in print(self.imageName) }

        // 正确写法：使用 weak
        onLoad = { [weak self] in
            guard let self = self else { return }
            print(self.imageName)
        }
    }

    deinit {
        print("ImageLoader \(imageName) 被销毁")
    }
}

var loader: ImageLoader? = ImageLoader(imageName: "avatar.png")
loader?.loadImage()
loader = nil
// ImageLoader avatar.png 被销毁
```
