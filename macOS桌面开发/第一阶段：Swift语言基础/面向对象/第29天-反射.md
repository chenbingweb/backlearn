## 1. Mirror 基础

### 基本用法

```swift
struct Person {
    var name: String
    var age: Int
}

let person = Person(name: "Alice", age: 30)

// 创建 Mirror
let mirror = Mirror(reflecting: person)

print("类型: \(mirror.displayStyle)")
print("属性数量: \(mirror.children.count)")

for child in mirror.children {
    if let label = child.label {
        print("\(label): \(child.value)")
    }
}
// name: Alice
// age: 30
```

## 2. 遍历属性

### 获取所有属性

```swift
class Employee {
    var name: String
    var salary: Double
    var department: String

    init(name: String, salary: Double, department: String) {
        self.name = name
        self.salary = salary
        self.department = department
    }
}

func describe<T>(_ value: T) {
    let mirror = Mirror(reflecting: value)

    print("=== \(mirror.description) ===")

    for child in mirror.children {
        print("\(child.label ?? ""): \(child.value)")
    }
}

describe(Employee(name: "Bob", salary: 50000, department: "Engineering"))
```

### 递归遍历

```swift
func dumpProperties<T>(_ value: T, indent: Int = 0) {
    let mirror = Mirror(reflecting: value)
    let prefix = String(repeating: "  ", count: indent)

    for child in mirror.children {
        if let label = child.label {
            let childMirror = Mirror(reflecting: child.value)

            if childMirror.children.count > 0 {
                print("\(prefix)\(label):")
                dumpProperties(child.value, indent: indent + 1)
            } else {
                print("\(prefix)\(label): \(child.value)")
            }
        }
    }
}
```

## 3. 动态类型检查

### 检查属性类型

```swift
struct Config {
    var timeout: Int = 30
    var url: String = "https://api.example.com"
    var debug: Bool = false
}

let config = Config()
let mirror = Mirror(reflecting: config)

for child in mirror.children {
    guard let label = child.label else { continue }

    let typeName = String(describing: type(of: child.value))

    switch child.value {
    case is Int:
        print("\(label) 是 Int: \(child.value)")
    case is String:
        print("\(label) 是 String: \(child.value)")
    case is Bool:
        print("\(label) 是 Bool: \(child.value)")
    default:
        print("\(label) 是其他类型: \(typeName)")
    }
}
```

## 4. 动态创建实例（受限）

### 局限性

```swift
// Swift 反射不能直接创建实例
// 需要使用无参初始化器

class Person {
    var name: String
    var age: Int

    init(name: String, age: Int) {
        self.name = name
        self.age = age
    }
}

// 无法通过反射调用带参数的 init
// let type = Person.self
// let instance = type.init(name: "Test", age: 25)  // 编译错误
```

### 使用 defaultInitializer

```swift
// 如果类型有无参初始化器
class EmptyPerson {
    var name: String = ""
    var age: Int = 0
}

// 可以创建实例
let type = EmptyPerson.self
let instance = type.init()
print(instance.name)  // ""
```

## 5. 实际应用

### 通用打印

```swift
func debugPrint<T>(_ value: T) {
    let mirror = Mirror(reflecting: value)

    var result = "\(mirror.displayStyle?.description ?? "Instance"): {\n"

    for child in mirror.children {
        let label = child.label ?? "(unknown)"
        let valueStr = String(describing: child.value)
        result += "  \(label): \(valueStr)\n"
    }

    result += "}"
    print(result)
}

struct Point {
    var x: Double
    var y: Double
}

debugPrint(Point(x: 10, y: 20))
```

### JSON 序列化辅助

```swift
func toDictionary<T>(_ value: T) -> [String: Any]? {
    let mirror = Mirror(reflecting: value)
    var dict = [String: Any]()

    for child in mirror.children {
        guard let label = child.label else { continue }

        if let dictValue = child.value as? [String: Any] {
            dict[label] = dictValue
        } else if let arrayValue = child.value as? [Any] {
            dict[label] = arrayValue
        } else {
            dict[label] = child.value
        }
    }

    return dict
}

struct User {
    var name: String
    var age: Int
}

if let dict = toDictionary(User(name: "Alice", age: 30)) {
    print(dict)
    // ["name": "Alice", "age": 30]
}
```

## 6. 反射的限制

### Swift 反射的局限

| 功能 | 支持情况 |
|------|---------|
| 读取属性名和值 | 支持 |
| 修改属性值 | 仅限 `nonisolated` 属性 |
| 调用方法 | 不支持 |
| 创建实例 | 仅限无参初始化器 |
| 获取方法名列表 | 有限支持 |

### 使用场景

```swift
// 适合场景：
// - 调试/日志输出
// - 通用序列化
// - 配置检查
// - 测试辅助

// 不适合场景：
// - 运行时动态创建复杂对象
// - 方法调用
// - 性能关键代码
```

## 7. Codable 与反射

### Codable 自动序列化

```swift
struct Product: Codable {
    var name: String
    var price: Double
    var quantity: Int
}

let product = Product(name: "iPhone", price: 999.99, quantity: 2)

// 编码
let encoder = JSONEncoder()
let data = try encoder.encode(product)
let json = String(data: data, encoding: .utf8)!
print(json)
// {"name":"iPhone","price":999.99,"quantity":2}

// 解码
let decoder = JSONDecoder()
let decoded = try decoder.decode(Product.self, from: data)
print(decoded.name)  // iPhone
```

## 实战练习

### 练习 1：属性验证器

```swift
protocol Validatable {
    func validate() -> Bool
}

struct User: Validatable {
    var username: String
    var email: String
    var age: Int

    func validate() -> Bool {
        let mirror = Mirror(reflecting: self)
        var errors: [String] = []

        for child in mirror.children {
            guard let label = child.label else { continue }

            if let stringValue = child.value as? String, stringValue.isEmpty {
                errors.append("\(label) 不能为空")
            }

            if let intValue = child.value as? Int, intValue < 0 {
                errors.append("\(label) 不能为负数")
            }
        }

        if errors.isEmpty {
            return true
        } else {
            print("验证失败: \(errors)")
            return false
        }
    }
}

let validUser = User(username: "alice", email: "alice@example.com", age: 25)
print(validUser.validate())  // true

let invalidUser = User(username: "", email: "", age: -5)
print(invalidUser.validate())
// 验证失败: ["username 不能为空", "email 不能为空", "age 不能为负数"]
```

### 练习 2：表格打印

```swift
func printTable<T>(values: [T]) {
    let mirrors = values.map { Mirror(reflecting: $0) }

    guard let first = mirrors.first, !first.children.isEmpty else { return }

    let labels = first.children.map { $0.label ?? "" }
    let widths = labels.map { $0.count }

    // 打印表头
    var header = "|"
    for (label, width) in zip(labels, widths) {
        header += " \(label.padding(toLength: width, withPad: " ", startingAt: 0)) |"
    }
    print(header)
    print(String(repeating: "-", count: header.count))

    // 打印数据行
    for mirror in mirrors {
        var row = "|"
        for (child, width) in zip(mirror.children, widths) {
            let valueStr = String(describing: child.value)
            row += " \(valueStr.padding(toLength: width, withPad: " ", startingAt: 0)) |"
        }
        print(row)
    }
}

struct Student {
    var name: String
    var score: Int
    var grade: String
}

let students = [
    Student(name: "Alice", score: 92, grade: "A"),
    Student(name: "Bob", score: 78, grade: "C"),
    Student(name: "Carol", score: 95, grade: "A")
]

printTable(values: students)
```

## 今日总结

- [ ] `Mirror(reflecting:)` 创建反射镜像
- [ ] `mirror.children` 遍历属性
- [ ] `child.label` 获取属性名，`child.value` 获取值
- [ ] Swift 反射是只读的，不能修改属性
- [ ] 反射不能调用方法或创建带参实例
- [ ] 反射适用于调试、序列化、验证
- [ ] 生产环境优先用 Codable
