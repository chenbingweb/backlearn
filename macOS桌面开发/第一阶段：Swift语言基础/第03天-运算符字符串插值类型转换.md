## 1. 算术运算符

### 基本运算

```swift
let a = 10
let b = 3

let sum = a + b        // 13
let difference = a - b // 7
let product = a * b    // 30
let quotient = a / b   // 3 (整数除法)
let remainder = a % b  // 1 (取模)
```

### 浮点数除法

```swift
let c = 10.0
let d = 3.0
let floatQuotient = c / d  // 3.333...
```

### 溢出运算

```swift
var maxInt = Int8.max  // 127

// 普通运算溢出会报错
// maxInt + 1  // 运行时错误

// 溢出运算符（回绕）
let overflow = maxInt &+ 1  // -128 (回绕到最小值)
let underflow = Int8.min &- 1  // 127

let multiplied = Int8.max &* 2  // -2
```

### 复合赋值

```swift
var score = 100
score += 10   // 110
score -= 5    // 105
score *= 2    // 210
score /= 3    // 70
score %= 8    // 6
```

## 2. 比较运算符

```swift
let x = 10
let y = 20

x == y    // false，等于
x != y    // true，不等于
x > y     // false，大于
x < y     // true，小于
x >= y    // false，大于等于
x <= y    // true，小于等于
```

### 三元运算符

```swift
let age = 20
let status = age >= 18 ? "成年" : "未成年"
print(status)  // "成年"

// 等价于
let status2: String
if age >= 18 {
    status2 = "成年"
} else {
    status2 = "未成年"
}
```

### 元组比较

```swift
// 元组可以比较（元素逐个比较）
(1, "zebra") < (2, "apple")     // true（1 < 2）
(3, "apple") < (3, "bird")      // true（3 == 3, "apple" < "bird"）
(4, "dog") == (4, "dog")        // true
```

## 3. 逻辑运算符

```swift
let hasTicket = true
let isVIP = false

// 逻辑与
let canEnter = hasTicket && isVIP    // false

// 逻辑或
let canWatch = hasTicket || isVIP    // true

// 逻辑非
let cannotEnter = !hasTicket          // false

// 组合使用
let condition1 = true
let condition2 = false
let condition3 = true

let result = condition1 && (condition2 || condition3)
// true && (false || true) = true
```

### 短路求值

```swift
func expensiveOperation() -> Bool {
    print("执行耗时操作")
    return true
}

let a = false
let b = a && expensiveOperation()
// expensiveOperation 不会执行，因为 a 已经是 false

let c = true
let d = c || expensiveOperation()
// expensiveOperation 不会执行，因为 c 已经是 true
```

## 4. 区间运算符

### 闭区间

```swift
// 包含两端
for index in 1...5 {
    print(index)  // 1, 2, 3, 4, 5
}

let range = 1...5
print(range.contains(3))   // true
print(range.contains(5))   // true
print(range.contains(6))   // false
```

### 半开区间

```swift
// 包含开头，不包含结尾
for index in 1..<5 {
    print(index)  // 1, 2, 3, 4
}

let array = [10, 20, 30, 40, 50]
for i in 0..<array.count {
    print(array[i])
}
```

### 单侧区间

```swift
let names = ["Anna", "Brian", "Jack", "Li", "Tom"]

// 从索引 2 开始到结束
for name in names[2...] {
    print(name)  // "Jack", "Li", "Tom"
}

// 从开始到索引 2
for name in names[...2] {
    print(name)  // "Anna", "Brian", "Jack"
}

// 从开始到索引 2（不包含）
for name in names[..<2] {
    print(name)  // "Anna", "Brian"
}
```

## 5. 字符串插值进阶

### 基本插值

```swift
let name = "Swift"
let version = 5
print("Hello, \(name) \(version)!")
```

### 表达式插值

```swift
let a = 10
let b = 20
print("\(a) + \(b) = \(a + b)")
print("最大值: \(max(a, b))")
```

### 自定义字符串插值

```swift
// Swift 5.0+ 支持自定义插值
extension String.StringInterpolation {
    mutating func appendInterpolation(format value: Double, specifier: String) {
        appendLiteral(String(format: specifier, value))
    }
}

let pi = 3.14159
print("Pi = \(format: pi, specifier: "%.2f")")
// Pi = 3.14
```

### 多行字符串插值

```swift
let user = "张三"
let score = 95

let report = """
    学生: \(user)
    成绩: \(score) 分
    评价: \(score >= 90 ? "优秀" : "良好")
    """

print(report)
```

## 6. 类型转换

### 整数转换

```swift
let int8Value: Int8 = 10
let int16Value = Int16(int8Value)     // 安全转换
let intValue = Int(int8Value)          // 安全转换

// 大类型转小类型可能溢出
let bigInt: Int16 = 1000
// let smallInt = Int8(bigInt)  // 运行时错误
```

### 数值与字符串转换

```swift
// 数值转字符串
let number = 42
let str1 = String(number)
let str2 = "\(number)"

// 字符串转数值（返回 Optional）
let numStr = "42"
if let num = Int(numStr) {
    print("转换成功: \(num)")
}

// 转换失败的情况
let invalidStr = "abc"
let num = Int(invalidStr)   // nil

// 提供默认值
let result = Int(invalidStr) ?? 0   // 0
```

### 浮点数转换

```swift
let intNum = 10
let doubleNum = Double(intNum)   // 10.0

let floatNum: Float = 3.14
let doubleFromFloat = Double(floatNum)

let decimal = 3.99
let integer = Int(decimal)       // 3（截断，不是四舍五入）

// 四舍五入
let rounded = Int(decimal + 0.5)  // 4
```

### Any 与类型转换

```swift
var anyValue: Any = 100
anyValue = "Hello"
anyValue = true

// 向下转型
let unknown: Any = "Hello"

if let string = unknown as? String {
    print("是字符串: \(string)")
}

if let number = unknown as? Int {
    print("是整数: \(number)")
} else {
    print("不是整数")
}
```

## 7. 空合运算符

```swift
// ?? 提供默认值
let optionalName: String? = nil
let displayName = optionalName ?? "匿名用户"
print(displayName)  // "匿名用户"

let realName: String? = "张三"
let name = realName ?? "匿名"
print(name)  // "张三"

// 链式使用
let defaultValue = someOptional ?? anotherOptional ?? "最终默认值"
```

## 8. 自定义运算符

```swift
// 定义前缀运算符
prefix operator √

prefix func √ (number: Double) -> Double {
    return sqrt(number)
}

print(√25)  // 5.0

// 定义中缀运算符
infix operator **: PowerPrecedence

precedencegroup PowerPrecedence {
    associativity: right
    higherThan: MultiplicationPrecedence
}

func ** (base: Double, exponent: Double) -> Double {
    return pow(base, exponent)
}

print(2 ** 3)  // 8.0
```

## 实战练习

### 练习 1：购物车计算

```swift
let price1 = 29.99
let price2 = 15.50
let price3 = 8.00

let quantity1 = 2
let quantity2 = 1
let quantity3 = 3

let subtotal = price1 * Double(quantity1) +
               price2 * Double(quantity2) +
               price3 * Double(quantity3)

let taxRate = 0.08
let tax = subtotal * taxRate
let total = subtotal + tax

print("小计: $\(String(format: "%.2f", subtotal))")
print("税费: $\(String(format: "%.2f", tax))")
print("总计: $\(String(format: "%.2f", total))")
```

### 练习 2：成绩评级系统

```swift
let score = 85

let grade: String
switch score {
case 90...100:
    grade = "A"
case 80..<90:
    grade = "B"
case 70..<80:
    grade = "C"
case 60..<70:
    grade = "D"
default:
    grade = "F"
}

let passStatus = score >= 60 ? "及格" : "不及格"
print("分数: \(score), 等级: \(grade), \(passStatus)")
```
