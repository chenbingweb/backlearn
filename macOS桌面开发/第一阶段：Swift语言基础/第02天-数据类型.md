## 1. 整数类型（Int）

### 基础用法

```swift
let age: Int = 25
let temperature = -10
let bigNumber = 1_000_000  // 使用下划线分隔，更易读

// 不同位数的整数
let small: Int8 = 127       // -128 ~ 127
let medium: Int16 = 32767   // -32768 ~ 32767
let large: Int32 = 2_147_483_647
let huge: Int64 = 9_223_372_036_854_775_807

// 无符号整数
let unsigned: UInt = 100    // 只能表示正数
let byte: UInt8 = 255       // 0 ~ 255
```

### 整数边界

```swift
print(Int8.min)     // -128
print(Int8.max)     // 127
print(Int.min)      // -9223372036854775808 (64位系统)
print(Int.max)      // 9223372036854775807
```

### 进制表示

```swift
let decimal = 17        // 十进制
let binary = 0b10001    // 二进制
let octal = 0o21        // 八进制
let hexadecimal = 0x11  // 十六进制
```

## 2. 浮点类型（Float、Double）

### 区别

| 类型 | 精度 | 占用 |
|------|------|------|
| Float | 6位小数 | 32位 |
| Double | 15位小数 | 64位 |

```swift
// 默认推断为 Double
let pi = 3.14159

// 显式声明
let floatValue: Float = 3.14
let doubleValue: Double = 3.14159265358979

// 科学计数法
let oneMillion = 1e6          // 1,000,000
let tinyNumber = 1e-6         // 0.000001

// 特殊值
let notANumber = Double.nan
let infinity = Double.infinity
```

### 数值字面量

```swift
let decimal = 12.1875
let exponent = 1.21875e1
let hexadecimalFloat = 0xC.3p0
```

## 3. 布尔类型（Bool）

```swift
let isActive = true
let isDeleted = false

// 条件判断
if isActive {
    print("用户已激活")
} else {
    print("用户未激活")
}

// 逻辑运算
let hasPermission = true
let isAdmin = false

let canAccess = isActive && hasPermission  // 与
let canEdit = isAdmin || hasPermission      // 或
let isBanned = !isActive                    // 非
```

### Swift 的特殊之处

```swift
// Swift 中不能用数字代替布尔值
// if 1 { }  // 错误！Int 不能转为 Bool
// if 0 { }  // 错误！

// 必须显式比较
let count = 10
if count > 0 {
    print("有数据")
}
```

## 4. 字符串（String）

### 创建字符串

```swift
let singleLine = "Hello, Swift!"
let multiline = """
    这是多行字符串
    可以包含换行
    缩进会被保留
    """

// 空字符串
let empty1 = ""
let empty2 = String()

// 检查是否为空
if empty1.isEmpty {
    print("字符串为空")
}
```

### 字符串操作

```swift
let str = "Hello, Swift!"

// 长度
print(str.count)           // 13

// 检查前缀后缀
print(str.hasPrefix("Hello"))   // true
print(str.hasSuffix("!"))       // true

// 大小写转换
print(str.uppercased())    // "HELLO, SWIFT!"
print(str.lowercased())    // "hello, swift!"

// 包含检查
print(str.contains("Swift"))  // true

// 替换
let newStr = str.replacingOccurrences(of: "Swift", with: "World")
print(newStr)              // "Hello, World!"
```

### 字符串索引

```swift
let greeting = "Hello, Swift!"

// Swift 的 String 使用 Unicode 安全索引
let start = greeting.startIndex
let firstChar = greeting[start]           // "H"

// 移动到下一个字符
let secondIndex = greeting.index(after: start)
let secondChar = greeting[secondIndex]    // "e"

// 偏移索引
let fifthIndex = greeting.index(start, offsetBy: 4)
let fifthChar = greeting[fifthIndex]      // "o"

// 范围截取
let commaIndex = greeting.firstIndex(of: ",")!
let substring = greeting[start..<commaIndex]
print(substring)           // "Hello"
```

### 字符串遍历

```swift
let message = "Swift"

// 遍历字符
for char in message {
    print(char)
}

// 遍历索引
for index in message.indices {
    print("\(message[index]) 在位置 \(index.utf16Offset(in: message))")
}
```

### 字符串拼接

```swift
// 使用 +
let firstName = "张"
let lastName = "三"
let fullName = firstName + lastName       // "张三"

// 使用 +=
var greeting = "Hello"
greeting += ", World!"                    // "Hello, World!"

// 使用 append
var hello = "Hello"
hello.append("!")                          // "Hello!"

// 使用 append(contentsOf:)
hello.append(contentsOf: " Swift")         // "Hello! Swift"
```

### 字符串插值

```swift
let name = "Swift"
let version = 5.0

// 基本插值
let info = "语言: \(name), 版本: \(version)"

// 表达式插值
let a = 10
let b = 20
let result = "\(a) + \(b) = \(a + b)"

// 格式化插值（Swift 5.0+）
let pi = 3.14159
let formatted = "Pi = \(pi, specifier: "%.2f")"
// Pi = 3.14
```

## 5. 类型转换

### 整数转换

```swift
let intValue = 42
let doubleFromInt = Double(intValue)     // 42.0

let doubleValue = 3.14
let intFromDouble = Int(doubleValue)     // 3（截断小数）

let int8Value: Int8 = 10
let int16Value = Int16(int8Value)        // 安全转换
```

### 字符串与数值转换

```swift
// 数值转字符串
let number = 42
let str1 = String(number)                // "42"
let str2 = "\(number)"                   // "42"

// 字符串转数值
let strNumber = "42"
if let intValue = Int(strNumber) {
    print("转换成功: \(intValue)")
} else {
    print("转换失败")
}

// 带默认值的转换
let invalidNumber = "abc"
let value = Int(invalidNumber) ?? 0      // 0
```

### 类型别名

```swift
typealias Integer = Int
typealias Float32 = Float

let x: Integer = 10
let y: Float32 = 3.14
```

## 6. 元组（Tuple）

```swift
// 定义元组
let person = ("张三", 25, "北京")

// 解构
let (name, age, city) = person
print(name)    // "张三"

// 部分解构
let (userName, _, _) = person

// 命名元组
let student = (name: "李四", score: 90, grade: "A")
print(student.name)    // "李四"
print(student.score)   // 90
```

## 实战练习

### 练习 1：温度转换器

```swift
let celsius = 25.0
let fahrenheit = celsius * 9 / 5 + 32
let kelvin = celsius + 273.15

print("摄氏: \(celsius)°C")
print("华氏: \(fahrenheit)°F")
print("开尔文: \(kelvin)K")
```

### 练习 2：字符串处理工具

```swift
let original = "  Hello, Swift World!  "

// 去除空白
let trimmed = original.trimmingCharacters(in: .whitespaces)

// 分割
let parts = trimmed.split(separator: " ")
print(parts)  // ["Hello,", "Swift", "World!"]

// 统计单词数
let words = trimmed.components(separatedBy: " ")
print("单词数: \(words.count)")
```

### 练习 3：用户信息格式化

```swift
let userName = "Alice"
let userAge = 25
let userScore = 95.5

let profile = """
    ========== 用户信息 ==========
    姓名: \(userName)
    年龄: \(userAge) 岁
    分数: \(userScore) 分
    评级: \(userScore >= 90 ? "优秀" : "良好")
    =============================
    """

print(profile)
```

## 今日总结

- [ ] Int、Float、Double 表示数值，注意精度区别
- [ ] Bool 只有 true/false，不能用数字代替
- [ ] String 支持丰富的操作：索引、遍历、拼接、插值
- [ ] 类型转换需要显式进行，可能失败
- [ ] 元组可以组合多个值，支持命名和解构
