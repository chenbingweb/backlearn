### 1. 数据模型

```swift
// Student.swift
struct Student {
    let id: String
    var name: String
    var scores: [String: Double]  // 科目: 成绩

    var totalScore: Double {
        return scores.values.reduce(0, +)
    }

    var averageScore: Double {
        return scores.isEmpty ? 0 : totalScore / Double(scores.count)
    }

    func score(for subject: String) -> Double {
        return scores[subject] ?? 0
    }
}

extension Student: CustomStringConvertible {
    var description: String {
        return "\(id)\t\(name)\t\(String(format: "%.1f", averageScore))"
    }
}
```

### 2. 管理系统

```swift
// GradeManager.swift
class GradeManager {
    private var students: [String: Student] = [:]  // 学号: 学生

    // 添加学生
    func addStudent(id: String, name: String) -> Bool {
        guard students[id] == nil else {
            print("学号 \(id) 已存在")
            return false
        }

        students[id] = Student(id: id, name: name, scores: [:])
        print("添加成功: \(name) (\(id))")
        return true
    }

    // 删除学生
    func removeStudent(id: String) -> Bool {
        guard students[id] != nil else {
            print("学号 \(id) 不存在")
            return false
        }

        students.removeValue(forKey: id)
        print("删除成功")
        return true
    }

    // 录入成绩
    func addScore(studentId: String, subject: String, score: Double) -> Bool {
        guard var student = students[studentId] else {
            print("学号 \(studentId) 不存在")
            return false
        }

        guard score >= 0 && score <= 100 else {
            print("成绩必须在 0-100 之间")
            return false
        }

        student.scores[subject] = score
        students[studentId] = student
        print("成绩录入成功")
        return true
    }

    // 查询学生
    func queryStudent(id: String) {
        guard let student = students[id] else {
            print("学号 \(id) 不存在")
            return
        }

        print("\n===== 学生信息 =====")
        print("学号: \(student.id)")
        print("姓名: \(student.name)")
        print("\n成绩详情:")
        for (subject, score) in student.scores.sorted(by: { $0.key < $1.key }) {
            print("  \(subject): \(score)")
        }
        print("\n总分: \(String(format: "%.1f", student.totalScore))")
        print("平均分: \(String(format: "%.1f", student.averageScore))")
        print("==================\n")
    }

    // 显示所有学生
    func listAllStudents() {
        if students.isEmpty {
            print("没有学生记录")
            return
        }

        print("\n===== 学生列表 =====")
        print("学号\t\t姓名\t\t平均分")
        print("-".repeating(count: 40))

        let sorted = students.values.sorted { $0.id < $1.id }
        for student in sorted {
            print(student)
        }
        print("==================\n")
    }

    // 排名
    func showRanking(by subject: String? = nil) {
        if students.isEmpty {
            print("没有学生记录")
            return
        }

        let sorted: [Student]
        if let subject = subject {
            // 按科目排名
            sorted = students.values.sorted { $0.score(for: subject) > $1.score(for: subject) }
            print("\n===== \(subject) 排名 =====")
        } else {
            // 按平均分排名
            sorted = students.values.sorted { $0.averageScore > $1.averageScore }
            print("\n===== 综合排名 =====")
        }

        print("排名\t学号\t\t姓名\t\t成绩")
        print("-".repeating(count: 50))

        for (index, student) in sorted.enumerated() {
            let score: Double
            if let subject = subject {
                score = student.score(for: subject)
            } else {
                score = student.averageScore
            }
            print("\(index + 1)\t\t\(student.id)\t\(student.name)\t\t\(String(format: "%.1f", score))")
        }
        print("==================\n")
    }

    // 统计信息
    func showStatistics(for subject: String) {
        let scores = students.values.map { $0.score(for: subject) }.filter { $0 > 0 }

        guard !scores.isEmpty else {
            print("没有 \(subject) 的成绩记录")
            return
        }

        let sum = scores.reduce(0, +)
        let avg = sum / Double(scores.count)
        let max = scores.max() ?? 0
        let min = scores.min() ?? 0

        print("\n===== \(subject) 统计 =====")
        print("人数: \(scores.count)")
        print("最高分: \(max)")
        print("最低分: \(min)")
        print("平均分: \(String(format: "%.1f", avg))")
        print("==================\n")
    }
}
```

```swift
// main.swift
func printMenu() {
    print("\n===== 学生成绩管理系统 =====")
    print("1. 添加学生")
    print("2. 删除学生")
    print("3. 录入成绩")
    print("4. 查询学生")
    print("5. 显示所有学生")
    print("6. 综合排名")
    print("7. 科目排名")
    print("8. 科目统计")
    print("0. 退出")
    print("========================")
    print("请选择功能: ", terminator: "")
}

func run() {
    let manager = GradeManager()

    // 预置一些测试数据
    manager.addStudent(id: "S001", name: "Alice")
    manager.addStudent(id: "S002", name: "Bob")
    manager.addStudent(id: "S003", name: "Charlie")

    manager.addScore(studentId: "S001", subject: "Math", score: 95)
    manager.addScore(studentId: "S001", subject: "English", score: 88)
    manager.addScore(studentId: "S002", subject: "Math", score: 78)
    manager.addScore(studentId: "S002", subject: "English", score: 92)
    manager.addScore(studentId: "S003", subject: "Math", score: 88)
    manager.addScore(studentId: "S003", subject: "English", score: 85)

    var running = true

    while running {
        printMenu()

        guard let choice = readLine() else { continue }

        switch choice {
        case "1":
            print("请输入学号: ", terminator: "")
            guard let id = readLine(), !id.isEmpty else {
                print("学号不能为空")
                continue
            }
            print("请输入姓名: ", terminator: "")
            guard let name = readLine(), !name.isEmpty else {
                print("姓名不能为空")
                continue
            }
            manager.addStudent(id: id, name: name)

        case "2":
            print("请输入学号: ", terminator: "")
            guard let id = readLine() else { continue }
            manager.removeStudent(id: id)

        case "3":
            print("请输入学号: ", terminator: "")
            guard let id = readLine() else { continue }
            print("请输入科目: ", terminator: "")
            guard let subject = readLine() else { continue }
            print("请输入成绩: ", terminator: "")
            guard let scoreStr = readLine(), let score = Double(scoreStr) else {
                print("成绩格式错误")
                continue
            }
            manager.addScore(studentId: id, subject: subject, score: score)

        case "4":
            print("请输入学号: ", terminator: "")
            guard let id = readLine() else { continue }
            manager.queryStudent(id: id)

        case "5":
            manager.listAllStudents()

        case "6":
            manager.showRanking()

        case "7":
            print("请输入科目: ", terminator: "")
            guard let subject = readLine() else { continue }
            manager.showRanking(by: subject)

        case "8":
            print("请输入科目: ", terminator: "")
            guard let subject = readLine() else { continue }
            manager.showStatistics(for: subject)

        case "0":
            print("感谢使用，再见！")
            running = false

        default:
            print("无效选择，请重新输入")
        }
    }
}

// 启动程序
run()
```

### 练习 1：添加文件持久化

```swift
import Foundation

extension GradeManager {
    func save(to file: String) {
        let data = students.values.map { [
            "id": $0.id,
            "name": $0.name,
            "scores": $0.scores
        ] }

        if let json = try? JSONSerialization.data(withJSONObject: data) {
            try? json.write(to: URL(fileURLWithPath: file))
        }
    }

    func load(from file: String) {
        guard let data = try? Data(contentsOf: URL(fileURLWithPath: file)),
              let json = try? JSONSerialization.jsonObject(with: data) as? [[String: Any]] else {
            return
        }

        for item in json {
            guard let id = item["id"] as? String,
                  let name = item["name"] as? String else { continue }

            addStudent(id: id, name: name)

            if let scores = item["scores"] as? [String: Double] {
                for (subject, score) in scores {
                    addScore(studentId: id, subject: subject, score: score)
                }
            }
        }
    }
}
```

### 练习 2：添加成绩等级

```swift
enum GradeLevel: String {
    case a = "A"
    case b = "B"
    case c = "C"
    case d = "D"
    case f = "F"

    init(score: Double) {
        switch score {
        case 90...100: self = .a
        case 80..<90: self = .b
        case 70..<80: self = .c
        case 60..<70: self = .d
        default: self = .f
        }
    }
}

extension Student {
    func gradeLevel(for subject: String) -> GradeLevel {
        return GradeLevel(score: score(for: subject))
    }
}
```
