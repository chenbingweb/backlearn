## 1. 数组（Array）

### 创建数组

```swift
// 空数组
var emptyArray: [Int] = []
var anotherEmpty = [String]()

// 带初始值
var numbers = [1, 2, 3, 4, 5]
var fruits = ["Apple", "Banana", "Orange"]

// 使用 Array 初始化器
var zeros = Array(repeating: 0, count: 5)
// [0, 0, 0, 0, 0]

// 推断类型
var mixed = [1, 2, 3]        // [Int]
var texts = ["a", "b"]       // [String]
```

### 访问元素

```swift
var colors = ["Red", "Green", "Blue", "Yellow"]

// 下标访问
let first = colors[0]        // "Red"
let second = colors[1]       // "Green"

// 安全访问
if let firstColor = colors.first {
    print(firstColor)        // "Red"
}

if let lastColor = colors.last {
    print(lastColor)         // "Yellow"
}

// 检查是否为空
if !colors.isEmpty {
    print("数组不为空")
}

// 获取数量
print(colors.count)          // 4
```

### 修改数组

```swift
var items = ["Book", "Pen", "Eraser"]

// 修改元素
items[0] = "Notebook"
// ["Notebook", "Pen", "Eraser"]

// 添加元素
items.append("Ruler")           // 末尾添加
items.insert("Pencil", at: 1)   // 指定位置插入
// ["Notebook", "Pencil", "Pen", "Eraser", "Ruler"]

// 批量添加
items += ["Sharpener", "Glue"]

// 删除元素
items.remove(at: 2)             // 删除索引2的元素
items.removeFirst()             // 删除第一个
items.removeLast()              // 删除最后一个
items.removeAll()               // 删除全部
```

### 数组遍历

```swift
let scores = [85, 92, 78, 95, 88]

// for-in 遍历
for score in scores {
    print(score)
}

// 带索引遍历
for (index, score) in scores.enumerated() {
    print("第 \(index + 1) 名: \(score) 分")
}

// forEach
scores.forEach { score in
    print(score)
}

// 简写
scores.forEach { print($0) }
```

### 数组常用方法

```swift
var numbers = [3, 1, 4, 1, 5, 9, 2, 6]

// 排序
let sorted = numbers.sorted()              // [1, 1, 2, 3, 4, 5, 6, 9]
let reversed = numbers.sorted(by: >)       // [9, 6, 5, 4, 3, 2, 1, 1]

// 原地排序
numbers.sort()                             // 修改原数组

// 反转
let reversedArray = numbers.reversed()

// 包含检查
let hasFive = numbers.contains(5)          // true

// 查找索引
if let index = numbers.firstIndex(of: 4) {
    print("4 在索引 \(index)")
}

// 过滤
let evens = numbers.filter { $0 % 2 == 0 }  // [2, 4, 6]

// 映射
let doubled = numbers.map { $0 * 2 }        // [2, 2, 4, 6, 8, 10, 12, 18]

// 累加
let sum = numbers.reduce(0, +)              // 31

// 切片
let slice = numbers[2...5]                  // [2, 3, 4, 5]
```

### 多维数组

```swift
// 二维数组
var matrix = [
    [1, 2, 3],
    [4, 5, 6],
    [7, 8, 9]
]

print(matrix[0][0])    // 1
print(matrix[1][2])    // 6

// 修改
matrix[0][0] = 100

// 遍历
for row in matrix {
    for element in row {
        print(element, terminator: " ")
    }
    print()
}
```

## 2. 集合（Set）

### 创建集合

```swift
// 空集合
var emptySet = Set<Int>()

// 从数组创建
var numbers = Set([1, 2, 3, 4, 5])

// 字面量创建（需要显式类型）
var fruits: Set<String> = ["Apple", "Banana", "Orange"]

// 重复元素自动去重
var uniqueNumbers = Set([1, 2, 2, 3, 3, 3])
// {1, 2, 3}
```

### 集合操作

```swift
var setA: Set = [1, 2, 3, 4, 5]
var setB: Set = [4, 5, 6, 7, 8]

// 并集
let union = setA.union(setB)
// {1, 2, 3, 4, 5, 6, 7, 8}

// 交集
let intersection = setA.intersection(setB)
// {4, 5}

// 差集
let difference = setA.subtracting(setB)
// {1, 2, 3}

// 对称差集
let symmetricDiff = setA.symmetricDifference(setB)
// {1, 2, 3, 6, 7, 8}
```

### 集合关系

```swift
let smallSet: Set = [1, 2]
let largeSet: Set = [1, 2, 3, 4]

// 子集
print(smallSet.isSubset(of: largeSet))        // true

// 超集
print(largeSet.isSuperset(of: smallSet))      // true

// 真子集
print(smallSet.isStrictSubset(of: largeSet))  // true

// 是否有交集
print(smallSet.isDisjoint(with: [5, 6]))      // true（没有共同元素）
```

### 修改集合

```swift
var colors: Set = ["Red", "Green", "Blue"]

// 添加
colors.insert("Yellow")

// 删除
colors.remove("Green")

// 安全删除（返回 Optional）
if let removed = colors.remove("Red") {
    print("删除了: \(removed)")
}

// 删除并返回任意元素
if let random = colors.popFirst() {
    print("随机删除: \(random)")
}

// 清空
colors.removeAll()
```

### 集合遍历

```swift
let vowels: Set = ["A", "E", "I", "O", "U"]

// 遍历（无序）
for vowel in vowels {
    print(vowel)
}

// 排序后遍历
for vowel in vowels.sorted() {
    print(vowel)
}
```

## 3. 数组 vs 集合

| 特性 | 数组 Array | 集合 Set |
|------|-----------|---------|
| 顺序 | 有序 | 无序 |
| 重复 | 允许 | 不允许 |
| 访问 | 索引访问 | 无索引 |
| 性能（查找）| O(n) | O(1) |
| 用途 | 有序列表 | 唯一性检查 |

## 实战练习

### 练习 1：数组操作

```swift
var tasks = ["买牛奶", "写代码", "健身", "看书", "买菜"]

// 添加任务
tasks.append("打电话")

// 完成第一个任务
tasks.removeFirst()

// 在指定位置插入
tasks.insert("开会", at: 1)

// 打印剩余任务
print("剩余 \(tasks.count) 个任务:")
for (index, task) in tasks.enumerated() {
    print("\(index + 1). \(task)")
}
```

### 练习 2：集合去重

```swift
let duplicateNumbers = [1, 2, 2, 3, 3, 3, 4, 4, 4, 4]
let unique = Array(Set(duplicateNumbers)).sorted()
print(unique)  // [1, 2, 3, 4]

// 保留顺序的去重
func uniqueOrdered<T: Hashable>(_ array: [T]) -> [T] {
    var seen = Set<T>()
    return array.filter { seen.insert($0).inserted }
}

let ordered = uniqueOrdered(duplicateNumbers)
print(ordered)  // [1, 2, 3, 4]
```

### 练习 3：学生成绩统计

```swift
let scores = [85, 92, 78, 95, 88, 92, 78, 85, 95, 100]

// 最高分、最低分、平均分
let maxScore = scores.max() ?? 0
let minScore = scores.min() ?? 0
let average = Double(scores.reduce(0, +)) / Double(scores.count)

// 去重后的分数段
let uniqueScores = Set(scores).sorted()

// 各分数出现次数
var scoreCount: [Int: Int] = [:]
for score in scores {
    scoreCount[score, default: 0] += 1
}

print("最高分: \(maxScore)")
print("最低分: \(minScore)")
print("平均分: \(String(format: "%.1f", average))")
print("不同分数: \(uniqueScores)")
print("分数分布: \(scoreCount)")
```

## 今日总结

- [ ] 数组有序、可重复，支持索引访问
- [ ] 集合无序、不重复，查找性能 O(1)
- [ ] 数组常用方法：sort、filter、map、reduce
- [ ] 集合支持数学运算：并集、交集、差集
- [ ] 根据场景选择：有序用 Array，唯一性用 Set
