## 1. if 语句

### 基本形式

```swift
let temperature = 25

if temperature > 30 {
    print("很热")
} else if temperature > 20 {
    print("舒适")    // 输出
} else {
    print("凉爽")
}
```

### 条件必须是 Bool

```swift
// Swift 不允许隐式转换
let count = 10

// 错误写法
// if count { }  // 编译错误

// 正确写法
if count > 0 {
    print("有数据")
}

// 判断 Optional
let name: String? = "Alice"
if name != nil {
    print(name!)
}
```

### 使用 let 绑定

```swift
let score: Int? = 85

if let actualScore = score {
    print("分数是 \(actualScore)")
} else {
    print("没有分数")
}

// 多个绑定
let firstName: String? = "张"
let lastName: String? = "三"

if let first = firstName, let last = lastName {
    print("全名: \(first)\(last)")
}

// 带条件
if let score = score, score >= 60 {
    print("及格了，分数: \(score)")
}
```

### 逗号分隔条件

```swift
let age = 25
let hasID = true

if age >= 18, hasID {
    print("可以进入")
}

// 等价于
if age >= 18 && hasID {
    print("可以进入")
}
```

## 2. switch 语句

### 基本用法

```swift
let grade = "A"

switch grade {
case "A":
    print("优秀")
case "B":
    print("良好")
case "C":
    print("及格")
case "D", "F":
    print("不及格")
default:
    print("未知等级")
}
```

### 不需要 break

```swift
// Swift 默认不会穿透（fallthrough）
let number = 5

switch number {
case 1:
    print("一")    // 不会执行
 case 2:
    print("二")    // 不会执行
case 5:
    print("五")    // 输出 "五"
    // 不需要 break，自动跳出
case 6:
    print("六")    // 不会执行
default:
    print("其他")
}
```

### 显式 fallthrough

```swift
let value = 5

switch value {
case 5:
    print("是 5")
    fallthrough     // 继续执行下一个 case
case 6:
    print("或是 6")   // 也输出
 default:
    print("结束")
}
// 输出:
// 是 5
// 或是 6
```

### 区间匹配

```swift
let score = 85

switch score {
case 0..<60:
    print("不及格")
case 60..<70:
    print("及格")
case 70..<80:
    print("中等")
case 80..<90:
    print("良好")    // 输出
case 90...100:
    print("优秀")
default:
    print("无效分数")
}
```

### 元组匹配

```swift
let point = (1, 1)

switch point {
case (0, 0):
    print("原点")
case (_, 0):
    print("x 轴上")
case (0, _):
    print("y 轴上")
case (-2...2, -2...2):
    print("在范围内")    // 输出
default:
    print("在范围外")
}
```

### 值绑定

```swift
let point = (2, 0)

switch point {
case (let x, 0):
    print("在 x 轴上，x = \(x)")
case (0, let y):
    print("在 y 轴上，y = \(y)")
case let (x, y):
    print("x = \(x), y = \(y)")
}
```

### where 子句

```swift
let point = (10, -5)

switch point {
case let (x, y) where x == y:
    print("x = y")
case let (x, y) where x == -y:
    print("x = -y")    // 输出
case let (x, y) where x > 0 && y > 0:
    print("第一象限")
default:
    print("其他")
}
```

### 枚举匹配

```swift
enum Direction {
    case north, south, east, west
}

let direction = Direction.east

switch direction {
case .north:
    print("北")
case .south:
    print("南")
case .east:
    print("东")     // 输出
case .west:
    print("西")
}
// 枚举不需要 default，如果覆盖所有 case
```

## 3. guard 语句

### 基本用法

```swift
func greet(person: [String: String]) {
    guard let name = person["name"] else {
        print("没有名字")
        return
    }

    print("Hello, \(name)!")

    guard let age = person["age"] else {
        print("不知道年龄")
        return
    }

    print("你 \(age) 岁了")
}

greet(person: ["name": "Alice"])
// Hello, Alice!
// 不知道年龄
```

### guard 与 if let 的区别

```swift
// if let - 处理存在的情况
func process1(value: Int?) {
    if let v = value {
        // 值存在时的逻辑
        print(v)
    }
    // 继续执行
}

// guard let - 处理不存在的情况
func process2(value: Int?) {
    guard let v = value else {
        // 值不存在时的逻辑
        return
    }
    // 值存在，v 可在后续使用
    print(v)
}
```

### guard 多个条件

```swift
func register(name: String?, age: Int?, password: String?) {
    guard let name = name, !name.isEmpty else {
        print("名字不能为空")
        return
    }

    guard let age = age, age >= 18 else {
        print("必须年满18岁")
        return
    }

    guard let password = password, password.count >= 6 else {
        print("密码至少6位")
        return
    }

    print("注册成功: \(name), \(age)岁")
}

register(name: "Alice", age: 25, password: "123456")
// 注册成功: Alice, 25岁
```

## 5. 可选链与空合

```swift
let user: [String: Any] = [
    "profile": [
        "name": "Alice",
        "address": [
            "city": "Beijing"
        ]
    ]
]

// 可选链
let city = (user["profile"] as? [String: Any])?["address"] as? [String: Any])?["city"] as? String
print(city ?? "未知城市")

// 使用空合运算符
let displayName = (user["profile"] as? [String: Any])?["name"] as? String ?? "匿名"
```

## 实战练习

### 练习 1：成绩评级

```swift
func gradeFor(score: Int) -> String {
    switch score {
    case 90...100:
        return "A"
    case 80..<90:
        return "B"
    case 70..<80:
        return "C"
    case 60..<70:
        return "D"
    case 0..<60:
        return "F"
    default:
        return "无效"
    }
}

let scores = [95, 82, 75, 60, 45, -5, 105]
for score in scores {
    print("\(score) 分 -> \(gradeFor(score: score))")
}
```

### 练习 2：登录验证

```swift
func login(username: String?, password: String?) -> String {
    guard let username = username, !username.isEmpty else {
        return "用户名不能为空"
    }

    guard let password = password, !password.isEmpty else {
        return "密码不能为空"
    }

    guard password.count >= 6 else {
        return "密码长度不能少于6位"
    }

    // 模拟验证
    if username == "admin" && password == "123456" {
        return "登录成功"
    } else {
        return "用户名或密码错误"
    }
}

print(login(username: nil, password: "123"))
print(login(username: "admin", password: "123"))
print(login(username: "admin", password: "123456"))
```

### 练习 3：季节判断

```swift
func seasonFor(month: Int) -> String {
    switch month {
    case 3...5:
        return "春季"
    case 6...8:
        return "夏季"
    case 9...11:
        return "秋季"
    case 1, 2, 12:
        return "冬季"
    default:
        return "无效月份"
    }
}

for month in 1...12 {
    print("\(month)月 -> \(seasonFor(month: month))")
}
```

## 今日总结

- [ ] if 条件必须是 Bool，不能隐式转换
- [ ] if let 用于 Optional 绑定，可带多个条件
- [ ] switch 默认不穿透，支持区间、元组、where
- [ ] guard 用于提前返回，让代码结构更清晰
- [ ] 枚举 switch 如果覆盖所有 case 不需要 default
