## 1. 函数基础

### 定义函数

```swift
// 无参数，无返回值
func sayHello() {
    print("Hello!")
}

sayHello()

// 有参数，无返回值
func greet(person: String) {
    print("Hello, \(person)!")
}

greet(person: "Alice")

// 有参数，有返回值
func add(a: Int, b: Int) -> Int {
    return a + b
}

let result = add(a: 3, b: 5)
print(result)  // 8

// 多返回值（使用元组）
func getMinMax(numbers: [Int]) -> (min: Int, max: Int) {
    return (numbers.min()!, numbers.max()!)
}

let range = getMinMax(numbers: [3, 1, 4, 1, 5])
print("最小: \(range.min), 最大: \(range.max)")
```

### 忽略返回值

```swift
func printAndReturn(message: String) -> String {
    print(message)
    return message
}

// 使用 _ 忽略返回值
_ = printAndReturn(message: "Hello")
```

## 2. 参数标签

### 默认参数标签

```swift
// 参数名同时作为参数标签
func greet(person: String, from hometown: String) {
    print("\(person) 来自 \(hometown)")
}

greet(person: "Bill", from: "Cupertino")
```

### 省略参数标签

```swift
// 使用 _ 省略参数标签
func multiply(_ a: Int, _ b: Int) -> Int {
    return a * b
}

let product = multiply(3, 5)  // 不需要标签
print(product)  // 15
```

### 自定义参数标签

```swift
// 参数标签和参数名不同
func greet(to person: String) {
    print("Hello, \(person)!")
}

greet(to: "Alice")

// 更复杂的例子
func transfer(amount money: Double, from sender: String, to receiver: String) {
    print("\(sender) 转账 \(money) 给 \(receiver)")
}

transfer(amount: 100.0, from: "Alice", to: "Bob")
```

## 3. 默认参数

### 基本用法

```swift
func greet(person: String, greeting: String = "Hello") {
    print("\(greeting), \(person)!")
}

greet(person: "Alice")                    // "Hello, Alice!"
greet(person: "Bob", greeting: "Hi")      // "Hi, Bob!"
```

### 多个默认参数

```swift
func createUser(
    name: String,
    age: Int = 18,
    city: String = "Beijing",
    isVIP: Bool = false
) {
    print("创建用户: \(name), \(age)岁, \(city), VIP: \(isVIP)")
}

createUser(name: "Alice")                           // 全部默认
createUser(name: "Bob", age: 25)                     // 部分默认
createUser(name: "Charlie", city: "Shanghai")        // 跳过中间参数
createUser(name: "David", isVIP: true)               // 使用参数名跳过
```

### 默认参数的顺序

```swift
// 建议将带默认值的参数放在后面
func badFunction(defaultParam: Int = 10, requiredParam: String) {
    // 调用时需要写标签
}

func goodFunction(requiredParam: String, defaultParam: Int = 10) {
    // 调用时更简洁
}
```

## 4. 可变参数

### 基本用法

```swift
func sum(_ numbers: Int...) -> Int {
    var total = 0
    for number in numbers {
        total += number
    }
    return total
}

print(sum(1, 2, 3))       // 6
print(sum(1, 2, 3, 4, 5))  // 15
print(sum())               // 0
```

### 可变参数类型

```swift
func joinStrings(_ strings: String..., separator: String = " ") -> String {
    return strings.joined(separator: separator)
}

print(joinStrings("Hello", "World"))           // "Hello World"
print(joinStrings("a", "b", "c", separator: ","))  // "a,b,c"
```

### 注意事项

```swift
// 一个函数只能有一个可变参数
// 可变参数必须放在最后（除非后面有默认参数）
func example(normal: Int, variadic: String..., flag: Bool = true) {
    // 可以这么定义
}
```

## 5. 输入输出参数（inout）

### 基本用法

```swift
func swapValues(_ a: inout Int, _ b: inout Int) {
    let temp = a
    a = b
    b = temp
}

var x = 10
var y = 20
swapValues(&x, &y)
print("x = \(x), y = \(y)")  // x = 20, y = 10
```

### 使用场景

```swift
func increment(_ value: inout Int, by amount: Int = 1) {
    value += amount
}

var counter = 0
increment(&counter)
print(counter)  // 1

increment(&counter, by: 5)
print(counter)  // 6
```

### 注意事项

```swift
// inout 参数不能有默认值
// inout 参数不能是可变参数
// 传入的必须是变量，不能是常量或字面量

// 错误示例
// let constant = 10
// increment(&constant)  // 错误！

// increment(&10)  // 错误！
```

## 6. 函数类型

### 作为变量

```swift
func add(a: Int, b: Int) -> Int {
    return a + b
}

// 函数类型: (Int, Int) -> Int
var mathOperation: (Int, Int) -> Int = add

let result = mathOperation(3, 5)
print(result)  // 8
```

### 作为参数

```swift
func printResult(_ operation: (Int, Int) -> Int, a: Int, b: Int) {
    let result = operation(a, b)
    print("结果: \(result)")
}

func multiply(a: Int, b: Int) -> Int {
    return a * b
}

printResult(add, a: 3, b: 5)       // 结果: 8
printResult(multiply, a: 3, b: 5)  // 结果: 15
```

### 作为返回值

```swift
func chooseOperation(add: Bool) -> (Int, Int) -> Int {
    if add {
        return { $0 + $1 }
    } else {
        return { $0 * $1 }
    }
}

let operation = chooseOperation(add: false)
print(operation(3, 5))  // 15
```

## 7. 嵌套函数

```swift
func calculate(_ a: Int, _ b: Int, operation: String) -> Int {

    func add() -> Int { return a + b }
    func subtract() -> Int { return a - b }
    func multiply() -> Int { return a * b }

    switch operation {
    case "+": return add()
    case "-": return subtract()
    case "*": return multiply()
    default: return 0
    }
}

print(calculate(10, 5, operation: "+"))  // 15
print(calculate(10, 5, operation: "-"))  // 5
print(calculate(10, 5, operation: "*"))  // 50
```

## 实战练习

### 练习 1：计算器函数

```swift
func calculate(_ a: Double, _ b: Double, operation: String = "+") -> Double {
    switch operation {
    case "+": return a + b
    case "-": return a - b
    case "*": return a * b
    case "/": return b != 0 ? a / b : 0
    default: return 0
    }
}

print(calculate(10, 3))           // 13.0
print(calculate(10, 3, operation: "*"))  // 30.0
print(calculate(10, 3, operation: "/"))  // 3.33...
```

### 练习 2：统计函数

```swift
func statistics(_ numbers: Double...) -> (sum: Double, avg: Double, min: Double, max: Double) {
    let sum = numbers.reduce(0, +)
    let avg = sum / Double(numbers.count)
    let min = numbers.min() ?? 0
    let max = numbers.max() ?? 0

    return (sum, avg, min, max)
}

let stats = statistics(85, 92, 78, 95, 88)
print("总和: \(stats.sum)")
print("平均: \(stats.avg)")
print("最低: \(stats.min)")
print("最高: \(stats.max)")
```

### 练习 3：配置构建器

```swift
func buildConfig(
    host: String = "localhost",
    port: Int = 8080,
    useSSL: Bool = false,
    timeout: Int = 30
) -> [String: Any] {
    return [
        "host": host,
        "port": port,
        "ssl": useSSL,
        "timeout": timeout
    ]
}

let defaultConfig = buildConfig()
let prodConfig = buildConfig(host: "api.example.com", useSSL: true)
let customConfig = buildConfig(host: "localhost", port: 3000, timeout: 60)

print(defaultConfig)
print(prodConfig)
print(customConfig)
```

## 今日总结

- [ ] 函数使用 `func` 定义，参数可以有标签和名称
- [ ] 使用 `_` 省略参数标签，使用自定义标签提高可读性
- [ ] 默认参数让函数调用更灵活
- [ ] 可变参数使用 `...`，一个函数只能有一个
- [ ] inout 参数可以修改外部变量的值
- [ ] 函数可以作为变量、参数和返回值
